# EN: ENTRE RIOS -- Urquiza's province. Capital Parana.
# ES: ENTRE RIOS -- la provincia de Urquiza. Capital, Parana.
ENR = {
	# EN: the blue of the Echague tricolour of 28 December 1833, the flag Urquiza carried to Caseros.
	#     Deliberately dark: the law banned celeste as the unitarian colour, so a dark blue is the
	#     political statement of that government and it also tells Entre Rios from the Coalition's
	#     light blue on the map.
	# ES: el azul de la tricolor de Echague del 28/12/1833, la bandera que Urquiza llevo a Caseros.
	#     Oscuro a proposito: la ley prohibia el celeste por unitario, asi que el azul intenso es la
	#     declaracion politica de ese gobierno, y de paso distingue a Entre Rios del celeste de la
	#     Coalicion en el mapa.
	color = { 26 42 120 }

	country_type = recognized

	tier = principality

	religion = catholic

	cultures = { platinean }

	capital = STATE_ENTRE_RIOS
}
