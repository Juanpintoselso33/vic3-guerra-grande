# EN: What each government does with its bank of the River Plate. The engine has no effect to set a
#     strait's access: its controller sets it -- the player in the strait panel, the AI through these
#     scores. Verified in game on 21/9/2026: the AI obeys set_strategy (Rosas closed his bank on
#     10/1/1836), and weight alone is not enough because the AI only re-rolls its strategies about
#     every ten years. So weight is 0 and the script sets them (gg_plata_rosas_* effects).
#     All four are administrative: Argentina's mod strategies are diplomatic, so these never replace
#     them. Only the Plata scores are touched; every other strait keeps vanilla's behaviour.
# ES: Que hace cada gobierno con su orilla del Plata. El motor no tiene efecto para fijar el acceso
#     de un estrecho: lo fija su controlador -- el jugador en el panel, la IA con estos puntajes.
#     Verificado en partida el 21/9/2026: la IA obedece a set_strategy, y el weight solo no alcanza
#     porque la IA re-elige estrategias cada unos diez anios. Por eso weight 0 y las fija el script.
#     Las cuatro son administrativas: las del mod para la Argentina son diplomaticas, asi que estas
#     nunca las reemplazan. Solo se tocan los puntajes del Plata; los demas estrechos quedan como
#     en vanilla.

# EN/ES: Rosas, river open -- 1836 until the siege, and after he yields / rio abierto
gg_ai_strategy_plata_open = {
	icon = "gfx/interface/icons/ai_strategy_icons/placate_population.dds"
	type = administrative
	strait_scores = {
		status_scores = {
			open = {
				value = 0
				if = { limit = { scope:strait = { type = strait_type:strait_rio_de_la_plata } } add = 1000 }
			}
		}
		permissions_scores = {
			all = {
				value = 0
				if = { limit = { scope:strait = { type = strait_type:strait_rio_de_la_plata } } add = 1000 }
			}
		}
	}
	possible = { c:ARG ?= this }
	weight = { value = 0 }
}
