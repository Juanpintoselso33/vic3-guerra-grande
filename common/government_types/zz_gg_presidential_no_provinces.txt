# EN: 23/9/2026 (headless run: Tucuman, Jujuy, Corrientes and La Rioja still showed "President") --
#     the engine does not pick the "best" valid government type, so vanilla keeps its generic
#     presidential types from ever matching a Peru-Bolivia member (NAND pbc_member, 02_presidential_
#     republics.txt). Same method here: these nine generic types are vanilla's, copied whole, plus one
#     line that leaves the Argentine province countries to gov_gg_provincia (governor).
#     Re-copy from vanilla on a game update; only the gg_has_province_governor line is ours (was gg_is_province_country until 0.1.2).
# ES: el motor no elige el "mejor" tipo valido, asi que vanilla impide a mano que los tipos presidenciales
#     genericos le toquen a un miembro de la Confederacion Peru-Boliviana. Mismo metodo: estos nueve son
#     los de vanilla, copiados enteros, mas una linea que deja a las provincias argentinas en
#     gov_gg_provincia (gobernador). Recopiar de vanilla si el juego se actualiza.

REPLACE:gov_technate_presidential = {
	transfer_of_power = dictatorial

	male_ruler = "RULER_TITLE_CHIEF_EXECUTIVE"
	female_ruler = "RULER_TITLE_CHIEF_EXECUTIVE"

	possible = {
		has_law_or_variant = law_type:law_presidential_republic
		has_law_or_variant = law_type:law_technocracy
		NAND = {
			is_involved_in_journal_entry = je_peru_bolivia
			has_variable = pbc_member
		}
		NOT = { gg_has_province_governor = yes } # EN/ES: Argentine provinces (and the Oriental Province, AUD-ORIENTAL-PROVINCE) have a governor / las provincias tienen gobernador
	}

	on_government_type_change = {
		change_to_dictatorial = yes
	}
	on_post_government_type_change = {
		post_change_to_dictatorial = yes
	}
}
