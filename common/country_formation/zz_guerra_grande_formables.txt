# EN: 24/9/2026 (Juan: «la Banda Oriental seguro, Paraguay tambien, y los otros que te pide si los tenes o ya casi
#     completaste la colonizacion») -- the old list asked for 21 states at 0.70, several of them frontier still being
#     colonized, and a one-province colony counted as a whole state. Now the real test is in possible
#     (common/scripted_triggers/gg_plata_formation.txt): the five Oriental states and Paraguay's two, all of them, plus
#     10 of the 14 historical provinces, where a frontier state held and at least 75% colonized makes good a missing one.
#     states lists those 21 so the formation panel shows them; the fraction is set so the engine's own count (21 x 0.33
#     = 6.93 -> 7 rounded up, 6 rounded down) never asks for more than the seven mandatory states, which possible
#     already demands: the engine gate can never block what possible allows.
# ES: la lista vieja pedia 21 estados al 0.70, varios de frontera todavia en colonizacion. Ahora la prueba real esta en
#     possible: los cinco orientales y los dos del Paraguay, todos, mas 10 de las 14 provincias historicas, donde un
#     estado de frontera propio y colonizado en al menos un 75% reemplaza a una que falte. states lista esos 21 para
#     que el panel los muestre; la fraccion hace que la cuenta del motor nunca pida mas que los siete obligatorios.
REPLACE:PLT = {
	states = {
		# EN/ES: mandatory / obligatorios
		STATE_MONTEVIDEO
		STATE_SAN_JOSE
		STATE_MALDONADO
		STATE_BANDA_ORIENTAL
		STATE_PAYSANDU
		STATE_ALTO_PARAGUAY
		STATE_BAJO_PARAGUAY
		# EN/ES: the fourteen historical provinces, 10 needed / las catorce provincias historicas, hacen falta 10
		STATE_BUENOS_AIRES
		STATE_SANTA_FE
		STATE_ENTRE_RIOS
		STATE_CORRIENTES
		STATE_CORDOBA
		STATE_SANTIAGO_DEL_ESTERO
		STATE_TUCUMAN
		STATE_SALTA
		STATE_JUJUY
		STATE_CATAMARCA
		STATE_LA_RIOJA
		STATE_SAN_JUAN
		STATE_MENDOZA
		STATE_SAN_LUIS
		# EN/ES: frontier, counted by the real check only when >= 75% colonized / frontera, cuenta en el chequeo real solo colonizada al 75%
		STATE_LA_PAMPA
		STATE_NEUQUEN
		STATE_RIO_NEGRO
		STATE_CHUBUT
		STATE_SANTA_CRUZ_ARG
		STATE_TIERRA_DEL_FUEGO
		STATE_MISIONES
		STATE_CHACO
		STATE_FORMOSA_ARG
		STATE_RIO_GRANDE_DO_SUL
	}

	# EN: 31 states x 0.54 -> 17 (7 mandatory + 10 provinces or settled frontier), so the panel shows the real size
	#     (Juan 25/9: 7 of 21 looked far too few). The exact rule stays in `possible`.
	# ES: 31 x 0.54 -> 17, para que el panel muestre el tamaño real; la regla exacta sigue en possible.
	required_states_fraction = 0.54

	ai_will_do = { always = yes }

	possible = {
		gg_arg_chain_pending = no # EN/ES: 24/9/2026 (Juan, AUD-ARG-1850) -- not while Argentina's calendar runs (to 13/10/1862): forming changes the tag and loses it / no mientras corre el calendario argentino: formar cambia el tag y lo pierde
		has_technology_researched = nationalism
		custom_tooltip = {
			text = gg_plt_form_mandatory_tt
			gg_plt_formation_mandatory = yes
		}
		custom_tooltip = {
			text = gg_plt_form_provinces_tt
			gg_plt_formation_provinces = yes
		}
	}
}
