# -------------------------------------------------------------------------------------------------
# EN: La Plata and Great Argentina on la Argentina's map (a hard requirement since 24/9/2026). Vanilla and
#     la Argentina (99_formable_countries.txt) ask for their own oriental states; this mod has five. Both are
#     la Argentina's definitions with the five oriental states: La Plata (its states, 0.70, nationalism) and
#     Great_ARG (with San Jose and Maldonado). Merged from the compatch (zzz_gg_x_arg_formables.txt, audit X3)
#     on 24/9/2026; the vanilla-map version of the standalone mod is gone.
# ES: La Plata y la Gran Argentina sobre el mapa de la Argentina (requisito desde el 24/9/2026): las
#     definiciones de la Argentina con los cinco estados orientales. Venian del compatch; la version para el
#     mapa de vanilla del mod solo se fue.
# -------------------------------------------------------------------------------------------------

	states = {

		STATE_MALDONADO
		STATE_BANDA_ORIENTAL
		STATE_PAYSANDU

		STATE_ALTO_PARAGUAY
		STATE_RIO_NEGRO
		STATE_NEUQUEN
		STATE_CHUBUT
		STATE_SANTA_CRUZ_ARG
		STATE_TIERRA_DEL_FUEGO
		STATE_SOUTH_ATLANTIC_ISLANDS
		STATE_LA_PAMPA
		STATE_TUCUMAN
		STATE_JUJUY
		STATE_CORRIENTES
		STATE_RIO_GRANDE_DO_SUL
	}

	required_states_fraction = 0.70

	ai_will_do = { always = yes }

	possible = {
		gg_arg_chain_pending = no # EN/ES: 24/9/2026 (Juan, AUD-ARG-1850) -- not while Argentina's calendar runs (to 13/10/1862): forming changes the tag and loses it / no mientras corre el calendario argentino: formar cambia el tag y lo pierde
		has_technology_researched = nationalism
	}
}
