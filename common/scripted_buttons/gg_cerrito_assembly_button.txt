gg_cerrito_assembly_button = {
	name = gg_cerrito_assembly_button

		custom_tooltip = {
			text = gg_cerrito_assembly_requires_once

		custom_tooltip = {
			text = gg_cerrito_assembly_requires_window
			game_date < 1851.7.1
		}
		hidden_trigger = {
			gg_cerrito_assembly_open = yes
		}
	}
