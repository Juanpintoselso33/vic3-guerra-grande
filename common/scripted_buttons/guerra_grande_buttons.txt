# Two buttons, now shown in je_gg_patrons (squadrons) and je_gg_siege (Italian legion), built like la Argentina's patagonia_button and vanilla's
# carlist war buttons: a visibility check, a gated possible block and a one-shot effect.

# The Anglo-French blockade of the Rio de la Plata (1845-1849). Only the side holding
# Montevideo can call the squadrons in, and only once.
guerra_grande_foreign_intervention_button = {
	name = guerra_grande_foreign_intervention_button
	desc = guerra_grande_foreign_intervention_button_desc

	visible = {
		# EN: AI keeps the native button; humans use the Overview control.
		# ES: La IA conserva el boton nativo; el jugador usa el de Panorama.
		is_player = no
		c:URU ?= this
		# EN/ES: playtest -- not before the split / no antes de la particion
		has_global_variable = guerra_grande_split_done
	}

	possible = { gg_ui_can_call_squadrons = yes }

	selected = {
		has_global_variable = gg_intervention_called
	}

	effect = { gg_ui_call_squadrons_effect = yes }

	ai_chance = {
		value = 50
	}
}

# Garibaldi's Italian Legion, raised among the Italian community of Montevideo in 1843.
guerra_grande_italian_legion_button = {
	name = guerra_grande_italian_legion_button
	desc = guerra_grande_italian_legion_button_desc

	visible = {
		c:URU ?= this
		# EN/ES: playtest -- not before the split / no antes de la particion
		has_global_variable = guerra_grande_split_done
	}

	possible = {
		c:URU ?= THIS
		custom_tooltip = {
			text = guerra_grande_requires_open_war
			has_global_variable = guerra_grande_open_war
		}
		# EN/ES: T4, FE-10 -- bring the legion forward, or reinforce it once / adelantar la legion, o reforzarla una vez
		hidden_trigger = {
			NOT = { has_variable = guerra_grande_italian_legion_reinforced }
		}
		custom_tooltip = {
			text = guerra_grande_requires_montevideo
			any_scope_state = {
				state_region = s:STATE_MONTEVIDEO
			}
		}
	}

	selected = {
		has_variable = guerra_grande_italian_legion_reinforced
	}

	effect = {
		# EN: visible counter, since debug_log writes nothing in this release build
		#     (verified: not even vanilla's own debug_log lines appear in game.log).
		#     Read from the journal's status text.
		# ES: contador visible, porque debug_log no escribe nada en este build de release
		#     (verificado: ni los debug_log propios de vanilla aparecen en game.log). Se lee
		#     desde el texto de estado del journal.
		# EN: self-healing guard -- this fix landed after some saves already ran
		#     on_game_started once, so the global init there never fires for them.
		# ES: guarda auto-reparable -- este arreglo llego despues de que algunas
		#     partidas ya corrieron on_game_started una vez, asi que la inicializacion
		#     de alla nunca les llega.
		if = {
			limit = { NOT = { has_variable = guerra_grande_italian_legion_raised } }
			set_variable = guerra_grande_italian_legion_raised
			create_character = {
				first_name = "Giuseppe"
				last_name = "Garibaldi"
				historical = yes
				is_general = yes
				interest_group = ig_armed_forces
				ideology = ideology_radical
				culture = cu:north_italian
				religion = rel:catholic
				traits = {
					charismatic
					brave
					experienced_offensive_planner
				}
			}
			add_modifier = {
				name = guerra_grande_italian_legion
				years = 10
				is_decaying = yes
			}
			# EN/ES: FN-3 -- with Costa Brava behind him the legion musters a battalion of its own
			#     / con Costa Brava atras, la legion forma su propio batallon
			if = {
				limit = { has_global_variable = gg_garibaldi_famous }
				gg_select_formation_hq_state = yes
				# EN: Without an owned state there is no valid HQ; skip this formation.
				# ES: Sin un estado propio no hay cuartel valido; se omite esta formacion.
				if = {
					limit = { exists = scope:gg_formation_hq_state }
					create_military_formation = {
						type = army
						hq_region = scope:gg_formation_hq_state.region
						combat_unit = {
							type = unit_type:combat_unit_type_line_infantry
							state_region = s:STATE_MONTEVIDEO
							count = 1
						}
					}
				}
			}
			trigger_event = { id = guerra_grande.5 popup = yes }
		}
		else = {
			# EN/ES: T4, FE-10 -- reinforce: money and French goodwill / reforzar: plata y buena voluntad francesa
			set_variable = guerra_grande_italian_legion_reinforced
			add_treasury = -20000
			gg_patron_change = { PATRON = francia DELTA = -5 }
			add_modifier = {
				name = guerra_grande_italian_legion
				years = 5
				is_decaying = yes
			}
		}
	}

	ai_chance = {
		value = 50
	}
}
