# EN: Capture the enforced side before on_war_end, then resolve the political settlement once.
# ES: Registrar el lado que impuso su objetivo antes de on_war_end y resolver una sola vez.
on_wargoal_enforced = {
	on_actions = { gg_pact_wargoal_enforced gg_cis_wargoal_enforced gg_ba_wargoal_enforced }
}

on_capitulation = {
	on_actions = { gg_pact_capitulation gg_cis_capitulation gg_ba_capitulation }
}

gg_pact_wargoal_enforced = {
	trigger = {
		has_global_variable = gg_war_against_buenos_aires
		exists = scope:diplomatic_play
		scope:diplomatic_play = { is_diplomatic_play_type = dp_gg_pacto_federal }
		OR = {
			AND = { c:ENR ?= this c:ARG ?= scope:target }
			AND = { c:ARG ?= this c:ENR ?= scope:target }
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_wargoal_enforced" # GG_TRAZA
		if = { limit = { c:ENR ?= this } set_global_variable = { name = gg_pact_war_urquiza_won value = 1 } }
		else = { set_global_variable = { name = gg_pact_war_rosas_won value = 1 } }
	}
}

# EN: Country scope. A capitulation says who lost even when the defending side owns no wargoal.
#     Requiring the new truce avoids confusing a simultaneous unrelated capitulation with Caseros.
# ES: Scope pais. La capitulacion identifica al derrotado aunque el defensor no tenga objetivo.
#     Exigir la tregua nueva evita confundir otra capitulacion simultanea con Caseros.
gg_pact_capitulation = {
	trigger = {
		has_global_variable = gg_war_against_buenos_aires
		OR = {
			AND = { c:ARG ?= this has_truce_with = c:ENR }
			AND = { c:ENR ?= this has_truce_with = c:ARG }
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_capitulation" # GG_TRAZA
		if = { limit = { c:ARG ?= this } set_global_variable = { name = gg_pact_war_urquiza_won value = 1 } }
		else = { set_global_variable = { name = gg_pact_war_rosas_won value = 1 } }
	}
}

gg_pact_on_war_end = {
	trigger = {
		has_global_variable = gg_war_against_buenos_aires
		is_diplomatic_play_type = dp_gg_pacto_federal
		c:ARG ?= scope:actor
		c:ENR ?= scope:target
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_on_war_end" # GG_TRAZA
		# EN: 23/9/2026 -- headless run 155901: on_war_end fires BEFORE on_wargoal_enforced and on_capitulation
		#     in the same tick, so the winner was not recorded yet and a won war closed the Pact as a stalemate
		#     ("two governments, no victor"). The reading waits one day (gg_caseros.6).
		# ES: on_war_end corre ANTES que on_wargoal_enforced y on_capitulation en el mismo instante: el ganador
		#     todavia no estaba anotado y una guerra ganada cerraba el Pacto en tablas. La lectura espera un dia.
		c:ARG ?= { trigger_event = { id = gg_caseros.6 days = 1 } }
	}
}

# EN: 23/9/2026 -- the second Cisplatine war (events/gg_cisplatina.txt): who won, recorded before the
#     day-late reading (gg_cisplatina.3), the same way as the Pact wars above.
# ES: la segunda guerra cisplatina: quien gano, anotado antes de la lectura del dia siguiente.

		has_global_variable = gg_cis_war
		exists = scope:diplomatic_play
		scope:diplomatic_play = { is_diplomatic_play_type = dp_gg_cisplatina }
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_cis_wargoal_enforced" # GG_TRAZA
		if = { limit = { c:BRZ ?= this } set_global_variable = gg_cis_brazil_won }
	}
}

	}

# EN: 23/9/2026 (Juan) -- a backdown in the Pact's play with Entre Rios decides it as a war would: whoever backs
#     down loses, and the settlement is read the next day (gg_caseros.6), as after on_war_end.
# ES: un retroceso en la jugada del Pacto con Entre Rios la decide como una guerra: el que retrocede pierde, y
#     el arreglo se lee al dia siguiente, como despues de on_war_end.
on_diplo_play_back_down = {
	on_actions = { gg_pact_back_down gg_ba_back_down }
}

gg_pact_back_down = {
	trigger = {
		has_global_variable = gg_war_against_buenos_aires
		is_diplomatic_play_type = dp_gg_pacto_federal
		exists = scope:actor
		OR = {
			c:ENR ?= scope:actor
			c:ARG ?= scope:actor
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_back_down" # GG_TRAZA
		if = {
			limit = { c:ENR ?= scope:actor }
			set_global_variable = { name = gg_pact_war_rosas_won value = 1 }
		}
		else = { set_global_variable = { name = gg_pact_war_urquiza_won value = 1 } }
		c:ARG ?= { trigger_event = { id = gg_caseros.6 days = 1 } }
	}
}

# EN: 23/9/2026 -- the wars between the Confederation and the State of Buenos Aires (events/gg_buenos_aires.txt):
#     who won, recorded before the day-late reading (gg_ba.9), the same way as the Pact's wars.
# ES: las guerras entre la Confederacion y el Estado de Buenos Aires: quien gano, anotado antes de la lectura del
#     dia siguiente, igual que las del Pacto.
gg_ba_wargoal_enforced = {
	trigger = {
		has_global_variable = gg_ba_split_active
		exists = scope:diplomatic_play
		scope:diplomatic_play = { is_diplomatic_play_type = dp_gg_buenos_aires }
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_ba_wargoal_enforced" # GG_TRAZA
		if = { limit = { global_var:gg_ba_side ?= this } set_global_variable = { name = gg_ba_ba_won value = 1 } }
		else_if = { limit = { global_var:gg_conf_side ?= this } set_global_variable = { name = gg_ba_conf_won value = 1 } }
	}
}

gg_ba_capitulation = {
	trigger = {
		has_global_variable = gg_ba_split_active
		OR = {
			AND = { global_var:gg_ba_side ?= this global_var:gg_conf_side ?= { has_truce_with = root } }
			AND = { global_var:gg_conf_side ?= this global_var:gg_ba_side ?= { has_truce_with = root } }
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_ba_capitulation" # GG_TRAZA
		if = { limit = { global_var:gg_ba_side ?= this } set_global_variable = { name = gg_ba_conf_won value = 1 } }
		else = { set_global_variable = { name = gg_ba_ba_won value = 1 } }
	}
}

gg_ba_on_war_end = {

		has_global_variable = gg_ba_split_active
