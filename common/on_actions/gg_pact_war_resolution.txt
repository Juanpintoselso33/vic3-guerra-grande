# EN: Capture the enforced side before on_war_end, then resolve the political settlement once.
# ES: Registrar el lado que impuso su objetivo antes de on_war_end y resolver una sola vez.
on_wargoal_enforced = {
	on_actions = { gg_pact_wargoal_enforced }
}

on_capitulation = {
	on_actions = { gg_pact_capitulation }
}

gg_pact_wargoal_enforced = {
	trigger = {
		has_global_variable = gg_war_against_buenos_aires
		exists = scope:diplomatic_play
		scope:diplomatic_play = { is_diplomatic_play_type = dp_gg_pacto_federal }
		OR = {
			AND = { c:ENR ?= this c:ARG ?= scope:target }
			AND = { c:ARG ?= this c:ENR ?= scope:target }
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_wargoal_enforced" # GG_TRAZA
		if = { limit = { c:ENR ?= this } set_global_variable = { name = gg_pact_war_urquiza_won value = 1 } }
		else = { set_global_variable = { name = gg_pact_war_rosas_won value = 1 } }
	}
}

# EN: Country scope. A capitulation says who lost even when the defending side owns no wargoal.
#     Requiring the new truce avoids confusing a simultaneous unrelated capitulation with Caseros.
# ES: Scope pais. La capitulacion identifica al derrotado aunque el defensor no tenga objetivo.
#     Exigir la tregua nueva evita confundir otra capitulacion simultanea con Caseros.
gg_pact_capitulation = {
	trigger = {
		has_global_variable = gg_war_against_buenos_aires
		OR = {
			AND = { c:ARG ?= this has_truce_with = c:ENR }
			AND = { c:ENR ?= this has_truce_with = c:ARG }
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_capitulation" # GG_TRAZA
		if = { limit = { c:ARG ?= this } set_global_variable = { name = gg_pact_war_urquiza_won value = 1 } }
		else = { set_global_variable = { name = gg_pact_war_rosas_won value = 1 } }
	}
}

gg_pact_on_war_end = {
	trigger = {
		has_global_variable = gg_war_against_buenos_aires
		is_diplomatic_play_type = dp_gg_pacto_federal
		c:ARG ?= scope:actor
		c:ENR ?= scope:target
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_on_war_end" # GG_TRAZA
		# EN: 23/9/2026 -- headless run 155901: on_war_end fires BEFORE on_wargoal_enforced and on_capitulation
		#     in the same tick, so the winner was not recorded yet and a won war closed the Pact as a stalemate
		#     ("two governments, no victor"). The reading waits one day (gg_caseros.6).
		# ES: on_war_end corre ANTES que on_wargoal_enforced y on_capitulation en el mismo instante: el ganador
		#     todavia no estaba anotado y una guerra ganada cerraba el Pacto en tablas. La lectura espera un dia.
		c:ARG ?= { trigger_event = { id = gg_caseros.6 days = 1 } }
	}
}
