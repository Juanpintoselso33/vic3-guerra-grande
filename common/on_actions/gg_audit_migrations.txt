# EN: One-shot, reload-safe repairs for saves produced before the 20 September audit fixes.
#     23/9/2026 (AUD-MIGR-NEWGAME, N1 of the design-vs-code audit) -- ONLY for old saves. Until today none of the three
#     had a version guard, so they ran in every new game too: the postwar one killed every Rivera of the Republic a
#     month after any punishing peace (a Rivera restored in 1846 who never left the Defensa, against the historical
#     lives and the postwar that brings him back), and was the only thing clearing the siege modifiers after a
#     reincorporation, a month late. A new game sets gg_new_game_no_migrations on its first day
#     (guerra_grande_on_game_start); a save from before this commit does not have it, so it is still repaired.
#     The jobs they were doing by accident now have their own places: the siege modifiers are cleared by
#     gg_wartime_modifiers_clear_once (gg_postwar_start and gg_oriental_incorporate), and the Coalition born of the
#     unitario rising of 1877 withdraws from Argentina's treaties when it is created (gg_unitario_rising).
# ES: Reparaciones idempotentes para guardados anteriores a las correcciones de la auditoria. SOLO para partidas
#     viejas: sin guarda corrian en toda partida nueva, mataban a todo Rivera de la Republica al mes de una paz con
#     castigo y eran lo unico que sacaba los modificadores del sitio despues de una reincorporacion. Una partida nueva
#     prende gg_new_game_no_migrations el primer dia; lo que hacian de casualidad tiene ahora su lugar propio.
on_monthly_pulse_country = {
	on_actions = {
		gg_audit_postwar_migration
		gg_pact_negotiated_migration
		gg_rebel_diplomacy_migration
		gg_urquiza_prelude_migration
		gg_oriental_left_out_migration
		gg_brazil_deal_migration
		gg_province_incorporation_migration
		gg_cis_note_migration
		gg_rosas_first_term_migration
		gg_paraguay_recognition_pulse
	}
}

gg_audit_postwar_migration = {
	trigger = {
		c:ARG ?= this
		has_global_variable = gg_peace_settled
		NOT = { has_global_variable = gg_audit_postwar_cleanup_v2 }
		NOT = { has_global_variable = gg_new_game_no_migrations } # EN/ES: old saves only / solo partidas viejas
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_audit_postwar_migration" # GG_TRAZA
		set_global_variable = gg_audit_postwar_cleanup_v2
		gg_wartime_modifiers_clear = yes
		# EN/ES: Old partitions generated extra Oribe copies; keep one canonical living character.
		# Las particiones viejas generaron copias de los Oribe; conservar un solo personaje vivo.
		gg_deduplicate_character_worldwide = { TEMPLATE = URU_Manuel_Oribe }
		gg_deduplicate_character_worldwide = { TEMPLATE = URU_Ignacio_Oribe }
		# EN/ES: Reconcile the three contradictory debt outcomes already saved by S05.
		every_country = {
			if = {
				limit = { has_global_variable = gg_peace_punished }
				if = { limit = { has_modifier = guerra_grande_owes_nobody } remove_modifier = guerra_grande_owes_nobody }
				if = { limit = { has_modifier = guerra_grande_debts_of_the_cerrito } remove_modifier = guerra_grande_debts_of_the_cerrito }
				if = {
					limit = { NOT = { has_modifier = gg_postwar_unrecognised_debts } }
					add_modifier = { name = gg_postwar_unrecognised_debts years = 15 is_decaying = yes }
				}
			}
			else_if = {
				limit = { has_global_variable = gg_peace_no_victors }
				if = { limit = { has_modifier = gg_postwar_unrecognised_debts } remove_modifier = gg_postwar_unrecognised_debts }
			}
		}
		# EN/ES: The punitive save contains Rivera's generated copy in Uruguay and the original abroad.
		# The obsolete postwar event must not leave two men, nor replace Flores again.
		if = {
			limit = { has_global_variable = gg_peace_punished exists = c:URU }
			c:URU = {
				every_scope_character = {
					limit = { has_template = URU_Fructuoso_Rivera }
					kill_character = yes
				}
			}
		}
	}
}

# EN: 24/9/2026 (Codex review of the Entre Rios prelude) -- a save started before the relay gg_argentina_cal.35 existed
#     never queued it (it is scheduled on the first day). Such a save has no gg_urquiza_prelude_seeded; in the window of

		game_date >= 1851.2.22
		game_date < 1851.4.17
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza_prelude_migration" # GG_TRAZA

# EN: 24/9/2026 (hotfix 0.1.2, AUD-ORIENTAL-PROVINCE; Juan's game, 22/12/1851) -- up to 0.1.1, gg_argentina.80.a
#     incorporated the thirteen provinces and left the Oriental Province out, and with no constitution (gg_constitution.1.b)
#     nothing took it later: Buenos Aires's provinces were all Argentine ground except the Banda Oriental. Such a save
#     is asked the question again, once, now that the answer includes it: .80.a absorbs it (gg_oriental_province_join),
#     .80.b keeps it as it is. Only after the constitution was answered (before that, the constitution takes it), with
#     no other province still held as a subject (then .80.b was the answer, and it stays consistent), and never where
#     gg_argentina.80 already knew of it (gg_pact_80_knows_oriental). A player's Oriental Province is not asked about.
#     Not guarded by gg_new_game_no_migrations: Juan's game was begun on 24/9 and has that mark. A game begun with
#     0.1.2 gets gg_pact_80_knows_oriental on its first day (guerra_grande_on_actions.txt; Codex review): a Pact settled
#     with no province to ask about sends the constitution straight away, and a later Oriental Province is not a repair.
# ES: hasta 0.1.1, gg_argentina.80.a incorporaba las trece provincias y dejaba afuera a la Provincia Oriental, y sin
#     constitucion nada la tomaba despues. A esa partida se le repite la pregunta, una vez, ahora que la respuesta la
#     incluye. Solo con la constitucion ya contestada, sin otra provincia todavia sujeta, y nunca si .80 ya la conocia.
gg_oriental_left_out_migration = {
	trigger = {
		c:ARG ?= this
		has_variable = gg_arg_rosas_seat
		has_global_variable = gg_pact_settled
		has_global_variable = gg_constitution_chosen
		NOT = { has_global_variable = gg_pact_80_knows_oriental }
		NOT = { has_global_variable = gg_pact_settlement_stalemate }
		c:URU ?= { gg_is_oriental_pact_province = yes }
		# EN/ES: Codex review -- a player's Oriental Province under an AI Argentina is asked about too (the answer offers him
		#     Argentina's seat, gg_constitution.5); with both in human hands nothing would change
		#     / con la Argentina en manos de la IA si; con las dos en manos de jugadores no cambiaria nada
		NAND = {
			is_player = yes
			c:URU = { is_player = yes }
		}
		NOT = {
			any_subject_or_below = {
				OR = {
					c:CLN ?= this
					gg_is_province_country = yes
				}
				is_subject_of = c:ARG
			}
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_oriental_left_out_migration" # GG_TRAZA
		set_global_variable = gg_pact_80_knows_oriental
		trigger_event = { id = gg_argentina.80 days = 0 popup = yes }
	}
}

# EN: 24/9/2026 (Juan, the Empire keeps the September contracts: common/scripted_triggers/gg_brazil_deal.txt) -- a save
#     from before this change: (1) a loan taken while gg_loan.2 wrote no mark (mod 247b25d to this commit) is read from
#     the modifier only gg_loan.2.a/.b add (guerra_grande_imperial_client / guerra_grande_maua_paper); (2) a convention
#     signed before gg_brazil_marches existed marks it, or its crossing (gg_notice.101) would wait for a mark that never
#     comes. Idempotent; not guarded by gg_new_game_no_migrations (saves of 23-24/9 have that mark and lack these).
# ES: una partida de antes de este cambio: (1) un emprestito tomado cuando gg_loan.2 no escribia la marca se lee del
#     modificador que solo ponen gg_loan.2.a/.b; (2) una convencion firmada antes de que existiera gg_brazil_marches la
#     marca, o el cruce esperaria una marca que no llega. Idempotente; sin la guarda de partida nueva.
gg_brazil_deal_migration = {
	trigger = {
		c:URU ?= this
		OR = {
			AND = {
				NOT = { has_global_variable = gg_brazil_deal }
				OR = {
					has_modifier = guerra_grande_imperial_client
					has_modifier = guerra_grande_maua_paper
				}
			}
			AND = {
				has_global_variable = gg_montevideo_free_navigation
				NOT = { has_global_variable = gg_brazil_marches }
			}
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_brazil_deal_migration" # GG_TRAZA
		if = {
			limit = {
				NOT = { has_global_variable = gg_brazil_deal }
				OR = {
					has_modifier = guerra_grande_imperial_client
					has_modifier = guerra_grande_maua_paper
				}
			}
			set_global_variable = gg_brazil_deal
		}
		if = {
			limit = {
				has_global_variable = gg_montevideo_free_navigation
				NOT = { has_global_variable = gg_brazil_marches }
			}
			gg_brazil_marches_set = { ROUTE = migration }
		}
	}
}

# EN: 24/9/2026 (Juan, AUD-PROV-INCORP; his game, 0.1.2, played as Argentina) -- up to 0.1.2 every annexation of a
#     province (gg_argentina.80.a on 16/5/1851, the constitution, the Coalition, the Buenos Aires chain) left its states
#     unincorporated, and so did the old bare annex of the Oriental Province. Such a save is repaired once, on the next
#     monthly pulse: Argentina's unincorporated states in a historical province (and the Banda Oriental once it rejoined,
#     gg_arg_historical_province_state) become incorporated (gg_incorporate_province_states). Only where the provinces
#     are one country already (the Pact settled and no province still Argentina's subject); a Pact that kept its
#     provinces is not touched. A game begun with this fix has gg_province_incorporation_fixed from its first day
#     (guerra_grande_on_game_start), so it never runs there. Not guarded by gg_new_game_no_migrations: Juan's game has it.
# ES: hasta 0.1.2 toda anexion de una provincia dejaba sus estados sin incorporar. Esa partida se repara una vez, en el
#     proximo pulso mensual: los estados no incorporados de la Argentina en una provincia historica pasan a incorporados.
#     Solo si las provincias ya son un solo pais. Una partida nueva tiene la marca desde el primer dia.
gg_province_incorporation_migration = {
	trigger = {
		c:ARG ?= this
		NOT = { has_global_variable = gg_province_incorporation_fixed }
		has_global_variable = gg_pact_settled
		NOT = {
			any_subject_or_below = {
				OR = {
					c:CLN ?= this
					gg_is_province_country = yes
					gg_is_oriental_pact_province = yes # EN/ES: Codex review -- a Pact that kept only the Oriental Province / un Pacto que solo conservo la Oriental
				}
				is_subject_of = c:ARG
			}
		}
		any_scope_state = {
			is_incorporated = no
			gg_arg_historical_province_state = yes
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_province_incorporation_migration" # GG_TRAZA
		set_global_variable = gg_province_incorporation_fixed
		gg_incorporate_province_states = yes
	}
}

# EN: 24/9/2026 (AUD-END-A1; Juan's game: gg_constitution.1.a on 16/6/1851, Rosas out of the seat on 17/6/1851) -- a save
#     that gave the Constitution of 1853 before the first term existed. The Restorer gets his term back: the Rosas seat
#     puts him in again (gg_rosas_keeps_the_seat, on this pulse or at the next ruler change) until it ends. THE DATE OF
#     THE ANSWER IS NOT IN THE SAVE, so the term runs to 31/5/1857: six years from the earliest day the old flow could
#     ask (the settlement of 1/5/1851 plus its thirty days). For Juan (16/6/1851) that is 16 days short, never long --
#     art. 75: the president «cesa en el poder el dia mismo en que expira su periodo». Only after the negotiated or
#     Rosas's own settlement (the roads where he held the seat), with Rosas alive in Argentina. Once per save
#     (gg_rosas_first_term_known, which the new answer writes too). Not guarded by gg_new_game_no_migrations: Juan's game
#     carries that mark.
# ES: una partida que dio la Constitucion de 1853 antes de que existiera el primer periodo. El Restaurador recupera su
#     periodo hasta el 31/5/1857: la fecha de la respuesta no esta en la partida, asi que se cuenta desde el primer dia en
#     que el flujo viejo pudo preguntar. Para Juan son 16 dias menos, nunca mas (art. 75). Una vez por partida.
# SOURCE / FUENTE: Constitucion de la Nacion Argentina (1 May 1853), arts. 74-75 (es.wikisource).
gg_rosas_first_term_migration = {
	trigger = {
		c:ARG ?= this
		has_variable = gg_arg_rosas_seat
		global_var:gg_constitution_chosen ?= 1
		NOT = { has_global_variable = gg_rosas_first_term_known }
		OR = {
			has_global_variable = gg_pact_settlement_constitutional
			has_global_variable = gg_pact_settlement_rosas
		}
		game_date < 1857.5.31
		any_scope_character = { has_template = ARG_Juan_Manuel_de_Rosas is_character_alive = yes }
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_rosas_first_term_migration" # GG_TRAZA
		set_global_variable = gg_rosas_first_term_known
		set_global_variable = gg_rosas_first_term_legacy
		gg_rosas_keeps_the_seat = yes
	}
}

# EN: 24/9/2026 (Juan, the independence of Paraguay; Codex review) -- the retry of gg_argentina_cal.36, and its
#     delivery to saves begun before it existed. From 17/7/1852 (day 6037, when the dated call comes) to 7/6/1856 (the
#     Argentine Congress's law), every month, on whatever country is ARG that month, until the relay delivers
#     (gg_arg_cal_done_36). A pulse, not a self-queued retry: a retry queued on an Argentina later annexed in the Buenos
#     Aires chain (gg_ba.20/.21) would be lost. Not an old-save repair only, so not guarded by gg_new_game_no_migrations.
#     The last check is ARG's last monthly pulse before 7/6/1856 (Codex, round 2): conditions that only come true after
#     it deliver nothing, which is the cutoff meant (the recognition was the Congress's by then).
# ES: el reintento de gg_argentina_cal.36 y su entrega a las partidas empezadas antes de que existiera. Desde el
#     17/7/1852 hasta el 7/6/1856, cada mes, sobre la ARG de ese mes, hasta que el relevo entregue. Un pulso y no un
#     reintento encolado: el encolado sobre una Argentina anexada despues en la cadena de Buenos Aires se perderia.

	trigger = {
		c:ARG ?= this
		NOT = { has_global_variable = gg_arg_cal_done_36 }
		game_date >= 1852.7.18 # EN/ES: the day after the dated call / el dia despues de la llamada fechada
		game_date < 1856.6.7
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_paraguay_recognition_pulse" # GG_TRAZA
		trigger_event = { id = gg_argentina_cal.36 days = 0 }
	}
}

# EN: 25/9/2026 (audit of Juan's Argentina game, 1851-1869) -- three repairs on the GLOBAL monthly pulse, owned by no
#     country (Argentina may be La Plata, Uruguay may be gone). None is guarded by gg_new_game_no_migrations: games
#     begun before today carry the bugs too.
#     - AUD-RULER-ROLE: once, the ruler role is taken from whoever holds it in a South American country he does not
#       rule (Oribe and Suarez in Argentina, 1853-1869; gg_bring_characters_to no longer carries it).

#       (gg_wartime_modifiers_clear now removes it from states).

	on_actions = {
		gg_stray_ruler_role_migration

	}
}
