# EN: One-shot, reload-safe repairs for saves produced before the 20 September audit fixes.
#     23/9/2026 (AUD-MIGR-NEWGAME, N1 of the design-vs-code audit) -- ONLY for old saves. Until today none of the three
#     had a version guard, so they ran in every new game too: the postwar one killed every Rivera of the Republic a
#     month after any punishing peace (a Rivera restored in 1846 who never left the Defensa, against the historical
#     lives and the postwar that brings him back), and was the only thing clearing the siege modifiers after a
#     reincorporation, a month late. A new game sets gg_new_game_no_migrations on its first day
#     (guerra_grande_on_game_start); a save from before this commit does not have it, so it is still repaired.
#     The jobs they were doing by accident now have their own places: the siege modifiers are cleared by
#     gg_wartime_modifiers_clear_once (gg_postwar_start and gg_oriental_incorporate), and the Coalition born of the
#     unitario rising of 1877 withdraws from Argentina's treaties when it is created (gg_unitario_rising).
# ES: Reparaciones idempotentes para guardados anteriores a las correcciones de la auditoria. SOLO para partidas
#     viejas: sin guarda corrian en toda partida nueva, mataban a todo Rivera de la Republica al mes de una paz con
#     castigo y eran lo unico que sacaba los modificadores del sitio despues de una reincorporacion. Una partida nueva
#     prende gg_new_game_no_migrations el primer dia; lo que hacian de casualidad tiene ahora su lugar propio.
on_monthly_pulse_country = {
	on_actions = {
		gg_audit_postwar_migration
		gg_pact_negotiated_migration
		gg_rebel_diplomacy_migration
		gg_urquiza_prelude_migration
		gg_oriental_left_out_migration
	}
}

gg_audit_postwar_migration = {
	trigger = {
		c:ARG ?= this
		has_global_variable = gg_peace_settled
		NOT = { has_global_variable = gg_audit_postwar_cleanup_v2 }
		NOT = { has_global_variable = gg_new_game_no_migrations } # EN/ES: old saves only / solo partidas viejas
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_audit_postwar_migration" # GG_TRAZA
		set_global_variable = gg_audit_postwar_cleanup_v2
		gg_wartime_modifiers_clear = yes
		# EN/ES: Old partitions generated extra Oribe copies; keep one canonical living character.
		# Las particiones viejas generaron copias de los Oribe; conservar un solo personaje vivo.
		gg_deduplicate_character_worldwide = { TEMPLATE = URU_Manuel_Oribe }
		gg_deduplicate_character_worldwide = { TEMPLATE = URU_Ignacio_Oribe }
		# EN/ES: Reconcile the three contradictory debt outcomes already saved by S05.
		every_country = {
			if = {
				limit = { has_global_variable = gg_peace_punished }
				if = { limit = { has_modifier = guerra_grande_owes_nobody } remove_modifier = guerra_grande_owes_nobody }
				if = { limit = { has_modifier = guerra_grande_debts_of_the_cerrito } remove_modifier = guerra_grande_debts_of_the_cerrito }
				if = {
					limit = { NOT = { has_modifier = gg_postwar_unrecognised_debts } }
					add_modifier = { name = gg_postwar_unrecognised_debts years = 15 is_decaying = yes }
				}
			}
			else_if = {
				limit = { has_global_variable = gg_peace_no_victors }
				if = { limit = { has_modifier = gg_postwar_unrecognised_debts } remove_modifier = gg_postwar_unrecognised_debts }
			}
		}
		# EN/ES: The punitive save contains Rivera's generated copy in Uruguay and the original abroad.
		# The obsolete postwar event must not leave two men, nor replace Flores again.
		if = {
			limit = { has_global_variable = gg_peace_punished exists = c:URU }
			c:URU = {
				every_scope_character = {
					limit = { has_template = URU_Fructuoso_Rivera }
					kill_character = yes
				}
			}
		}
	}
}

gg_pact_negotiated_migration = {
	trigger = {
		c:ARG ?= this
		has_global_variable = gg_confederacion_negociada
		NOT = { has_global_variable = gg_new_game_no_migrations } # EN/ES: old saves only / solo partidas viejas
		NOT = { has_global_variable = gg_pact_settled }
	}
	effect = { debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_negotiated_migration"  gg_pact_resolve = yes }
}

gg_rebel_diplomacy_migration = {
	trigger = {
		c:CLN ?= this
		NOT = { has_variable = gg_rebel_diplomacy_reset }
		NOT = { has_global_variable = gg_new_game_no_migrations } # EN/ES: old saves only / solo partidas viejas
	}
	effect = { debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_rebel_diplomacy_migration"  gg_argentina_rebel_diplomacy_reset = yes }
}

# EN: 24/9/2026 (Codex review of the Entre Rios prelude) -- a save started before the relay gg_argentina_cal.35 existed
#     never queued it (it is scheduled on the first day). Such a save has no gg_urquiza_prelude_seeded; in the window of
#     the letter (22/2/1851 to 16/4/1851: at least one monthly pulse falls in it) this queues the relay, which reads its
#     own conditions and retries. Not guarded by gg_new_game_no_migrations: saves begun on 23-24/9 have that mark and
#     still lack the relay.
# ES: una partida empezada antes de que existiera el relevo nunca lo encolo. Sin gg_urquiza_prelude_seeded, en la
#     ventana de la carta (22/2 a 16/4/1851: cae al menos un pulso mensual) se encola el relevo, que lee sus condiciones
#     y reintenta. Sin la guarda de partida nueva: las partidas del 23-24/9 la tienen y tampoco tienen el relevo.
gg_urquiza_prelude_migration = {
	trigger = {
		c:ARG ?= this
		NOT = { has_global_variable = gg_urquiza_prelude_seeded }
		NOT = { has_global_variable = gg_arg_cal_done_35 }
		game_date >= 1851.2.22
		game_date < 1851.4.17
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza_prelude_migration" # GG_TRAZA
		set_global_variable = gg_urquiza_prelude_seeded
		trigger_event = { id = gg_argentina_cal.35 days = 0 }
	}
}

# EN: 24/9/2026 (hotfix 0.1.2, AUD-ORIENTAL-PROVINCE; Juan's game, 22/12/1851) -- up to 0.1.1, gg_argentina.80.a
#     incorporated the thirteen provinces and left the Oriental Province out, and with no constitution (gg_constitution.1.b)
#     nothing took it later: Buenos Aires's provinces were all Argentine ground except the Banda Oriental. Such a save
#     is asked the question again, once, now that the answer includes it: .80.a absorbs it (gg_oriental_province_join),
#     .80.b keeps it as it is. Only after the constitution was answered (before that, the constitution takes it), with
#     no other province still held as a subject (then .80.b was the answer, and it stays consistent), and never where
#     gg_argentina.80 already knew of it (gg_pact_80_knows_oriental). A player's Oriental Province is not asked about.
#     Not guarded by gg_new_game_no_migrations: Juan's game was begun on 24/9 and has that mark. A game begun with
#     0.1.2 gets gg_pact_80_knows_oriental on its first day (guerra_grande_on_actions.txt; Codex review): a Pact settled
#     with no province to ask about sends the constitution straight away, and a later Oriental Province is not a repair.
# ES: hasta 0.1.1, gg_argentina.80.a incorporaba las trece provincias y dejaba afuera a la Provincia Oriental, y sin
#     constitucion nada la tomaba despues. A esa partida se le repite la pregunta, una vez, ahora que la respuesta la
#     incluye. Solo con la constitucion ya contestada, sin otra provincia todavia sujeta, y nunca si .80 ya la conocia.
gg_oriental_left_out_migration = {
	trigger = {
		c:ARG ?= this
		has_variable = gg_arg_rosas_seat
		has_global_variable = gg_pact_settled
		has_global_variable = gg_constitution_chosen
		NOT = { has_global_variable = gg_pact_80_knows_oriental }
		NOT = { has_global_variable = gg_pact_settlement_stalemate }
		c:URU ?= { gg_is_oriental_pact_province = yes }
		# EN/ES: Codex review -- a player's Oriental Province under an AI Argentina is asked about too (the answer offers him
		#     Argentina's seat, gg_constitution.5); with both in human hands nothing would change
		#     / con la Argentina en manos de la IA si; con las dos en manos de jugadores no cambiaria nada
		NAND = {
			is_player = yes
			c:URU = { is_player = yes }
		}
		NOT = {
			any_subject_or_below = {
				OR = {
					c:CLN ?= this
					gg_is_province_country = yes
				}
				is_subject_of = c:ARG
			}
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_oriental_left_out_migration" # GG_TRAZA
		set_global_variable = gg_pact_80_knows_oriental
		trigger_event = { id = gg_argentina.80 days = 0 popup = yes }
	}
}
