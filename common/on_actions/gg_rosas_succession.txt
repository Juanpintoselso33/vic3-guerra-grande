# EN: 23/9/2026 (Juan) -- while Rosas rules without a constitution, the month he dies ends it: the modifier goes,
#     and the question between federales and unitarios opens (gg_constitution.2).
# ES: mientras Rosas gobierne sin constitucion, el mes en que muere lo termina.
on_monthly_pulse_country = {
	on_actions = { gg_rosas_succession_monthly }
}

gg_rosas_succession_monthly = {
	trigger = {
		c:ARG ?= this
		has_modifier = gg_pais_sin_constitucion
		NOT = { any_country = { any_scope_character = { has_template = ARG_Juan_Manuel_de_Rosas is_character_alive = yes } } }
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_rosas_succession_monthly" # GG_TRAZA
		remove_modifier = gg_pais_sin_constitucion
		set_global_variable = gg_rosas_succession_crisis
		trigger_event = { id = gg_constitution.2 days = 0 popup = yes }
	}
}

# EN/ES: the war between federales and unitarios ends (by arms or by a backdown): read the day after
#     / termina la guerra entre federales y unitarios: se lee al dia siguiente
on_war_end = {
	on_actions = { gg_unitario_on_war_end }
}
on_diplo_play_back_down = {
	on_actions = { gg_unitario_on_war_end }
}
gg_unitario_on_war_end = {
	trigger = {
		has_global_variable = gg_unitario_war
		is_diplomatic_play_type = dp_gg_pacto_federal
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_on_war_end" # GG_TRAZA
		c:ARG ?= { trigger_event = { id = gg_constitution.3 days = 1 } }
	}
}
