# EN: 23/9/2026 (Juan) -- while Rosas rules without a constitution, the month he dies ends it: the modifier goes,
#     and the question between federales and unitarios opens (gg_constitution.2).
# ES: mientras Rosas gobierne sin constitucion, el mes en que muere lo termina.
on_monthly_pulse_country = {
	on_actions = { gg_rosas_succession_monthly gg_constitution_monthly gg_constitution_choice_migration gg_constitution_legacy_bootstrap gg_pact_80_monthly }
}

# EN: 24/9/2026 (AUD-CONST-CHOICE) -- the monthly retry of the constitution: while it is pending (gg_constitution_schedule)
#     and not yet shown, every month the Rosas seat asks again; gg_constitution_try holds it off while Buenos Aires is at
#     war with Brazil. A pulse cannot be lost the way a queued event can.
# ES: el reintento mensual de la constitucion: mientras este pendiente y no haya salido, la sede de Rosas vuelve a
#     preguntar cada mes; gg_constitution_try la frena mientras Buenos Aires este en guerra con el Brasil.
# EN: 24/9/2026 (AUD-END-A3) -- gg_argentina.80 held off by the Cisplatine question (gg_argentina_vassals_settle): asked
#     on the first pulse after it is settled. A pulse cannot be lost the way a queued event can.
# ES: gg_argentina.80 frenado por la cuestion cisplatina: se pregunta en el primer pulso despues de saldada.
gg_pact_80_monthly = {
	trigger = {
		has_variable = gg_arg_rosas_seat
		has_global_variable = gg_pact_80_pending
		NOT = { gg_constitution_brazil_war = yes }
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_80_monthly" # GG_TRAZA
		remove_global_variable = gg_pact_80_pending
		trigger_event = { id = gg_argentina.80 days = 0 popup = yes }
	}
}

gg_constitution_monthly = {
	trigger = {
		has_variable = gg_arg_rosas_seat
		has_global_variable = gg_constitution_pending
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_monthly" # GG_TRAZA
		gg_constitution_try = yes
	}
}

# EN: 24/9/2026 (AUD-CONST-CHOICE) -- saves that already answered gg_constitution.1 with the one option 0.1.1 left
#     Rosas (b, the sum of public power, after his victory or with two governments). The answer was forced, so a
#     PLAYER Buenos Aires is asked again, once, with the options of today: same timing rules (after the war with
#     Brazil). gg_constitution_offer_v2 marks the saves whose question already had them: a real choice is never
#     reopened. Only while Rosas is alive and nothing has moved since (the modifier is still there, no succession
#     crisis). Not guarded by gg_new_game_no_migrations: the games Juan is playing (started 23-24/9) carry that mark.
# ES: partidas que ya contestaron con la unica opcion que 0.1.1 le dejaba a Rosas. La respuesta fue forzada: a una
#     Buenos Aires JUGADOR se le vuelve a preguntar, una vez, con las opciones de hoy, con las mismas reglas (despues
#     de la guerra con el Brasil). Una eleccion de verdad nunca se reabre.
gg_constitution_choice_migration = {
	trigger = {
		has_variable = gg_arg_rosas_seat
		is_player = yes
		global_var:gg_constitution_chosen ?= 2
		OR = {
			has_global_variable = gg_pact_settlement_rosas
			has_global_variable = gg_pact_settlement_stalemate
		}
		NOT = { has_global_variable = gg_constitution_offer_v2 }
		NOT = { has_global_variable = gg_constitution_reoffered }
		NOT = { has_global_variable = gg_rosas_succession_crisis }
		has_modifier = gg_pais_sin_constitucion
		gg_constitution_rosas_alive = yes # EN/ES: Codex -- the question is his to answer / la pregunta es suya
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_choice_migration" # GG_TRAZA
		set_global_variable = gg_constitution_reoffered
		remove_global_variable = gg_constitution_chosen
		set_global_variable = gg_constitution_pending
		gg_constitution_try = yes
	}
}

gg_rosas_succession_monthly = {
	trigger = {
		c:ARG ?= this
		has_modifier = gg_pais_sin_constitucion
		NOT = { has_global_variable = gg_ba_split_active } # EN/ES: 23/9 -- waits while Buenos Aires is apart / espera mientras Buenos Aires esta aparte
		NOT = { any_country = { any_scope_character = { has_template = ARG_Juan_Manuel_de_Rosas is_character_alive = yes } } }
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_rosas_succession_monthly" # GG_TRAZA
		remove_modifier = gg_pais_sin_constitucion
		set_global_variable = gg_rosas_succession_crisis
		trigger_event = { id = gg_constitution.2 days = 0 popup = yes }
	}
}

# EN: 24/9/2026 (AUD-CONST-CHOICE, Codex) -- a save from 0.1.1 settled the Pact and queued gg_constitution.1 thirty days
#     ahead, with no pending mark. If that copy comes during a war with Brazil its trigger drops it; so while the Pact
#     is settled and the constitution neither answered nor pending, this puts the question on the pending path (a month
#     of wait, then the monthly retry). In a new game it can only run between the settlement and gg_argentina.80's
#     answer, and does what that answer does anyway. Once per save.
# ES: una partida de 0.1.1 cerro el Pacto y encolo gg_constitution.1 a treinta dias, sin marca pendiente. Si esa copia
#     llega en guerra con el Brasil, su trigger la tira; esto pone la pregunta en el camino pendiente. Una vez.
gg_constitution_legacy_bootstrap = {
	trigger = {
		has_variable = gg_arg_rosas_seat
		OR = {
			has_global_variable = gg_pact_settlement_rosas
			has_global_variable = gg_pact_settlement_stalemate
			has_global_variable = gg_pact_settlement_urquiza
			has_global_variable = gg_pact_settlement_constitutional
			has_global_variable = gg_pact_settlement_mandate_collapsed
		}
		NOT = { has_global_variable = gg_constitution_chosen }
		NOT = { has_global_variable = gg_constitution_pending }
		NOT = { has_global_variable = gg_constitution_offer_v2 }
		NOT = { has_global_variable = gg_constitution_legacy_seeded }
		NOT = { has_global_variable = gg_pact_settled_v3 } # EN/ES: AUD-END-B3 -- only a save settled before the new flow / solo una partida cerrada antes del flujo nuevo
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_legacy_bootstrap" # GG_TRAZA
		set_global_variable = gg_constitution_legacy_seeded
		set_global_variable = gg_constitution_pending
		set_global_variable = { name = gg_constitution_wait days = 29 }
	}
}

# EN/ES: the war between federales and unitarios ends (by arms or by a backdown): read the day after
#     / termina la guerra entre federales y unitarios: se lee al dia siguiente
on_war_end = {
	on_actions = { gg_unitario_on_war_end }
}
on_diplo_play_back_down = {
	on_actions = { gg_unitario_on_war_end }
}
gg_unitario_on_war_end = {
	trigger = {
		has_global_variable = gg_unitario_war
		is_diplomatic_play_type = dp_gg_pacto_federal
		# EN/ES: 23/9 (Codex review of 743b851) -- only Buenos Aires against the league, not another Pact war
		#     / solo Buenos Aires contra la liga, no otra guerra del Pacto
		exists = c:CLN
		initiator_is = c:ARG
		target_is = c:CLN
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_on_war_end" # GG_TRAZA
		set_global_variable = gg_unitario_war_ended # EN/ES: AUD-CONST-PEACE -- with no winner recorded, a white peace / sin ganador anotado, paz blanca
		c:ARG ?= { trigger_event = { id = gg_constitution.3 days = 1 } }
	}
}

# EN: 23/9/2026 (Codex audit, AUD-CONST-PEACE) -- who won the war between federales and unitarios, recorded the way
#     the Pact's wars record it (common/on_actions/gg_pact_war_resolution.txt): on_war_end runs before these in the
#     same tick, so gg_constitution.3 reads them the day after. Only Buenos Aires holds a war goal (the play's target
#     cannot add any): its war goal enforced, or the league capitulating or backing down, is a federal victory;
#     Buenos Aires capitulating or backing down, a unitario one. Nothing recorded is a white peace.
# ES: quien gano la guerra entre federales y unitarios, anotado como en las guerras del Pacto; gg_constitution.3 lo
#     lee al dia siguiente. Nada anotado es paz blanca.
on_wargoal_enforced = {
	on_actions = { gg_unitario_wargoal_enforced }
}
on_capitulation = {
	on_actions = { gg_unitario_capitulation }
}
on_diplo_play_back_down = {
	on_actions = { gg_unitario_back_down }
}
gg_unitario_wargoal_enforced = {
	trigger = {
		has_global_variable = gg_unitario_war
		exists = scope:diplomatic_play
		scope:diplomatic_play = { is_diplomatic_play_type = dp_gg_pacto_federal }
		c:ARG ?= this
		c:CLN ?= scope:target
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_wargoal_enforced" # GG_TRAZA
		set_global_variable = { name = gg_unitario_federales_won value = 1 }
	}
}
gg_unitario_capitulation = {
	trigger = {
		has_global_variable = gg_unitario_war
		OR = {
			AND = { c:ARG ?= this has_truce_with = c:CLN }
			AND = { c:CLN ?= this has_truce_with = c:ARG }
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_capitulation" # GG_TRAZA
		if = { limit = { c:ARG ?= this } set_global_variable = { name = gg_unitario_unitarios_won value = 1 } }
		else = { set_global_variable = { name = gg_unitario_federales_won value = 1 } }
	}
}
gg_unitario_back_down = {
	trigger = {
		has_global_variable = gg_unitario_war
		is_diplomatic_play_type = dp_gg_pacto_federal
		exists = c:CLN
		initiator_is = c:ARG
		target_is = c:CLN
		exists = scope:actor
		OR = {
			c:ARG ?= scope:actor
			c:CLN ?= scope:actor
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_back_down" # GG_TRAZA
		if = { limit = { c:ARG ?= scope:actor } set_global_variable = { name = gg_unitario_unitarios_won value = 1 } }
		else = { set_global_variable = { name = gg_unitario_federales_won value = 1 } }
	}
}
