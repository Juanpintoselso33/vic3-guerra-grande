# EN: 23/9/2026 (Juan) -- while Rosas rules without a constitution, the month he dies ends it: the modifier goes,
#     and the question between federales and unitarios opens (gg_constitution.2).
# ES: mientras Rosas gobierne sin constitucion, el mes en que muere lo termina.
on_monthly_pulse_country = {
	on_actions = { gg_rosas_succession_monthly }
}

gg_rosas_succession_monthly = {
	trigger = {
		c:ARG ?= this
		has_modifier = gg_pais_sin_constitucion
		NOT = { has_global_variable = gg_ba_split_active } # EN/ES: 23/9 -- waits while Buenos Aires is apart / espera mientras Buenos Aires esta aparte
		NOT = { any_country = { any_scope_character = { has_template = ARG_Juan_Manuel_de_Rosas is_character_alive = yes } } }
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_rosas_succession_monthly" # GG_TRAZA
		remove_modifier = gg_pais_sin_constitucion
		set_global_variable = gg_rosas_succession_crisis
		trigger_event = { id = gg_constitution.2 days = 0 popup = yes }
	}
}

# EN/ES: the war between federales and unitarios ends (by arms or by a backdown): read the day after
#     / termina la guerra entre federales y unitarios: se lee al dia siguiente
on_war_end = {
	on_actions = { gg_unitario_on_war_end }
}
on_diplo_play_back_down = {
	on_actions = { gg_unitario_on_war_end }
}
gg_unitario_on_war_end = {
	trigger = {
		has_global_variable = gg_unitario_war
		is_diplomatic_play_type = dp_gg_pacto_federal
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_on_war_end" # GG_TRAZA
		set_global_variable = gg_unitario_war_ended # EN/ES: AUD-CONST-PEACE -- with no winner recorded, a white peace / sin ganador anotado, paz blanca
		c:ARG ?= { trigger_event = { id = gg_constitution.3 days = 1 } }
	}
}

# EN: 23/9/2026 (Codex audit, AUD-CONST-PEACE) -- who won the war between federales and unitarios, recorded the way
#     the Pact's wars record it (common/on_actions/gg_pact_war_resolution.txt): on_war_end runs before these in the
#     same tick, so gg_constitution.3 reads them the day after. Only Buenos Aires holds a war goal (the play's target
#     cannot add any): its war goal enforced, or the league capitulating or backing down, is a federal victory;
#     Buenos Aires capitulating or backing down, a unitario one. Nothing recorded is a white peace.
# ES: quien gano la guerra entre federales y unitarios, anotado como en las guerras del Pacto; gg_constitution.3 lo
#     lee al dia siguiente. Nada anotado es paz blanca.
on_wargoal_enforced = {
	on_actions = { gg_unitario_wargoal_enforced }
}
on_capitulation = {
	on_actions = { gg_unitario_capitulation }
}
on_diplo_play_back_down = {
	on_actions = { gg_unitario_back_down }
}
gg_unitario_wargoal_enforced = {
	trigger = {
		has_global_variable = gg_unitario_war
		exists = scope:diplomatic_play
		scope:diplomatic_play = { is_diplomatic_play_type = dp_gg_pacto_federal }
		c:ARG ?= this
		c:CLN ?= scope:target
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_wargoal_enforced" # GG_TRAZA
		set_global_variable = { name = gg_unitario_federales_won value = 1 }
	}
}
gg_unitario_capitulation = {
	trigger = {
		has_global_variable = gg_unitario_war
		OR = {
			AND = { c:ARG ?= this has_truce_with = c:CLN }
			AND = { c:CLN ?= this has_truce_with = c:ARG }
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_capitulation" # GG_TRAZA
		if = { limit = { c:ARG ?= this } set_global_variable = { name = gg_unitario_unitarios_won value = 1 } }
		else = { set_global_variable = { name = gg_unitario_federales_won value = 1 } }
	}
}
gg_unitario_back_down = {
	trigger = {
		has_global_variable = gg_unitario_war
		is_diplomatic_play_type = dp_gg_pacto_federal
		exists = scope:actor
		OR = {
			c:ARG ?= scope:actor
			c:CLN ?= scope:actor
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_back_down" # GG_TRAZA
		if = { limit = { c:ARG ?= scope:actor } set_global_variable = { name = gg_unitario_unitarios_won value = 1 } }
		else = { set_global_variable = { name = gg_unitario_federales_won value = 1 } }
	}
}
