# on_actions merge with the ones already defined by the game and by la Argentina,
# so nothing here overwrites vanilla content.

# Hook the list, never the effect: two mods defining 'effect' on on_game_started fight and the
# last one loaded wins. Adding to its on_actions list merges instead. Same pattern BPM uses.
on_game_started = {
	on_actions = {
		guerra_grande_on_game_start
	}
}

guerra_grande_on_game_start = {
	effect = {
		# EN/ES: T3a -- patrons, the Defensa's line and the Cerrito's balance / padrinos, linea y balanza
		gg_systems_init = yes
		# EN: initialise the debug button-press counters. Vanilla itself always sets a global
		#     variable to 0 before the first change_global_variable on it (e.g.
		#     common/on_actions/00_code_on_actions.txt: grunderzeit_german_countries) --
		#     change_global_variable on a never-set variable is not guaranteed to start from 0.
		# ES: inicializa los contadores de debug de botones apretados. La propia vanilla siempre
		#     pone una variable global en 0 antes del primer change_global_variable sobre ella
		#     (ej. common/on_actions/00_code_on_actions.txt: grunderzeit_german_countries) --
		#     change_global_variable sobre una variable nunca puesta no arranca garantizado en 0.
		set_global_variable = { name = gg_debug_intervention_presses value = 0 }
		set_global_variable = { name = gg_debug_italian_presses value = 0 }
		set_global_variable = { name = gg_debug_french_presses value = 0 }
		set_global_variable = { name = gg_debug_montonera_presses value = 0 }
		set_global_variable = { name = gg_debug_blockade_presses value = 0 }
		# EN: PROVISIONAL, see the modifier's own header in
		#     common/static_modifiers/gg_rosas_provisional_lock.txt for the full reasoning. 20
		#     years comfortably covers 1836 to Caseros (3 February 1852) with slack either side.
		# ES: PROVISORIO, ver el encabezado del propio modificador en
		#     common/static_modifiers/gg_rosas_provisional_lock.txt para el razonamiento completo.
		#     20 anios cubren con margen de 1836 a Caseros (3 de febrero de 1852).
		c:ARG ?= {
			add_modifier = {
				name = gg_rosas_provisional_lock
				years = 20
			}
		}
		c:URU ?= {
			# Manuel Oribe is the caudillo of the countryside, not a moderate: he gets
			# la Argentina's own autonomist ideology (local caudillos, tenant farmers).
			random_scope_character = {
				limit = { has_template = URU_Manuel_Oribe }
				set_ideology = ideology:ideology_autonomist
			}
			# EN/ES: T5 -- phase 0 on their historical dates (days from 1 January 1836).
			trigger_event = { id = gg_prewar.1 days = 2 popup = yes }     # 1836.01.03 El Moderador
			trigger_event = { id = gg_prewar.2 days = 74 popup = yes }    # 1836.03.15 Farrapos
			trigger_event = { id = gg_prewar.3 days = 160 popup = yes }   # 1836.06.09 cuentas de Rivera
			trigger_event = { id = gg_prewar.4 days = 532 popup = yes }   # 1837.06.16 departamentos
			trigger_event = { id = gg_prewar.5 days = 660 popup = yes }   # 1837.10.22 Yucutuja
			trigger_event = { id = gg_prewar.6 days = 776 popup = yes }   # 1838.02.15 el Yi
			trigger_event = { id = gg_prewar.7 days = 840 popup = yes }   # 1838.04.20 bloqueo frances
			trigger_event = { id = gg_prewar.8 days = 877 popup = yes }   # 1838.05.27 Universidad
			trigger_event = { id = gg_prewar.9 days = 896 popup = yes }   # 1838.06.15 Palmar
			# EN/ES: T6 -- Ley N. 242 of the Defensa / de la Defensa, 1842.12.12
			trigger_event = { id = gg_abolition.1 days = 2537 }
			# 19 September 1836: the battle of Carpinteria, where the ribbons are born.
			trigger_event = { id = guerra_grande.1 days = 262 popup = yes }
		}
		# EN: In 1836 the Republic has two borders at once, and the men of the time said so in
		#     those words: Juan Manuel de la Sota's Catecismo geografico distinguishes the
		#     "limites de hecho" -- the Cuareim and the Yaguaron, where Rivera's army had halted
		#     by the agreement of 25 December 1828 -- from the "limites de derecho", "mas
		#     extensos", those of San Ildefonso (1777). In 1837 Oribe was still instructing
		#     Villademoros in Rio to claim as far as the river Ibicui, plus the navigation of the
		#     Laguna Merin and of the channels leading to San Pedro and Porto Alegre. On the
		#     ground Brazil had held the Campos Neutrales between the Chuy and San Miguel passes
		#     since the war of 1801. So the claim is a right, not a possession -- and the limits
		#     treaty of 12 October 1851 is the moment it dies: article 1 renounced the San
		#     Ildefonso reserve outright, and article 3 left Uruguay a dry coast on the lagoon
		#     and the river for the next 58 years, until the treaty of 1909.
		# ES: En 1836 la Republica tiene dos fronteras a la vez, y los hombres de la epoca lo
		#     decian con esas palabras: el Catecismo geografico de Juan Manuel de la Sota
		#     distingue los "limites de hecho" -- el Cuareim y el Yaguaron, donde el ejercito de
		#     Rivera se habia detenido por el acuerdo del 25 de diciembre de 1828 -- de los
		#     "limites de derecho", "mas extensos", los de San Ildefonso (1777). En 1837 Oribe
		#     todavia instruia a Villademoros en Rio a reclamar hasta el rio Ibicui, y la
		#     navegacion de la Laguna Merin y de los canales que llevan a San Pedro y Porto
		#     Alegre. En el terreno, Brasil ocupaba los Campos Neutrales entre los pasos del Chuy
		#     y de San Miguel desde la guerra de 1801. El reclamo es un derecho, no una posesion,
		#     y el tratado de limites del 12 de octubre de 1851 es el momento en que muere: el
		#     articulo 1 renuncio a la reserva de San Ildefonso y el articulo 3 dejo al Uruguay
		#     costa seca sobre la laguna y el rio por 58 anios, hasta el tratado de 1909.
		s:STATE_RIO_GRANDE_DO_SUL ?= {
			add_claim = c:URU
		}
	}
}

# Vanilla moves the carlist bar from on_battle_won; the Guerra Grande bar works the same way.
on_battle_won = {
	on_actions = {
		guerra_grande_battle_score
	}
}

on_empty_battle_won = {
	on_actions = {
		guerra_grande_battle_score
	}
}

guerra_grande_battle_score = {
	trigger = {
		has_global_variable = guerra_grande_open_war
		has_journal_entry = je_guerra_grande
		scope:enemy_country ?= {
			has_journal_entry = je_guerra_grande
		}
	}

	effect = {
		if = {
			limit = {
				c:CER ?= this
			}
			change_global_variable = {
				name = guerra_grande_blanco_score
				add = 1
			}
			je:je_guerra_grande ?= {
				add_progress = { value = 1 name = guerra_grande_war_score_bar }
			}
		}
		else_if = {
			limit = {
				c:URU ?= this
			}
			change_global_variable = {
				name = guerra_grande_colorado_score
				add = 1
			}
			je:je_guerra_grande ?= {
				add_progress = { value = -1 name = guerra_grande_war_score_bar }
			}
		}
	}
}

# EN: HOTFIX 2026-09-16. Our peace menu (gg_peace.1/.2) only fired off our OWN battle-score lead
#     reaching 10, checked in the journal's monthly pulse -- a slower, separate clock from the
#     real war. The real dp_guerra_grande diplomatic play resolves through vanilla's own
#     occupation/war-score mechanics, which can end much faster: reported in play, the Defensa's
#     army melted under AI control, Juan attacked once and won outright, and NO peace event fired
#     at all -- only gg_loan.6 (the debt reckoning, which doesn't need the enemy to still exist).
#     This is a hotfix, not the real fix: it reacts to the real war ending and fires the correct
#     peace menu immediately, bypassing the lead-10 gate, so a decisive military win is never
#     silently missed. The actual redesign (unifying the two clocks properly) is deferred --
#     Juan asked for a flowchart of all guerra_grande events first, next week.
# ES: HOTFIX 2026-09-16. Nuestro menu de paz (gg_peace.1/.2) solo disparaba cuando NUESTRO
#     contador de batallas llegaba a 10 de ventaja, chequeado en el pulso mensual del journal: un
#     reloj mas lento y separado del de la guerra real. La jugada diplomatica dp_guerra_grande de
#     verdad se resuelve por el mecanismo normal de ocupacion/war-score de vanilla, que puede
#     terminar mucho mas rapido: en una partida real, el ejercito de la Defensa se derritio bajo
#     control de la IA, Juan ataco una vez y gano de una, y NO disparo ningun evento de paz --
#     solo gg_loan.6 (el ajuste de deudas, que no necesita que el enemigo siga existiendo).
#     Esto es un hotfix, no el arreglo de fondo: reacciona a que la guerra real termine y dispara
#     el menu de paz correcto de inmediato, saltandose el piso de 10, para que una victoria militar
#     decisiva nunca se pierda en silencio. El rediseño de fondo (unificar los dos relojes bien)
#     queda para despues -- Juan pidio primero un diagrama de flujo de todos los eventos, la
#     semana que viene.
on_war_end = {
	on_actions = {
		guerra_grande_on_war_end
	}
}

guerra_grande_on_war_end = {
	trigger = {
		# Root = the diplomatic play that just concluded.
		# EN: BUG FOUND 2026-09-16 (real conquest tested, no popup fired). Gating on
		#     guerra_grande_open_war raced against the journal entry's own complete condition,
		#     which polls territorial control continuously and can clear open_war (via
		#     on_complete_all_involved's hidden_effect) before or in the same tick this on_action
		#     runs -- so a real, decisive conquest silently failed this trigger with no error.
		#     guerra_grande_split_done is set once at the split and never cleared afterward, so it
		#     cannot lose that race.
		# ES: BUG ENCONTRADO 2026-09-16 (conquista real probada, no salto el popup). Depender de
		#     guerra_grande_open_war corria una carrera contra la propia condicion complete del
		#     journal, que sondea el control territorial todo el tiempo y puede apagar open_war
		#     (via el hidden_effect de on_complete_all_involved) antes o en el mismo tick en que
		#     corre este on_action -- asi que una conquista real y decisiva fallaba este trigger en
		#     silencio, sin error. guerra_grande_split_done se prende una sola vez en el corte y
		#     nunca se apaga despues, asi que no puede perder esa carrera.
		is_diplomatic_play_type = dp_guerra_grande
		has_global_variable = guerra_grande_split_done
		NOT = { has_global_variable = gg_peace_settled }
	}

	effect = {
		# EN: no winner/loser scope exists at on_war_end (only scope:actor/scope:target, the
		#     initiator and the target), so the winner is read off the same world-state check
		#     the journal's own complete/fail conditions already use: who still exists, and who
		#     holds what.
		# ES: on_war_end no expone quien gano (solo scope:actor/scope:target, el iniciador y el
		#     objetivo), asi que el ganador se lee del mismo chequeo de estado del mundo que ya
		#     usan las condiciones complete/fail del journal: quien sigue existiendo, y quien
		#     tiene que.
		if = {
			limit = {
				OR = {
					NOT = { exists = c:URU }
					c:URU ?= {
						NOR = {
							any_scope_state = { state_region = s:STATE_SAN_JOSE }
							any_scope_state = { state_region = s:STATE_MONTEVIDEO }
						}
					}
				}
				exists = c:CER
			}
			random_country = {
				limit = { has_variable = guerra_grande_cerrito_government }
				trigger_event = { id = gg_peace.2 days = 3 popup = yes }
			}
		}
		else_if = {
			limit = {
				OR = {
					NOT = { exists = c:CER }
					c:CER ?= { NOT = { any_scope_state = { state_region = s:STATE_MONTEVIDEO } } }
				}
				exists = c:URU
			}
			random_country = {
				limit = { has_variable = guerra_grande_defensa_government }
				trigger_event = { id = gg_peace.1 days = 3 popup = yes }
			}
		}
	}
}
