# EN: T1, C4 -- only the white peace comes through here: both governments still exist and nobody
#     annexed anybody. A war that ended with an annexation is closed by the journal (C11). The menu
#     goes to whoever leads (a tie goes to the Defensa, FA-1) and has no 'wait' option.
# ES: T1, C4 -- por aca solo pasa la paz blanca: los dos gobiernos existen y nadie anexo a nadie.
#     Una guerra terminada con anexion la cierra el journal. El menu va al que va arriba (el
#     empate va a la Defensa) y no tiene opcion de esperar.
guerra_grande_on_war_end = {
	trigger = {
		# Root = the diplomatic play that just concluded.
		is_diplomatic_play_type = dp_guerra_grande
		has_global_variable = guerra_grande_split_done
		# EN/ES: 23/9 -- not when the siege's locks close an old save's engine war (gg_siege_locks.txt)
		#     / no cuando los candados del sitio cierran la guerra de una partida vieja
		NOT = { has_global_variable = gg_siege_war_migrated }
		NOT = { has_global_variable = gg_peace_settled }
		exists = c:CER
		exists = c:URU
	}

	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|guerra_grande_on_war_end" # GG_TRAZA
		# EN/ES: review 2026-09-17 -- also when a menu is already pending: its f option must go
		#     / tambien con un menu pendiente: su opcion f tiene que desaparecer
		set_global_variable = gg_white_peace
		if = {
			limit = { NOT = { has_global_variable = gg_peace_offered } }
			set_global_variable = gg_peace_offered
			if = {
			limit = { guerra_grande_blanco_lead > 0 }
			random_country = {
				limit = { has_variable = guerra_grande_cerrito_government }
				trigger_event = { id = gg_peace.2 days = 3 popup = yes }
			}
		}
		else = {
			random_country = {
				limit = { has_variable = guerra_grande_defensa_government }
				trigger_event = { id = gg_peace.1 days = 3 popup = yes }
			}
		}
		}
	}
}
