# Vanilla moves the carlist bar from on_battle_won; the Guerra Grande bar works the same way.
on_battle_won = {
	on_actions = {
		guerra_grande_battle_score
	}
}

on_empty_battle_won = {
	on_actions = {
		guerra_grande_battle_score
	}
}

guerra_grande_battle_score = {
	trigger = {
		has_global_variable = guerra_grande_open_war
		has_journal_entry = je_guerra_grande
		scope:enemy_country ?= {
			has_journal_entry = je_guerra_grande
		}
	}

	effect = {
		# EN: T2, C3 -- the score and the real war move together: every battle won costs the loser
		#     war support in the Guerra Grande. -5 is provisional until T10.
		# ES: T2, C3 -- el puntaje y la guerra real se mueven juntos: cada batalla ganada le cuesta
		#     apoyo a la guerra al perdedor. El -5 es provisorio hasta T10.
		every_scope_war = {
			limit = { gg_is_the_war = yes }
			add_war_war_support = {
				target = scope:enemy_country
				value = -5
			}
		}
		if = {
			limit = {
				c:CER ?= this
			}
			change_global_variable = {
				name = guerra_grande_blanco_score
				add = 1
			}
			je:je_guerra_grande ?= {
				set_bar_progress = { name = guerra_grande_war_score_bar value = gg_war_score_bar_value }
			}
		}
		else_if = {
			limit = {
				c:URU ?= this
			}
			change_global_variable = {
				name = guerra_grande_colorado_score
				add = 1
			}
			je:je_guerra_grande ?= {
				set_bar_progress = { name = guerra_grande_war_score_bar value = gg_war_score_bar_value }
			}
		}
	}
}

#     This is a hotfix, not the real fix: it reacts to the real war ending and fires the correct
#     peace menu immediately, bypassing the lead-10 gate, so a decisive military win is never
#     silently missed. The actual redesign (unifying the two clocks properly) is deferred --

# ES: HOTFIX 2026-09-16. Nuestro menu de paz (gg_peace.1/.2) solo disparaba cuando NUESTRO
#     contador de batallas llegaba a 10 de ventaja, chequeado en el pulso mensual del journal: un

on_war_end = {
	on_actions = {
		guerra_grande_on_war_end
	}
}

# EN: T1, C4 -- only the white peace comes through here: both governments still exist and nobody
#     annexed anybody. A war that ended with an annexation is closed by the journal (C11). The menu
#     goes to whoever leads (a tie goes to the Defensa, FA-1) and has no 'wait' option.
# ES: T1, C4 -- por aca solo pasa la paz blanca: los dos gobiernos existen y nadie anexo a nadie.
#     Una guerra terminada con anexion la cierra el journal. El menu va al que va arriba (el
#     empate va a la Defensa) y no tiene opcion de esperar.
guerra_grande_on_war_end = {
	trigger = {
		# Root = the diplomatic play that just concluded.
		is_diplomatic_play_type = dp_guerra_grande
		has_global_variable = guerra_grande_split_done
		NOT = { has_global_variable = gg_peace_settled }
		exists = c:CER
		exists = c:URU
	}

	effect = {
		# EN/ES: review 2026-09-17 -- also when a menu is already pending: its f option must go
		#     / tambien con un menu pendiente: su opcion f tiene que desaparecer
		set_global_variable = gg_white_peace
		if = {
			limit = { NOT = { has_global_variable = gg_peace_offered } }
			set_global_variable = gg_peace_offered
			if = {
			limit = { guerra_grande_blanco_lead > 0 }
			random_country = {
				limit = { has_variable = guerra_grande_cerrito_government }
				trigger_event = { id = gg_peace.2 days = 3 popup = yes }
			}
		}
		else = {
			random_country = {
				limit = { has_variable = guerra_grande_defensa_government }
				trigger_event = { id = gg_peace.1 days = 3 popup = yes }
			}
		}
		}
	}
}

	#     duro el sitio, y la Asamblea General del Cerrito confirmo a Oribe en 1845.

		OR = {
			AND = {
				c:URU ?= this
				NOT = { has_global_variable = guerra_grande_split_done }
				OR = {
					has_global_variable = gg_rivera_presidency
					has_variable = guerra_grande_blanco_path
				}
			}
			has_variable = guerra_grande_defensa_government
			has_variable = guerra_grande_cerrito_government
		}
		OR = {
			NOT = { has_global_variable = gg_peace_settled }
			has_global_variable = gg_president_railroad
		}
	}
	effect = {
		trigger_event = { id = gg_phase1.10 days = 1 }
	}
}
