# EN: 24/9/2026 (Juan) -- the provinces subject to Argentina join its foreign wars
#     (common/scripted_effects/gg_arg_provinces_war.txt). Two doors:
#     1. a war involving Argentina starts, as initiator, target or backer, from any play (the mod's or an engine/AI one
#        with any war goal; root = the diplomatic play, as in gg_pact_sides_on_actions.txt): a hidden event on
#        Argentina a week later, so a war that ends at once (a scripted play closed the same days) drags nobody in,
#        and the war is standing when the provinces join. Not at the play stage: while the play lasts a backer can
#        be named in new war goals (Brazil could claim Corrientes), and the Workshop's war rework 3474730999 keeps
#        its play-stage variant (add_*_backers) commented out;
#     2. Argentina's monthly pulse, the net: a province that became a subject mid-war, and a war created with
#        war = yes that may not pass through the play's war start (dp_gg_cisplatina).
# ES: las provincias sujetas a la Argentina entran a sus guerras exteriores. Dos puertas: una semana despues de que
#     empieza una guerra de la Argentina (asi una guerra que termina enseguida no arrastra a nadie), y el pulso
#     mensual de la Argentina como red (una provincia que pasa a ser sujeta en plena guerra, una guerra creada sin
#     etapa de jugada).
on_diplo_play_war_start = {
	on_actions = { gg_arg_provinces_war_start }
}

gg_arg_provinces_war_start = {
	trigger = {
		# EN/ES: Argentina on either side, whoever started it and whatever the goal (a Brazilian AI play to humiliate
		#     Argentina, Juan's game of 24/9/2026, is the case) / la Argentina de cualquier lado, lo haya empezado quien
		#     sea y con el objetivo que sea (el caso es una jugada del Brasil de la IA para humillarla, 24/9/2026)
		# EN/ES: 25/9 (AUD-PLT-ARGONLY) -- the Argentine state, also as La Plata / Great Argentina (gg_is_argentine_state);
		#     a target is matched from the country side (is_diplomatic_play_target), the play's target link may be a state
		#     / el estado argentino, tambien como La Plata o la Gran Argentina
		OR = {
			initiator ?= { gg_is_argentine_state = yes }
			any_scope_play_involved = { gg_is_argentine_state = yes }
			any_country = {
				gg_is_argentine_state = yes
				is_diplomatic_play_target = yes
			}
		}
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_provinces_war_start" # GG_TRAZA
		every_country = {
			limit = { gg_is_argentine_state = yes }
			trigger_event = { id = gg_arg_allies.1 days = 7 }
		}
	}
}

on_monthly_pulse_country = {
	on_actions = { gg_arg_provinces_war_monthly }
}

gg_arg_provinces_war_monthly = {
	trigger = {
		gg_is_argentine_state = yes # EN/ES: 25/9 (AUD-PLT-ARGONLY) -- also La Plata / Great Argentina formed from Argentina / tambien La Plata o la Gran Argentina
		is_at_war = yes
		any_scope_war = { gg_arg_war_is_foreign = yes }
		any_subject_or_below = { gg_arg_province_marches = yes }
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_provinces_war_monthly" # GG_TRAZA
		gg_arg_provinces_join_foreign_wars = yes
	}
}

# EN: 24/9/2026 (Codex review) -- and a war a province marched into that has since turned into one between Argentines
#     lets it go (gg_arg_provinces_leave_argentine_wars). Only when some province carries the list.
# ES: y una guerra en la que marcho una provincia y que despues se volvio una guerra entre argentinos la deja ir.
on_monthly_pulse_country = {
	on_actions = { gg_arg_provinces_leave_monthly }
}

gg_arg_provinces_leave_monthly = {
	trigger = {
		gg_is_argentine_state = yes # EN/ES: 25/9 (AUD-PLT-ARGONLY) -- also La Plata / Great Argentina formed from Argentina / tambien La Plata o la Gran Argentina
		is_at_war = yes
		any_scope_war = { NOT = { gg_arg_war_is_foreign = yes } }
		any_country = { has_variable_list = gg_arg_marched_against }
	}
	effect = {
		debug_log = "GG_OA|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_provinces_leave_monthly" # GG_TRAZA
		gg_arg_provinces_leave_argentine_wars = yes
	}
}
