# EN: 25/9/2026 (Juan: «las dos») -- puts on and takes off the war's demographic modifiers (static_modifiers/
#     gg_war_demography.txt) on one state. State scope. Idempotent: called every half year by gg_war_demography_pulse,
#     which is also the migration for games in progress (a save in the middle of the war gets them within six
#     months; one past the peace never does, because gg_wartime_cleared is set).
#     Montevideo gets the refuge only while the Cerrito does not hold it; the countryside is ravaged whoever holds it.
# ES: pone y saca los modificadores demograficos de la guerra en un estado. Idempotente; cada medio año, y sirve de
#     migracion para partidas en curso. Montevideo tiene el refugio solo mientras no la tiene el Cerrito.
gg_war_demography_sync = {
	if = {
		limit = {
			gg_war_demography_on = yes
			state_region = s:STATE_MONTEVIDEO
			owner = { NOT = { c:CER ?= this } }
		}
		if = {
			limit = { NOT = { has_modifier = gg_war_montevideo_refuge } }
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_war_demography_sync|refuge on" # GG_TRAZA
			add_modifier = { name = gg_war_montevideo_refuge }
		}
	}
	else_if = {
		limit = { has_modifier = gg_war_montevideo_refuge }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_war_demography_sync|refuge off" # GG_TRAZA
		remove_modifier = gg_war_montevideo_refuge
	}
	if = {
		limit = {
			gg_war_demography_on = yes
			gg_is_oriental_countryside_state = yes
		}
		if = {
			limit = { NOT = { has_modifier = gg_war_countryside_ravaged } }
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_war_demography_sync|ravaged on" # GG_TRAZA
			add_modifier = { name = gg_war_countryside_ravaged }
		}
	}
	else_if = {
		limit = { has_modifier = gg_war_countryside_ravaged }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_war_demography_sync|ravaged off" # GG_TRAZA
		remove_modifier = gg_war_countryside_ravaged
	}
}

# EN: the peace takes them off the same day, with the rest of the wartime modifiers (called from
#     gg_wartime_modifiers_clear); the half-yearly sync would otherwise leave them up to six months.
# ES: la paz los saca el mismo dia, con el resto de los modificadores de guerra.
gg_war_demography_clear = {
	every_state = {
		limit = {
			OR = {
				has_modifier = gg_war_montevideo_refuge
				has_modifier = gg_war_countryside_ravaged
			}
		}
		if = { limit = { has_modifier = gg_war_montevideo_refuge } remove_modifier = gg_war_montevideo_refuge }
		if = { limit = { has_modifier = gg_war_countryside_ravaged } remove_modifier = gg_war_countryside_ravaged }
	}
}

# EN: the population of Montevideo and of the other four Oriental states, once a year until 1862, so a headless run
#     reads the share without a save (GG_POP|date|montevideo= N campana= N). Written only with -debug_mode.
# ES: la poblacion de Montevideo y de los otros cuatro estados orientales, una vez por año hasta 1862, para que una
#     corrida sin pantalla lea la proporcion sin partida guardada. Solo escribe con -debug_mode.
gg_trace_oriental_population = {
	set_global_variable = { name = gg_debug_pop_mvd value = 0 }
	set_global_variable = { name = gg_debug_pop_rest value = 0 }
	every_state = {
		limit = { state_region = s:STATE_MONTEVIDEO }
		change_global_variable = { name = gg_debug_pop_mvd add = state_population }
	}
	every_state = {
		limit = { gg_is_oriental_countryside_state = yes }
		change_global_variable = { name = gg_debug_pop_rest add = state_population }
	}
	debug_log = "GG_POP|[TimeKeeper.GetCurrentDate.GetString]|montevideo= [GetGlobalVariable('gg_debug_pop_mvd').GetValue|0] campana= [GetGlobalVariable('gg_debug_pop_rest').GetValue|0]"
	remove_global_variable = gg_debug_pop_mvd
	remove_global_variable = gg_debug_pop_rest
}
