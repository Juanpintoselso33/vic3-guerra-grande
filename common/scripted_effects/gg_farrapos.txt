# EN: THE FARRAPOS (Ragamuffin War, 1835-1845). 23/9/2026 (Juan: integration I2; AUD-FARRAPOS-A option 1:
#     no engine war until Ponche Verde, like the siege of Montevideo). One global, gg_farrapos_refuge (0-3),
#     is how strong the Oriental refuge of the Riograndense Republic is. It moves two things, both visible:
#       - on Brazil, the troops tied down in Rio Grande (gg_farrapos_tied_down_0-3): army offence down and
#         military goods dearer, more with a stronger refuge. This is what kept the Empire out of the Plata
#         until 1845. It ends at Ponche Verde;
#       - on PNI, the Republic's strength (gg_farrapos_refuge_1-3).
#     Who moves it: gg_prewar.2 (the 1836 border), gg_prewar.9 (Cangue, 1838), gg_phase1.8 (San Fructuoso,
#     1841) and gg_farrapos.6 (Caxias's pact with Oribe, 1842). The end is always Ponche Verde (gg_farrapos.2,
#     25/2/1845).
# ES: LOS FARRAPOS. Sin guerra del motor hasta Ponche Verde. Una global, gg_farrapos_refuge (0-3), es la
#     fuerza del refugio oriental. Mueve dos cosas visibles: en el Brasil, las tropas atadas en Rio Grande
#     (que lo tuvieron fuera del Plata hasta 1845; se terminan en Ponche Verde); en PNI, la fuerza de la
#     Republica. La mueven la frontera de 1836, Cangue, San Fructuoso y el pacto de Caxias con Oribe.
# -------------------------------------------------------------------------------------------------

# EN/ES: from guerra_grande_on_game_start. Scheduled on Brazil, which never changes tag, and directly: NOT
#     through the gg_cal relays / sobre el Brasil, que no cambia de tag, y directo: no por los relevos gg_cal
gg_farrapos_start = {
	if = {
		limit = {
			exists = c:PNI
			exists = c:BRZ
		}
		set_global_variable = { name = gg_farrapos_refuge value = 1 }
		gg_farrapos_refuge_apply = yes
		c:BRZ = {
			# EN: 9/11/1842: (1842-1836)*365 + (313-1) = 2190 + 312 = 2502
			trigger_event = { id = gg_farrapos.5 days = 2502 }   # Caxias takes command / Caxias asume
			# EN: 25/2/1845: (1845-1836)*365 + (56-1) = 3285 + 55 = 3340
			trigger_event = { id = gg_farrapos.2 days = 3340 }   # Ponche Verde
		}
	}
}

# EN/ES: the modifiers that match the refuge, on Brazil and on PNI (hidden; the tooltip is
#     gg_farrapos_refuge_change's) / los modificadores que corresponden al refugio
gg_farrapos_refuge_apply = {
	c:BRZ ?= {
		gg_farrapos_tied_down_clear = yes
		if = {
			limit = { NOT = { has_global_variable = gg_farrapos_ponche_verde } }
			if = {
				limit = { global_var:gg_farrapos_refuge >= 3 }
				add_modifier = { name = gg_farrapos_tied_down_3 }
			}
			else_if = {
				limit = { global_var:gg_farrapos_refuge = 2 }
				add_modifier = { name = gg_farrapos_tied_down_2 }
			}
			else_if = {
				limit = { global_var:gg_farrapos_refuge = 1 }
				add_modifier = { name = gg_farrapos_tied_down_1 }
			}
			else = {
				add_modifier = { name = gg_farrapos_tied_down_0 }
			}
		}
	}
	c:PNI ?= {
		if = {
			limit = { has_modifier = gg_farrapos_refuge_1 }
			remove_modifier = gg_farrapos_refuge_1
		}
		if = {
			limit = { has_modifier = gg_farrapos_refuge_2 }
			remove_modifier = gg_farrapos_refuge_2
		}
		if = {
			limit = { has_modifier = gg_farrapos_refuge_3 }
			remove_modifier = gg_farrapos_refuge_3
		}
		if = {
			limit = { global_var:gg_farrapos_refuge = 1 }
			add_modifier = { name = gg_farrapos_refuge_1 }
		}
		else_if = {
			limit = { global_var:gg_farrapos_refuge = 2 }
			add_modifier = { name = gg_farrapos_refuge_2 }
		}
		else_if = {
			limit = { global_var:gg_farrapos_refuge >= 3 }
			add_modifier = { name = gg_farrapos_refuge_3 }
		}
	}
}

# EN/ES: Brazil scope; also called at Ponche Verde / scope Brasil; tambien en Ponche Verde
gg_farrapos_tied_down_clear = {
	if = {
		limit = { has_modifier = gg_farrapos_tied_down_0 }
		remove_modifier = gg_farrapos_tied_down_0
	}
	if = {
		limit = { has_modifier = gg_farrapos_tied_down_1 }
		remove_modifier = gg_farrapos_tied_down_1
	}
	if = {
		limit = { has_modifier = gg_farrapos_tied_down_2 }
		remove_modifier = gg_farrapos_tied_down_2
	}
	if = {
		limit = { has_modifier = gg_farrapos_tied_down_3 }
		remove_modifier = gg_farrapos_tied_down_3
	}
}

# EN: DELTA = -3 (closed), -1, 0 or 1. Shown to the player: a global has no tooltip of its own (CLAUDE.md).
# ES: se muestra al jugador; una global no tiene tooltip propio.
gg_farrapos_refuge_change = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_farrapos_refuge_change| DELTA= $DELTA$ " # GG_TRAZA
	custom_tooltip = {
		text = gg_tt_farrapos_refuge_$DELTA$
		if = {
			limit = { NOT = { has_global_variable = gg_farrapos_refuge } }
			set_global_variable = { name = gg_farrapos_refuge value = 1 }
		}
		change_global_variable = { name = gg_farrapos_refuge add = $DELTA$ }
		if = {
			limit = { global_var:gg_farrapos_refuge > 3 }
			set_global_variable = { name = gg_farrapos_refuge value = 3 }
		}
		else_if = {
			limit = { global_var:gg_farrapos_refuge < 0 }
			set_global_variable = { name = gg_farrapos_refuge value = 0 }
		}
	}
	custom_tooltip = gg_tt_farrapos_refuge_means
	hidden_effect = { gg_farrapos_refuge_apply = yes }
}
