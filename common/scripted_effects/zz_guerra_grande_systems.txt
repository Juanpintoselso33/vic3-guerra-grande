# EN: The Guerra Grande systems (docs/guerra-grande-flujo/to-be.md, C19, FG-1, FG-2).
#     T3a: variables and their single writers only -- nothing reads them yet. Every change to a
#     patron, to the Defensa's line or to the Cerrito's balance goes through one of these
#     effects, so the invariant check (I14) can prove nobody writes them anywhere else.
#     Starting values are provisional until the balance pass (T10).
# ES: Los sistemas de la Guerra Grande (docs/guerra-grande-flujo/to-be.md, C19, FG-1, FG-2).
#     T3a: solo las variables y quien las escribe -- todavia nadie las lee. Todo cambio a un
#     padrino, a la linea de la Defensa o a la balanza del Cerrito pasa por uno de estos
#     efectos, asi el chequeo de invariantes (I14) puede probar que nadie las escribe en otro
#     lado. Los valores iniciales son provisorios hasta la pasada de balance (T10).

# EN: called once from guerra_grande_on_game_start.
# ES: se llama una sola vez desde guerra_grande_on_game_start.
gg_systems_init = {
	# EN/ES: patrons, 0-100 / padrinos, 0-100
	set_global_variable = { name = gg_patron_rosas value = 50 }
	set_global_variable = { name = gg_patron_francia value = 20 }
	set_global_variable = { name = gg_patron_britania value = 10 }
	set_global_variable = { name = gg_patron_brasil value = 20 }
	set_global_variable = { name = gg_patron_urquiza value = 40 }
	# EN: the Defensa's line: 0 = no Defensa yet, 1 = Pacheco y Obes, 2 = Herrera y Obes.
	# ES: la linea de la Defensa: 0 = todavia no hay Defensa, 1 = Pacheco y Obes, 2 = Herrera y Obes.
	set_global_variable = { name = gg_defensa_line value = 0 }
	# EN: the Cerrito's balance, -100 (caudillos) to 100 (doctores).
	# ES: la balanza del Cerrito, -100 (caudillos) a 100 (doctores).
	set_global_variable = { name = gg_cerrito_balance value = 0 }
	# EN: T5 -- phase 0 events score before Carpinteria creates the journal, so the score starts
	#     here; the journal's own immediate only sets it if it is still missing.
	# ES: T5 -- los eventos de la fase 0 puntuan antes de que Carpinteria cree el journal, asi que
	#     el puntaje arranca aca; el immediate del journal solo lo pone si todavia falta.
	set_global_variable = { name = guerra_grande_blanco_score value = 0 }
	set_global_variable = { name = guerra_grande_colorado_score value = 0 }
}

# EN: usage: gg_score_blanco = { AMOUNT = 2 }. AMOUNT must be a positive literal: the colorado
#     side writes -$AMOUNT$ for the bar, so a negative number or a named value would not parse.
#     To take score away from one side, give it to the other. Moves the score, and the journal bar if the
#     journal already exists (before Carpinteria it does not; .1 syncs the bar when it opens).
# ES: uso: gg_score_blanco = { AMOUNT = 2 }. AMOUNT tiene que ser un numero positivo escrito tal
#     cual: el lado colorado escribe -$AMOUNT$ para la barra. Para restarle a un bando, se le suma
#     al otro. Mueve el puntaje, y la barra si el journal ya existe
#     (antes de Carpinteria no existe; .1 sincroniza la barra al abrirlo).
gg_score_blanco = {
	change_global_variable = { name = guerra_grande_blanco_score add = $AMOUNT$ }
	je:je_guerra_grande ?= {
		add_progress = { value = $AMOUNT$ name = guerra_grande_war_score_bar }
	}
}

gg_score_colorado = {
	change_global_variable = { name = guerra_grande_colorado_score add = $AMOUNT$ }
	je:je_guerra_grande ?= {
		add_progress = { value = -$AMOUNT$ name = guerra_grande_war_score_bar }
	}
}

# EN: usage: gg_patron_change = { PATRON = francia DELTA = -15 }
#     PATRON is one of rosas, francia, britania, brasil, urquiza. The result is kept in 0-100.
# ES: uso: gg_patron_change = { PATRON = francia DELTA = -15 }
#     PATRON es rosas, francia, britania, brasil o urquiza. El resultado queda entre 0 y 100.
gg_patron_change = {
	change_global_variable = { name = gg_patron_$PATRON$ add = $DELTA$ }
	if = {
		limit = { global_var:gg_patron_$PATRON$ > 100 }
		set_global_variable = { name = gg_patron_$PATRON$ value = 100 }
	}
	else_if = {
		limit = { global_var:gg_patron_$PATRON$ < 0 }
		set_global_variable = { name = gg_patron_$PATRON$ value = 0 }
	}
	debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_PATRON $PATRON$ $DELTA$"
}

# EN: usage: gg_defensa_line_set = { LINE = 2 }  (1 = Pacheco y Obes, 2 = Herrera y Obes)
# ES: uso: gg_defensa_line_set = { LINE = 2 }  (1 = Pacheco y Obes, 2 = Herrera y Obes)
gg_defensa_line_set = {
	set_global_variable = { name = gg_defensa_line value = $LINE$ }
	debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_LINE $LINE$"
}

# EN: usage: gg_cerrito_balance_change = { DELTA = 10 }  (positive = towards the doctores)
# ES: uso: gg_cerrito_balance_change = { DELTA = 10 }  (positivo = hacia los doctores)
gg_cerrito_balance_change = {
	change_global_variable = { name = gg_cerrito_balance add = $DELTA$ }
	if = {
		limit = { global_var:gg_cerrito_balance > 100 }
		set_global_variable = { name = gg_cerrito_balance value = 100 }
	}
	else_if = {
		limit = { global_var:gg_cerrito_balance < -100 }
		set_global_variable = { name = gg_cerrito_balance value = -100 }
	}
	debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_BALANCE $DELTA$"
}
