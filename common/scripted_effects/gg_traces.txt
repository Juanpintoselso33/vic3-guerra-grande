# EN: 23/9/2026 (Juan: "add logs everywhere, otherwise you won't understand anything") -- the arbiter's
#     numbers at the moment it offers the peace, so a headless run says WHY a government was judged
#     beaten, not only that it was. Written only with -debug_mode (debug_log). BRANCH names the branch.
# ES: los numeros del arbitro en el momento en que ofrece la paz, para que una corrida sin pantalla diga
#     POR QUE se dio por batido a un gobierno. Solo escribe con -debug_mode.
gg_trace_arbiter = {
	set_global_variable = { name = gg_debug_arb_bal_bl value = gg_war_balance_blanco }
	set_global_variable = { name = gg_debug_arb_bal_co value = gg_war_balance_colorado }
	set_global_variable = { name = gg_debug_arb_gap value = gg_exhaustion_gap }
	debug_log = "GG_ARB|[TimeKeeper.GetCurrentDate.GetString]|$BRANCH$|balanza_blancos= [GetGlobalVariable('gg_debug_arb_bal_bl').GetValue|0] balanza_colorados= [GetGlobalVariable('gg_debug_arb_bal_co').GetValue|0] rio_cerrado_racha= [GetGlobalVariable('gg_plata_closed_streak').GetValue|0] defensa_agotada= [GetGlobalVariable('gg_defensa_exhausted').GetValue|0] cerrito_agotado= [GetGlobalVariable('gg_cerrito_exhausted').GetValue|0] brecha_agotamiento= [GetGlobalVariable('gg_debug_arb_gap').GetValue|0] puntaje_blanco= [GetGlobalVariable('guerra_grande_blanco_score').GetValue|0] puntaje_colorado= [GetGlobalVariable('guerra_grande_colorado_score').GetValue|0]"
	remove_global_variable = gg_debug_arb_bal_bl
	remove_global_variable = gg_debug_arb_bal_co
	remove_global_variable = gg_debug_arb_gap
}
