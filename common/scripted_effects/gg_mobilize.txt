# EN: 23/9 -- an army created in this same effect cannot be mobilized yet ("Scope formation cannot be mobilized",
#     headless run 161928, the Ejercito Grande). The owner queues it (country scope, where it was created) and
#     gg_troops.10 mobilizes it the next day and sends it to a front of its war against ENEMY.
# ES: un ejercito recien creado todavia no se puede movilizar: el duenio lo encola y gg_troops.10 lo moviliza al
#     dia siguiente y lo manda al frente de su guerra contra ENEMY.
gg_mobilize_next_day = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_mobilize_next_day| ARMY= $ARMY$  ENEMY= $ENEMY$ " # GG_TRAZA
	# EN/ES: 25/9/2026 (AUD-ERR-DEPLOY-BRZ-DIV) -- a new wave forgets the last wave's front (gg_troops.10)
	#     / una tanda nueva olvida el frente de la anterior
	if = {
		limit = { NOT = { has_variable = gg_mob_pending } has_global_variable = gg_mob_front_ref }
		remove_global_variable = gg_mob_front_ref
	}
	add_to_variable_list = { name = gg_mob_queue target = $ARMY$ }
	set_variable = { name = gg_mob_enemy value = $ENEMY$ }
	if = {
		limit = { NOT = { has_variable = gg_mob_pending } }
		set_variable = gg_mob_pending
		trigger_event = { id = gg_troops.10 days = 1 }
	}
}

# EN: 23/9 -- the engine refuses to mobilize an army with no commander (its check is "all commanders busy",
#     true with none): the Ejercito Grande's foreign contingents were raised without one (headless run
#     172225, "Scope formation cannot be mobilized"). Each gets its documented chief, created once per
#     country and reused (the full and the limited terms both call this). Run in the owner's scope.
# ES: el motor no moviliza un ejercito sin comandante (su chequeo es "todos ocupados", que se cumple con
#     ninguno): los contingentes se levantaban sin jefe. Cada uno recibe su jefe documentado, creado una
#     vez por pais y reutilizado. Se corre en el scope del duenio.
# SOURCE / FUENTE: es.wikipedia.org/wiki/Ejército_Grande (Brazilian 1st Division under brigadier Manuel
#     Marques de Sousa; Oriental contingent under colonel Cesar Diaz, named by decree of 5/11/1851);
#     perio.unlp.edu.ar, Tomo 01 de la Batalla de Caseros (Ruiz Moreno 2006).
gg_commander_marques_de_sousa = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_commander_marques_de_sousa|" # GG_TRAZA
	if = {
		limit = { NOT = { any_scope_character = { has_variable = gg_marques_de_sousa } } }
		create_character = {
			first_name = "Manuel"
			# EN/ES: 24/9 (AUD-ERR-LOAD) -- a name the engine cannot find as a loc key is replaced by a random one
			#     ("Missing localization key for character last name"): the key lives in guerra_grande_characters_l_*
			#     / un apellido sin clave de localizacion lo reemplaza el motor por uno al azar
			last_name = gg_name_marques_de_sousa
			historical = yes
			birth_date = 1804.6.13
			is_general = yes
			interest_group = ig_armed_forces
			ideology = ideology_moderate
			culture = cu:brazilian
			religion = rel:catholic
			traits = { experienced_offensive_planner }
			save_scope_as = gg_new_commander
		}
		scope:gg_new_commander = { set_variable = gg_marques_de_sousa }
	}
	random_scope_character = {
		limit = { has_variable = gg_marques_de_sousa }
		transfer_to_formation = $ARMY$
	}
}
