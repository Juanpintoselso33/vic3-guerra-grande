# EN: The reincorporation and the second Cisplatine war (events/gg_cisplatina.txt).
# ES: La reincorporacion y la segunda guerra cisplatina (events/gg_cisplatina.txt).

# EN: Argentina has answered (gg_cisplatina.1): Rosas's offer now goes to the Cerrito, through the calendar
#     relay so the usual delivery and retry rules hold (gg_cal_fire, DONE gg_cal_done_11).
# ES: Argentina ya contesto: la oferta de Rosas va al Cerrito por el relevo del calendario.
gg_cis_offer_go = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_cis_offer_go|" # GG_TRAZA
	set_global_variable = gg_cis_offer_decided
	hidden_effect = {
		gg_cal_fire = {
			DONE = gg_cal_done_11
			EVENT = gg_loan.4
			SIDE = guerra_grande_cerrito_government
			RETRY = gg_cal.11
		}
	}
}

# EN: Brazil goes to war (gg_cisplatina.2). The engine war goal is a humiliation: the front and the
#     battles are the engine's, the settlement is the script's (gg_cisplatina.3).
# ES: El Brasil va a la guerra. El objetivo del motor es una humillacion: el frente y las batallas son
#     del motor, el arreglo lo aplica el script.
gg_cis_open_war = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_cis_open_war|" # GG_TRAZA
	if = {
		limit = { exists = c:ARG NOT = { has_global_variable = gg_cis_war } }
		set_global_variable = gg_cis_war
		create_diplomatic_play = {
			name = gg_cis_play_name
			target_country = c:ARG
			type = dp_gg_cisplatina
			war = yes
		}
		# EN/ES: 23/9 (Juan) -- Buenos Aires hears of it and answers (gg_cisplatina.6) / Buenos Aires se entera
		c:ARG ?= { trigger_event = { id = gg_cisplatina.6 days = 0 popup = yes } }
	}
}

# EN: Free the Republic: the Oriental states (marked gg_provincia_oriental by the reincorporation,
#     gg_peace.2.g) leave the Confederation as the Republica Oriental del Uruguay. Montevideo first, so the
#     new country is born with its capital.
# ES: Liberar la Republica: los estados orientales (marcados gg_provincia_oriental) salen de la
#     Confederacion como la Republica Oriental del Uruguay. Montevideo primero, para que nazca con su capital.
gg_cis_release = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_cis_release|" # GG_TRAZA
	c:ARG ?= {
		random_scope_state = {
			limit = { state_region = s:STATE_MONTEVIDEO has_modifier = gg_provincia_oriental }
			save_scope_as = gg_cis_first
		}
		if = {
			limit = { NOT = { exists = scope:gg_cis_first } }
			random_scope_state = {
				limit = { has_modifier = gg_provincia_oriental }
				save_scope_as = gg_cis_first
			}
		}
	}
	if = {
		limit = { exists = scope:gg_cis_first NOT = { exists = c:URU } }
		create_country = {
			tag = URU
			origin = c:ARG
			state = scope:gg_cis_first
		}
		gg_argentina_new_country_people = { TAG = URU }
		c:ARG ?= {
			every_scope_state = {
				limit = { has_modifier = gg_provincia_oriental }
				set_state_owner = c:URU
			}
		}
		c:URU ?= {
			every_scope_state = { remove_modifier = gg_provincia_oriental }
			if = { limit = { has_state_in_state_region = STATE_MONTEVIDEO } set_capital = STATE_MONTEVIDEO }
		}
	}
	clear_saved_scope = gg_cis_first
}

# EN: Take the Cisplatina back into the Empire: the Oriental states pass to Brazil.
# ES: Recuperar la Cisplatina para el Imperio: los estados orientales pasan al Brasil.

	c:ARG ?= {

			limit = { has_modifier = gg_provincia_oriental }

}

# EN: 23/9/2026 (Juan: "all the characters of Uruguay, it is an integration, let it work as one") -- when the
#     Republic is absorbed, EVERY character comes with it, politicians and interest group leaders included, to TARGET,
#     before the annexation. (The first version filtered by role with is_admiral, which is not a trigger: error.log.)
# ES: cuando se absorbe la Republica, TODOS sus personajes vienen con ella, politicos y jefes de grupos de interes
#     incluidos, antes de la anexion. (La primera version filtraba por rol con is_admiral, que no es trigger.)
gg_bring_characters_to = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_bring_characters_to|" # GG_TRAZA
	every_scope_character = {
		transfer_character = $TARGET$
	}
}

# EN: 23/9/2026 (Juan) -- the Banda Oriental comes back to the Confederation. Scope: the Oriental country
#     (the Cerrito's government, after it absorbed the Defensa). Called by Oribe's menu when Buenos Aires is
#     not a player, or by Buenos Aires's own choice (gg_cisplatina.5).
# ES: la Banda Oriental vuelve a la Confederacion. Scope: el pais oriental. La llama el menu de Oribe si
#     Buenos Aires no es jugador, o la eleccion de Buenos Aires (gg_cisplatina.5).
gg_oriental_incorporate = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_oriental_incorporate|" # GG_TRAZA
	# EN: 23/9/2026 (AUD-SEAT-ANNEX) -- both governments in human hands: nobody's country is annexed; the Banda Oriental
	#     stays a province of the Pact, a subject (the vanilla rule for Cuba when both are players).
	# ES: los dos gobiernos en manos de jugadores: no se anexa el pais de nadie; queda provincia del Pacto, sujeta.
	if = {
		limit = { is_player = yes c:ARG ?= { is_player = yes } }
		gg_oriental_make_province = yes
	}
	else = {
		set_global_variable = gg_reincorporated
		set_global_variable = gg_oriental_in_pact
		# EN/ES: 23/9 (AUD-MIGR-NEWGAME) -- the siege is over: its modifiers go now, before Brazil hears of it (the
		#     postwar that cleared them does not run after a reincorporation) / el sitio termino: sus modificadores se van ya
		gg_wartime_modifiers_clear_once = yes
		c:BRZ ?= { trigger_event = { id = gg_cisplatina.2 days = 30 popup = yes } }
		every_scope_state = {
			add_modifier = {
				name = gg_provincia_oriental
				years = 30
			}
		}
		save_scope_as = gg_oriental_country
		# EN/ES: 23/9 (Juan) -- the Oriental army comes too: its size read before the annexation (the engine
		#     drops the annexed country's formations) / el ejercito oriental viene tambien: su tamanio antes de anexar
		if = { limit = { army_size >= 20 } set_global_variable = { name = gg_oriental_battalions value = 20 } }
		else_if = { limit = { army_size >= 12 } set_global_variable = { name = gg_oriental_battalions value = 12 } }
		else_if = { limit = { army_size >= 6 } set_global_variable = { name = gg_oriental_battalions value = 6 } }
		else = { set_global_variable = { name = gg_oriental_battalions value = 3 } }
		c:ARG ?= {
			c:BRZ ?= { change_relations = { country = prev value = -100 } }
			c:GBR ?= { change_relations = { country = prev value = -50 } }
			c:FRA ?= { change_relations = { country = prev value = -50 } }
			scope:gg_oriental_country = { gg_bring_characters_to = { TARGET = c:ARG } }
			annex = scope:gg_oriental_country # EN/ES: AUD-PROV-INCORP -- incorporation only through the filter below / la incorporacion solo por el filtro de abajo
			gg_incorporate_province_states = yes # EN/ES: AUD-PROV-INCORP -- the provinces come home incorporated / las provincias vuelven incorporadas
			gg_oriental_army_raise = yes
		}
	}
}

# EN: 23/9/2026 (Juan) -- the Army of the Oriental Province, in Argentina's scope, after the annexation:
#     the same size band as the Cerrito's army, under Oribe. / el Ejercito de la Provincia Oriental, bajo Oribe.
gg_oriental_army_raise = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_oriental_army_raise|" # GG_TRAZA
	random_scope_state = {
		limit = { state_region = s:STATE_MONTEVIDEO }
		save_scope_as = gg_oriental_hq
	}
	if = {
		limit = { exists = scope:gg_oriental_hq }
		if = {
			limit = { global_var:gg_oriental_battalions >= 20 }
			create_military_formation = { type = army hq_region = scope:gg_oriental_hq.region name = gg_oriental_army save_scope_as = gg_oriental_army_scope
				combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = s:STATE_MONTEVIDEO count = 12 }
				combat_unit = { type = unit_type:combat_unit_type_hussars state_region = s:STATE_MONTEVIDEO count = 8 } }
		}
		else_if = {
			limit = { global_var:gg_oriental_battalions >= 12 }
			create_military_formation = { type = army hq_region = scope:gg_oriental_hq.region name = gg_oriental_army save_scope_as = gg_oriental_army_scope
				combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = s:STATE_MONTEVIDEO count = 7 }
				combat_unit = { type = unit_type:combat_unit_type_hussars state_region = s:STATE_MONTEVIDEO count = 5 } }
		}
		else_if = {
			limit = { global_var:gg_oriental_battalions >= 6 }
			create_military_formation = { type = army hq_region = scope:gg_oriental_hq.region name = gg_oriental_army save_scope_as = gg_oriental_army_scope
				combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = s:STATE_MONTEVIDEO count = 4 }
				combat_unit = { type = unit_type:combat_unit_type_hussars state_region = s:STATE_MONTEVIDEO count = 2 } }
		}
		else = {
			create_military_formation = { type = army hq_region = scope:gg_oriental_hq.region name = gg_oriental_army save_scope_as = gg_oriental_army_scope
				combat_unit = { type = unit_type:combat_unit_type_line_infantry state_region = s:STATE_MONTEVIDEO count = 2 }
				combat_unit = { type = unit_type:combat_unit_type_hussars state_region = s:STATE_MONTEVIDEO count = 1 } }
		}
		random_scope_character = {
			limit = { has_template = URU_Manuel_Oribe exists = scope:gg_oriental_army_scope }
			if = { limit = { NOT = { has_role = character_role_general } } add_character_role = general }
			transfer_to_formation = scope:gg_oriental_army_scope
		}
	}
	clear_saved_scope = gg_oriental_hq
}

gg_oriental_make_province = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_oriental_make_province|" # GG_TRAZA
	set_global_variable = gg_oriental_province
	set_global_variable = gg_oriental_in_pact
	c:BRZ ?= { trigger_event = { id = gg_cisplatina.2 days = 30 popup = yes } }
	save_scope_as = gg_oriental_country
	# EN/ES: a subject pact only with the peace signed (I17) / pacto de sujecion solo con la paz firmada
	# EN: 24/9/2026 (hotfix 0.1.1, AUD-ORIENTAL-FREED) -- was type = vassal, which the engine drops at once: a vassal
	#     needs an unrecognized overlord, and every recognized-overlord type needs Buenos Aires to outrank the
	#     Republic (both were minor_power in Juan's game). Its own type has no rank rule (zz_gg_subject_types.txt).
	#     Guarded on not being a subject already: gg_peace.10 may have rebuilt it after the tag change.
	# ES: era type = vassal, que el motor tira enseguida (el vasallaje pide un soberano no reconocido, y los otros
	#     tipos piden que Buenos Aires tenga mas rango que la Republica). Tipo propio, sin la regla del rango.
	if = {
		limit = {
			has_global_variable = gg_peace_settled
			is_subject = no
		}
		c:ARG ?= {
			create_diplomatic_pact = {
				country = scope:gg_oriental_country
				type = gg_pact_province
			}
			gg_subject_watch = { SUBJECT = scope:gg_oriental_country } # EN/ES: 24/9 (AUD-VASSAL-RECOGNIZED) -- thirty-day check / chequeo a los treinta dias
			change_relations = { country = scope:gg_oriental_country value = 80 }
		}
	}
	c:BRZ ?= { change_relations = { country = scope:gg_oriental_country value = -80 } }
}

# EN: 24/9/2026 (hotfix 0.1.2, AUD-ORIENTAL-PROVINCE; Juan, playing Argentina: «when I annexed the whole Pact, the
#     Provincia Oriental was not annexed»). The Oriental Province joins the one country with the other provinces.
#     Scope: Argentina. Called from gg_argentina_provinces_incorporate, so every road that incorporates the Pact takes
#     it: gg_argentina.80.a, the constitution (gg_constitution_one_country) and the seat hand-over (gg_constitution.6).
#     Same player rule as the provinces (AUD-SEAT-ANNEX, vanilla's Cuba): not a player's -> absorbed; a player's and
#     Argentina not -> he is offered Argentina's seat and this runs again the day after; both players -> it stays a
#     province of the Pact.
# ES: la Provincia Oriental entra al pais unico con las demas provincias. Scope: la Argentina. La llama
#     gg_argentina_provinces_incorporate, asi que la toma todo camino que incorpora el Pacto. Misma regla de jugadores
#     que las provincias.
gg_oriental_province_join = {
	if = {
		limit = {
			# EN/ES: Codex review -- only Argentina takes it: gg_constitution_one_country is also reached from the State
			#     of Buenos Aires (gg_ba.4) / solo la Argentina la toma: tambien se llega desde el Estado de Buenos Aires
			c:ARG ?= this
			c:URU ?= { gg_is_oriental_pact_province = yes }
		}
		if = {
			limit = { c:URU = { is_player = no } }
			gg_oriental_province_absorb = yes
		}
		else_if = {
			limit = { is_player = no }
			c:URU = { trigger_event = { id = gg_constitution.5 days = 0 popup = yes } }
		}
	}
}

# EN: The absorption itself, as the full reincorporation does it (gg_oriental_incorporate; Juan 23/9: «all the
#     characters of Uruguay, it is an integration, let it work as one»). Scope: Argentina.
#       - its army's size is read before the annexation (the engine drops the annexed country's formations) and the
#         Army of the Oriental Province is raised after it, under Oribe (gg_oriental_army_raise);
#       - its states keep the mark of the Oriental Province (gg_provincia_oriental), which the Cisplatine question
#         reads to free them or hand them to Brazil (gg_cis_release, gg_cis_annex);
#       - every character, Oribe included, passes to Buenos Aires before the annexation (gg_bring_characters_to);
#       - the marks move from «province of the Pact» (gg_oriental_province) to «reincorporated» (gg_reincorporated):
#         the Pact's row goes on showing Oribe as its governor, and Brazil's war (dp_gg_cisplatina) stays possible.
#     A bare annex and then gg_incorporate_province_states (AUD-PROV-INCORP), as gg_oriental_incorporate: its states are
#     incorporated Argentine ground at once, and nothing outside the Banda Oriental is.
# ES: la absorcion, como la reincorporacion completa: el tamanio del ejercito antes de anexar y el Ejercito de la
#     Provincia Oriental despues, bajo Oribe; los estados con la marca de la Provincia Oriental (la cuestion
#     cisplatina los lee); todos los personajes a Buenos Aires antes de anexar; las marcas pasan de «provincia del
#     Pacto» a «reincorporada», asi la fila sigue mostrando a Oribe y la guerra del Brasil sigue siendo posible.
gg_oriental_province_absorb = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_oriental_province_absorb|" # GG_TRAZA
	c:URU = {
		save_scope_as = gg_oriental_country
		if = { limit = { army_size >= 20 } set_global_variable = { name = gg_oriental_battalions value = 20 } }
		else_if = { limit = { army_size >= 12 } set_global_variable = { name = gg_oriental_battalions value = 12 } }
		else_if = { limit = { army_size >= 6 } set_global_variable = { name = gg_oriental_battalions value = 6 } }
		else = { set_global_variable = { name = gg_oriental_battalions value = 3 } }
		every_scope_state = {
			limit = { NOT = { has_modifier = gg_provincia_oriental } }
			add_modifier = { name = gg_provincia_oriental years = 30 }
		}
		gg_bring_characters_to = { TARGET = c:ARG }
	}
	remove_global_variable = gg_oriental_province
	set_global_variable = gg_reincorporated
	set_global_variable = gg_oriental_in_pact
	annex = scope:gg_oriental_country # EN/ES: AUD-PROV-INCORP -- incorporation only through the filter below / la incorporacion solo por el filtro de abajo
	gg_incorporate_province_states = yes # EN/ES: AUD-PROV-INCORP -- the provinces come home incorporated / las provincias vuelven incorporadas
	gg_oriental_army_raise = yes
	clear_saved_scope = gg_oriental_country
}

# EN: 24/9/2026 (Juan: «if the Provincia Oriental exists it must join the war on Rosas's side; Oribe was Rosas's ally,
#     and Urquiza had to beat him first in October 1851»). Urquiza's war (dp_gg_pacto_federal, Buenos Aires against
#     Entre Rios): the Oriental Province enters it on Buenos Aires's side, the mirror of Corrientes joining Entre Rios
#     (gg_pact_join_urquiza_side). Scope: any. Called the day after the war starts (gg_argentina.89) and by the Pact's
#     weekly pulse (gg_pact_governors_update), for a province made after the war began. A player's Oriental Province is
#     not forced in. The subject type does not join its overlord's wars on its own (join_overlord_wars = no); this is
#     the one war it is brought into.
# SOURCE / FUENTE: design repo SOURCES_peace.md §1 (the peace of 8 October 1851, Oribe before Urquiza) and
#     SOURCES_loans.md (Capitulacion del Pantanoso, 8/10/1851).
# ES: la guerra de Urquiza: la Provincia Oriental entra del lado de Buenos Aires, el espejo de Corrientes con Entre
#     Rios. Se llama al dia siguiente de empezar la guerra y en el pulso semanal del Pacto. La de un jugador no se fuerza.
gg_oriental_province_joins_rosas = {
	if = {
		limit = {
			exists = c:ENR
			c:ARG ?= { has_war_with = c:ENR }
			c:URU ?= {
				gg_is_oriental_pact_province = yes
				is_player = no
				NOT = { has_war_with = c:ENR }
			}
		}
		c:URU = {
			if = { limit = { has_truce_with = c:ENR } end_truce = c:ENR }
		}
		c:ENR = {
			if = { limit = { has_truce_with = c:URU } end_truce = c:URU }
			every_scope_war = {
				limit = {
					# EN/ES: the Pact war itself: Buenos Aires attacks, Entre Rios defends / la guerra del Pacto misma
					attacker_warleader ?= { c:ARG ?= this }
					defender_warleader ?= { c:ENR ?= this }
					NOT = { is_war_participant = c:URU }
				}
				join_war = { target = c:URU side = c:ARG }
				debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_oriental_province_joins_rosas| JOINED=war " # GG_TRAZA
			}
		}
	}
}

# EN: 23/9/2026 (Juan: "a war at once, me without the Pact, against a whole Brazil") -- Brazil protests and waits

# EN: 24/9/2026 (Juan: Brazil protested and Buenos Aires never heard of it) -- Brazil's answer to the reincorporation
#     reaches Buenos Aires (gg_cisplatina.9). Scope: Brazil. KIND 1 = the protest (gg_cisplatina.2.a), 2 = the rupture
#     (.b/.c/.d, or a save still waiting, gg_cisplatina.10). The mark is set before the event is scheduled (its trigger
#     is checked then), and only once.
# ES: la respuesta del Brasil llega a Buenos Aires. KIND 1 protesta, 2 ruptura. La marca va antes de agendar, una vez.
gg_cis_note_send = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_cis_note_send| KIND= $KIND$ " # GG_TRAZA
	if = {
		limit = { NOT = { has_global_variable = gg_cis_note_kind } }
		set_global_variable = { name = gg_cis_note_kind value = $KIND$ }
		c:ARG ?= { trigger_event = { id = gg_cisplatina.9 days = 0 popup = yes } }
	}
}
