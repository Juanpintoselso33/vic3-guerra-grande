# EN: Shared native action with execution-time revalidation.
# ES: Accion nativa compartida con validacion al ejecutarla.
# EN: C10 (22/9/2026) -- calling the squadrons no longer brings them by itself, and it no longer adds
#     30 to a sea formula (that formula is gone). It is a plea to London and Paris: their patience
#     with a shut river drops from eighteen months to twelve (gg_plata_patience_threshold), and the
#     ultimatum to Rosas comes that much sooner. The patrons still rise, because the patrons are
#     money and legions, not the sea. Calling before 1845 still costs sovereignty (Varela's offer).
# ES: C10 -- llamar a las escuadras ya no las trae por si solo ni suma 30 a una formula del mar (esa
#     formula ya no existe). Es un pedido a Londres y Paris: su paciencia con un rio cerrado baja de
#     dieciocho meses a doce, y el ultimatum a Rosas llega antes. Los padrinos igual suben, porque
#     los padrinos son plata y legiones, no el mar. Llamarlas antes de 1845 sigue costando soberania.
gg_ui_call_squadrons_effect = {
	if = {
		limit = { gg_ui_can_call_squadrons = yes has_global_variable = guerra_grande_split_done }
		set_global_variable = gg_intervention_called
		custom_tooltip = gg_plata_squadrons_called_tt
		gg_patron_change = { PATRON = francia DELTA = 30 }
		gg_patron_change = { PATRON = britania DELTA = 30 }
		if = {
			limit = { year < 1845 }
			add_modifier = {
				name = gg_intervention_early_price
				years = 10
				is_decaying = yes
			}
		}
		trigger_event = { id = gg_powers.1 days = 1826 }
	}
}
