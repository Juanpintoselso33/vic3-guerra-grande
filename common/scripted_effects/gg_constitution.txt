	# EN/ES: the unitario provinces that are Buenos Aires's subjects bring their ground first; a player's province is
	#     not annexed (AUD-SEAT-ANNEX): it stays a subject until the constitution offers the seat
	#     / primero su tierra; la provincia de un jugador no se anexa: sigue sujeta hasta la constitucion
	# EN/ES: 25/9 (AUD-CHAR-ANNEX) -- its men come along; a bare annex deletes them (Urquiza, Mitre, Sarmiento lost with Entre Rios) / sus hombres vienen con ella; la anexion pelada los borra

	c:ARG ?= { gg_incorporate_province_states = yes } # EN/ES: AUD-PROV-INCORP -- the provinces come home incorporated / las provincias vuelven incorporadas
	if = {
		limit = { exists = s:STATE_TUCUMAN.region_state:ARG NOT = { exists = c:CLN } }

			tag = CLN
			origin = c:ARG
			state = s:STATE_TUCUMAN.region_state:ARG
		}
		if = { limit = { exists = s:STATE_CORRIENTES.region_state:ARG } s:STATE_CORRIENTES.region_state:ARG = { set_state_owner = c:CLN } }
		if = { limit = { exists = s:STATE_SALTA.region_state:ARG } s:STATE_SALTA.region_state:ARG = { set_state_owner = c:CLN } }
		if = { limit = { exists = s:STATE_JUJUY.region_state:ARG } s:STATE_JUJUY.region_state:ARG = { set_state_owner = c:CLN } }
		if = { limit = { exists = s:STATE_CATAMARCA.region_state:ARG } s:STATE_CATAMARCA.region_state:ARG = { set_state_owner = c:CLN } }

	else = { c:ARG ?= { trigger_event = { id = gg_constitution.3 days = 1 } } }

# EN/ES: scope: the Rosas seat (Argentina). Asks now if nothing holds it off / pregunta ahora si nada lo frena
gg_constitution_try = {
	if = {
		limit = {
			has_variable = gg_arg_rosas_seat
			has_global_variable = gg_constitution_pending
			NOT = { has_global_variable = gg_constitution_chosen }
			NOT = { has_global_variable = gg_constitution_wait }
			NOT = { has_global_variable = gg_constitution_asking }
			NOT = { has_global_variable = gg_rosas_succession_crisis } # EN/ES: his death already reopened the question / la muerte ya reabrio la cuestion
		}
		# EN: 24/9/2026 (Codex) -- Rosas died while the question waited, after his victory or with two governments: nobody
		#     is left to give it or to refuse it. It becomes the question of his death: the country without a constitution
		#     (answer 2) and its modifier, and gg_rosas_succession_monthly opens gg_constitution.2 on its next pulse.
		# ES: Rosas murio mientras la pregunta esperaba: no queda quien la de ni quien la niegue. Pasa a ser la cuestion de
		#     su muerte: el pais sin constitucion, y el pulso de la sucesion abre gg_constitution.2.
		if = {
			limit = {
				OR = {
					has_global_variable = gg_pact_settlement_rosas
					has_global_variable = gg_pact_settlement_stalemate
				}
				gg_constitution_rosas_alive = no
			}
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_try| ROSAS_DEAD=yes" # GG_TRAZA
			remove_global_variable = gg_constitution_pending
			set_global_variable = { name = gg_constitution_chosen value = 2 }
			if = { limit = { NOT = { has_modifier = gg_pais_sin_constitucion } } add_modifier = { name = gg_pais_sin_constitucion } }
		}
		else_if = {
			limit = { gg_constitution_brazil_war = yes }
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_try| WAIT=brazil_war" # GG_TRAZA
		}
		else = {
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_try| ASK=yes" # GG_TRAZA
			set_global_variable = { name = gg_constitution_asking days = 20 }
			trigger_event = { id = gg_constitution.1 days = 0 popup = yes }
		}
	}
}

# EN: 24/9/2026 (Juan: «a confederation with every province kept as a subject», and the same day: «also the provinces
#     Argentina controls directly»; AUD-CONST-CONFED) -- THE CONFEDERATION (gg_constitution.1.g). No constitution: the
#     Pact of 4/1/1831 is the only charter, a treaty between governments that recognise one another's «libertad,
#     independencia, representacion y derechos» (art. 1). Every historical province is its own country again, a
#     Province of the Pact (subject_type_gg_pact_province) under Buenos Aires, which keeps the foreign relations the
#     provinces delegated to its governor; Buenos Aires keeps its own province, La Pampa, Rio Negro and the Chaco
#     (gg_confed_province_state). Scope: Argentina (the Rosas seat). Idempotent: gg_constitution.11 runs the ground
#     half again the next day, for a pact the engine only reads after this effect.
#     THE MACHINERY IS THE MOD'S OWN, NOT A NEW ONE: the province countries that stand are kept or taken in
#     (gg_constitution_confed_keep: gg_pact_protectorate_to_province for a protectorate of the war goal); the ground
#     Argentina holds goes through gg_province_split_off with Argentina as the origin -- to the province if it stands
#     as a subject, or a new country under the governor Buenos Aires installs (gg_province_install_governor,
#     gg_pact_gov_federal_*), with the clones killed and Argentina's army law (gg_argentina_new_country_people,
#     gg_argentina_inherit_army_model).

#     already a subject only has its protectorate turned into the Province of the Pact, as the settlement already does.
#     THE ORIENTAL PROVINCE: kept as it stands if it is one (it already is a Province of the Pact); one absorbed at
#     gg_argentina.80.a is not rebuilt -- the Banda Oriental never signed the Pact of 1831, and its return is the
#     Cisplatine question's (gg_cis_release).

#     reconocen «libertad, independencia, representacion y derechos» (art. 1). Cada provincia historica vuelve a ser un
#     pais, provincia del Pacto bajo Buenos Aires, que conserva las relaciones exteriores que las provincias delegaron en
#     su gobernador; Buenos Aires se queda con su provincia, La Pampa, Rio Negro y el Chaco. La maquinaria es la del mod:
#     las provincias-pais que existen se conservan o se reciben; el suelo que tiene la Argentina pasa por
#     gg_province_split_off con la Argentina como origen. La provincia de un jugador nunca se fuerza. La Provincia

#     independent one at peace with Buenos Aires and not a player's is taken in (Corrientes, Entre Rios and the
#     Coalition after the negotiated confederation). A subject of another power, or one at war with Buenos Aires,
#     stays out; the trace says which.

#     que no es de un jugador, se recibe. Una sujeta de otra potencia o en guerra con Buenos Aires queda afuera.
gg_constitution_confed_keep = {
	if = {
		limit = { c:$TAG$ ?= { is_subject_of = c:ARG is_subject_type = subject_type_gg_pact_province } }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_confed_keep| TAG= $TAG$ KEPT=yes " # GG_TRAZA

	#     / cualquier otro tipo de sujecion pasa a provincia del Pacto, como en gg_pact_protectorate_to_province

		limit = {
			c:$TAG$ ?= { is_subject_of = c:ARG }
			NOT = { has_war_with = c:$TAG$ }
		}
		c:$TAG$ = {
			make_independent = yes
			if = { limit = { has_truce_with = c:ARG } end_truce = c:ARG }
		}
		if = { limit = { has_truce_with = c:$TAG$ } end_truce = c:$TAG$ }
		create_diplomatic_pact = { country = c:$TAG$ type = gg_pact_province }
		gg_subject_watch = { SUBJECT = c:$TAG$ }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_confed_keep| TAG= $TAG$ CONVERTED=yes " # GG_TRAZA
	}
	else_if = {
		limit = {
			c:$TAG$ ?= { is_subject = no is_player = no }
			NOT = { has_war_with = c:$TAG$ }
		}
		c:$TAG$ = {
			if = { limit = { has_truce_with = c:ARG } end_truce = c:ARG }
		}
		if = { limit = { has_truce_with = c:$TAG$ } end_truce = c:$TAG$ }
		create_diplomatic_pact = { country = c:$TAG$ type = gg_pact_province }
		gg_subject_watch = { SUBJECT = c:$TAG$ }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_confed_keep| TAG= $TAG$ JOINED=yes " # GG_TRAZA
	}
	else_if = {
		limit = { c:$TAG$ ?= { is_subject = no is_player = yes } }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_confed_keep| TAG= $TAG$ PLAYER_KEPT=yes " # GG_TRAZA
	}
	else_if = {
		limit = { exists = c:$TAG$ }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_confed_keep| TAG= $TAG$ KEPT_OUT=yes " # GG_TRAZA
	}
}

# EN: The ground half. Scope: Argentina. Each province's state Argentina holds goes to that province (or makes it, as a
#     Province of the Pact); the Misiones lands follow Corrientes (as gg_argentina_revive_corrientes gives them) and
#     Tarija follows Salta (as gg_province_split_beaten does). Entre Rios is made only after its pronouncement date
#     (1/5/1851), so its own birth never finds its state taken.
# ES: la mitad del suelo. Cada estado provincial de la Argentina pasa a su provincia (o la crea); Misiones sigue a
#     Corrientes y Tarija a Salta, como en el resto del mod. Entre Rios se crea solo despues del 1/5/1851.
gg_constitution_confed_ground = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_confed_ground|" # GG_TRAZA
	# EN/ES: Codex review -- the seat of the government that keeps the foreign relations is Buenos Aires: a capital moved
	#     to a province would keep that province out of the split (gg_province_split_off never splits the origin's capital)
	#     / la sede es Buenos Aires: una capital mudada a una provincia la dejaria afuera de la particion
	if = {
		limit = {
			has_state_in_state_region = STATE_BUENOS_AIRES
			NOT = { capital ?= { state_region = s:STATE_BUENOS_AIRES } }
		}
		set_capital = STATE_BUENOS_AIRES
	}
	# EN/ES: Codex review -- a Coalition under Buenos Aires is a league, not a province: it splits into its provinces as
	#     after 1841 (gg_argentina_split_coalition; gg_provinces.1 remakes the remainder's pact the next day). A player's
	#     Coalition stays as it is / una Coalicion bajo Buenos Aires se parte en sus provincias; la de un jugador no
	c:CLN ?= {
		if = {
			limit = {
				is_subject_of = c:ARG
				is_player = no
			}
			gg_argentina_split_coalition = yes
		}
	}
	save_scope_as = gg_split_origin
	gg_province_split_off = { STATE = STATE_CORRIENTES TAG = CRR KEY = corrientes }
	if = {
		limit = { game_date >= 1851.5.1 }
		gg_province_split_off = { STATE = STATE_ENTRE_RIOS TAG = ENR KEY = entre_rios }
	}
	gg_province_split_off = { STATE = STATE_SANTA_FE TAG = SFE KEY = santa_fe }
	gg_province_split_off = { STATE = STATE_CORDOBA TAG = CBA KEY = cordoba }
	gg_province_split_off = { STATE = STATE_SANTIAGO_DEL_ESTERO TAG = SDE KEY = santiago_del_estero }
	gg_province_split_off = { STATE = STATE_TUCUMAN TAG = TUC KEY = tucuman }
	gg_province_split_off = { STATE = STATE_SALTA TAG = SLT KEY = salta }
	gg_province_split_off = { STATE = STATE_JUJUY TAG = JUJ KEY = jujuy }
	gg_province_split_off = { STATE = STATE_CATAMARCA TAG = CTM KEY = catamarca }
	gg_province_split_off = { STATE = STATE_LA_RIOJA TAG = LRJ KEY = la_rioja }
	gg_province_split_off = { STATE = STATE_MENDOZA TAG = MZA KEY = mendoza }
	gg_province_split_off = { STATE = STATE_SAN_JUAN TAG = SJN KEY = san_juan }
	gg_province_split_off = { STATE = STATE_SAN_LUIS TAG = SNL KEY = san_luis }
	every_scope_state = {
		limit = {
			OR = {
				state_region = s:STATE_MISIONES
				state_region = s:STATE_PARANA
				state_region = s:STATE_SANTA_CATARINA
			}
			c:CRR ?= { is_subject_of = c:ARG }
		}
		set_state_owner = c:CRR
	}
	every_scope_state = {
		limit = {
			state_region = s:STATE_TARIJA
			c:SLT ?= { is_subject_of = c:ARG }
		}
		set_state_owner = c:SLT
	}
	clear_saved_scope = gg_split_origin
	# EN/ES: the trace the night sweep reads / la traza que lee el barrido nocturno
	if = {
		limit = { NOT = { any_scope_state = { gg_confed_province_state = yes } } }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_confed_ground| ARG_ONLY_BUENOS_AIRES=yes " # GG_TRAZA
	}
	else = {
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_confed_ground| ARG_ONLY_BUENOS_AIRES=no " # GG_TRAZA
	}
	if = {
		limit = {
			any_subject_or_below = {
				is_subject_of = c:ARG
				OR = {
					gg_is_province_country = yes
					c:CLN ?= this
				}
				NOT = { is_subject_type = subject_type_gg_pact_province }
			}
		}
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_confed_ground| NOT_PACT_PROVINCE=yes " # GG_TRAZA
	}
}
