# EN: 23/9/2026 (Juan: "a first, simple version; lean it a little towards the federales") -- the Restorer's death
#     with no constitution (gg_constitution.2) splits the Confederation along an imagined historical line. UNITARIOS:
#     Corrientes (it rose against Rosas three times) and the heart of the 1840 Coalicion del Norte (Tucuman, Salta,
#     Jujuy, Catamarca) -- the league reborn under its own tag, CLN. FEDERALES: everyone else with Buenos Aires,
#     Entre Rios and the Banda Oriental included. An engine war (dp_gg_pacto_federal); the winner writes the
#     constitution and the autocracy ends either way (gg_constitution.3). A constituent convention where provinces
#     join a side by the player's choices is for a later release.
#     A CONSTITUTION MAKES A FEDERATION OR A UNITARY STATE, NOT A CONFEDERATION (Juan, 23/9: "a confederation is not a
#     federation"): either one incorporates the provinces into one country; only the Pact without a constitution
#     keeps them as protectorates.
# ES: la muerte del Restaurador sin constitucion parte la Confederacion por una linea historica imaginaria. El que
#     gana escribe la constitucion y la autocracia termina igual. Una constitucion hace una federacion o un estado
#     unitario, no una confederacion: las dos incorporan las provincias; solo el Pacto sin constitucion las deja
#     como protectorados.
gg_unitario_rising = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_unitario_rising|" # GG_TRAZA
	# EN/ES: the unitario provinces that are Buenos Aires's subjects bring their ground first; a player's province is
	#     not annexed (AUD-SEAT-ANNEX): it stays a subject until the constitution offers the seat
	#     / primero su tierra; la provincia de un jugador no se anexa: sigue sujeta hasta la constitucion
	if = { limit = { c:CRR ?= { is_subject_of = c:ARG is_player = no } } c:ARG = { annex = c:CRR } }
	if = { limit = { c:TUC ?= { is_subject_of = c:ARG is_player = no } } c:ARG = { annex = c:TUC } }
	if = { limit = { c:SLT ?= { is_subject_of = c:ARG is_player = no } } c:ARG = { annex = c:SLT } }
	if = { limit = { c:JUJ ?= { is_subject_of = c:ARG is_player = no } } c:ARG = { annex = c:JUJ } }
	if = { limit = { c:CTM ?= { is_subject_of = c:ARG is_player = no } } c:ARG = { annex = c:CTM } }
	c:ARG ?= { gg_incorporate_province_states = yes } # EN/ES: AUD-PROV-INCORP -- the provinces come home incorporated / las provincias vuelven incorporadas
	if = {
		limit = { exists = s:STATE_TUCUMAN.region_state:ARG NOT = { exists = c:CLN } }
		create_country = {
			tag = CLN
			origin = c:ARG
			state = s:STATE_TUCUMAN.region_state:ARG
		}
		if = { limit = { exists = s:STATE_CORRIENTES.region_state:ARG } s:STATE_CORRIENTES.region_state:ARG = { set_state_owner = c:CLN } }
		if = { limit = { exists = s:STATE_SALTA.region_state:ARG } s:STATE_SALTA.region_state:ARG = { set_state_owner = c:CLN } }
		if = { limit = { exists = s:STATE_JUJUY.region_state:ARG } s:STATE_JUJUY.region_state:ARG = { set_state_owner = c:CLN } }
		if = { limit = { exists = s:STATE_CATAMARCA.region_state:ARG } s:STATE_CATAMARCA.region_state:ARG = { set_state_owner = c:CLN } }
		gg_argentina_new_country_people = { TAG = CLN }
		# EN/ES: 23/9 (AUD-MIGR-NEWGAME) -- the rebel leaves the treaties it inherited, as the Coalition of 1840 does;
		#     until today only the old-save migration did it, a month late / el rebelde sale de los tratados heredados
		c:CLN ?= { gg_argentina_rebel_diplomacy_reset = yes }
		c:CLN ?= { set_capital = STATE_TUCUMAN }
		set_global_variable = gg_unitario_war
		# EN/ES: AUD-CONST-PEACE -- the winner is recorded when it ends / el ganador se anota al terminar
		set_global_variable = { name = gg_unitario_federales_won value = 0 }
		set_global_variable = { name = gg_unitario_unitarios_won value = 0 }
		if = { limit = { has_global_variable = gg_unitario_war_ended } remove_global_variable = gg_unitario_war_ended }
		c:ARG ?= { gg_argentina_open_war = { TAG = CLN } }
	}
	# EN/ES: nobody left to rise: the federales write it in peace / nadie se alza: los federales la escriben en paz
	else = { c:ARG ?= { trigger_event = { id = gg_constitution.3 days = 1 } } }
}

# EN: 24/9/2026 (Juan, playing Rosas on 0.1.1; AUD-CONST-CHOICE) -- THE CONSTITUTION WAITS FOR THE WAR WITH BRAZIL.
#     In his game the Pact ended on 21/12/1851 and gg_constitution.1 came on 21/1/1852 in the middle of the second
#     Cisplatine war (gg_cis_war, dp_gg_cisplatina, opened 19/6/1851). Juan: the constitution comes after that war.
#     WHICH WAR (decided by Claude): a war or diplomatic play with Brazil -- the scripted Cisplatine one or any other
#     between the two -- not every foreign war: the Guerra Grande itself, an AI border war with Chile or Bolivia or a
#     Pact war with a province would hold it off for years, and the question Juan named is the war with the Empire.
#     HOW IT CANNOT BE LOST: the settlement no longer queues gg_constitution.1 thirty days ahead. It sets the global
#     gg_constitution_pending (globals survive a revolution that replaces Argentina) and a thirty-day wait; a hidden
#     relay (gg_constitution.10) asks after the month, and the monthly pulse of the Rosas seat
#     (gg_constitution_monthly, gg_rosas_succession.txt) asks again every month until gg_constitution.1 has been shown
#     (it clears the mark in its immediate). gg_constitution_asking (20 days) stops the relay and the pulse from both
#     asking in the same month; if the event was lost it expires and the next pulse asks.
# ES: LA CONSTITUCION ESPERA A LA GUERRA CON EL BRASIL. En su partida el Pacto termino el 21/12/1851 y gg_constitution.1
#     llego el 21/1/1852 en plena segunda guerra cisplatina. Juan: la constitucion viene despues de esa guerra.
#     QUE GUERRA (decidido por Claude): guerra o jugada diplomatica con el Brasil, la cisplatina del mod u otra, no toda
#     guerra exterior: la Guerra Grande, una guerra de frontera de la IA o una guerra del Pacto la frenarian por anios.
#     COMO NO SE PIERDE: el arreglo ya no agenda gg_constitution.1 a treinta dias. Prende la global
#     gg_constitution_pending (las globales sobreviven a una revolucion que reemplaza a la Argentina) y una espera de
#     treinta dias; un relevo oculto pregunta al mes, y el pulso mensual de la sede de Rosas vuelve a preguntar cada mes
#     hasta que el evento salio (lo borra en su immediate).
# -------------------------------------------------------------------------------------------------
gg_constitution_schedule = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_schedule|" # GG_TRAZA
	if = {
		limit = { NOT = { has_global_variable = gg_constitution_chosen } }
		set_global_variable = gg_constitution_pending
		set_global_variable = { name = gg_constitution_wait days = 29 } # EN/ES: expires the day before the relay / vence el dia antes del relevo
		trigger_event = { id = gg_constitution.10 days = 30 }
	}
}

# EN/ES: scope: the Rosas seat (Argentina). Asks now if nothing holds it off / pregunta ahora si nada lo frena
gg_constitution_try = {
	if = {
		limit = {
			has_variable = gg_arg_rosas_seat
			has_global_variable = gg_constitution_pending
			NOT = { has_global_variable = gg_constitution_chosen }
			NOT = { has_global_variable = gg_constitution_wait }
			NOT = { has_global_variable = gg_constitution_asking }
			NOT = { has_global_variable = gg_rosas_succession_crisis } # EN/ES: his death already reopened the question / la muerte ya reabrio la cuestion
		}
		# EN: 24/9/2026 (Codex) -- Rosas died while the question waited, after his victory or with two governments: nobody
		#     is left to give it or to refuse it. It becomes the question of his death: the country without a constitution
		#     (answer 2) and its modifier, and gg_rosas_succession_monthly opens gg_constitution.2 on its next pulse.
		# ES: Rosas murio mientras la pregunta esperaba: no queda quien la de ni quien la niegue. Pasa a ser la cuestion de
		#     su muerte: el pais sin constitucion, y el pulso de la sucesion abre gg_constitution.2.
		if = {
			limit = {
				OR = {
					has_global_variable = gg_pact_settlement_rosas
					has_global_variable = gg_pact_settlement_stalemate
				}
				gg_constitution_rosas_alive = no
			}
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_try| ROSAS_DEAD=yes" # GG_TRAZA
			remove_global_variable = gg_constitution_pending
			set_global_variable = { name = gg_constitution_chosen value = 2 }
			if = { limit = { NOT = { has_modifier = gg_pais_sin_constitucion } } add_modifier = { name = gg_pais_sin_constitucion } }
		}
		else_if = {
			limit = { gg_constitution_brazil_war = yes }
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_try| WAIT=brazil_war" # GG_TRAZA
		}
		else = {
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_try| ASK=yes" # GG_TRAZA
			set_global_variable = { name = gg_constitution_asking days = 20 }
			trigger_event = { id = gg_constitution.1 days = 0 popup = yes }
		}
	}
}

# EN: AUD-CONST-CHOICE -- what the event offers, one trace line per option shown (the options read the same triggers),
#     and whether Buenos Aires is at war with Brazil as it asks (read with the engine's own has_war_with, not with the

	if = { limit = { gg_constitution_offers_1853 = yes } debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_offer_log| OFFER=a" } # GG_TRAZA

	if = { limit = { gg_constitution_offers_pact_congress = yes } debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_offer_log| OFFER=c" } # GG_TRAZA
	if = { limit = { gg_constitution_offers_state_constitution = yes } debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_offer_log| OFFER=e" } # GG_TRAZA
	if = {
		limit = { exists = c:BRZ has_war_with = c:BRZ }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_offer_log| BRAZIL_WAR=yes" # GG_TRAZA
	}
	else = { debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution_offer_log| BRAZIL_WAR=no" } # GG_TRAZA
}
