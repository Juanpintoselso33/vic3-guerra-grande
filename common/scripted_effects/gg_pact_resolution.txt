# EN/ES: defined first: gg_pact_resolve below uses it / definido primero: lo usa gg_pact_resolve
# EN: 23/9/2026 (Juan, AUD-RULES-CLIENT, decisiones.md) -- when Rosas's government falls to Urquiza, the Republic made
#     Rosas's client by the Cerrito's peace (gg_peace.2.e, modifier gg_peace_rosista_client) is Urquiza's to decide:
#     free it or keep the protection (gg_argentina.86, player or AI). A puppet made any other way is not touched.
# ES: cuando el gobierno de Rosas cae ante Urquiza, la Republica hecha cliente de Rosas la decide Urquiza: liberarla o
#     conservar la proteccion. Otra sujecion no se toca.
gg_rosista_client_release = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_rosista_client_release|" # GG_TRAZA
	if = {
		limit = {
			exists = c:ARG
			c:URU ?= {
				has_modifier = gg_peace_rosista_client
				is_subject_of = c:ARG
			}
		}
		c:ARG = { trigger_event = { id = gg_argentina.86 days = 0 popup = yes } }
	}
}

# EN: 24/9/2026 (AUD-BA-STALEMATE-ENR, decided by Claude while Juan slept, decisiones.md) -- «two governments» means
#     the Pact war ends in a draw (Juan, 23/9: "la guerra termina en tablas"). When the engine's war was actually won
#     by Rosas (Entre Rios capitulated or backed down), its war goal (make_protectorate, dp_gg_pacto_federal) had
#     already made Entre Rios a protectorate, and the rule's reading left that in place: night sweep 24/9, trace
#     20260924-055131 -- capitulation 3/10/1851, stalemate the same day, gg_ba.101 found Entre Rios a subject on
#     11/9/1852 and there were never two governments. The rule's draw undoes the protectorate the war just made, the
#     mirror of the Rosas rule creating the one the war did not make (gg_pact_resolve below). Country scope (Argentina).
#     Only a protectorate this Pact war enforced (gg_pact_war_enforced_rosas): an Entre Rios made a subject any other way
#     keeps gg_ba.101's «no two governments» guard (buenos-aires-separada-plan.md, section 10).
# ES: «dos gobiernos» es la guerra del Pacto terminada en tablas. Si Rosas gano de verdad la guerra del motor, su
#     objetivo ya habia hecho protectorado a Entre Rios y la lectura de la regla lo dejaba: nunca habia dos gobiernos.
#     Las tablas deshacen el protectorado que acaba de imponer esta guerra (espejo de la regla de Rosas, que crea el que
#     la guerra no impuso). Solo el de esta guerra: otra sujecion conserva la guarda de gg_ba.101.
gg_pact_stalemate_frees_enr = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_stalemate_frees_enr|" # GG_TRAZA
	c:ENR ?= {
		if = {
			limit = {
				is_subject_of = c:ARG
				NOT = { has_war_with = c:ARG }
			}
			make_independent = yes
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_stalemate_frees_enr| FREED=yes " # GG_TRAZA
		}
	}
}

# EN: Country scope (Argentina). One idempotent settlement writer for the Federal Pact.
# ES: Scope pais (Argentina). Un escritor idempotente para todos los cierres del Pacto Federal.
gg_pact_resolve = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_resolve|" # GG_TRAZA
	# EN: 22/9/2026, Caseros option (c) -- while the siege of Buenos Aires lasts (at least six months
	#     from the choice) no ending is written: the request is recorded and gg_caseros.1 resolves it
	#     when the siege ends. Rosas keeps the capital and his government meanwhile. DESIGN: the
	#     six-month minimum is a counterfactual; Rosas did not take this road on 3/2/1852.
	# ES: mientras dura el sitio de Buenos Aires (seis meses como minimo desde la eleccion) no se
	#     escribe ningun final: el pedido queda anotado y gg_caseros.1 lo resuelve al terminar el sitio.
	#     DISENO: el minimo de seis meses es contrafactico.
	if = {
		limit = {
			has_global_variable = gg_caseros_siege
			NOT = { has_global_variable = gg_pact_settled }
		}
		set_global_variable = gg_pact_resolve_pending
	}
	else_if = {
		limit = { NOT = { has_global_variable = gg_pact_settled } }
		set_global_variable = gg_pact_settled
		# EN: 23/9/2026 -- the Pact's game rule reads the war its way (common/game_rules/zz_gg_game_rules.txt),
		#     unless Buenos Aires itself chose the negotiated confederation.
		# ES: la regla del Pacto lee la guerra a su modo, salvo que Buenos Aires eligiera la confederacion negociada.
		if = {
			limit = {
				gg_pact_forced_war = yes
				NOT = { has_global_variable = gg_confederacion_negociada }
			}
			set_global_variable = { name = gg_pact_war_urquiza_won value = 0 }
			set_global_variable = { name = gg_pact_war_rosas_won value = 0 }
			set_global_variable = { name = gg_pact_war_white_peace value = 0 }
			if = { limit = { gg_pact_historical = yes } set_global_variable = { name = gg_pact_war_urquiza_won value = 1 } }
			else_if = {
				limit = { gg_pact_rosas = yes }
				set_global_variable = { name = gg_pact_war_rosas_won value = 1 }
				# EN/ES: 23/9 (Codex review of 743b851) -- the rule's victory has no enforced war goal: Entre Rios becomes
				#     the protectorate the war would have made it, before the provinces are asked about
				#     / la victoria de la regla no impuso objetivo: Entre Rios queda el protectorado que la guerra le daba
				if = {
					limit = { c:ENR ?= { is_subject = no } NOT = { has_war_with = c:ENR } }
					create_diplomatic_pact = { country = c:ENR type = protectorate }
				}
			}
			else = {
				set_global_variable = { name = gg_pact_war_white_peace value = 1 }
				# EN/ES: 24/9 (AUD-BA-STALEMATE-ENR) -- the draw undoes the protectorate the war just enforced, now and again
				#     tomorrow in case the engine applies it after this reading (gg_caseros.7)
				#     / las tablas deshacen el protectorado que acaba de imponer la guerra, ahora y otra vez manana
				if = {
					limit = { has_global_variable = gg_pact_war_enforced_rosas }
					gg_pact_stalemate_frees_enr = yes
				}
			}
		}
		if = {
			limit = { has_global_variable = gg_confederacion_negociada }
			set_global_variable = gg_pact_settlement_constitutional
			trigger_event = { id = gg_argentina.82 days = 0 popup = yes }
		}
		else_if = {
			limit = { global_var:gg_pact_war_urquiza_won = 1 }
			set_global_variable = gg_pact_settlement_urquiza
			remove_global_variable = gg_president_railroad
			gg_ruler_set = { TEMPLATE = ARG_Justo_Jose_de_Urquiza }
			gg_rosista_client_release = yes # EN/ES: 23/9 (AUD-RULES-CLIENT) -- Rosas fell / cayo Rosas
			# EN: Caseros (a) or (c) -- Rosas falls and goes into exile (historically he embarked on
			#     HMS Centaur for England, February 1852). The Confederation remains.
			# ES: Caseros (a) o (c) -- cae Rosas y se exilia (se embarco en el Centaur rumbo a
			#     Inglaterra). La Confederacion queda.
			if = {
				limit = {
					OR = {
						global_var:gg_caseros_choice ?= 1
						global_var:gg_caseros_choice ?= 3
					}
				}
				random_scope_character = {
					limit = { has_template = ARG_Juan_Manuel_de_Rosas }
					exile_character_with_role_cleanup = yes
				}
			}
			trigger_event = { id = gg_argentina.83 days = 0 popup = yes }
			gg_caseros_notify_urquiza = { EVENT = gg_caseros.5 }
		}
		else_if = {
			limit = { global_var:gg_pact_war_rosas_won = 1 }
			set_global_variable = gg_pact_settlement_rosas
			# EN: 24/9/2026 (Juan, points 1 and 3; gg_pact_sides.txt) -- the allies of Entre Rios end as Entre Rios does,
			#     protectorates of Buenos Aires (the war goal, make_protectorate, only reached Entre Rios). Written here, in
			#     the Pact's one settlement writer (inventario I19). The weekly observer of the Pact then installs the
			#     federal governors and splits a beaten Coalition into its provinces, as after 1841.
			# ES: los aliados de Entre Rios terminan como Entre Rios, protectorados de Buenos Aires. El observador semanal
			#     pone despues a los gobernadores federales y parte a la Coalicion batida.
			if = {
				limit = {
					has_global_variable = gg_corrientes_with_urquiza
					c:CRR ?= { is_subject = no }
					NOT = { has_war_with = c:CRR }
				}
				create_diplomatic_pact = { country = c:CRR type = protectorate }
				gg_delegation_restored = { KEY = corrientes COUNT = 1 }
			}
			if = {
				limit = {
					has_global_variable = gg_coalition_with_urquiza
					c:CLN ?= { is_subject = no }
					NOT = { has_war_with = c:CLN }
				}
				create_diplomatic_pact = { country = c:CLN type = protectorate }
			}
			if = {
				limit = { has_global_variable = gg_delegation_gone_entre_rios }
				gg_delegation_restored = { KEY = entre_rios COUNT = 1 }
			}
			trigger_event = { id = gg_argentina.84 days = 0 popup = yes }
			trigger_event = { id = gg_caseros.7 days = 1 } # EN/ES: 24/9 (Codex, round 2) -- the protectorates again tomorrow / otra vez manana
		}
		else_if = {
			limit = { global_var:gg_pact_war_white_peace = 1 }
			set_global_variable = gg_pact_settlement_stalemate
			# EN: 24/9/2026 (Juan, point 4) -- two governments: Rosas keeps Buenos Aires, the rest goes to Urquiza's
			#     Confederation (gg_pact_sides.txt), now and again tomorrow in case the engine enforces a war goal after
			#     this reading (gg_caseros.7 also undoes the war's protectorate over Entre Rios, AUD-BA-STALEMATE-ENR).
			# ES: dos gobiernos: Rosas conserva Buenos Aires y el resto pasa a la Confederacion de Urquiza; ahora y otra
			#     vez manana, por si el motor impone un objetivo despues de esta lectura.
			gg_pact_stalemate_partition = yes
			trigger_event = { id = gg_caseros.7 days = 1 }
			trigger_event = { id = gg_argentina.85 days = 0 popup = yes }
			gg_caseros_notify_urquiza = { EVENT = gg_caseros.4 }
		}
		else = {
			set_global_variable = gg_pact_settlement_mandate_collapsed
			remove_global_variable = gg_president_railroad
			gg_ruler_set = { TEMPLATE = ARG_Justo_Jose_de_Urquiza }
			gg_rosista_client_release = yes # EN/ES: 23/9 (AUD-RULES-CLIENT) -- Rosas fell / cayo Rosas
			trigger_event = { id = gg_argentina.30 days = 0 popup = yes }
		}
		# EN/ES: 22/9/2026 (Juan) -- the provinces held as protectorates are settled now, whatever the settlement.
		#     24/9/2026: after the settlement is written, not before: with two governments the partition above has
		#     already taken them, and gg_argentina.80 has nothing to ask / despues de escrito el cierre: con dos
		#     gobiernos la particion ya se las llevo, y gg_argentina.80 no tiene nada que preguntar
		gg_argentina_vassals_settle = yes
	}
	# EN/ES: 24/9 -- a later call (the journal's on_complete the day after the war's reading) writes nothing; said so
	#     in the trace, which read like a second settlement / una llamada posterior no escribe nada; se dice en la traza
	else = {
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_resolve| ALREADY_SETTLED=yes " # GG_TRAZA
		# EN: 24/9/2026 (Codex review, round 2) -- Rosas won: the day after (gg_caseros.7), the protectorates the reading
		#     could not make because the war was still ending that same tick (end_play in gg_urquiza.3 / gg_caseros.1
		#     runs in the same effect) are made now. Idempotent; written here, in the Pact's one settlement writer (I19).
		# ES: gano Rosas: al dia siguiente se hacen los protectorados que la lectura no pudo hacer porque la guerra
		#     todavia terminaba en ese mismo instante. Idempotente.
		if = {
			limit = { has_global_variable = gg_pact_settlement_rosas }
			if = {
				limit = {
					gg_pact_rosas = yes
					NOT = { has_global_variable = gg_confederacion_negociada }
					c:ENR ?= { is_subject = no }
					NOT = { has_war_with = c:ENR }
				}
				create_diplomatic_pact = { country = c:ENR type = protectorate }
			}
			if = {
				limit = {
					has_global_variable = gg_corrientes_with_urquiza
					c:CRR ?= { is_subject = no }
					NOT = { has_war_with = c:CRR }
				}
				create_diplomatic_pact = { country = c:CRR type = protectorate }
				gg_delegation_restored = { KEY = corrientes COUNT = 1 }
			}
			if = {
				limit = {
					has_global_variable = gg_coalition_with_urquiza
					c:CLN ?= { is_subject = no }
					NOT = { has_war_with = c:CLN }
				}
				create_diplomatic_pact = { country = c:CLN type = protectorate }
			}
		}
	}
}
