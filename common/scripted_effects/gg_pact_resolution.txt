# EN/ES: defined first: gg_pact_resolve below uses it / definido primero: lo usa gg_pact_resolve
# EN: 23/9/2026 (Juan, AUD-RULES-CLIENT, decisiones.md) -- when Rosas's government falls to Urquiza, the Republic made
#     Rosas's client by the Cerrito's peace (gg_peace.2.e, modifier gg_peace_rosista_client) is Urquiza's to decide:
#     free it or keep the protection (gg_argentina.86, player or AI). A puppet made any other way is not touched.
# ES: cuando el gobierno de Rosas cae ante Urquiza, la Republica hecha cliente de Rosas la decide Urquiza: liberarla o
#     conservar la proteccion. Otra sujecion no se toca.
gg_rosista_client_release = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_rosista_client_release|" # GG_TRAZA
	if = {
		limit = {
			exists = c:ARG
			c:URU ?= {
				has_modifier = gg_peace_rosista_client
				is_subject_of = c:ARG
			}
		}
		c:ARG = { trigger_event = { id = gg_argentina.86 days = 0 popup = yes } }
	}
}

# EN: Country scope (Argentina). One idempotent settlement writer for the Federal Pact.
# ES: Scope pais (Argentina). Un escritor idempotente para todos los cierres del Pacto Federal.
gg_pact_resolve = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_resolve|" # GG_TRAZA
	# EN: 22/9/2026, Caseros option (c) -- while the siege of Buenos Aires lasts (at least six months
	#     from the choice) no ending is written: the request is recorded and gg_caseros.1 resolves it
	#     when the siege ends. Rosas keeps the capital and his government meanwhile. DESIGN: the
	#     six-month minimum is a counterfactual; Rosas did not take this road on 3/2/1852.
	# ES: mientras dura el sitio de Buenos Aires (seis meses como minimo desde la eleccion) no se
	#     escribe ningun final: el pedido queda anotado y gg_caseros.1 lo resuelve al terminar el sitio.
	#     DISENO: el minimo de seis meses es contrafactico.
	if = {
		limit = {
			has_global_variable = gg_caseros_siege
			NOT = { has_global_variable = gg_pact_settled }
		}
		set_global_variable = gg_pact_resolve_pending
	}
	else_if = {
		limit = { NOT = { has_global_variable = gg_pact_settled } }
		set_global_variable = gg_pact_settled
		# EN: 23/9/2026 -- the Pact's game rule reads the war its way (common/game_rules/zz_gg_game_rules.txt),
		#     unless Buenos Aires itself chose the negotiated confederation.
		# ES: la regla del Pacto lee la guerra a su modo, salvo que Buenos Aires eligiera la confederacion negociada.
		if = {
			limit = {
				gg_pact_forced_war = yes
				NOT = { has_global_variable = gg_confederacion_negociada }
			}
			set_global_variable = { name = gg_pact_war_urquiza_won value = 0 }
			set_global_variable = { name = gg_pact_war_rosas_won value = 0 }
			set_global_variable = { name = gg_pact_war_white_peace value = 0 }
			if = { limit = { gg_pact_historical = yes } set_global_variable = { name = gg_pact_war_urquiza_won value = 1 } }
			else_if = {
				limit = { gg_pact_rosas = yes }
				set_global_variable = { name = gg_pact_war_rosas_won value = 1 }
				# EN/ES: 23/9 (Codex review of 743b851) -- the rule's victory has no enforced war goal: Entre Rios becomes
				#     the protectorate the war would have made it, before the provinces are asked about
				#     / la victoria de la regla no impuso objetivo: Entre Rios queda el protectorado que la guerra le daba
				if = {
					limit = { c:ENR ?= { is_subject = no } NOT = { has_war_with = c:ENR } }
					create_diplomatic_pact = { country = c:ENR type = protectorate }
				}
			}
			else = { set_global_variable = { name = gg_pact_war_white_peace value = 1 } }
		}
		# EN/ES: 22/9/2026 (Juan) -- the provinces held as protectorates are annexed now, whatever the
		#     settlement / las que quedaron protectorado se anexan ahora, sea cual sea el cierre
		gg_argentina_vassals_settle = yes
		if = {
			limit = { has_global_variable = gg_confederacion_negociada }
			set_global_variable = gg_pact_settlement_constitutional
			trigger_event = { id = gg_argentina.82 days = 0 popup = yes }
		}
		else_if = {
			limit = { global_var:gg_pact_war_urquiza_won = 1 }
			set_global_variable = gg_pact_settlement_urquiza
			remove_global_variable = gg_president_railroad
			gg_ruler_set = { TEMPLATE = ARG_Justo_Jose_de_Urquiza }
			gg_rosista_client_release = yes # EN/ES: 23/9 (AUD-RULES-CLIENT) -- Rosas fell / cayo Rosas
			# EN: Caseros (a) or (c) -- Rosas falls and goes into exile (historically he embarked on
			#     HMS Centaur for England, February 1852). The Confederation remains.
			# ES: Caseros (a) o (c) -- cae Rosas y se exilia (se embarco en el Centaur rumbo a
			#     Inglaterra). La Confederacion queda.
			if = {
				limit = {
					OR = {
						global_var:gg_caseros_choice ?= 1
						global_var:gg_caseros_choice ?= 3
					}
				}
				random_scope_character = {
					limit = { has_template = ARG_Juan_Manuel_de_Rosas }
					exile_character_with_role_cleanup = yes
				}
			}
			trigger_event = { id = gg_argentina.83 days = 0 popup = yes }
			gg_caseros_notify_urquiza = { EVENT = gg_caseros.5 }
		}
		else_if = {
			limit = { global_var:gg_pact_war_rosas_won = 1 }
			set_global_variable = gg_pact_settlement_rosas
			if = {
				limit = { has_global_variable = gg_delegation_gone_entre_rios }
				gg_delegation_restored = { KEY = entre_rios COUNT = 1 }
			}
			trigger_event = { id = gg_argentina.84 days = 0 popup = yes }
		}
		else_if = {
			limit = { global_var:gg_pact_war_white_peace = 1 }
			set_global_variable = gg_pact_settlement_stalemate
			trigger_event = { id = gg_argentina.85 days = 0 popup = yes }
			gg_caseros_notify_urquiza = { EVENT = gg_caseros.4 }
		}
		else = {
			set_global_variable = gg_pact_settlement_mandate_collapsed
			remove_global_variable = gg_president_railroad
			gg_ruler_set = { TEMPLATE = ARG_Justo_Jose_de_Urquiza }
			gg_rosista_client_release = yes # EN/ES: 23/9 (AUD-RULES-CLIENT) -- Rosas fell / cayo Rosas
			trigger_event = { id = gg_argentina.30 days = 0 popup = yes }
		}
	}
}
