# EN: AUD-ARG-1850 (23/9/2026) -- Argentina must stay the SAME country object while its dated calendar is
#     pending (common/static_modifiers/gg_arg_identity.txt has the whole story). Scope: any; runs on the
#     first day and every month (guerra_grande_on_actions.txt), so existing saves get it on their next
#     monthly pulse. The last event queued on c:ARG at game start is gg_ba.105, 12/10/1862.
#     Watchdog: the first run keeps the ARG object in global gg_arg_identity_country. If a later month finds
#     c:ARG is another object, the engine replaced the country anyway (a revolution that was already under way in
#     an old save, or a path this guard does not know): the trace says so on the spot (GG_EF ...
#     ARG_REPLACED), and the new object is kept so it is reported once per replacement.
#     One replacement is the mod's own and expected: gg_ba.21 (Pavon, 1861) turns the surviving Entre Rios into
#     ARG after the old one is annexed; the line is logged then too, and the new ARG gets the modifier next month.
# ES: la Argentina tiene que seguir siendo el MISMO objeto mientras su calendario fechado este pendiente.
#     Corre el primer dia y cada mes; las partidas en curso lo reciben en el proximo pulso mensual. El
#     vigia guarda el objeto ARG en una global; si un mes c:ARG es otro objeto, el motor reemplazo el pais igual y
#     la traza lo dice en el acto (ARG_REPLACED).
gg_arg_identity_guard = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_identity_guard|" # GG_TRAZA
	c:ARG ?= {
		if = {
			limit = {
				game_date < 1862.10.13
				NOT = { has_modifier = gg_arg_caudillos_grip }
			}
			add_modifier = { name = gg_arg_caudillos_grip }
		}
		else_if = {
			limit = {
				game_date >= 1862.10.13
				has_modifier = gg_arg_caudillos_grip
			}
			remove_modifier = gg_arg_caudillos_grip
		}
		# EN: the identity is the country OBJECT kept in a global, not a variable on the country: a victorious
		#     revolutionary country inherits its origin's country variables (Codex 23/9, seen in a 1.13 save),
		#     so a mark on ARG would survive the replacement it is meant to catch.
		# ES: la identidad es el OBJETO pais guardado en una global, no una variable del pais: el pais
		#     revolucionario hereda las variables del original, y una marca sobre ARG sobreviviria al reemplazo.
		if = {
			limit = { NOT = { has_global_variable = gg_arg_identity_country } }
			set_global_variable = { name = gg_arg_identity_country value = this }
		}
		else_if = {
			limit = { NOT = { global_var:gg_arg_identity_country ?= this } }
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_identity_guard| ARG_REPLACED" # GG_TRAZA
			ruler ?= {
				debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_identity_guard| ARG_REPLACED ruler= [THIS.GetCharacter.GetFirstNameNoFormatting] [THIS.GetCharacter.GetLastNameNoFormatting]" # GG_TRAZA
			}
			set_global_variable = { name = gg_arg_identity_country value = this }
		}
	}
}
