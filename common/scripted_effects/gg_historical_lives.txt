# EN: 23/9/2026 (playtest) -- the men the war turns on do not die before their historical date. Rivera
#     died in an observer game in 1840 and a nobody governed Uruguay. Same effect vanilla uses for Pedro II.
#     Death dates (Wikipedia ES): Rosas 14/3/1877, Oribe 12/11/1857, Rivera 13/1/1854, Joaquin Suarez
#     26/11/1868, Urquiza 11/4/1870, William Brown 3/3/1857, Lavalleja 22/10/1853.
# ES: los hombres sobre los que gira la guerra no mueren antes de su fecha historica. Rivera murio en
#     una partida de observador en 1840 y goberno el Uruguay un cualquiera.
#
# EN: PERFORMANCE (23/9/2026, AUD-PERF-2). It used to scan every character in the world seven times a month
#     until 1936. Now, per man:
#     - before his date, the monthly pass looks only in South American countries (capital in
#       geographic_region_south_america), which is every place the mod creates one of these men (Argentina,
#       both Uruguayan governments, Entre Rios, Brazil for Rivera's return). The flag stays on a character
#       wherever he is moved afterwards, so men already flagged keep it abroad, as before;
#     - on the first monthly pulse on or after his date -- the same day the old code first cleared it -- ONE
#       world scan clears it, and the mark gg_hlife_over_<template> ends the work for him. After 14/3/1877
#       the monthly cost is fourteen reads of a date and a mark.
#     Saves: no migration. The marks are missing in an old save, so each man gets his one clearing scan.
# ES: RENDIMIENTO. Antes eran siete barridos mundiales por mes hasta 1936. Ahora, por hombre: antes de su
#     fecha, el pulso mensual mira solo los paises sudamericanos (donde el mod crea a estos hombres; la
#     marca de inmortal viaja con el personaje); el primer pulso en o despues de su fecha -- el mismo dia
#     que antes -- UN barrido mundial se la quita y la marca gg_hlife_over_<plantilla> cierra el trabajo.
#     Sin migracion: en una partida vieja falta la marca y cada hombre recibe su unico barrido.
gg_historical_life = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_historical_life| TEMPLATE= $TEMPLATE$ " # GG_TRAZA
	if = {
		limit = { game_date < $DIES$ }
		every_country = {
			limit = { capital ?= { is_in_geographic_region = geographic_region_south_america } }
			every_scope_character = {
				limit = { has_template = $TEMPLATE$ }
				set_character_immortal = yes
			}
		}
	}
	else_if = {
		limit = { NOT = { has_global_variable = gg_hlife_over_$TEMPLATE$ } }
		every_country = {
			every_scope_character = {
				limit = { has_template = $TEMPLATE$ }
				set_character_immortal = no
			}
		}
		# EN/ES: Codex 23/9 -- and the exile pool: the old monthly pass cleared him once he came back;
		#     with one pass he must be cleared where he is / y el exilio: el pase unico lo limpia donde este
		every_character_in_exile_pool = {
			limit = { has_template = $TEMPLATE$ }
			set_character_immortal = no
		}
		set_global_variable = gg_hlife_over_$TEMPLATE$
	}
}
gg_historical_lives = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_historical_lives|" # GG_TRAZA
	gg_historical_life = { TEMPLATE = ARG_Juan_Manuel_de_Rosas DIES = 1877.3.14 }
	gg_historical_life = { TEMPLATE = URU_Manuel_Oribe DIES = 1857.11.12 }

	gg_historical_life = { TEMPLATE = ARG_Justo_Jose_de_Urquiza DIES = 1870.4.11 }

	gg_historical_life = { TEMPLATE = URU_Juan_Antonio_Lavalleja DIES = 1853.10.22 }
}

# EN: 23/9/2026 (headless playtests) -- while Rosas's era lasts, Argentina's head of state is Rosas, as
#     Uruguay's follows history (gg_president_historical). In two headless runs Argentina's ruler became
#     Echague in November 1836 with no law changed and no script of ours involved (the engine's pick,
#     cause still under a probe). The era ends with the Pact's settlement (the same marks as
#     gg_x_arg_rosas_era_over, common/scripted_triggers/gg_rosas_era.txt, which also ends the era with a constitution).
# ES: mientras dura la era de Rosas, el jefe de Estado argentino es Rosas, como el uruguayo sigue la
#     historia. En dos corridas sin pantalla el gobernante paso a ser Echague en noviembre de 1836 sin
#     cambio de ley ni script nuestro. La era termina con el cierre del Pacto.
gg_rosas_keeps_the_seat = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_rosas_keeps_the_seat|" # GG_TRAZA
	c:ARG ?= {
		if = {
			limit = {
				has_variable = gg_arg_rosas_seat
				NOR = {
					has_global_variable = gg_pact_settlement_urquiza
					has_global_variable = gg_pact_settlement_constitutional
					has_global_variable = gg_pact_settlement_mandate_collapsed
				}
				any_scope_character = { has_template = ARG_Juan_Manuel_de_Rosas is_character_alive = yes }
				NOT = { ruler ?= { has_template = ARG_Juan_Manuel_de_Rosas } }
			}
			gg_ruler_set = { TEMPLATE = ARG_Juan_Manuel_de_Rosas }
		}
	}
}
