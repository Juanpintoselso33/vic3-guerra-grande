# EN: WHO GOVERNS EACH OF THE THIRTEEN, for the Federal Pact's list (gui/gg_pact_overview.gui). Rewritten
#     22/9/2026 for the thirteen rows (it used to cover three: Corrientes, «the North», Entre Rios).
#     One global variable per province, gg_pact_gov_<key>, cleared and set again every week:
#       1. if the province stands as a country (CRR, ENR, TUC...), its ruler;
#       2. otherwise the HISTORICAL governor for the date, while he lives (gg_pact_gov_hist_<key>).
#     A governor who is dead stays dead: the row shows the governorship vacant rather than a name the
#     date does not support. The dates and templates come from
#     docs/guerra-grande-argentina/gobernadores-provincias.md -- no governor without a source.
#     gg_pact_gov_federal_<key> is the man Buenos Aires installs when it beats the province
#     (gg_province_install_governor): the same list, except where a unitarian government held the
#     province at that date -- Rosas did not leave the man he had just beaten in the chair.
# ES: QUIEN GOBIERNA CADA UNA DE LAS TRECE, para la lista del Pacto. Reescrito el 22/9/2026 para trece
#     filas (antes eran tres). Una global por provincia, gg_pact_gov_<clave>, borrada y vuelta a poner
#     cada semana: 1) si la provincia es pais, su gobernante; 2) si no, el gobernador HISTORICO para la
#     fecha, mientras viva. Un gobernador muerto sigue muerto: la fila muestra la gobernacion vacante
#     antes que un nombre que la fecha no respalda. Fechas y templates: gobernadores-provincias.md.
#     gg_pact_gov_federal_<clave> es el que pone Buenos Aires cuando bate a la provincia: la misma
#     lista, salvo donde a esa fecha gobernaba un unitario -- Rosas no dejaba en la silla al que batia.
# -------------------------------------------------------------------------------------------------

# EN/ES: find a living man with this template, anywhere / busca un vivo con este template, donde sea
gg_pact_gov_from_template = {
	every_country = {
		every_scope_character = {
			limit = { has_template = $TEMPLATE$ is_character_alive = yes }
			set_global_variable = { name = $VAR$ value = this }
		}
	}
}

# EN: The governor for the date, if he lives. For characters the game or another event already
#     creates (Rosas's roster, Urquiza, Echague, Heredia, Beron de Astrada, Avellaneda...).
# ES: El gobernador de la fecha, si vive. Para personajes que ya crea el juego u otro evento.
gg_pact_gov_pick = {
	gg_pact_gov_from_template = { TEMPLATE = $TEMPLATE$ VAR = gg_pact_gov_$KEY$ }
	# EN/ES: Codex 22/9 -- alive in exile still counts: shown, not moved (another event owns him)
	#     / vivo en el exilio igual cuenta: se lo muestra, no se lo mueve
	if = {
		limit = { NOT = { has_global_variable = gg_pact_gov_$KEY$ } }
		random_character_in_exile_pool = {
			limit = { has_template = $TEMPLATE$ }
			set_global_variable = { name = gg_pact_gov_$KEY$ value = this }
		}
	}
}

# EN: The same, and if he was NEVER born, he is born now -- once -- in whoever holds the province
#     (Argentina, the Coalition or a province country; never Bolivia). The mark gg_gov_born_<template>
#     is what keeps a dead governor dead: without it the next week would raise him again.
# ES: Lo mismo, y si NUNCA nacio, nace ahora -- una vez -- en quien tenga la provincia. La marca
#     gg_gov_born_<template> es lo que mantiene muerto a un gobernador muerto.
gg_pact_gov_pick_or_create = {
	gg_pact_gov_from_template = { TEMPLATE = $TEMPLATE$ VAR = gg_pact_gov_$KEY$ }
	if = {
		limit = { has_global_variable = gg_pact_gov_$KEY$ }
		if = {
			limit = { NOT = { has_global_variable = gg_gov_born_$TEMPLATE$ } }
			set_global_variable = gg_gov_born_$TEMPLATE$
		}
	}
	# EN/ES: Codex 22/9 -- a governor of the date who is in exile comes home to whoever holds the province
	#     / un gobernador de la fecha que esta en el exilio vuelve a quien tenga la provincia
	else_if = {
		limit = { any_character_in_exile_pool = { has_template = $TEMPLATE$ } }
		s:$STATE$ = {
			random_scope_state = {
				limit = { owner ?= { gg_is_argentine_holder = yes } }
				owner = { save_scope_as = gg_gov_holder }
			}
		}
		if = {
			limit = { exists = scope:gg_gov_holder }
			random_character_in_exile_pool = {
				limit = { has_template = $TEMPLATE$ }
				transfer_character = scope:gg_gov_holder
				set_global_variable = { name = gg_pact_gov_$KEY$ value = this }
			}
			clear_saved_scope = gg_gov_holder
		}
	}
	else_if = {
		limit = { NOT = { has_global_variable = gg_gov_born_$TEMPLATE$ } }
		# EN/ES: the mark is set on birth, not before: with no Argentine holder of the province nobody is
		#     born, and he must stay bornable / la marca va al nacer: sin duenio argentino no nace nadie
		s:$STATE$ = {
			random_scope_state = {
				limit = { owner ?= { gg_is_argentine_holder = yes } }
				owner = {
					save_scope_as = gg_gov_holder
					create_character = {
						template = $TEMPLATE$
						on_created = {
							set_global_variable = { name = gg_pact_gov_$KEY$ value = this }
							set_global_variable = gg_gov_born_$TEMPLATE$
							# EN: a governor born inside Buenos Aires's roster gets no command there: the
							#     list is not a way to hand Rosas his rivals' generals (Brizuela governs La
							#     Rioja from 1837; the Coalition gives him back his command in 1840).
							# ES: un gobernador que nace en el plantel de Buenos Aires no recibe mando ahi:
							#     la lista no es una forma de darle a Rosas los generales de sus rivales.
							if = {
								limit = { c:ARG ?= scope:gg_gov_holder is_general = yes }
								remove_character_role = general
							}
						}
					}
					clear_saved_scope = gg_gov_holder
				}
			}
		}
	}
}

# EN/ES: one row: the province country's ruler, else the historical governor / una fila
gg_pact_gov_province = {
	if = { limit = { has_global_variable = gg_pact_gov_$KEY$ } remove_global_variable = gg_pact_gov_$KEY$ }
	if = {
		limit = { exists = c:$TAG$ c:$TAG$ = { exists = ruler } }
		c:$TAG$ = { ruler = { set_global_variable = { name = gg_pact_gov_$KEY$ value = this } } }
	}
	else = { gg_pact_gov_hist_$KEY$ = yes }
}

#     but not here, so between two windows the row reads «interim government». Where the sources
#     disagree the comment says which date was taken and why (two sources agreeing wins).

#     1820-62» (R).

#     ventanas la fila dice «gobierno interino». Donde las fuentes difieren, el comentario dice que
#     fecha se tomo y por que (gana la que dos fuentes comparten).
