# EN: 25/9/2026 (Juan, AUD-CASEROS-WAR) -- who carried the day of Caseros (gg_caseros.10) and what carried it. WINNER and
#     LOSER are the two countries, VALUE is gg_caseros_battle (1 Urquiza, 2 Rosas), SIGN turns the reckoning to the
#     winner's side (1 Urquiza, -1 Rosas). The caller logs GG_OPC gg_caseros.10.a/.b. What carried it is the part of
#     the reckoning (gg_caseros_day_battles/_ground/_numbers/_spirit) that weighed most for the winner; a near-tie
#     (gg_caseros_tie) was carried by fortune. The winner and the loser see it for a year.
# ES: quien se llevo la jornada de Caseros y que se la dio: la parte de la cuenta que mas peso para el ganador; un
#     empate lo decidio la suerte. Ganador y perdedor lo ven durante un ano.
# -------------------------------------------------------------------------------------------------
gg_caseros_day_decide = {
	set_global_variable = { name = gg_caseros_battle value = $VALUE$ }
	# EN/ES: 26/9/2026 (AUD-PACT-HOLD-MILESTONES) -- the disarray of Buenos Aires's ranks after the crossing ends with the day:
	#     from here the day's own modifiers and the holds of gg_pact_front_hold say what is left of each army
	#     / el desbande de las filas de Buenos Aires termina con la jornada: desde aca mandan los de la jornada y las retenciones
	c:ARG ?= { if = { limit = { has_modifier = gg_rosas_army_disarray } remove_modifier = gg_rosas_army_disarray } }
	set_global_variable = { name = gg_caseros_w_battles value = global_var:gg_caseros_c_battles }
	set_global_variable = { name = gg_caseros_w_ground value = global_var:gg_caseros_c_ground }
	set_global_variable = { name = gg_caseros_w_numbers value = global_var:gg_caseros_c_numbers }
	set_global_variable = { name = gg_caseros_w_spirit value = global_var:gg_caseros_c_spirit }
	change_global_variable = { name = gg_caseros_w_battles multiply = $SIGN$ }
	change_global_variable = { name = gg_caseros_w_ground multiply = $SIGN$ }
	change_global_variable = { name = gg_caseros_w_numbers multiply = $SIGN$ }
	change_global_variable = { name = gg_caseros_w_spirit multiply = $SIGN$ }
	if = {
		limit = { has_global_variable = gg_caseros_tie }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=fortune " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost years = 1 } }
	}
	else_if = {
		limit = {
			global_var:gg_caseros_w_battles >= global_var:gg_caseros_w_ground
			global_var:gg_caseros_w_battles >= global_var:gg_caseros_w_numbers
			global_var:gg_caseros_w_battles >= global_var:gg_caseros_w_spirit
		}
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=battles " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried_battles years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost_battles years = 1 } }
	}
	else_if = {
		limit = {
			global_var:gg_caseros_w_ground >= global_var:gg_caseros_w_numbers
			global_var:gg_caseros_w_ground >= global_var:gg_caseros_w_spirit
		}
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=ground " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried_ground years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost_ground years = 1 } }
	}
	else_if = {
		limit = { global_var:gg_caseros_w_numbers >= global_var:gg_caseros_w_spirit }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=numbers " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried_numbers years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost_numbers years = 1 } }
	}
	else = {
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=spirit " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried_spirit years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost_spirit years = 1 } }
	}
}

# EN: 26/9/2026 (Juan, AUD-PACT-HOLD-LEAK) -- THE HOLD ON THE GENERAL, NOT ON HIS ORDER. Every general the hold restrains now
#     (gg_pact_front_hold) carries gg_pact_line_held (advancement speed -100 %: an advancing general never launches a battle);
#     every other general that still carries it loses it. Daily from gg_caseros.12. Why not the order:
#     the engine keeps an order already given and hands every general the default order «advance» (defines NMilitary
#     DEFAULT_ORDER), and no script effect sets a general's order in 1.13 (the effects on commanders are set_commander_rank,
#     create_military_formation, transfer_to_formation, unassign_from_formation, teleport_to_front, mobilize_army): in the
#     caseros runs on 271e2ec the probe found 1-2 held generals advancing for months (HELD_ADV), Urquiza's field army attacked
#     in July 1851 and lost 71 % of its battles before the Ejercito Grande (caseros_5), and the day went to Buenos Aires
#     (SCORE=-22). GG_DEBUG gg_debug_line_held: how many generals carry it after this pass (the probe's LINE=). Any scope.
# ES: LA RETENCION SOBRE EL GENERAL, NO SOBRE SU ORDEN. Todo general que la retencion frena lleva gg_pact_line_held (velocidad
#     de avance -100 %: un general que avanza nunca da batalla); el que ya no esta frenado la pierde. Cada dia desde
#     gg_caseros.12 y antes de cada sonda. Por que no la orden: el motor conserva la orden ya dada y a todo general le da la
#     orden por defecto «avanzar», y en 1.13 ningun efecto de script fija la orden de un general. En las corridas de caseros
#     sobre 271e2ec la sonda encontro 1-2 generales retenidos avanzando durante meses, el ejercito de Urquiza ataco en julio
#     de 1851 y perdio el 71 % de sus batallas antes del Ejercito Grande, y la jornada fue de Buenos Aires.
# -------------------------------------------------------------------------------------------------
# EN/ES: Argentina's scope: start the daily line (gg_caseros.12) if the war of the Pronouncement is on and the pulse is not
#     running; gg_pact_line_running lasts three days and each pulse renews it, so a lost pulse is restarted by the next caller
#     (gg_caseros.9 daily until the day, gg_caseros.8 monthly: also an older save past Caseros) / arranca la linea si no corre
gg_pact_line_start = {
	if = {
		limit = {
			has_global_variable = gg_war_against_buenos_aires
			NOT = { has_global_variable = gg_pact_settled }
			NOT = { has_global_variable = gg_pact_line_running }
			exists = c:ENR
			c:ARG ?= { has_war_with = c:ENR }
		}
		set_global_variable = { name = gg_pact_line_running days = 3 }
		c:ARG = { trigger_event = { id = gg_caseros.12 days = 0 } }
	}
}

gg_pact_hold_line_sync = {
	set_global_variable = { name = gg_debug_line_held value = 0 }
	every_country = {
		limit = {
			any_scope_general = {
				OR = {
					gg_pact_front_hold = yes
					has_modifier = gg_pact_line_held
				}
			}
		}
		every_scope_general = {
			if = {
				limit = { gg_pact_front_hold = yes }
				# EN/ES: refreshed each day for three days (Codex review): if the pulse ever stops, or the general retires out of
				#     every_scope_general, the line runs out on its own / se renueva cada dia por tres: si el pulso se corta, se vence sola
				if = {
					limit = { has_modifier = gg_pact_line_held }
					remove_modifier = gg_pact_line_held
				}
				add_modifier = { name = gg_pact_line_held days = 3 }
				change_global_variable = { name = gg_debug_line_held add = 1 }
			}
			else_if = {
				limit = { has_modifier = gg_pact_line_held }
				remove_modifier = gg_pact_line_held
			}
		}
	}
}

		set_global_variable = { name = gg_debug_probe_held_free value = 0 }
		set_global_variable = { name = gg_debug_probe_nf_adv value = 0 }

			limit = {
				OR = {
					AND = { gg_pact_rosas_side = yes has_war_with = c:ENR }
					AND = { gg_pact_urquiza_side = yes has_war_with = c:ARG }
				}
			}

				limit = { character_has_advance_order = yes gg_pact_front_hold = yes }

				}
			}

				limit = { character_has_advance_order = yes NOT = { front ?= { always = yes } } }
				change_global_variable = { name = gg_debug_probe_nf_adv add = 1 }
			}
		}

		change_global_variable = { name = gg_debug_probe_occ_enr multiply = 100 }
		c:ARG = {
			random_scope_war = {
				limit = { is_war_participant = c:ENR }
				save_scope_as = gg_probe_war
			}
		}
		if = {
			limit = { exists = scope:gg_probe_war }

			c:ENR = { set_global_variable = { name = gg_debug_probe_lost_enr value = "size_weighted_lost_battles_fraction(scope:gg_probe_war)" } }

# -------------------------------------------------------------------------------------------------
# EN: 26/9/2026 (Juan, AUD-PACT-ROSAS-SMALL) -- WHY BUENOS AIRES CUTS ITS ARMY, for the trace. Between 1836 and 1851 Buenos
#     Aires's barracks fall from 6 to 0-3 and no script touches them; the hypothesis is the engine's AI: the wish of the
#     playset (Kuromi's/Knowledgeable AI, kai_script_values.txt: tope num_states x 4) and the money (1845: money 0). One line
#     with what closes it: BRK_ARG/CC_ARG Argentina's barracks and conscription-centre levels, BRK_BA those of the Buenos
#     Aires state, REG_ARG its standing battalions (army_size), WANT_ARG the army its AI wants (the default strategy's
#     wanted_army_size_script_value plus gg_pact_field_army_wanted when it holds a Plata strategy, STRAT_ARG 1), GOLD_ARG and
#     DEBT_ARG (hundredths of the ceiling), SL the battalions left in the army of Santos Lugares. Monthly in the Pact war
#     (WHEN=pact, gg_pact_war_probe) and once a year from 1836 (WHEN=year, gg_arg_army_sample). Trace only; changes nothing.
# ES: POR QUE BUENOS AIRES RECORTA SU EJERCITO, para la traza. Entre 1836 y 1851 los cuarteles de Buenos Aires bajan de 6 a
#     0-3 sin que ningun script los toque; la hipotesis es la IA del motor: el deseo del playset y la plata. Una linea con lo
#     que la cierra: cuarteles y centros de la Argentina y del estado de Buenos Aires, permanentes, deseo de la IA, oro,
#     deuda y lo que queda del ejercito de Santos Lugares. Cada mes en la guerra del Pacto y una vez por anio desde 1836.
# -------------------------------------------------------------------------------------------------
gg_arg_army_probe = {
	c:ARG ?= {
		set_global_variable = { name = gg_debug_arg_brk value = 0 }
		set_global_variable = { name = gg_debug_arg_cc value = 0 }
		set_global_variable = { name = gg_debug_arg_brk_ba value = 0 }
		set_global_variable = { name = gg_debug_arg_strat value = 0 }
		set_global_variable = { name = gg_debug_arg_sl value = 0 }
		every_scope_building = {
			limit = { is_building_type = building_barrack }
			change_global_variable = { name = gg_debug_arg_brk add = this.level }
		}
		every_scope_building = {
			limit = { is_building_type = building_conscription_center }
			change_global_variable = { name = gg_debug_arg_cc add = this.level }
		}
		s:STATE_BUENOS_AIRES.region_state:ARG ?= {
			every_scope_building = {
				limit = { is_building_type = building_barrack }
				change_global_variable = { name = gg_debug_arg_brk_ba add = this.level }
			}
		}
		set_global_variable = { name = gg_debug_arg_reg value = army_size }
		set_global_variable = { name = gg_debug_arg_want value = wanted_army_size_script_value }
		if = {
			limit = {
				OR = {
					has_strategy = gg_ai_strategy_plata_open
					has_strategy = gg_ai_strategy_plata_closed
					has_strategy = gg_ai_strategy_plata_enemies
				}
			}
			set_global_variable = { name = gg_debug_arg_strat value = 1 }
			change_global_variable = { name = gg_debug_arg_want add = gg_pact_field_army_wanted }
		}
		set_global_variable = { name = gg_debug_arg_gold value = gold_reserves }
		set_global_variable = { name = gg_debug_arg_debt value = scaled_debt }
		change_global_variable = { name = gg_debug_arg_debt multiply = 100 }
		every_in_global_list = {
			variable = gg_sl_probe_army
			every_combat_unit = { change_global_variable = { name = gg_debug_arg_sl add = 1 } }
		}
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_army_probe| WHEN=$WHEN$ BRK_ARG=[GetGlobalVariable('gg_debug_arg_brk').GetValue|0] BRK_BA=[GetGlobalVariable('gg_debug_arg_brk_ba').GetValue|0] CC_ARG=[GetGlobalVariable('gg_debug_arg_cc').GetValue|0] REG_ARG=[GetGlobalVariable('gg_debug_arg_reg').GetValue|0] WANT_ARG=[GetGlobalVariable('gg_debug_arg_want').GetValue|0] STRAT_ARG=[GetGlobalVariable('gg_debug_arg_strat').GetValue|0] GOLD_ARG=[GetGlobalVariable('gg_debug_arg_gold').GetValue|0] DEBT_ARG=[GetGlobalVariable('gg_debug_arg_debt').GetValue|0] SL=[GetGlobalVariable('gg_debug_arg_sl').GetValue|0] " # GG_TRAZA
	}
}

# EN: 26/9/2026 (Juan, AUD-PACT-ARMY-MELTS) -- the province's AI takes its own strategy, whose wanted army is the army it has
#     while the war of the Pronouncement lasts (gg_pact_field_army_wanted). The AI re-picks strategies rarely and weight alone
#     is not enough (gg_rio_plata.txt, verified 21/9/2026), so the script sets it: when the field army musters, when the
#     Ejercito Grande is raised and every week of the war (gg_caseros.9, under every rule of the Pact). Country scope (Entre Rios or Corrientes).
# ES: la IA de la provincia toma su estrategia, cuyo ejercito deseado es el que tiene mientras dura la guerra. La IA re-elige
#     estrategias pocas veces y el peso solo no alcanza, asi que la fija el script: al reunirse el ejercito de campania, al
#     levantarse el Ejercito Grande y cada mes de la guerra. Scope de pais.
gg_pact_army_keep = {
	# EN/ES: 26/9/2026 (AUD-PACT-ARMY-FLOOR) -- the floor the wish is measured against, once: the barracks the province has at
	#     the Pronouncement's muster (or, for a save already in the war, the first week after this fix) / el piso, una vez: los
	#     cuarteles al reunirse el ejercito del Pronunciamiento (o, en una partida ya en guerra, la primera semana)
	if = {
		limit = {
			NOT = { has_variable = gg_pact_army_floor }
			is_subject = no
			has_global_variable = gg_war_against_buenos_aires
			OR = {
				c:ENR ?= this
				c:CRR ?= this
			}
		}
		set_variable = { name = gg_pact_army_floor value = gg_pact_field_army_barracks }
		set_global_variable = { name = gg_debug_army_floor value = var:gg_pact_army_floor }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_army_keep| FLOOR=[GetGlobalVariable('gg_debug_army_floor').GetValue|0] " # GG_TRAZA
	}
	# EN/ES: never a province Rosas made his subject; the war mark, not has_war_with, which the
	#     war declared in the same effect may not have set yet / nunca una provincia sujeta; la marca de la guerra, no
	#     has_war_with, que la guerra declarada en el mismo efecto puede no haber puesto todavia
	if = {
		limit = {
			is_player = no
			is_subject = no
			has_global_variable = gg_war_against_buenos_aires
			c:ENR ?= this
			NOT = { has_strategy = gg_ai_strategy_entre_rios }
		}
		set_strategy = gg_ai_strategy_entre_rios
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_army_keep| STRATEGY=entre_rios " # GG_TRAZA
	}
	else_if = {
		limit = {
			is_player = no
			is_subject = no
			has_global_variable = gg_war_against_buenos_aires
			c:CRR ?= this
			NOT = { has_strategy = gg_ai_strategy_corrientes }
		}
		set_strategy = gg_ai_strategy_corrientes
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_army_keep| STRATEGY=corrientes " # GG_TRAZA
	}
}

# EN: 26/9/2026 (AUD-PACT-ARMY-FLOOR) -- the mod raises COUNT battalions for this province (the Ejercito Grande's 18, the Oriental
#     division's 3): its floor grows by them, from the larger of the floor and the barracks it has now. Called BEFORE the
#     formation is created, so the new levels are never counted twice. Country scope (Entre Rios).
# ES: el mod le levanta COUNT batallones a la provincia: su piso crece en ellos, desde el mayor entre el piso y los cuarteles
#     que tiene. Antes de crear la formacion, para no contar dos veces. Scope de pais.
gg_pact_army_floor_add = {
	if = {
		limit = {
			NOT = { has_variable = gg_pact_army_floor }
		}
		set_variable = { name = gg_pact_army_floor value = gg_pact_field_army_barracks }
	}
	else_if = {
		limit = { gg_pact_field_army_barracks > var:gg_pact_army_floor }
		set_variable = { name = gg_pact_army_floor value = gg_pact_field_army_barracks }
	}
	change_variable = { name = gg_pact_army_floor add = $COUNT$ }
	set_global_variable = { name = gg_debug_army_floor value = var:gg_pact_army_floor }
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_army_floor_add| COUNT= $COUNT$ FLOOR=[GetGlobalVariable('gg_debug_army_floor').GetValue|0] " # GG_TRAZA
}
