# EN/ES: loads before gg_pact_resolution.txt, which uses it / carga antes de gg_pact_resolution.txt, que lo usa
# EN: Tell the player of Entre Rios (Urquiza's side) what happened at Caseros; Buenos Aires gets its own
#     events. Only players: the AI does not read.
# ES: Avisarle al jugador de Entre Rios lo que paso en Caseros; Buenos Aires tiene sus propios eventos.
gg_caseros_notify_urquiza = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_notify_urquiza| EVENT= $EVENT$ " # GG_TRAZA
	c:ENR ?= {
		if = {
			limit = { is_player = yes }
			trigger_event = { id = $EVENT$ popup = yes }
		}
	}
}

# -------------------------------------------------------------------------------------------------
# EN: 25/9/2026 (Juan, AUD-CASEROS-WAR) -- who carried the day of Caseros (gg_caseros.10) and what carried it. WINNER and
#     LOSER are the two countries, VALUE is gg_caseros_battle (1 Urquiza, 2 Rosas), SIGN turns the reckoning to the
#     winner's side (1 Urquiza, -1 Rosas). The caller logs GG_OPC gg_caseros.10.a/.b. What carried it is the part of
#     the reckoning (gg_caseros_day_battles/_ground/_numbers/_spirit) that weighed most for the winner; a near-tie
#     (gg_caseros_tie) was carried by fortune. The winner and the loser see it for a year.
# ES: quien se llevo la jornada de Caseros y que se la dio: la parte de la cuenta que mas peso para el ganador; un
#     empate lo decidio la suerte. Ganador y perdedor lo ven durante un ano.
# -------------------------------------------------------------------------------------------------
gg_caseros_day_decide = {
	set_global_variable = { name = gg_caseros_battle value = $VALUE$ }
	# EN/ES: 26/9/2026 (AUD-PACT-HOLD-MILESTONES) -- the disarray of Buenos Aires's ranks after the crossing ends with the day:
	#     from here the day's own modifiers and the holds of gg_pact_front_hold say what is left of each army
	#     / el desbande de las filas de Buenos Aires termina con la jornada: desde aca mandan los de la jornada y las retenciones
	c:ARG ?= { if = { limit = { has_modifier = gg_rosas_army_disarray } remove_modifier = gg_rosas_army_disarray } }
	set_global_variable = { name = gg_caseros_w_battles value = global_var:gg_caseros_c_battles }
	set_global_variable = { name = gg_caseros_w_ground value = global_var:gg_caseros_c_ground }
	set_global_variable = { name = gg_caseros_w_numbers value = global_var:gg_caseros_c_numbers }
	set_global_variable = { name = gg_caseros_w_spirit value = global_var:gg_caseros_c_spirit }
	change_global_variable = { name = gg_caseros_w_battles multiply = $SIGN$ }
	change_global_variable = { name = gg_caseros_w_ground multiply = $SIGN$ }
	change_global_variable = { name = gg_caseros_w_numbers multiply = $SIGN$ }
	change_global_variable = { name = gg_caseros_w_spirit multiply = $SIGN$ }
	if = {
		limit = { has_global_variable = gg_caseros_tie }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=fortune " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost years = 1 } }
	}
	else_if = {
		limit = {
			global_var:gg_caseros_w_battles >= global_var:gg_caseros_w_ground
			global_var:gg_caseros_w_battles >= global_var:gg_caseros_w_numbers
			global_var:gg_caseros_w_battles >= global_var:gg_caseros_w_spirit
		}
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=battles " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried_battles years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost_battles years = 1 } }
	}
	else_if = {
		limit = {
			global_var:gg_caseros_w_ground >= global_var:gg_caseros_w_numbers
			global_var:gg_caseros_w_ground >= global_var:gg_caseros_w_spirit
		}
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=ground " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried_ground years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost_ground years = 1 } }
	}
	else_if = {
		limit = { global_var:gg_caseros_w_numbers >= global_var:gg_caseros_w_spirit }
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=numbers " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried_numbers years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost_numbers years = 1 } }
	}
	else = {
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros_day_decide| BY=spirit " # GG_TRAZA
		$WINNER$ ?= { add_modifier = { name = gg_caseros_day_carried_spirit years = 1 } }
		$LOSER$ ?= { add_modifier = { name = gg_caseros_day_lost_spirit years = 1 } }
	}
}

# -------------------------------------------------------------------------------------------------
# EN: 26/9/2026 (Juan, AUD-PACT-HOLD-MILESTONES) -- THE WAR OF THE PRONOUNCEMENT AS IT STANDS, for the trace (monthly, from the
#     tilt, gg_caseros.8). The six caseros runs of 26/9 left no record of the war between the Pronouncement and Caseros:
#     who fought, who advanced, how many battalions each side kept. Each side is its countries at war with the other's
#     leader (Rosas: Argentina and its provinces; Urquiza: Entre Rios and Corrientes): battalions under arms, the engine's
#     occupation and lost battles (in hundredths), and ADV_* the AI generals of each side with an advance order on a front
#     against the other (gg_pact_front_hold must keep ADV_ROSAS at 0 before the day and ADV_URQ at 0 before the crossing:
#     if not, the engine ignores the orders' `possible` for the AI). Trace only; changes nothing.
# ES: LA GUERRA DEL PRONUNCIAMIENTO COMO ESTA, para la traza, cada mes. Las corridas del 26/9 no dejaron registro de la
#     guerra entre el Pronunciamiento y Caseros. Cada lado: batallones, ocupacion y batallas perdidas (en centesimos), y
#     ADV_* los generales de la IA de cada lado con orden de avance contra el otro (la retencion tiene que dejar ADV_ROSAS en
#     0 antes de la jornada y ADV_URQ en 0 antes del cruce). Solo traza; no cambia nada.
# -------------------------------------------------------------------------------------------------
gg_pact_war_probe = {
	if = {
		limit = {
			exists = c:ARG
			exists = c:ENR
			c:ARG = { has_war_with = c:ENR }
		}
		set_global_variable = { name = gg_debug_probe_bn_arg value = 0 }
		set_global_variable = { name = gg_debug_probe_bn_enr value = 0 }
		set_global_variable = { name = gg_debug_probe_adv_rosas value = 0 }
		set_global_variable = { name = gg_debug_probe_adv_urq value = 0 }
		set_global_variable = { name = gg_debug_probe_lost_arg value = 0 }
		set_global_variable = { name = gg_debug_probe_lost_enr value = 0 }
		every_country = {
			limit = {
				gg_pact_rosas_side = yes
				has_war_with = c:ENR
			}
			change_global_variable = { name = gg_debug_probe_bn_arg add = army_size_including_raised_conscripts }
			every_scope_general = {
				limit = {
					character_has_advance_order = yes
					gg_pact_general_faces = { SIDE = gg_pact_urquiza_side }
				}
				change_global_variable = { name = gg_debug_probe_adv_rosas add = 1 }
			}
		}
		every_country = {
			limit = {
				gg_pact_urquiza_side = yes
				has_war_with = c:ARG
			}
			change_global_variable = { name = gg_debug_probe_bn_enr add = army_size_including_raised_conscripts }
			every_scope_general = {
				limit = {
					character_has_advance_order = yes
					gg_pact_general_faces = { SIDE = gg_pact_rosas_side }
				}
				change_global_variable = { name = gg_debug_probe_adv_urq add = 1 }
			}
		}
		c:ARG = { set_global_variable = { name = gg_debug_probe_occ_arg value = enemy_occupation } }
		c:ENR = { set_global_variable = { name = gg_debug_probe_occ_enr value = enemy_occupation } }
		change_global_variable = { name = gg_debug_probe_occ_arg multiply = 100 }
		change_global_variable = { name = gg_debug_probe_occ_enr multiply = 100 }
		c:ARG = {
			random_scope_war = {
				limit = { is_war_participant = c:ENR }
				save_scope_as = gg_probe_war
			}
		}
		if = {
			limit = { exists = scope:gg_probe_war }
			c:ARG = { set_global_variable = { name = gg_debug_probe_lost_arg value = "size_weighted_lost_battles_fraction(scope:gg_probe_war)" } }
			c:ENR = { set_global_variable = { name = gg_debug_probe_lost_enr value = "size_weighted_lost_battles_fraction(scope:gg_probe_war)" } }
			change_global_variable = { name = gg_debug_probe_lost_arg multiply = 100 }
			change_global_variable = { name = gg_debug_probe_lost_enr multiply = 100 }
		}
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_war_probe| BN_ARG=[GetGlobalVariable('gg_debug_probe_bn_arg').GetValue|0] BN_ENR=[GetGlobalVariable('gg_debug_probe_bn_enr').GetValue|0] OCC_ARG=[GetGlobalVariable('gg_debug_probe_occ_arg').GetValue|0] OCC_ENR=[GetGlobalVariable('gg_debug_probe_occ_enr').GetValue|0] LOST_ARG=[GetGlobalVariable('gg_debug_probe_lost_arg').GetValue|0] LOST_ENR=[GetGlobalVariable('gg_debug_probe_lost_enr').GetValue|0] ADV_ROSAS=[GetGlobalVariable('gg_debug_probe_adv_rosas').GetValue|0] ADV_URQ=[GetGlobalVariable('gg_debug_probe_adv_urq').GetValue|0] " # GG_TRAZA
		# EN: 26/9/2026 (Juan, AUD-PACT-ARMY-MELTS) -- WHERE THE BATTALIONS GO. BN_* above counts raised conscripts; here each part
		#     apart: REG_* standing battalions (army_size) of each side, CRR Corrientes's battalions, BRK/CC the barracks and
		#     conscription-centre levels of Entre Rios, WANT the army its AI wants (the default strategy plus
		#     gg_pact_field_army_wanted), STRAT 1 if it has its own strategy, GOLD its gold reserves and DEBT its debt (hundredths
		#     of the ceiling), EG/BRZDIV/ORIDIV the battalions left in the Ejercito Grande and the allied divisions of the
		#     convention (Brazil's, the Oriental one). If BN_ENR falls with BRK and REG, the AI disbanded barracks; if REG holds and
		#     BN falls, conscripts were sent home. Trace only.
		# ES: DONDE VAN LOS BATALLONES. BN_* de arriba cuenta los conscriptos levantados; aca cada parte por separado: batallones
		#     permanentes de cada lado, los de Corrientes, niveles de cuartel y de centro de conscripcion de Entre Rios, el
		#     ejercito que quiere su IA, si tiene su estrategia, reservas y deuda, y lo que queda en el Ejercito Grande y en las
		#     divisiones aliadas. Si BN_ENR cae con BRK y REG, la IA disolvio cuarteles; si REG se sostiene, volvieron conscriptos.
		set_global_variable = { name = gg_debug_probe_reg_arg value = 0 }
		set_global_variable = { name = gg_debug_probe_reg_enr value = 0 }
		set_global_variable = { name = gg_debug_probe_bn_crr value = 0 }
		set_global_variable = { name = gg_debug_probe_brk_enr value = 0 }
		set_global_variable = { name = gg_debug_probe_cc_enr value = 0 }
		set_global_variable = { name = gg_debug_probe_want_enr value = 0 }
		set_global_variable = { name = gg_debug_probe_strat_enr value = 0 }
		set_global_variable = { name = gg_debug_probe_gold_enr value = 0 }
		set_global_variable = { name = gg_debug_probe_debt_enr value = 0 }
		set_global_variable = { name = gg_debug_probe_bn_eg value = 0 }
		set_global_variable = { name = gg_debug_probe_bn_brzdiv value = 0 }
		set_global_variable = { name = gg_debug_probe_bn_oridiv value = 0 }
		# EN/ES: 26/9/2026 (AUD-PACT-ROSAS-SMALL) -- Buenos Aires's barracks and conscription centres: REG_ARG is 0 in four of
		#     the six caseros runs on b690118 / cuarteles y centros de conscripcion de Buenos Aires: REG_ARG es 0 en cuatro de seis
		set_global_variable = { name = gg_debug_probe_brk_arg value = 0 }
		set_global_variable = { name = gg_debug_probe_cc_arg value = 0 }
		c:ARG ?= {
			every_scope_building = {
				limit = { is_building_type = building_barrack }
				change_global_variable = { name = gg_debug_probe_brk_arg add = this.level }
			}
			every_scope_building = {
				limit = { is_building_type = building_conscription_center }
				change_global_variable = { name = gg_debug_probe_cc_arg add = this.level }
			}
		}
		every_country = {
			limit = {
				gg_pact_rosas_side = yes
				has_war_with = c:ENR
			}
			change_global_variable = { name = gg_debug_probe_reg_arg add = army_size }
		}
		c:CRR ?= {
			if = {
				limit = { has_war_with = c:ARG }
				set_global_variable = { name = gg_debug_probe_bn_crr value = army_size_including_raised_conscripts }
			}
		}
		c:ENR = {
			set_global_variable = { name = gg_debug_probe_reg_enr value = army_size }
			every_scope_building = {
				limit = { is_building_type = building_barrack }
				change_global_variable = { name = gg_debug_probe_brk_enr add = this.level }
			}
			every_scope_building = {
				limit = { is_building_type = building_conscription_center }
				change_global_variable = { name = gg_debug_probe_cc_enr add = this.level }
			}
			set_global_variable = { name = gg_debug_probe_want_enr value = wanted_army_size_script_value }
			# EN/ES: the province's share only if its strategy is on (Codex review) / la parte de la provincia solo con su estrategia
			if = {
				limit = { has_strategy = gg_ai_strategy_entre_rios }
				set_global_variable = { name = gg_debug_probe_strat_enr value = 1 }
				change_global_variable = { name = gg_debug_probe_want_enr add = gg_pact_field_army_wanted }
			}
			set_global_variable = { name = gg_debug_probe_gold_enr value = gold_reserves }
			set_global_variable = { name = gg_debug_probe_debt_enr value = scaled_debt }
			change_global_variable = { name = gg_debug_probe_debt_enr multiply = 100 }
		}
		every_in_global_list = {
			variable = gg_eg_probe_ejercito_grande
			every_combat_unit = { change_global_variable = { name = gg_debug_probe_bn_eg add = 1 } }
		}
		every_in_global_list = {
			variable = gg_eg_probe_brazil
			every_combat_unit = { change_global_variable = { name = gg_debug_probe_bn_brzdiv add = 1 } }
		}
		every_in_global_list = {
			variable = gg_eg_probe_oriental
			every_combat_unit = { change_global_variable = { name = gg_debug_probe_bn_oridiv add = 1 } }
		}
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_war_probe| ARMY REG_ARG=[GetGlobalVariable('gg_debug_probe_reg_arg').GetValue|0] REG_ENR=[GetGlobalVariable('gg_debug_probe_reg_enr').GetValue|0] CRR=[GetGlobalVariable('gg_debug_probe_bn_crr').GetValue|0] BRK_ENR=[GetGlobalVariable('gg_debug_probe_brk_enr').GetValue|0] CC_ENR=[GetGlobalVariable('gg_debug_probe_cc_enr').GetValue|0] WANT_ENR=[GetGlobalVariable('gg_debug_probe_want_enr').GetValue|0] STRAT_ENR=[GetGlobalVariable('gg_debug_probe_strat_enr').GetValue|0] GOLD_ENR=[GetGlobalVariable('gg_debug_probe_gold_enr').GetValue|0] DEBT_ENR=[GetGlobalVariable('gg_debug_probe_debt_enr').GetValue|0] EG=[GetGlobalVariable('gg_debug_probe_bn_eg').GetValue|0] BRZDIV=[GetGlobalVariable('gg_debug_probe_bn_brzdiv').GetValue|0] ORIDIV=[GetGlobalVariable('gg_debug_probe_bn_oridiv').GetValue|0] BRK_ARG=[GetGlobalVariable('gg_debug_probe_brk_arg').GetValue|0] CC_ARG=[GetGlobalVariable('gg_debug_probe_cc_arg').GetValue|0] " # GG_TRAZA
	}
}

# EN: 26/9/2026 (Juan, AUD-PACT-ARMY-MELTS) -- the province's AI takes its own strategy, whose wanted army is the army it has
#     while the war of the Pronouncement lasts (gg_pact_field_army_wanted). The AI re-picks strategies rarely and weight alone
#     is not enough (gg_rio_plata.txt, verified 21/9/2026), so the script sets it: when the field army musters, when the
#     Ejercito Grande is raised and every week of the war (gg_caseros.9, under every rule of the Pact). Country scope (Entre Rios or Corrientes).
# ES: la IA de la provincia toma su estrategia, cuyo ejercito deseado es el que tiene mientras dura la guerra. La IA re-elige
#     estrategias pocas veces y el peso solo no alcanza, asi que la fija el script: al reunirse el ejercito de campania, al
#     levantarse el Ejercito Grande y cada mes de la guerra. Scope de pais.
gg_pact_army_keep = {
	# EN/ES: never a province Rosas made his subject; the war mark, not has_war_with, which the
	#     war declared in the same effect may not have set yet / nunca una provincia sujeta; la marca de la guerra, no
	#     has_war_with, que la guerra declarada en el mismo efecto puede no haber puesto todavia
	if = {
		limit = {
			is_player = no
			is_subject = no
			has_global_variable = gg_war_against_buenos_aires
			c:ENR ?= this
			NOT = { has_strategy = gg_ai_strategy_entre_rios }
		}
		set_strategy = gg_ai_strategy_entre_rios
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_army_keep| STRATEGY=entre_rios " # GG_TRAZA
	}
	else_if = {
		limit = {
			is_player = no
			is_subject = no
			has_global_variable = gg_war_against_buenos_aires
			c:CRR ?= this
			NOT = { has_strategy = gg_ai_strategy_corrientes }
		}
		set_strategy = gg_ai_strategy_corrientes
		debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_pact_army_keep| STRATEGY=corrientes " # GG_TRAZA
	}
}
