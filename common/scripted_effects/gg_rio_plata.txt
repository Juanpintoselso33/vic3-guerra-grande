# EN: THE RIVER PLATE QUESTION, effects (rio-plata-spec.md). What the script does is small on
#     purpose: it tells Rosas's AI what to do with his bank, touches the fort when history does,
#     counts months, and opens the intervention when the powers run out of patience. Everything else
#     -- who can pass, what trade is lost, what London and Paris think of a shut river -- is the
#     engine's.
# ES: LA CUESTION DEL PLATA, efectos. Lo que hace el script es chico a proposito: le dice a la IA de
#     Rosas que hacer con su orilla, toca el fuerte cuando lo hace la historia, cuenta meses y abre la
#     intervencion cuando a las potencias se les acaba la paciencia. Todo lo demas -- quien pasa, que
#     comercio se pierde, que piensan Londres y Paris de un rio cerrado -- lo pone el motor.

# EN: Rosas's stance, remembered as a global so the monthly pulse can put it back if the AI re-rolls
#     its strategies. 0 open, 1 closed to every foreign flag, 2 closed to enemies only (Purvis).
#     A human Rosas sets his bank himself in the strait panel; the stance still records his word.
# ES: La postura de Rosas, guardada como global para que el pulso mensual la reponga si la IA
#     re-elige estrategias. 0 abierto, 1 cerrado a toda bandera, 2 cerrado solo a enemigos (Purvis).
#     Un Rosas humano fija su orilla en el panel del estrecho; la postura igual registra su palabra.
gg_plata_rosas_stance_apply = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_plata_rosas_stance_apply|" # GG_TRAZA
	c:ARG ?= {
		if = {
			limit = { is_player = no }
			if = {
				limit = { global_var:gg_plata_rosas_stance = 1 NOT = { has_strategy = gg_ai_strategy_plata_closed } }
				set_strategy = gg_ai_strategy_plata_closed
			}
			else_if = {
				limit = { global_var:gg_plata_rosas_stance = 2 NOT = { has_strategy = gg_ai_strategy_plata_enemies } }
				set_strategy = gg_ai_strategy_plata_enemies
			}
			else_if = {
				limit = { global_var:gg_plata_rosas_stance = 0 NOT = { has_strategy = gg_ai_strategy_plata_open } }
				set_strategy = gg_ai_strategy_plata_open
			}
		}
	}
}

# EN: A. The fort is the key. Removing it lifts every restriction on the Buenos Aires bank.
# ES: A. El fuerte es la llave. Sacarlo levanta toda restriccion en la orilla de Buenos Aires.

# EN: Rosas returns the passports: the intervention begins. Its dated events (Brown's squadron, the
#     blockade of Buenos Aires, Hood, Howden, Gore and Gros, Southern, Lepredour) wait for this and
#     for their own distance from it (gg_plata_intervention_at).
# ES: Rosas devuelve los pasaportes: empieza la intervencion. Sus eventos fechados esperan esto y su
#     propia distancia desde aca.
gg_plata_intervention_start = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_plata_intervention_start|" # GG_TRAZA
	# EN/ES: 22/9/2026 -- London out of the river (Southern, or not standing behind the Defensa by its
	#     level): France intervenes alone / Londres fuera del rio (Southern, o sin respaldar a la
	#     Defensa por su nivel): Francia interviene sola
	if = {
		limit = { gg_plata_london_out = yes }
		custom_tooltip = gg_plata_intervention_start_france_tt
	}
	else = {
		custom_tooltip = gg_plata_intervention_start_tt
	}
	hidden_effect = {
		if = {
			limit = { has_global_variable = gg_plata_ultimatum_pending }
			remove_global_variable = gg_plata_ultimatum_pending
		}
		# EN/ES: decided BEFORE the intervention mark, which makes the intervening powers Guarantors
		#     (Codex 22/9) / se decide ANTES de la marca, que vuelve Garantes a las potencias que intervienen
		if = {
			limit = { gg_plata_london_out = yes }
			set_global_variable = gg_plata_france_alone
		}
		set_global_variable = gg_plata_intervention
		set_global_variable = { name = gg_plata_intervention_months value = 0 }
		# EN/ES: a broken guarantee brings its own notice (gg_plata.26) / una garantia rota trae su aviso
		if = {
			limit = { NOT = { has_global_variable = gg_plata_word_broken } }
			gg_plata_notify = { EVENT = gg_plata.23 }
		}
		# EN: Codex review 22/9 -- the clock lives in its own hidden event, not in the war journal: a truce
		#     or the end of the war must not stop it. 23/9/2026 (B1, run 214510) -- it ran on Britain only,
		#     and a British revolution (10/1/1847) replaced the country object and dropped the queued tick:
		#     the clock stopped at month 31 for good. Now it runs on Rosas's Argentina too (a tag that never
		#     changes), counting once a month through gg_plata_clock_ticked, and gg_plata_situation_monthly
		#     re-arms it if both copies ever die (gg_plata_clock_alive).
		# ES: el reloj vive en su propio evento oculto. Corria solo sobre Gran Bretania, y una revolucion
		#     britanica reemplazo el objeto pais y se llevo el tic agendado: el reloj quedo en 31 para
		#     siempre. Ahora corre tambien sobre la Argentina de Rosas, cuenta una vez por mes, y el pulso
		#     del Plata lo rearma si mueren las dos copias.
		set_global_variable = { name = gg_plata_clock_alive days = 65 }
		c:ARG ?= { trigger_event = { id = gg_plata.4 days = 30 } }
		c:GBR ?= { trigger_event = { id = gg_plata.4 days = 30 } }
		# EN: Codex review 22/9 -- decision B lets the intervention come BEFORE 1845, so its links are
		#     also scheduled from here, at their historical distance from the ultimatum. The dated
		#     copies stay: whichever arrives first fires (every relay has its DONE mark).
		# ES: la decision B deja que la intervencion llegue ANTES de 1845, asi que sus eslabones se
		#     agendan tambien desde aca, a su distancia historica del ultimatum. Las copias fechadas
		#     quedan: sale la que llegue primero (cada relevo tiene su marca DONE).
		# EN/ES: 23/9/2026 (Codex, B1) -- on Rosas's Argentina, not Britain: a British revolution dropped these copies
		#     too, and a late intervention has no dated copy left after 1856 / en la Argentina de Rosas, no en Gran Bretania
		c:ARG ?= {
			trigger_event = { id = gg_cal.31 days = 60 }     # the blockade of Buenos Aires, 18/9/1845
			trigger_event = { id = gg_cal.25 days = 1370 }   # Lepredour, August 1850
			# EN/ES: the British and joint links, only when London is in / los eslabones britanicos y
			#     conjuntos, solo con Londres adentro
			if = {
				limit = { NOT = { has_global_variable = gg_plata_france_alone } }
				trigger_event = { id = gg_cal.10 days = 1 }      # Brown, 2/8/1845
				trigger_event = { id = gg_cal.12 days = 365 }    # Hood, July 1846
				trigger_event = { id = gg_cal.15 days = 700 }    # Howden, July 1847
				trigger_event = { id = gg_cal.19 days = 940 }    # Gore and Gros, March 1848
				trigger_event = { id = gg_cal.23 days = 1095 }   # Southern, November 1849
			}
		}
		c:ARG ?= {
			if = {
				limit = {
					exists = s:STATE_ENTRE_RIOS
					NOT = { has_global_variable = gg_plata_france_alone }
				}
				trigger_event = { id = gg_argentina_cal.21 days = 60 }    # the blockade from Buenos Aires
				trigger_event = { id = gg_argentina_cal.22 days = 120 }   # Obligado
				trigger_event = { id = gg_argentina_cal.23 days = 310 }   # Quebracho
				trigger_event = { id = gg_argentina_cal.24 days = 365 }   # Hood from Buenos Aires
			}
		}
		c:ARG ?= {
			save_scope_as = gg_plata_rosas
			# EN/ES: London is out once it has recognised the right of blockade / Londres ya no esta si reconocio el bloqueo
			if = {
				limit = { NOT = { has_global_variable = gg_plata_france_alone } }
				c:GBR ?= { change_relations = { country = scope:gg_plata_rosas value = -30 } }
			}
			c:FRA ?= { change_relations = { country = scope:gg_plata_rosas value = -30 } }
		}
	}
}

# EN: Rosas yields (22/9/2026, the powers' guarantee): the river opens, London and Paris stand as
#     its guarantors, and the Cerrito loses the Argentine backing the blockade had bought. Under the
#     guarantee the Cerrito cannot ask for the closure, and if Rosas shuts the river anyway the
#     squadrons come at once (gg_plata.26). It ends with Southern without a war (gg_plata.27).
# ES: Rosas cede: se abre el rio, Londres y Paris quedan como garantes, y el Cerrito pierde el apoyo
#     argentino que le habia dado el bloqueo. Bajo la garantia el Cerrito no puede pedir el cierre,
#     y si Rosas cierra igual, las escuadras vienen enseguida. Termina con Southern sin guerra.
gg_plata_ultimatum_yield = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_plata_ultimatum_yield|" # GG_TRAZA
	gg_plata_rosas_open = yes
	# EN/ES: no guarantee once the war is over: nothing would be left to end it
	#     / sin garantia con la guerra terminada: no quedaria nada que la termine
	if = {
		limit = { NOT = { has_global_variable = gg_peace_settled } }
		if = {
			limit = { gg_plata_london_out = yes }
			custom_tooltip = gg_plata_guarantee_france_tt
		}
		else = {
			custom_tooltip = gg_plata_guarantee_tt
		}
	}
	c:CER ?= {
		if = {
			limit = { has_modifier = guerra_grande_rosista_support }
			custom_tooltip = gg_plata_yield_cerrito_tt
			remove_modifier = guerra_grande_rosista_support
			hidden_effect = { set_global_variable = { name = gg_plata_yield_cost_cerrito days = 60 } }
		}
	}
	hidden_effect = {
		if = {
			limit = { has_global_variable = gg_plata_ultimatum_pending }
			remove_global_variable = gg_plata_ultimatum_pending
		}
		set_global_variable = { name = gg_plata_closed_total value = 0 }
		# EN: River audit 22/9 -- the notice mark is left alone. Setting it to «open» here announced a
		#     closure (gg_plata.20) the next month whenever the shore had not opened yet -- an AI takes
		#     weeks to apply the strategy, a human has two months -- although Rosas had just yielded.
		#     gg_plata.25 announces the pledge; gg_plata.21 announces the opening when it happens.
		# ES: la marca del aviso no se toca. Ponerla en «abierto» aca anunciaba un cierre (gg_plata.20) al
		#     mes siguiente si la orilla todavia no se habia abierto, aunque Rosas acababa de ceder.
		#     gg_plata.25 anuncia la promesa; gg_plata.21 anuncia la apertura cuando ocurre.
		if = {
			limit = { NOT = { has_global_variable = gg_peace_settled } }
			# EN/ES: 22/9/2026 -- London out of the river when Rosas gives way: Paris alone guarantees it.
			#     Decided BEFORE the guarantee mark, which makes the guarantors Guarantors (Codex 22/9)
			#     / Londres fuera: garantiza solo Paris; se decide ANTES de la marca de garantia
			if = {
				limit = {
					gg_plata_london_out = yes
					NOT = { has_global_variable = gg_plata_britain_recognizes }
				}
				set_global_variable = gg_plata_guarantee_france_only
			}
			set_global_variable = gg_plata_guarantee
			# EN/ES: two months for a human Rosas to open his shore in the strait panel before a closure
			#     counts as a broken word / dos meses para que un Rosas jugador abra su orilla en el panel
			set_global_variable = { name = gg_plata_guarantee_grace days = 60 }
			gg_plata_notify = { EVENT = gg_plata.25 }
		}
	}
}

# EN: Once a month, from je_guerra_grande's pulse while the war is open. Frozen in a truce.
# ES: Una vez por mes, desde el pulso de je_guerra_grande con la guerra abierta. Congelado en tregua.
gg_plata_monthly = {
	debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_plata_monthly|" # GG_TRAZA
	gg_plata_fort_french_supply = yes
	# EN/ES: the stance survives the AI's own re-roll / la postura sobrevive a la re-eleccion de la IA
	gg_plata_rosas_stance_apply = yes
	every_country = {
		limit = {
			has_variable = guerra_grande_defensa_government
			is_player = no
			NOT = { has_strategy = gg_ai_strategy_plata_defensa }
		}
		set_strategy = gg_ai_strategy_plata_defensa
	}

	# EN/ES: 22/9/2026 -- nor during the two months a human Rosas has to open his shore after yielding:
	#     that closure is a promise pending, not a blockade / ni durante los dos meses de gracia
	if = {
		limit = {
			NOT = { has_global_variable = gg_truce }
			NOT = { has_global_variable = gg_plata_guarantee_grace }
		}
		if = {
			limit = { gg_plata_closed = yes }
			# EN/ES: 23/9/2026 (Codex) -- only a month already at the cap goes past it: scored before the counter moves
			#     / solo un mes que ya estaba en el techo pasa de el: se puntua antes de mover el contador
			if = { limit = { global_var:gg_plata_months >= 16 } gg_siege_river_scores = { STEP = 1 } }
			if = {
				limit = { global_var:gg_plata_months < 16 }
				change_global_variable = { name = gg_plata_months add = 1 }
			}
			# EN/ES: 23/9/2026 -- the run of hunger holds while the powers stand by the city (gg_plata_hunger_held)
			#     / la racha del hambre se sostiene mientras las potencias sostienen la plaza
			if = {
				limit = { NOT = { gg_plata_hunger_held = yes } }
				change_global_variable = { name = gg_plata_closed_streak add = 1 }
			}
			# EN/ES: 23/9 -- in the siege, what goes past the cap stays in the score (scored above, before the counter moves)
			#     / en el sitio lo que pasa del techo queda en el puntaje (se puntua arriba)
			# EN/ES: 22/9/2026 (Codex) -- the patience only runs while some power counts by its level: no
			#     months banked unseen / la paciencia corre solo si alguna potencia cuenta por su nivel
			if = {
				limit = {
					OR = {
						gg_plata_france_counts = yes
						gg_plata_britain_counts = yes
						# EN/ES: 23/9/2026 (D3) -- after the settlement, from 1850, the Empire counts
						#     / despues del arreglo, desde 1850, cuenta el Imperio
						gg_plata_brazil_counts = yes
					}
				}
				change_global_variable = { name = gg_plata_closed_total add = 1 }
			}
			# EN/ES: 23/9/2026 (D3) -- a closure after the settlement costs the Littoral's goodwill
			#     / un cierre despues del arreglo le cuesta al Litoral
			if = {
				limit = { gg_plata_settled = yes }
				gg_plata_littoral_cost = yes
			}
		}
		else = {
			if = { limit = { global_var:gg_plata_months <= -16 } gg_siege_river_scores = { STEP = -1 } }
			if = {
				limit = { global_var:gg_plata_months > -16 }
				change_global_variable = { name = gg_plata_months add = -1 }
			}
			set_global_variable = { name = gg_plata_closed_streak value = 0 }
		}
		if = {
			limit = { gg_plata_intervention_due = yes }
			gg_plata_ultimatum_send = yes
		}
		# EN/ES: 23/9/2026 (D3) -- the Empire's patience is spent: its squadron forces the passage
		#     / se acabo la paciencia del Imperio: su escuadra fuerza el paso
		if = {
			limit = { gg_plata_brazil_answer_due = yes }
			gg_plata_brazil_forces = yes
		}
	}
	# EN: Codex review 22/9 -- once Rosas has opened the river (he yielded, or Brown's option a), the
	#     Cerrito may ask again, as the old system allowed when the sea opened.
	# ES: cuando Rosas abrio el rio (cedio, o la opcion a de Brown), el Cerrito puede volver a pedir.
	# EN/ES: 22/9/2026 -- not under the powers' guarantee: then the request stays spent
	#     / no bajo la garantia de las potencias: ahi el pedido queda gastado
	if = {
		limit = {
			global_var:gg_plata_rosas_stance = 0
			gg_plata_open = yes
			NOT = { has_global_variable = gg_plata_guarantee }
		}
		c:CER ?= {
			if = {
				limit = { has_variable = guerra_grande_river_blockade_asked }
				remove_variable = guerra_grande_river_blockade_asked
			}
		}
	}

	# EN: FN-2 -- while the powers intervene, the Defensa sails a lent squadron; when the
	#     intervention ends they take their ships back. (Before 22/9/2026 this hung off the rejected
	#     60-point sea, together with a blockade-resistance modifier that is gone.)
	# ES: FN-2 -- mientras intervienen las potencias, la Defensa navega una escuadra prestada; cuando
	#     termina, se llevan sus barcos. (Antes colgaba del mar de 60 puntos, con un modificador de
	#     resistencia al bloqueo que ya no existe.)
	if = {
		limit = {
			has_global_variable = gg_plata_intervention
			NOT = { has_global_variable = gg_plata_intervention_over }
		}
		c:URU ?= {
			# EN/ES: S02 migration -- a dead formation cannot keep the creation guard true.
			if = {
				limit = {
					has_variable = gg_lent_squadron
					OR = {
						NOT = { has_variable = gg_lent_squadron_ref }
						NOT = { exists = var:gg_lent_squadron_ref }
					}
				}
				remove_variable = gg_lent_squadron
				if = { limit = { has_variable = gg_lent_squadron_ref } remove_variable = gg_lent_squadron_ref }
			}
			if = {
				limit = { NOT = { has_variable = gg_lent_squadron } }
				gg_select_formation_hq_state = yes
				clear_saved_scope = gg_lent_fleet
				# EN/ES: without an owned state there is no valid HQ / sin estado propio no hay cuartel
				if = {
					limit = { exists = scope:gg_formation_hq_state }
					create_military_formation = {
						save_scope_as = gg_lent_fleet
						type = fleet
						hq_region = scope:gg_formation_hq_state.region
						name = gg_lent_squadron_name
						ship = {
							type = ship_type:ship_type_frigate
							state_region = s:STATE_MONTEVIDEO
							count = 4
						}
					}
				}
				# EN/ES: the marks go after the creation, so a failed one is retried next month
				#     / las marcas van despues de crearla, asi un fallo se reintenta el mes que viene
				if = {
					limit = { exists = scope:gg_lent_fleet }
					set_variable = gg_lent_squadron
					set_variable = { name = gg_lent_squadron_ref value = scope:gg_lent_fleet }
				}
			}
			# EN: 22/9/2026 (Juan) -- France keeps the ships it lends: while the squadron sails for the
			#     Defensa, its naval goods cost the city nothing (the closest effect in 1.13: there is no
			#     per-formation upkeep; naval administration wages are not covered). Renewed monthly.
			# ES: Francia sostiene los barcos que presta: mientras la escuadra navega para la Defensa,
			#     sus bienes navales no le cuestan nada a la plaza (lo mas cercano en 1.13). Mensual.
			if = {
				limit = { has_variable = gg_lent_squadron }
				add_modifier = { name = gg_lent_squadron_french_supply months = 2 }
			}
			# EN/ES: Codex 22/9 -- a sunk squadron that could not be raised again takes the supply with it
			#     / una escuadra hundida que no pudo rehacerse se lleva el sosten
			else_if = {
				limit = { has_modifier = gg_lent_squadron_french_supply }
				remove_modifier = gg_lent_squadron_french_supply
			}
		}
	}
	else = {
		c:URU ?= {
			# EN/ES: Codex 22/9 -- France's supply leaves with her ships / el sosten frances se va con sus barcos
			if = { limit = { has_modifier = gg_lent_squadron_french_supply } remove_modifier = gg_lent_squadron_french_supply }
			if = {
