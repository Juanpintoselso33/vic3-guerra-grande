# EN: THE RIVER PLATE QUESTION, effects (rio-plata-spec.md). What the script does is small on
#     purpose: it tells Rosas's AI what to do with his bank, touches the fort when history does,
#     counts months, and opens the intervention when the powers run out of patience. Everything else
#     -- who can pass, what trade is lost, what London and Paris think of a shut river -- is the
#     engine's.
# ES: LA CUESTION DEL PLATA, efectos. Lo que hace el script es chico a proposito: le dice a la IA de
#     Rosas que hacer con su orilla, toca el fuerte cuando lo hace la historia, cuenta meses y abre la
#     intervencion cuando a las potencias se les acaba la paciencia. Todo lo demas -- quien pasa, que
#     comercio se pierde, que piensan Londres y Paris de un rio cerrado -- lo pone el motor.

# EN: Rosas's stance, remembered as a global so the monthly pulse can put it back if the AI re-rolls
#     its strategies. 0 open, 1 closed to every foreign flag, 2 closed to enemies only (Purvis).
#     A human Rosas sets his bank himself in the strait panel; the stance still records his word.
# ES: La postura de Rosas, guardada como global para que el pulso mensual la reponga si la IA
#     re-elige estrategias. 0 abierto, 1 cerrado a toda bandera, 2 cerrado solo a enemigos (Purvis).
#     Un Rosas humano fija su orilla en el panel del estrecho; la postura igual registra su palabra.
gg_plata_rosas_stance_apply = {
	c:ARG ?= {
		if = {
			limit = { is_player = no }
			if = {
				limit = { global_var:gg_plata_rosas_stance = 1 NOT = { has_strategy = gg_ai_strategy_plata_closed } }
				set_strategy = gg_ai_strategy_plata_closed
			}
			else_if = {
				limit = { global_var:gg_plata_rosas_stance = 2 NOT = { has_strategy = gg_ai_strategy_plata_enemies } }
				set_strategy = gg_ai_strategy_plata_enemies
			}
			else_if = {
				limit = { global_var:gg_plata_rosas_stance = 0 NOT = { has_strategy = gg_ai_strategy_plata_open } }
				set_strategy = gg_ai_strategy_plata_open
			}
		}
	}
}

gg_plata_rosas_close = {
	custom_tooltip = gg_plata_rosas_close_tt
	set_global_variable = { name = gg_plata_rosas_stance value = 1 }
	gg_plata_rosas_stance_apply = yes
}
gg_plata_rosas_open = {
	custom_tooltip = gg_plata_rosas_open_tt
	set_global_variable = { name = gg_plata_rosas_stance value = 0 }
	gg_plata_rosas_stance_apply = yes
}
gg_plata_rosas_neutrals_pass = {
	custom_tooltip = gg_plata_rosas_neutrals_tt
	set_global_variable = { name = gg_plata_rosas_stance value = 2 }
	gg_plata_rosas_stance_apply = yes
}

# EN: A. The fort is the key. Removing it lifts every restriction on the Buenos Aires bank.
# ES: A. El fuerte es la llave. Sacarlo levanta toda restriccion en la orilla de Buenos Aires.
gg_plata_fort_remove = {
	custom_tooltip = gg_plata_fort_remove_tt
	p:x365A5B.state ?= {
		if = {
			limit = { has_building = building_naval_fortification }
			remove_building = building_naval_fortification
		}
	}
}
gg_plata_fort_restore = {
	custom_tooltip = gg_plata_fort_restore_tt
	p:x365A5B.state ?= {
		if = {
			limit = { NOT = { has_building = building_naval_fortification } }
			create_building = {
				building = "building_naval_fortification"
				level = 1
			}
		}
	}
}

# EN: B. The powers' patience is spent: Ouseley and Deffaudis carry the ultimatum to Rosas.
# ES: B. Se acabo la paciencia: Ouseley y Deffaudis le llevan el ultimatum a Rosas.
gg_plata_ultimatum_send = {
	set_global_variable = gg_plata_ultimatum_pending
	c:ARG ?= { trigger_event = { id = gg_plata.10 popup = yes } }
}

# EN: Rosas returns the passports: the intervention begins. Its dated events (Brown's squadron, the
#     blockade of Buenos Aires, Hood, Howden, Gore and Gros, Southern, Lepredour) wait for this and
#     for their own distance from it (gg_plata_intervention_at).
# ES: Rosas devuelve los pasaportes: empieza la intervencion. Sus eventos fechados esperan esto y su
#     propia distancia desde aca.
gg_plata_intervention_start = {
	custom_tooltip = gg_plata_intervention_start_tt
	hidden_effect = {
		if = {
			limit = { has_global_variable = gg_plata_ultimatum_pending }
			remove_global_variable = gg_plata_ultimatum_pending
		}
		set_global_variable = gg_plata_intervention
		set_global_variable = { name = gg_plata_intervention_months value = 0 }
		# EN: Codex review 22/9 -- the clock lives in its own hidden event on Britain (a tag that never
		#     changes), not in the war journal: a truce or the end of the war must not stop it.
		# ES: el reloj vive en su propio evento oculto sobre Gran Bretania (un tag que nunca cambia), no
		#     en el journal de la guerra: ni una tregua ni el fin de la guerra lo pueden frenar.
		c:GBR ?= { trigger_event = { id = gg_plata.4 days = 30 } }
		# EN: Codex review 22/9 -- decision B lets the intervention come BEFORE 1845, so its links are
		#     also scheduled from here, at their historical distance from the ultimatum. The dated
		#     copies stay: whichever arrives first fires (every relay has its DONE mark).
		# ES: la decision B deja que la intervencion llegue ANTES de 1845, asi que sus eslabones se
		#     agendan tambien desde aca, a su distancia historica del ultimatum. Las copias fechadas
		#     quedan: sale la que llegue primero (cada relevo tiene su marca DONE).
		c:GBR ?= {
			trigger_event = { id = gg_cal.10 days = 1 }      # Brown, 2/8/1845
			trigger_event = { id = gg_cal.31 days = 60 }     # the blockade of Buenos Aires, 18/9/1845
			trigger_event = { id = gg_cal.12 days = 365 }    # Hood, July 1846
			trigger_event = { id = gg_cal.15 days = 700 }    # Howden, July 1847
			trigger_event = { id = gg_cal.19 days = 940 }    # Gore and Gros, March 1848
			trigger_event = { id = gg_cal.23 days = 1095 }   # Southern, November 1849
			trigger_event = { id = gg_cal.25 days = 1370 }   # Lepredour, August 1850
		}
		c:ARG ?= {
			if = {
				limit = { exists = s:STATE_ENTRE_RIOS }
				trigger_event = { id = gg_argentina_cal.21 days = 60 }    # the blockade from Buenos Aires
				trigger_event = { id = gg_argentina_cal.22 days = 120 }   # Obligado
				trigger_event = { id = gg_argentina_cal.23 days = 310 }   # Quebracho
				trigger_event = { id = gg_argentina_cal.24 days = 365 }   # Hood from Buenos Aires
			}
		}
		c:ARG ?= {
			save_scope_as = gg_plata_rosas
			c:GBR ?= { change_relations = { country = scope:gg_plata_rosas value = -30 } }
			c:FRA ?= { change_relations = { country = scope:gg_plata_rosas value = -30 } }
		}
	}
}

# EN: Rosas yields: the river opens and the patience starts again from zero.
# ES: Rosas cede: el rio se abre y la paciencia vuelve a empezar de cero.
gg_plata_ultimatum_yield = {
	gg_plata_rosas_open = yes
	hidden_effect = {
		if = {
			limit = { has_global_variable = gg_plata_ultimatum_pending }
			remove_global_variable = gg_plata_ultimatum_pending
		}
		set_global_variable = { name = gg_plata_closed_total value = 0 }
	}
}

# EN: Once a month, from je_guerra_grande's pulse while the war is open. Frozen in a truce.
# ES: Una vez por mes, desde el pulso de je_guerra_grande con la guerra abierta. Congelado en tregua.
gg_plata_monthly = {
	# EN/ES: the stance survives the AI's own re-roll / la postura sobrevive a la re-eleccion de la IA
	gg_plata_rosas_stance_apply = yes
	every_country = {
		limit = {
			has_variable = guerra_grande_defensa_government
			is_player = no
			NOT = { has_strategy = gg_ai_strategy_plata_defensa }
		}
		set_strategy = gg_ai_strategy_plata_defensa
	}

	if = {
		limit = { NOT = { has_global_variable = gg_truce } }
		if = {
			limit = { gg_plata_closed = yes }
			if = {
				limit = { global_var:gg_plata_months < 16 }
				change_global_variable = { name = gg_plata_months add = 1 }
			}
			change_global_variable = { name = gg_plata_closed_streak add = 1 }
			change_global_variable = { name = gg_plata_closed_total add = 1 }
		}
		else = {
			if = {
				limit = { global_var:gg_plata_months > -16 }
				change_global_variable = { name = gg_plata_months add = -1 }
			}
			set_global_variable = { name = gg_plata_closed_streak value = 0 }
		}
		if = {
			limit = { gg_plata_intervention_due = yes }
			gg_plata_ultimatum_send = yes
		}
	}
	# EN: Codex review 22/9 -- once Rosas has opened the river (he yielded, or Brown's option a), the
	#     Cerrito may ask again, as the old system allowed when the sea opened.
	# ES: cuando Rosas abrio el rio (cedio, o la opcion a de Brown), el Cerrito puede volver a pedir.
	if = {
		limit = {
			global_var:gg_plata_rosas_stance = 0
			gg_plata_open = yes
		}
		c:CER ?= {
			if = {
				limit = { has_variable = guerra_grande_river_blockade_asked }
				remove_variable = guerra_grande_river_blockade_asked
			}
		}
	}

	# EN: FN-2 -- while the powers intervene, the Defensa sails a lent squadron; when the
	#     intervention ends they take their ships back. (Before 22/9/2026 this hung off the rejected
	#     60-point sea, together with a blockade-resistance modifier that is gone.)
	# ES: FN-2 -- mientras intervienen las potencias, la Defensa navega una escuadra prestada; cuando
	#     termina, se llevan sus barcos. (Antes colgaba del mar de 60 puntos, con un modificador de
	#     resistencia al bloqueo que ya no existe.)
	if = {
		limit = {
			has_global_variable = gg_plata_intervention
			NOT = { has_global_variable = gg_plata_intervention_over }
		}
		c:URU ?= {
			# EN/ES: S02 migration -- a dead formation cannot keep the creation guard true.
			if = {
				limit = {
					has_variable = gg_lent_squadron
					OR = {
						NOT = { has_variable = gg_lent_squadron_ref }
						NOT = { exists = var:gg_lent_squadron_ref }
					}
				}
				remove_variable = gg_lent_squadron
				if = { limit = { has_variable = gg_lent_squadron_ref } remove_variable = gg_lent_squadron_ref }
			}
			if = {
				limit = { NOT = { has_variable = gg_lent_squadron } }
				gg_select_formation_hq_state = yes
				clear_saved_scope = gg_lent_fleet
				# EN/ES: without an owned state there is no valid HQ / sin estado propio no hay cuartel
				if = {
					limit = { exists = scope:gg_formation_hq_state }
					create_military_formation = {
						save_scope_as = gg_lent_fleet
						type = fleet
						hq_region = scope:gg_formation_hq_state.region
						name = gg_lent_squadron_name
						ship = {
							type = ship_type:ship_type_frigate
							state_region = s:STATE_MONTEVIDEO
							count = 4
						}
					}
				}
				# EN/ES: the marks go after the creation, so a failed one is retried next month
				#     / las marcas van despues de crearla, asi un fallo se reintenta el mes que viene
				if = {
					limit = { exists = scope:gg_lent_fleet }
					set_variable = gg_lent_squadron
					set_variable = { name = gg_lent_squadron_ref value = scope:gg_lent_fleet }
				}
			}
		}
	}
	else = {
		c:URU ?= {
			if = {
				limit = { has_variable = gg_lent_squadron }
				remove_variable = gg_lent_squadron
				if = {
					limit = { has_variable = gg_lent_squadron_ref }
					# EN/ES: lent, so they go home to France / prestados, vuelven a Francia
					if = {
						limit = { exists = c:FRA exists = var:gg_lent_squadron_ref }
						var:gg_lent_squadron_ref ?= {
							every_scope_ship = { set_ship_owner_multiple = c:FRA }
						}
						# EN/ES: Codex review 22/9 -- on the country that receives the ships, as vanilla
						#     does / en el pais que recibe los barcos, como vanilla
						c:FRA = { clear_ownership_transfer_fleet = yes }
					}
					remove_variable = gg_lent_squadron_ref
				}
			}
		}
	}
}

# EN: The situation's monthly upkeep (22/9/2026 redesign), from je_gg_patrons: the
#     phases' own effects (two months, renewed, so they fall by themselves when the phase ends), and
#     the faction flags the Overview draws.
# ES: El mantenimiento mensual de la situacion, desde je_gg_patrons: los efectos de
#     cada fase (dos meses, renovados, asi caen solos cuando termina la fase) y las banderas.
gg_plata_situation_monthly = {
	if = {
		limit = { gg_plata_phase = 1 }
		p:x302DDA.state ?= { add_modifier = { name = gg_plata_phase_blockade months = 2 } }
	}
	if = {
		limit = { gg_plata_phase = 2 }
		c:ARG ?= { add_modifier = { name = gg_plata_phase_ultimatum months = 2 } }
	}
	if = {
		limit = { gg_plata_phase = 3 }
		p:x365A5B.state ?= { add_modifier = { name = gg_plata_phase_intervention_port months = 2 } }
		# EN/ES: Britain pays only while its squadron is in the river / Gran Bretana paga solo mientras su
		#     escuadra esta en el rio
		if = {
			limit = { NOT = { has_global_variable = gg_plata_britain_settled } }
			c:GBR ?= { add_modifier = { name = gg_plata_phase_intervention_powers months = 2 } }
		}
		c:FRA ?= { add_modifier = { name = gg_plata_phase_intervention_powers months = 2 } }
	}
	gg_plata_flags_sync = yes
}

# EN: The countries the Overview draws: the two governments (the Cerrito only once it exists; the
#     Defensa falls back on Uruguay before the partition) and who holds each shore of the key. Run at
#     game start and every month, so the journal never reads an unset variable.
# ES: Los paises que dibuja el panel: los dos gobiernos (el Cerrito solo cuando existe; la Defensa
#     cae en el Uruguay antes de la particion) y quien tiene cada orilla de la llave. Corre al empezar
#     y cada mes, asi el journal nunca lee una variable sin fijar.
gg_plata_flags_sync = {
	if = {
		limit = { any_country = { has_variable = guerra_grande_defensa_government } }
		random_country = {
			limit = { has_variable = guerra_grande_defensa_government }
			set_global_variable = { name = gg_flag_defensa value = this }
		}
	}
	else = {
		c:URU ?= { set_global_variable = { name = gg_flag_defensa value = this } }
	}
	if = {
		limit = { any_country = { has_variable = guerra_grande_cerrito_government } }
		random_country = {
			limit = { has_variable = guerra_grande_cerrito_government }
			set_global_variable = { name = gg_flag_cerrito value = this }
		}
	}
	else_if = {
		limit = { has_global_variable = gg_flag_cerrito }
		remove_global_variable = gg_flag_cerrito
	}
	p:x365A5B.state ?= { owner = { set_global_variable = { name = gg_plata_ba_owner value = this } } }
	p:x302DDA.state ?= { owner = { set_global_variable = { name = gg_plata_mvd_owner value = this } } }
}
