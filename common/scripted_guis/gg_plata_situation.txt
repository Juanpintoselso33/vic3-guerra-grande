# EN: Who stands where in the River Plate situation (GUI only; no effects).
# ES: Quien esta donde en la situacion del Plata (solo interfaz; sin efectos).
gg_ui_plata_left_leader = {
	scope = country
	is_shown = { c:ARG ?= this }
}
# EN/ES: before the ultimatum and after the settlement, the Defensa (or the Republic before the split)
#     / antes del ultimatum y despues del arreglo, la Defensa (o la Republica antes de la particion)
gg_ui_plata_right_defensa = {
	scope = country
	is_shown = {
		OR = {
			gg_plata_phase < 2
			gg_plata_phase = 4
		}
		OR = {
			has_variable = guerra_grande_defensa_government
			AND = {
				NOT = { has_global_variable = guerra_grande_split_done }
				c:URU ?= this
			}
		}
	}
}
# EN/ES: from the ultimatum to the settlement, London / del ultimatum al arreglo, Londres
gg_ui_plata_right_powers = {
	scope = country
	is_shown = {
		gg_plata_phase >= 2
		gg_plata_phase < 4
		c:GBR ?= this
	}
}
