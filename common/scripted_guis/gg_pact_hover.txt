# EN: Tooltip-only evaluation; never executes gameplay changes.
# ES: Evaluacion exclusiva de tooltips; no ejecuta cambios en la partida.
gg_pact_hover_jujuy = {
 scope = country
 effect = {
  if = { limit = { has_global_variable = gg_delegation_gone_coalicion_jujuy } custom_tooltip = gg_pact_state_withdrawn }
  else_if = {
   limit = { has_global_variable = gg_reclamo_coalicion_jujuy }
   custom_tooltip = gg_pact_state_claim
   if = {
    limit = { has_global_variable = gg_pact_claim_source_coalicion_jujuy }
   if = { limit = { global_var:gg_pact_claim_source_coalicion_jujuy = 65 } custom_tooltip = gg_pact_source_65 }
   if = { limit = { global_var:gg_pact_claim_source_coalicion_jujuy = 68 } custom_tooltip = gg_pact_source_68 }
   if = { limit = { global_var:gg_pact_claim_source_coalicion_jujuy = 66 } custom_tooltip = gg_pact_source_66 }
   }
  }
  else = { custom_tooltip = gg_pact_state_delegated }
 }
}
