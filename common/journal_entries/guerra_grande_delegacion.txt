# -------------------------------------------------------------------------------------------------
# EN: THE REVOCABLE DELEGATION. The Argentine side's journal, and the same object seen from both
#     sides -- exactly the shape the Defensa and the Cerrito already have. Rosas watches a number he
#     needs to keep up; Corrientes, the Coalition and Entre Rios watch the same number and need it
#     down. One bar, one truth, two readings.
#     WHY IT IS THE HEART. Rosas' power over the other provinces was a delegation of foreign
#     relations that each legislature could take back. Tucuman took it back on 7 April 1840 and Entre
#     Rios on 1 May 1851. Without this, Obligado, Alcaraz, the Pronunciamiento and Caseros are four
#     separate episodes; with it they are one campaign.
# ES: LA DELEGACION REVOCABLE. El journal del lado argentino, y el mismo objeto visto desde los dos
#     lados -- exactamente la forma que ya tienen la Defensa y el Cerrito. Rosas mira un numero que
#     necesita sostener; Corrientes, la Coalicion y Entre Rios miran el mismo numero y lo necesitan
#     abajo. Una barra, una verdad, dos lecturas.
#     POR QUE ES EL CORAZON. El poder de Rosas sobre las demas provincias era una delegacion de las
#     relaciones exteriores que cada legislatura podia retirar. Tucuman la retiro el 7/4/1840 y Entre
#     Rios el 1/5/1851. Sin esto, Obligado, Alcaraz, el Pronunciamiento y Caseros son cuatro
#     episodios sueltos; con esto son una campania.
# -------------------------------------------------------------------------------------------------
je_gg_delegation = {
	group = je_group_guerra_grande
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	widget = {
		gui = "gui/gg_pact_map.gui"
		name = "gg_pact_map"
		container = "custom_widget_container_je_icon"
	}

	should_be_pinned_by_default_involved = yes

	# EN: The map shows mandate, claim and withdrawal separately; no aggregate bar.
	# ES: El mapa distingue mandato, reclamo y retiro; sin barra agregada.

	# EN/ES: Rosas y los tres que le pueden retirar la delegacion / Rosas and the three who can
	#     withdraw it from him
	should_be_involved = {
		OR = {
			has_variable = gg_arg_rosas_seat
			c:CRR ?= this
			c:CLN ?= this
			c:ENR ?= this
		}
	}

	should_show_when_not_involved = {
		always = no
	}

	# EN: Provincial states live on the map; no duplicate list or premature war report.
	# ES: Estados provinciales en el mapa; sin lista duplicada ni parte de guerra prematuro.
	status_desc = {
		first_valid = {
			triggered_desc = { desc = gg_ui_empty trigger = { always = yes } }
		}
	}

	# EN: the siege's monthly cost, and the redraw. Only Argentina runs it: this is a global count,
	#     and running it on every involved country would charge the month once per seat. The cost is
	#     one point -- a twelfth of a province -- for every month Oribe is in front of Montevideo,
	#     which is what nine years of holding him actually did to Rosas.
	# ES: el costo mensual del sitio, y el redibujo. Lo corre solo la Argentina: esto es una cuenta
	#     global, y correrla en cada pais involucrado cobraria el mes una vez por asiento. El costo es
	#     un punto -- un doceavo de provincia -- por cada mes que Oribe este frente a Montevideo, que
	#     es lo que nueve anios de sostenerlo le hicieron a Rosas de verdad.
	# EN: THE RESTORATIONS MOVED TO THE WEEK, and the reason is a screenshot. Juan beat Corrientes,
	#     it capitulated and became a protectorate -- exactly what the design intends -- and the panel
	#     went on showing it as gone, twelve of thirteen. Nothing was wrong with the test: the test
	#     lived in the MONTHLY pulse, so the panel could lie for up to thirty days, and the moment a
	#     player looks at it is the moment he wins, not the first of next month. A number that is
	#     right eventually is a number nobody trusts.
	#     The siege's cost stays monthly, because it is a monthly cost. Noticing that a province came
	#     back is not a cost and has no business waiting.
	# ES: LAS RESTAURACIONES PASAN A LA SEMANA, y la razon es una captura. Juan batio a Corrientes,
	#     claudico y quedo de protectorado --que es exactamente lo que el diseno busca-- y el panel la
	#     seguia mostrando como ida, doce de trece. La prueba no estaba mal: estaba en el pulso
	#     MENSUAL, asi que el panel podia mentir hasta treinta dias, y el momento en que un jugador lo
	#     mira es el momento en que gana, no el primero del mes que viene. Un numero que tiene razon
	#     eventualmente es un numero en el que nadie confia.
	#     El costo del sitio se queda mensual, porque es un costo mensual. Darse cuenta de que una
	#     provincia volvio no es un costo y no tiene por que esperar.
	on_weekly_pulse = {
		effect = {
			gg_pact_sync = yes
			# EN: Shared journals have no country root; resolve the mandate holder.
			# ES: El diario compartido no tiene pais raiz; resolver quien tiene el mandato.
			every_country = {
				limit = { has_variable = gg_arg_rosas_seat }
			hidden_effect = {
				# EN: Restore a withdrawn mandate after subjugation or full territorial recovery.
				#     The effect clears the withdrawal flag, preventing repeat weekly notices.
				# ES: Restituir el mandato retirado tras subordinacion o recuperacion territorial.
				#     El efecto borra la marca de retiro y evita avisos semanales repetidos.
				if = {
					limit = {
						has_variable = gg_arg_rosas_seat
						OR = {
							# EN: EITHER IT IS GONE, OR IT IS BACK UNDER THE PACT. Until 20/9/2026 a province
							#     only counted as recovered when its country had been wiped off the map, which
							#     is not what Rosas ever did: after Pago Largo he put a federal governor in
							#     Corrientes and the province went on existing. Now the war is fought for a
							#     protectorate -- a country whose foreign relations another conducts, which is
							#     the delegation itself -- so winning it is the second way home.
							# ES: O ya no esta, O volvio al pacto. Hasta el 20/9/2026 una provincia solo contaba
							#     como recuperada si su pais habia desaparecido del mapa, que no es lo que hizo
							#     Rosas nunca: despues de Pago Largo puso un gobernador federal en Corrientes y
							#     la provincia siguio existiendo. Ahora la guerra se pelea por un protectorado
							#     -- un pais cuyas relaciones exteriores lleva otro, que es la delegacion misma --
							#     asi que ganarla es el segundo camino de vuelta.
							AND = {
								NOT = { exists = c:CRR }
								owns_entire_state_region = STATE_CORRIENTES
							}
							c:CRR ?= { is_subject_of = c:ARG }
						}
						has_global_variable = gg_delegation_gone_corrientes
					}
					gg_delegation_restored = { KEY = corrientes COUNT = 1 POINTS = 12 }
					set_global_variable = { name = gg_corrientes_recovery days = 730 }
					if = {
						limit = { c:CRR ?= { is_subject_of = c:ARG } }
						trigger_event = { id = gg_argentina.51 days = 0 popup = yes }
					}
					else = { trigger_event = { id = gg_argentina.52 days = 0 popup = yes } }
				}
				# EN/ES: seis provincias que vuelven juntas son UNA noticia, no seis / six at once is one
				#     piece of news, not six
				if = {
					limit = {
						has_variable = gg_arg_rosas_seat
						OR = {
							# EN/ES: la misma regla que arriba: o ya no esta y la tiene Buenos Aires, o volvio al
							#     pacto como protectorado / same rule as above
							AND = {
								NOT = { exists = c:CLN }
								OR = { NOT = { has_global_variable = gg_delegation_gone_coalicion_tucuman } owns_entire_state_region = STATE_TUCUMAN }
								OR = { NOT = { has_global_variable = gg_delegation_gone_coalicion_salta } owns_entire_state_region = STATE_SALTA }
								OR = { NOT = { has_global_variable = gg_delegation_gone_coalicion_jujuy } owns_entire_state_region = STATE_JUJUY }
								OR = { NOT = { has_global_variable = gg_delegation_gone_coalicion_catamarca } owns_entire_state_region = STATE_CATAMARCA }
								OR = { NOT = { has_global_variable = gg_delegation_gone_coalicion_la_rioja } owns_entire_state_region = STATE_LA_RIOJA }
								OR = { NOT = { has_global_variable = gg_delegation_gone_cordoba } owns_entire_state_region = STATE_CORDOBA }
							}
							c:CLN ?= { is_subject_of = c:ARG }
						}
						has_global_variable = gg_delegation_gone_coalicion_tucuman
					}
					if = {
						limit = { c:CLN ?= { is_subject_of = c:ARG } }
						trigger_event = { id = gg_argentina.53 days = 0 popup = yes }
					}
					else = { trigger_event = { id = gg_argentina.54 days = 0 popup = yes } }
					gg_delegation_restored = { KEY = coalicion_tucuman COUNT = 1 POINTS = 12 }
					gg_delegation_restored = { KEY = coalicion_salta COUNT = 1 POINTS = 12 }
					gg_delegation_restored = { KEY = coalicion_jujuy COUNT = 1 POINTS = 12 }
					gg_delegation_restored = { KEY = coalicion_catamarca COUNT = 1 POINTS = 12 }
					gg_delegation_restored = { KEY = coalicion_la_rioja COUNT = 1 POINTS = 12 }
					gg_delegation_restored = { KEY = cordoba COUNT = 1 POINTS = 12 }
				}
				if = {
					limit = {
						has_variable = gg_arg_rosas_seat
						OR = {
							# EN/ES: la misma regla que arriba: o ya no esta y la tiene Buenos Aires, o volvio al
							#     pacto como protectorado / same rule as above
							AND = {
								NOT = { exists = c:ENR }
								owns_entire_state_region = STATE_ENTRE_RIOS
							}
							c:ENR ?= { is_subject_of = c:ARG }
						}
						has_global_variable = gg_delegation_gone_entre_rios
					}
					gg_delegation_restored = { KEY = entre_rios COUNT = 1 POINTS = 12 }
					if = {
						limit = { c:ENR ?= { is_subject_of = c:ARG } }
						trigger_event = { id = gg_argentina.55 days = 0 popup = yes }
					}
					else = { trigger_event = { id = gg_argentina.56 days = 0 popup = yes } }
				}
			}
					}
		}
	}

	on_monthly_pulse = {
		effect = {
			# EN: Shared journals have no country root; resolve the mandate holder.
			# ES: El diario compartido no tiene pais raiz; resolver quien tiene el mandato.
			every_country = {
				limit = { has_variable = gg_arg_rosas_seat }
			if = {
				limit = {
					has_variable = gg_arg_rosas_seat
					has_global_variable = guerra_grande_split_done
					NOT = { has_global_variable = gg_peace_settled }
					exists = c:CER
				}
					# EN: THE SIEGE COUNTS ITS OWN MONTHS AND STOPS EATING THE PROVINCES. It used to take a
					#     point a month off the same number the provinces moved, and that is precisely what
					#     made the bar unreadable: it fell with nothing happening on the map. Now it counts
					#     what it is -- how long Oribe has been standing in front of Montevideo -- and the
					#     journal prints that on a line of its own.
					# ES: EL SITIO CUENTA SUS PROPIOS MESES Y DEJA DE COMERSE LAS PROVINCIAS. Le sacaba un
					#     punto por mes al mismo numero que movian las provincias, y eso es exactamente lo
					#     que hacia ilegible la barra: bajaba sin que pasara nada en el mapa. Ahora cuenta lo
					#     que es -- cuanto hace que Oribe esta parado frente a Montevideo -- y el journal lo
					#     imprime en una linea propia.
					change_global_variable = {
						name = gg_oribe_siege_months
						add = 1
					}
			}
			# EN/ES: ya no hay barra que redibujar ni costo que aplicar: el sitio cuenta sus meses arriba
			#     / no bar to redraw and no cost to charge: the siege counts its months above
					}
		}
	}

	# EN: ONE AUTHORITY, AND IT IS THE POINTS. This used to complete on gg_delegating_provinces while
	#     the monthly wear only moved gg_delegation_points, so the two could disagree: the bar could
	#     read zero, the text still say seven provinces delegate, the journal never close, and the
	#     points keep falling past -13, -14, -100 with nothing watching. That is not a rounding
	#     difference, it is two clocks telling different times to the same player.
	#     The points govern the bar, the text and the ending. The province count stays as what it is
	#     -- the discrete number of legislatures that have formally withdrawn -- and it is still what
	#     the status line names, because "eight and a half provinces delegate" is not a sentence
	#     anybody would say.
	# ES: UNA SOLA AUTORIDAD, Y SON LOS PUNTOS. Esto cerraba con gg_delegating_provinces mientras el
	#     desgaste mensual solo movia gg_delegation_points, asi que los dos podian discrepar: la barra
	#     podia marcar cero, el texto seguir diciendo que delegan siete provincias, el journal no
	#     cerrar nunca, y los puntos seguir bajando a -13, -14, -100 sin que nadie mirara. No es una
	#     diferencia de redondeo: son dos relojes dandole la hora distinta al mismo jugador.
	#     Los puntos gobiernan la barra, el texto y el final. La cuenta de provincias queda como lo
	#     que es -- el numero discreto de legislaturas que retiraron formalmente la delegacion -- y
	#     sigue siendo lo que nombra la linea de estado, porque "delegan ocho provincias y media" no
	#     es una frase que diria nadie.
	complete = {
		custom_tooltip = {
			text = je_gg_delegation_complete_tt
			# EN/ES: Independent Argentine closure: a recorded settlement, or a majority withdrawal
			# after the final political window opens. No Oriental siege counter participates.
			OR = {
				has_global_variable = gg_pact_settled
				AND = { year >= 1851 global_var:gg_delegating_provinces <= 6 }
			}
		}
	}

	# EN: AND AN ENDING, because until now the entry simply vanished: no Rosas falling, no Caseros, no
	#     war ending. A journal that completes and says nothing is a journal that taught the player a
	#     mechanic for fifteen years and then shrugged.
	# ES: Y UN DESENLACE, porque hasta ahora la entrada sencillamente desaparecia: ni caia Rosas, ni
	#     se abria Caseros, ni terminaba ninguna guerra. Un journal que cumple y no dice nada es un
	#     journal que le enseno una mecanica al jugador durante quince anios y despues se encogio de
	#     hombros.
	on_complete_all_involved = {
		if = {
			limit = { has_variable = gg_arg_rosas_seat }
			gg_pact_resolve = yes
		}
	}

	weight = 100
}
