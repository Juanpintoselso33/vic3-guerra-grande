# Journal entry for the Guerra Grande. Structure copied from vanilla's je_the_first_carlist_war:
# a battle-driven double sided bar, two scripted buttons, monthly flavour events, and a
# completion condition that fires when one of the two rival governments stops existing.
je_guerra_grande = {
	# EN/ES: 22/9/2026 -- listed under Unique Journal Entries in the country selection screen
	#     / listado en los journals unicos de la pantalla de seleccion de pais
	is_shown_in_lobby = {
		c:URU ?= this
	}
	group = je_group_guerra_grande
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	widget = {
		gui = "gui/gg_war_overview.gui"
		name = "gg_war_overview"
		container = "custom_widget_container_je_icon"
	}
	# EN/ES: the bar's title, right above it (CMF container 3 sits just before the bars)
	#     / el titulo de la barra, justo arriba (el contenedor 3 del CMF va antes de las barras)
	widget = {
		gui = "gui/gg_war_overview.gui"
		name = "gg_war_balance_head"
		container = "custom_widget_container_3"
	}
	widget = {
		gui = "gui/gg_war_overview.gui"
		name = "gg_war_details"
		container = "custom_widget_container_4"
	}

	# EN: two fields, not one. The "_uninvolved_or_context" one is the only one vanilla ever uses,
	#     and it does not cover a country that is involved -- which is exactly our case, since this
	#     entry declares should_be_involved. The counterpart exists in the engine even though no
	#     vanilla entry uses it, which is why grepping the game files never finds it.
	# ES: son dos campos, no uno. El "_uninvolved_or_context" es el unico que usa vanilla, y no
	#     cubre a un pais involucrado, que es justo nuestro caso porque esta JE declara
	#     should_be_involved. El otro existe en el motor aunque ninguna JE de vanilla lo use, por
	#     eso buscando en los archivos del juego no aparece nunca.
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes

	scripted_progress_bar = guerra_grande_war_score_bar
	# EN: the Cerrito's own two. The mirror of a foreign legion is not another foreign legion:
	#     the Defensa bought men from outside with foreign money and an open port, so the Cerrito
	#     raises its own off the countryside it already holds, and its one road to victory is
	#     shutting the sea. The blockade is the only button on this entry the other side can take
	#     off the board -- calling in the European squadrons ends it, as it did in August 1845.
	# ES: los dos propios del Cerrito. El espejo de una legion extranjera no es otra legion
	#     extranjera: la Defensa compraba hombres de afuera con plata ajena y puerto abierto, asi
	#     que el Cerrito levanta los suyos de la campania que ya tiene, y su unico camino a la
	#     victoria es cerrar el mar. El bloqueo es el unico boton de esta entrada que el otro bando
	#     puede sacar del tablero: llamar a las escuadras europeas lo termina, como en agosto de 1845.
	scripted_button = guerra_grande_montonera_levy_button

	# EN: by tag or by mark. If the player took the Cerrito, the tags swap around -- the mark each
	#     government carries is what identifies it either way.
	# ES: por tag o por marca. Si el jugador se fue al Cerrito los tags se dan vuelta: la marca que
	#     lleva cada gobierno es lo que lo identifica igual.
	should_be_involved = {
		OR = {
			c:URU ?= this
			c:CER ?= this
			has_variable = guerra_grande_cerrito_government
			has_variable = guerra_grande_defensa_government
		}
	}

	should_show_when_not_involved = {
		always = no
	}

	immediate = {
		gg_journal_hide_engine_parts = yes
		gg_ui_prewar_leaders_sync = yes
		if = {
			limit = { NOT = { has_global_variable = guerra_grande_blanco_score } }
			set_global_variable = {
				name = guerra_grande_blanco_score
				value = 0
			}
		}
		if = {
			limit = { NOT = { has_global_variable = guerra_grande_colorado_score } }
			set_global_variable = {
				name = guerra_grande_colorado_score
				value = 0
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_guerra_grande_status_factions
				trigger = {
					NOT = { has_global_variable = guerra_grande_open_war }
				}
			}
			# EN: playtest 2026-09-17 -- the race between the two governments' exhaustion decided who got
			#     to dictate the peace, and the player could not see it. Now the journal prints it.
			# ES: la carrera entre los agotamientos decidia quien dictaba la paz, y el jugador no la veia.
			triggered_desc = {
				desc = je_guerra_grande_status_siege
				trigger = {
					always = yes
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			# EN/ES: 22/9/2026 -- the balance's sources and the phase's own effects / fuentes y efectos de fase
			# EN: Codex review 22/9/2026 -- after the split the entry has one instance per government and this
			#     pulse runs twice a month; the second rollover copied the zeroed counters over "last month".
			#     A 20-day tick makes it roll once a month, like gg_siege_tick below.
			# ES: tras la particion hay una instancia por gobierno y este pulso corre dos veces por mes; el
			#     segundo rollover pisaba "el mes pasado" con ceros. La marca de 20 dias lo deja en uno por mes.
			if = {
				limit = { NOT = { has_global_variable = gg_src_tick } }
				set_global_variable = { name = gg_src_tick days = 20 }
				gg_war_sources_rollover = yes
			}
			gg_war_phase_effects = yes
			# EN: playtest 2026-09-17 -- self-healing: if the split happened and the post-partition
			#     calendar never ran (a scheduled event whose trigger was checked too early was
			#     dropped), run it now. This also repairs saves made before the fix.
			# ES: auto-reparacion: si la particion ocurrio y el calendario posterior nunca corrio
			#     (un evento agendado cuyo trigger se chequeo demasiado temprano se descartaba), se
			#     corre ahora. Tambien repara partidas guardadas antes del arreglo.
			if = {
				limit = {
					has_global_variable = guerra_grande_split_done
					NOT = { has_global_variable = gg_partition_calendar_done }
				}
				trigger_event = { id = gg_intro.5 days = 0 }
			}
			# EN/ES: playtest -- keep the two bars in step / mantener las dos barras al dia
			# EN: Codex review 2026-09-19 -- a 45-line monthly census of modifiers lived here, written on
			#     2026-09-16 to chase the Defensa's "evaporating" forces. It printed to debug_log, which
			#     does not write in a release build, so it was ~200 pulses of work per game for nothing.
			#     The bug it was chasing was found and fixed (the naval HQ and the buttons the AI never
			#     pressed), so the instrument goes. If it is ever needed again it is in the history.
			# ES: aca vivia un censo mensual de modificadores de 45 lineas, escrito el 16/9 para
			#     perseguir las fuerzas de la Defensa que "se evaporaban". Escribia en debug_log, que en
			#     una build de release no escribe, asi que eran unos 200 pulsos de trabajo por partida
			#     para nada. El error que perseguia ya se encontro (el HQ naval y los botones que la IA
			#     no apretaba nunca), asi que el instrumento se va. Si hace falta, esta en el historial.
			# EN: the pre-war needle moves ONLY when the player chooses. It used to drift on a
			#     monthly roll -- and it drifted the wrong way, pushing the bar towards the
			#     colorados whenever the country was on the blanco path, because that roll did not
			#     follow the sign convention the rest of the mod uses (blanco positive). Worse, the
			#     drift never touched the score, so the needle and the numbers printed in this very
			#     entry disagreed. A bar that moves on its own and contradicts its own tooltip
			#     teaches the player nothing, so there is no drift: what you see is what you chose.
			# ES: la aguja previa a la guerra se mueve SOLO cuando el jugador elige. Antes derivaba
			#     con una tirada mensual, y derivaba para el lado equivocado: empujaba la barra
			#     hacia los colorados cuando el pais iba por el camino blanco, porque esa tirada no
			#     seguia la convencion de signo del resto del mod (blanco positivo). Peor: la deriva
			#     nunca tocaba el marcador, asi que la aguja y los numeros que imprime esta misma
			#     entrada se contradecian. Una barra que se mueve sola y se contradice a si misma no
			#     le ensenia nada al jugador, asi que no hay deriva: lo que ves es lo que elegiste.
			# EN: playtest 2026-09-19 -- this monthly roll threw two one-option events ("Las azoteas",
			#     "Se acabaron las haciendas") whose only effect was the mood of the pops: an OK button
			#     in the middle of the siege. Cut, with the events.
			# ES: esta tirada mensual sacaba dos eventos de una sola opcion cuyo unico efecto era el
			#     humor de los pops: un boton de OK en medio del sitio. Se fueron, con los eventos.
			# EN: the bar has to buy something. Five battles of lead give the leading government a
			#     real advantage and grind the other down; ten decide the war and bring the losing
			#     side to the table. Both modifiers are refreshed monthly, so they fade on their
			#     own the moment the other side catches up.
			# ES: la barra tiene que comprar algo. Cinco batallas de ventaja le dan al gobierno que
			#     va ganando una ventaja de verdad y desgastan al otro; diez deciden la guerra y
			#     sientan al que pierde a la mesa. Los dos modificadores se renuevan cada mes, asi
			#     que se caen solos apenas el otro empata.
			if = {
				limit = { has_global_variable = guerra_grande_open_war }
				if = {
					# EN: T10 balance 2026-09-19 -- these read the lead the DECISIONS built, not the war
					#     balance that now includes the siege and the sea. With the composite they would sit
					#     on one side for the six years the sea stays open, and +10% offense for six years is
					#     not an edge, it is a different war. The bar and the arbiter do read the composite.
					# ES: estos leen la ventaja que construyeron las DECISIONES, no la balanza que ahora
					#     incluye el sitio y el mar: con la balanza quedarian prendidos de un lado los seis
					#     anios que el mar esta abierto. La barra y el arbitro si leen la balanza.
					limit = { guerra_grande_blanco_lead >= 5 }
					c:CER ?= { add_modifier = { name = guerra_grande_upper_hand months = 2 } }
					c:URU ?= { add_modifier = { name = guerra_grande_losing_ground months = 2 } }
				}
				else_if = {
					limit = { guerra_grande_colorado_lead >= 5 }
					c:URU ?= { add_modifier = { name = guerra_grande_upper_hand months = 2 } }
					c:CER ?= { add_modifier = { name = guerra_grande_losing_ground months = 2 } }
				}
				# EN/ES: T1, C1 -- the peace is no longer scheduled here: the weekly arbiter does it.
				# EN: C10 (22/9/2026) -- the river, once a month: the signed counter, the powers' patience and
				#     the lent squadron, all in gg_plata_monthly (common/scripted_effects/gg_rio_plata.txt). The
				#     20-day tick makes it count once a month however many times this pulse runs.
				# ES: C10 -- el rio, una vez por mes: el contador con signo, la paciencia de las potencias y la
				#     escuadra prestada, todo en gg_plata_monthly. La marca de 20 dias hace que cuente una vez
				#     por mes, corra este pulso las veces que corra.
				if = {
					limit = { NOT = { has_global_variable = gg_siege_tick } }
					set_global_variable = { name = gg_siege_tick days = 20 }
					gg_plata_monthly = yes
					# EN: Codex review 2026-09-19 -- the Cerrito's two mechanics used to live inside scripted
					#     buttons, and the AI never presses a button. With the player on the Defensa, the
					#     Cerrito never raised the montonera, never called Brown's squadron and never
					#     declared the blockade: the sea stayed open by itself and the siege did nothing.
					#     An AI Cerrito does both, in the order a government would: the horsemen as soon
					#     as the war is on, the blockade once it has a war to blockade for.
					# ES: las dos mecanicas del Cerrito vivian adentro de botones, y la IA no aprieta
					#     botones. Con el jugador en la Defensa, el Cerrito no levantaba la montonera, no
					#     llamaba a la escuadra de Brown y no declaraba el bloqueo: el mar quedaba abierto
					#     solo y el sitio no hacia nada. Un Cerrito de la IA hace las dos, en el orden en
					#     que las haria un gobierno: los jinetes apenas hay guerra, el bloqueo despues.
					c:CER ?= {
						if = {
							limit = {
								is_player = no
								has_global_variable = guerra_grande_open_war
								NOT = { has_variable = guerra_grande_montonera_levy_raised }
								# EN/ES: 22/9/2026 -- the button's guard: the levy is raised in San Jose
								#     / la guarda del boton: la leva se levanta en San Jose
								any_scope_state = { state_region = s:STATE_SAN_JOSE }
							}
							gg_montonera_levy = yes
						}
						if = {
							limit = {
								is_player = no
								has_global_variable = guerra_grande_open_war
								NOT = { has_variable = guerra_grande_river_blockade_asked }
								# EN/ES: 22/9/2026 -- as the button: Brown's squadron, or from 1850 / como el boton
								OR = {
									NOT = { has_global_variable = gg_brown_fleet_lost }
									year >= 1850
								}
								# EN/ES: 22/9/2026 -- the same gates as the button: the powers' guarantee, a year
								#     after a refusal, and Rosas's backing at 40 / las mismas guardas que el boton
								NOT = { has_global_variable = gg_plata_guarantee }
								NOT = { has_variable = gg_plata_request_refused }
								global_var:gg_patron_rosas >= 40
								# EN/ES: C10 -- the same gates as the Cerrito's button: 1843, and a bank Rosas can
								#     shut / las mismas guardas que el boton: 1843, y una orilla que Rosas pueda cerrar
								year >= 1843
								gg_plata_ba_held_by_rosas = yes
								gg_plata_ba_fort = yes
								OR = {
									NOT = { has_global_variable = gg_plata_intervention }
									has_global_variable = gg_plata_intervention_over
								}
							}
							gg_river_blockade = yes
						}
					}
					# EN: T3b, C19 -- the patrons, once a month. A side whose backing is under 40 bleeds
					#     war support; France and Brazil pay the Defensa, Rosas pays the Cerrito.
					# ES: T3b, C19 -- los padrinos, una vez por mes. El bando con menos de 40 de sosten
					#     pierde apoyo a la guerra; Francia y Brasil le pagan a la Defensa, Rosas al Cerrito.
					# EN: playtest 2026-09-17 -- exhaustion is counted, not snapshotted, and it has a floor.
					#     A government is exhausted when its war support is at -25 or worse; three months in a
					#     row of that is what opens the enemy's peace menu. And nobody is allowed to slide to
					#     -100, which is where the engine capitulates on its own and the winner takes everything.
					# ES: el agotamiento se cuenta, no se mira de golpe, y tiene piso. Un gobierno esta agotado
					#     con el apoyo en -25 o peor; tres meses seguidos abren el menu de paz del enemigo. Y a
					#     nadie se lo deja caer a -100, que es donde el motor capitula solo.
					c:CER ?= {
						every_scope_war = {
							limit = { gg_is_the_war = yes }
							if = {
								limit = { has_war_support = { target = c:CER value <= -25 } }
								change_global_variable = { name = gg_cerrito_exhausted add = 1 }
							}
							else = {
								set_global_variable = { name = gg_cerrito_exhausted value = 0 }
							}
							if = {
								limit = { has_war_support = { target = c:CER value <= -60 } }
								add_war_war_support = { target = c:CER value = 15 }
							}
						}
					}
					# EN: Codex review 2026-09-19 -- the closing used to read enemy_occupation at the moment the
					#     journal closed, and that number counts ground taken by ANY enemy: a Brazilian war
					#     somewhere else could be read as the Cerrito's victory. It is written down here
					#     instead, month by month and only while this war is the one being fought, so the
					#     closing reads who took whose ground in the Guerra Grande.
					# ES: el cierre leia enemy_occupation en el momento de cerrar, y ese numero cuenta el
					#     suelo tomado por CUALQUIER enemigo: una guerra con el Brasil en otro lado podia
					#     leerse como victoria del Cerrito. Se anota aca, mes a mes y solo mientras la
					#     guerra que se pelea sea esta, asi el cierre lee quien tomo el suelo de quien.
					# EN: Codex review 2026-09-19, second pass -- wrapping this in the war's scope was not
					#     enough: enemy_occupation belongs to the country and adds up the ground taken by
					#     ANY enemy, whatever war it came from. The engine has no per-attacker occupation
					#     in script (attacker_occupation_progress is a war-goal contestion type, not a
					#     trigger), so the number is only trusted while the Guerra Grande is the ONLY war
					#     either government is fighting. With another war open the marks are left alone.
					# ES: envolverlo en el scope de la guerra no alcanzaba: enemy_occupation es del pais y
					#     suma el suelo tomado por CUALQUIER enemigo, venga de la guerra que venga. El motor
					#     no expone ocupacion por bando en script, asi que el numero solo se cree mientras
					#     la Guerra Grande sea la UNICA guerra de los dos gobiernos. Con otra guerra abierta
					#     las marcas se dejan como estan.
					if = {
						limit = {
							c:CER ?= {
								any_scope_war = { gg_is_the_war = yes }
								NOT = { any_scope_war = { NOT = { gg_is_the_war = yes } } }
							}
							c:URU ?= {
								NOT = { any_scope_war = { NOT = { gg_is_the_war = yes } } }
							}
						}
						if = {
							limit = { c:URU ?= { enemy_occupation >= 0.5 } }
							set_global_variable = { name = gg_defensa_overrun value = 1 }
						}
						else = {
							set_global_variable = { name = gg_defensa_overrun value = 0 }
						}
						if = {
							limit = { c:CER ?= { enemy_occupation >= 0.5 } }
							set_global_variable = { name = gg_cerrito_overrun value = 1 }
						}
						else = {
							set_global_variable = { name = gg_cerrito_overrun value = 0 }
						}
					}
					# EN: Codex review 2026-09-19, third pass -- when another war opens, the block above stops
					#     writing, and "leaving the marks as they are" meant a 1 from last month could decide
					#     the closing months later. A reading that cannot be trusted is not kept: both go to
					#     zero, and the war has to earn them again once it is the only one being fought.
					# ES: cuando se abre otra guerra el bloque de arriba deja de escribir, y "dejar las marcas
					#     como estan" queria decir que un 1 del mes pasado podia decidir el cierre meses
					#     despues. Una lectura que no se puede creer no se guarda: las dos van a cero, y la
					#     guerra tiene que volver a ganarselas cuando sea la unica que se pelea.
					else = {
						set_global_variable = { name = gg_cerrito_overrun value = 0 }
						set_global_variable = { name = gg_defensa_overrun value = 0 }
					}
					c:URU ?= {
						every_scope_war = {
							limit = { gg_is_the_war = yes }
							if = {
								limit = { has_war_support = { target = c:URU value <= -25 } }
								change_global_variable = { name = gg_defensa_exhausted add = 1 }
							}
							else = {
								set_global_variable = { name = gg_defensa_exhausted value = 0 }
							}
							if = {
								limit = { has_war_support = { target = c:URU value <= -60 } }
								add_war_war_support = { target = c:URU value = 15 }
							}
						}
					}
					if = {
						limit = { gg_defensa_backing < 40 }
						c:URU ?= {
							every_scope_war = {
								limit = { gg_is_the_war = yes }
								add_war_war_support = { target = c:URU value = gg_defensa_drain }
							}
						}
					}
					if = {
						limit = { gg_cerrito_backing < 40 }
						c:CER ?= {
							every_scope_war = {
								limit = { gg_is_the_war = yes }
								add_war_war_support = { target = c:CER value = gg_cerrito_drain }
							}
						}
					}
					c:URU ?= { add_treasury = gg_defensa_subsidy }
					c:CER ?= { add_treasury = gg_cerrito_subsidy }
				}
			}
			# EN: Codex review 2026-09-19 -- the bar is drawn at the END of the pulse. It used to be drawn
			#     first, before the siege and relief counters moved, so the player saw a number a month
			#     behind the one the weekly arbiter was already reading.
			# ES: la barra se dibuja al FINAL del pulso. Antes se dibujaba primero, antes de mover los
			#     contadores del sitio y del alivio, asi que el jugador veia un numero un mes atrasado
			#     respecto del que ya leia el arbitro semanal.
			scope:journal_entry = {
				set_bar_progress = { name = guerra_grande_war_score_bar value = gg_war_score_bar_value }
			}
		}
	}

	# EN: T1/T2, C2 -- the weekly arbiter: score, war support, the Cerrito's occupation and the siege.
	#     The only place that offers a negotiated peace (I2). It sets gg_peace_offered in the
	#     same effect that schedules the menu (I6), so no other path can offer a second one.
	#     From 1851 the war is tired and half the lead is enough (FA-2).
	# ES: T1, C2 -- el arbitro semanal (solo puntaje; T2 suma apoyo a la guerra, ocupacion y
	#     asedio). El unico lugar que ofrece una paz negociada. Prende gg_peace_offered en el mismo
	#     efecto que agenda el menu, asi ningun otro camino ofrece un segundo. Desde 1851 la guerra
	#     esta cansada y alcanza la mitad de la ventaja.
	on_weekly_pulse = {
		effect = {
			# EN: S19 -- should_be_involved only seeds the entry at activation (1836, before the Cerrito
			#     existed): both governments are added explicitly, every week, which also migrates saves.
			# ES: S19 -- should_be_involved solo siembra la entrada al abrirse (1836, antes del Cerrito):
			#     los dos gobiernos se agregan a mano cada semana, lo que tambien migra las partidas.
			gg_war_journal_sync = yes
			# EN/ES: 22/9/2026 -- only our own widgets; the engine's status box, long text, countries and
			#     conditions hidden (CMF), also for journals already open / solo lo nuestro (CMF)
			gg_journal_hide_engine_parts = yes
			gg_ui_prewar_leaders_sync = yes
			if = {
				limit = {
					has_global_variable = guerra_grande_open_war
					has_global_variable = guerra_grande_split_done
					NOT = { has_global_variable = gg_peace_offered }
					NOT = { has_global_variable = gg_peace_settled }
					NOT = { has_global_variable = gg_peace_cooldown }
					# EN/ES: Codex review 2026-09-19 -- a chain of the powers is already at the table
					#     / hay una cadena de las potencias sentada a la mesa
					NOT = { has_global_variable = gg_peace_chain_pending }
					# EN/ES: T7, I15 -- Howden's truce / la tregua de Howden
					NOT = { has_global_variable = gg_truce }
					exists = c:CER
					exists = c:URU
				}
				# EN/ES: T2 -- who is ripe (common/scripted_triggers) / quien esta maduro
				# EN/ES: review 2026-09-17 -- a Cerrito that asked for terms (F4-3 b/c) goes first
				#     / un Cerrito que pidio condiciones va primero
				if = {
					limit = { has_global_variable = gg_cerrito_negotiates }
					set_global_variable = gg_peace_offered
