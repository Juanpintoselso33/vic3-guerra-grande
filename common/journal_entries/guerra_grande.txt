# Journal entry for the Guerra Grande. Structure copied from vanilla's je_the_first_carlist_war:
# a battle-driven double sided bar, two scripted buttons, monthly flavour events, and a
# completion condition that fires when one of the two rival governments stops existing.
je_guerra_grande = {
	group = je_group_guerra_grande
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	# EN: two fields, not one. The "_uninvolved_or_context" one is the only one vanilla ever uses,
	#     and it does not cover a country that is involved -- which is exactly our case, since this
	#     entry declares should_be_involved. The counterpart exists in the engine even though no
	#     vanilla entry uses it, which is why grepping the game files never finds it.
	# ES: son dos campos, no uno. El "_uninvolved_or_context" es el unico que usa vanilla, y no
	#     cubre a un pais involucrado, que es justo nuestro caso porque esta JE declara
	#     should_be_involved. El otro existe en el motor aunque ninguna JE de vanilla lo use, por
	#     eso buscando en los archivos del juego no aparece nunca.
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes

	scripted_progress_bar = guerra_grande_war_score_bar
	# EN: the Cerrito's own two. The mirror of a foreign legion is not another foreign legion:
	#     the Defensa bought men from outside with foreign money and an open port, so the Cerrito
	#     raises its own off the countryside it already holds, and its one road to victory is
	#     shutting the sea. The blockade is the only button on this entry the other side can take
	#     off the board -- calling in the European squadrons ends it, as it did in August 1845.
	# ES: los dos propios del Cerrito. El espejo de una legion extranjera no es otra legion
	#     extranjera: la Defensa compraba hombres de afuera con plata ajena y puerto abierto, asi
	#     que el Cerrito levanta los suyos de la campania que ya tiene, y su unico camino a la
	#     victoria es cerrar el mar. El bloqueo es el unico boton de esta entrada que el otro bando
	#     puede sacar del tablero: llamar a las escuadras europeas lo termina, como en agosto de 1845.
	scripted_button = guerra_grande_montonera_levy_button

	# EN: by tag or by mark. If the player took the Cerrito, the tags swap around -- the mark each
	#     government carries is what identifies it either way.
	# ES: por tag o por marca. Si el jugador se fue al Cerrito los tags se dan vuelta: la marca que
	#     lleva cada gobierno es lo que lo identifica igual.
	should_be_involved = {
		OR = {
			c:URU ?= this
			c:CER ?= this
			has_variable = guerra_grande_cerrito_government
			has_variable = guerra_grande_defensa_government
		}
	}

	should_show_when_not_involved = {
		always = no
	}

	immediate = {
		if = {
			limit = { NOT = { has_global_variable = guerra_grande_blanco_score } }
			set_global_variable = {
				name = guerra_grande_blanco_score
				value = 0
			}
		}
		if = {
			limit = { NOT = { has_global_variable = guerra_grande_colorado_score } }
			set_global_variable = {
				name = guerra_grande_colorado_score
				value = 0
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_guerra_grande_status_factions
				trigger = {
					NOT = { has_global_variable = guerra_grande_open_war }
				}
			}
			triggered_desc = {
				desc = je_guerra_grande_status_siege
				trigger = {
					always = yes
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			# EN: playtest 2026-09-17 -- self-healing: if the split happened and the post-partition
			#     calendar never ran (a scheduled event whose trigger was checked too early was
			#     dropped), run it now. This also repairs saves made before the fix.
			# ES: auto-reparacion: si la particion ocurrio y el calendario posterior nunca corrio
			#     (un evento agendado cuyo trigger se chequeo demasiado temprano se descartaba), se
			#     corre ahora. Tambien repara partidas guardadas antes del arreglo.
			if = {
				limit = {
					has_global_variable = guerra_grande_split_done
					NOT = { has_global_variable = gg_partition_calendar_done }
				}
				trigger_event = { id = gg_intro.5 days = 0 }
			}
			# EN/ES: playtest 2026-09-17 -- the bar is read from the score every month, on this
			#     country's own journal, so it never drifts / la barra se lee del puntaje cada mes
			scope:journal_entry = {
				set_bar_progress = { name = guerra_grande_war_score_bar value = gg_war_score_bar_value }
			}
			# EN/ES: playtest -- keep the two bars in step / mantener las dos barras al dia
			if = { limit = { NOT = { has_global_variable = gg_debug_intervention_presses } } set_global_variable = { name = gg_debug_intervention_presses value = 0 } }
			if = { limit = { NOT = { has_global_variable = gg_debug_italian_presses } } set_global_variable = { name = gg_debug_italian_presses value = 0 } }
			if = { limit = { NOT = { has_global_variable = gg_debug_french_presses } } set_global_variable = { name = gg_debug_french_presses value = 0 } }
			if = { limit = { NOT = { has_global_variable = gg_debug_montonera_presses } } set_global_variable = { name = gg_debug_montonera_presses value = 0 } }
			if = { limit = { NOT = { has_global_variable = gg_debug_blockade_presses } } set_global_variable = { name = gg_debug_blockade_presses value = 0 } }
			# EN/ES: instrumented 2026-09-16 -- Juan reported the Defensa's forces "evaporating"
			#     across two playthroughs and asked for a way to log what is happening. First
			#     attempt used .HasModifier('x')/.IsSet as text-interpolation functions; NEITHER
			#     EXISTS in 1.13.11 (verified: .GetModifier returns an object, not a bool; no
			#     .IsSet usage anywhere in the game's own localization). Rewritten to branch in
			#     script with has_modifier as a TRIGGER (verified today against
			#     common/scripted_buttons/caudillo_buttons.txt) and log a plain string per branch,
			#     instead of trusting an unverified loc function -- the same mistake this session
			#     spent all day catching in other people's code.
			if = {
				limit = { has_global_variable = guerra_grande_open_war }
				c:URU ?= {
					if = {
						limit = { has_modifier = guerra_grande_walls_of_montevideo }
						debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_CENSUS Defensa has walls_of_montevideo"
					}
					else = {
						debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_CENSUS Defensa MISSING walls_of_montevideo"
					}
					if = {
						limit = { has_modifier = gg_siege_the_sea_lane_stays_open }
						debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_CENSUS Defensa has sea_lane_stays_open"
					}
					else = {
						debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_CENSUS Defensa MISSING sea_lane_stays_open"
					}
				}
				if = {
					limit = { gg_sea_open = yes }
					debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_CENSUS sea OPEN"
				}
				else = {
					debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_CENSUS sea SHUT"
				}
				c:CER ?= {
					if = {
						limit = { has_modifier = gg_siege_no_breaching_guns }
						debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_CENSUS Cerrito has no_breaching_guns"
					}
					if = {
						limit = { has_modifier = gg_siege_the_land_frontier_stays_open }
						debug_log = "[TimeKeeper.GetCurrentDate.GetString]: GG_CENSUS Cerrito has land_frontier_stays_open"
					}
				}
			}
			# EN: the pre-war needle moves ONLY when the player chooses. It used to drift on a
			#     monthly roll -- and it drifted the wrong way, pushing the bar towards the
			#     colorados whenever the country was on the blanco path, because that roll did not
			#     follow the sign convention the rest of the mod uses (blanco positive). Worse, the
			#     drift never touched the score, so the needle and the numbers printed in this very
			#     entry disagreed. A bar that moves on its own and contradicts its own tooltip
			#     teaches the player nothing, so there is no drift: what you see is what you chose.
			# ES: la aguja previa a la guerra se mueve SOLO cuando el jugador elige. Antes derivaba
			#     con una tirada mensual, y derivaba para el lado equivocado: empujaba la barra
			#     hacia los colorados cuando el pais iba por el camino blanco, porque esa tirada no
			#     seguia la convencion de signo del resto del mod (blanco positivo). Peor: la deriva
			#     nunca tocaba el marcador, asi que la aguja y los numeros que imprime esta misma
			#     entrada se contradecian. Una barra que se mueve sola y se contradice a si misma no
			#     le ensenia nada al jugador, asi que no hay deriva: lo que ves es lo que elegiste.
			# Flavour only while the war is actually being fought.
			if = {
				limit = { has_global_variable = guerra_grande_open_war }
				c:URU ?= { random_list = {
					17 = {}
					1 = { trigger_event = guerra_grande.6 }
					1 = { trigger_event = guerra_grande.7 }
				} }
			}
			# EN: the bar has to buy something. Five battles of lead give the leading government a
			#     real advantage and grind the other down; ten decide the war and bring the losing
			#     side to the table. Both modifiers are refreshed monthly, so they fade on their
			#     own the moment the other side catches up.
			# ES: la barra tiene que comprar algo. Cinco batallas de ventaja le dan al gobierno que
			#     va ganando una ventaja de verdad y desgastan al otro; diez deciden la guerra y
			#     sientan al que pierde a la mesa. Los dos modificadores se renuevan cada mes, asi
			#     que se caen solos apenas el otro empata.
			if = {
				limit = { has_global_variable = guerra_grande_open_war }
				if = {
					limit = { guerra_grande_blanco_lead >= 5 }
					c:CER ?= { add_modifier = { name = guerra_grande_upper_hand months = 2 } }
					c:URU ?= { add_modifier = { name = guerra_grande_losing_ground months = 2 } }
				}
				else_if = {
					limit = { guerra_grande_colorado_lead >= 5 }
					c:URU ?= { add_modifier = { name = guerra_grande_upper_hand months = 2 } }
					c:CER ?= { add_modifier = { name = guerra_grande_losing_ground months = 2 } }
				}
				# EN/ES: T1, C1 -- the peace is no longer scheduled here: the weekly arbiter does it.
				# EN: T2, C15 -- the siege counter. The sea counts as shut while the Defensa carries the
				#     river blockade and no squadron came; T3b replaces this with gg_sea_open. The
				#     20-day tick makes it count once a month however many times this pulse runs.
				# ES: T2, C15 -- el contador del asedio. El mar cuenta como cerrado mientras la Defensa
				#     tiene el bloqueo fluvial y no vino ninguna escuadra; T3b lo cambia por
				#     gg_sea_open. La marca de 20 dias hace que cuente una vez por mes, corra este
				#     pulso las veces que corra.
				if = {
					limit = { NOT = { has_global_variable = gg_siege_tick } }
					set_global_variable = { name = gg_siege_tick days = 20 }
					if = {
						# EN/ES: T3b, C15/C21 -- the sea is the one condition (I16) / el mar es la unica condicion
						# EN: FN-1 -- the month counts with the declared blockade (the Cerrito's button) OR with a
						#     real one: the Confederation's squadron sitting on Montevideo's water. The engine
						#     cannot be ordered to blockade, so both count.
						# ES: FN-1 -- el mes cuenta con el bloqueo declarado (el boton del Cerrito) O con uno real:
						#     la escuadra sobre el agua de Montevideo. Al motor no se le puede ordenar bloquear.
						limit = {
							OR = {
								c:URU ?= { has_modifier = guerra_grande_river_plate_blockade }
								c:URU ?= {
									any_scope_state = {
										state_region = s:STATE_MONTEVIDEO
										blockade_level > 0
									}
								}
							}
							NOT = { gg_sea_open = yes }
							NOT = { has_global_variable = gg_truce }
						}
						change_global_variable = { name = gg_siege_months add = 1 }
					}
					else_if = {
						limit = {
							global_var:gg_siege_months > 0
							NOT = { has_global_variable = gg_truce }
						}
						change_global_variable = { name = gg_siege_months add = -1 }
					}
					# EN: FN-2 -- the Anglo-French station: while the sea is open the city resists a blockade and
					#     keeps a lent squadron; when the backing falls the powers take their ships back.
					# ES: FN-2 -- la estacion anglo-francesa: con el mar abierto la ciudad resiste el bloqueo y
					#     tiene escuadra prestada; si el apoyo cae, las potencias se llevan sus barcos.
					if = {
						limit = { gg_sea_open = yes }
						c:URU ?= {
							if = {
								limit = { NOT = { has_variable = gg_lent_squadron } }
								set_variable = gg_lent_squadron
								create_military_formation = {
									save_scope_as = gg_lent_fleet
									type = fleet
									hq_region = sr:region_la_plata
									name = gg_lent_squadron_name
									ship = {
										type = ship_type:ship_type_frigate
										state_region = s:STATE_MONTEVIDEO
										count = 4
									}
								}
								# EN/ES: kept so the powers can take their ships back / guardada para que se las lleven
								set_variable = { name = gg_lent_squadron_ref value = scope:gg_lent_fleet }
							}
							every_scope_state = {
								limit = {
									state_region = s:STATE_MONTEVIDEO
									NOT = { has_modifier = gg_naval_anglo_french_station }
								}
								add_modifier = { name = gg_naval_anglo_french_station }
							}
						}
					}
					else = {
						c:URU ?= {
							# EN/ES: the station is gone and so are its ships / se fue la estacion y se van sus barcos
							if = {
								limit = { has_variable = gg_lent_squadron }
								remove_variable = gg_lent_squadron
								if = {
									limit = { has_variable = gg_lent_squadron_ref }
									var:gg_lent_squadron_ref ?= { disband_military_formation = yes }
									remove_variable = gg_lent_squadron_ref
								}
							}
							every_scope_state = {
								limit = { has_modifier = gg_naval_anglo_french_station }
								remove_modifier = gg_naval_anglo_french_station
							}
						}
					}
					# EN: T3b, FE-9 -- when the squadrons hold the river, the blockade is lifted.
					# ES: T3b, FE-9 -- cuando las escuadras sostienen el rio, el bloqueo se levanta.
					if = {
						limit = { gg_sea_open = yes }
						c:URU ?= {
							if = {
								limit = { has_modifier = guerra_grande_river_plate_blockade }
								remove_modifier = guerra_grande_river_plate_blockade
							}
						}
						# EN/ES: review -- the Cerrito may ask again if the sea shuts later
						#     / el Cerrito puede volver a pedirlo si el mar se cierra despues
						c:CER ?= {
							if = {
								limit = { has_variable = guerra_grande_river_blockade_asked }
								remove_variable = guerra_grande_river_blockade_asked
							}
						}
					}
					# EN: T3b, C19 -- the patrons, once a month. A side whose backing is under 40 bleeds
					#     war support; France and Brazil pay the Defensa, Rosas pays the Cerrito.
					# ES: T3b, C19 -- los padrinos, una vez por mes. El bando con menos de 40 de sosten
					#     pierde apoyo a la guerra; Francia y Brasil le pagan a la Defensa, Rosas al Cerrito.
					# EN: playtest 2026-09-17 -- exhaustion is counted, not snapshotted, and it has a floor.
					#     A government is exhausted when its war support is at -25 or worse; three months in a
					#     row of that is what opens the enemy's peace menu. And nobody is allowed to slide to
					#     -100, which is where the engine capitulates on its own and the winner takes everything.
					# ES: el agotamiento se cuenta, no se mira de golpe, y tiene piso. Un gobierno esta agotado
					#     con el apoyo en -25 o peor; tres meses seguidos abren el menu de paz del enemigo. Y a
					#     nadie se lo deja caer a -100, que es donde el motor capitula solo.
					c:CER ?= {
						every_scope_war = {
							limit = { gg_is_the_war = yes }
							if = {
								limit = { has_war_support = { target = c:CER value <= -25 } }
								change_global_variable = { name = gg_cerrito_exhausted add = 1 }
							}
							else = {
								set_global_variable = { name = gg_cerrito_exhausted value = 0 }
							}
							if = {
								limit = { has_war_support = { target = c:CER value <= -60 } }
								add_war_war_support = { target = c:CER value = 15 }
							}
						}
					}
					c:URU ?= {
						every_scope_war = {
							limit = { gg_is_the_war = yes }
							if = {
								limit = { has_war_support = { target = c:URU value <= -25 } }
								change_global_variable = { name = gg_defensa_exhausted add = 1 }
							}
							else = {
								set_global_variable = { name = gg_defensa_exhausted value = 0 }
							}
							if = {
								limit = { has_war_support = { target = c:URU value <= -60 } }
								add_war_war_support = { target = c:URU value = 15 }
							}
						}
					}
					if = {
						limit = { gg_defensa_backing < 40 }
						c:URU ?= {
							every_scope_war = {
								limit = { gg_is_the_war = yes }
								add_war_war_support = { target = c:URU value = gg_defensa_drain }
							}
						}
					}
					if = {
						limit = { gg_cerrito_backing < 40 }
						c:CER ?= {
							every_scope_war = {
								limit = { gg_is_the_war = yes }
								add_war_war_support = { target = c:CER value = gg_cerrito_drain }
							}
						}
					}
					# EN/ES: review -- the same for the Defensa's caudillos / lo mismo para los caudillos de la Defensa
					if = {
						limit = { global_var:gg_defensa_balance <= -30 }
						c:URU ?= {
							every_scope_war = {
								limit = { gg_is_the_war = yes }
								add_war_war_support = { target = c:URU value = 2 }
							}
						}
					}
					# EN/ES: T7, FG-2 -- with the caudillos in charge, the siege holds / con los caudillos, el sitio aguanta
					if = {
						limit = { global_var:gg_cerrito_balance <= -30 }
						c:CER ?= {
							every_scope_war = {
								limit = { gg_is_the_war = yes }
								add_war_war_support = { target = c:CER value = 2 }
							}
						}
					}
					c:URU ?= { add_treasury = gg_defensa_subsidy }
					c:CER ?= { add_treasury = gg_cerrito_subsidy }
				}
			}
		}
	}

	# EN: T1/T2, C2 -- the weekly arbiter: score, war support, the Cerrito's occupation and the siege.
	#     The only place that offers a negotiated peace (I2). It sets gg_peace_offered in the
	#     same effect that schedules the menu (I6), so no other path can offer a second one.
	#     From 1851 the war is tired and half the lead is enough (FA-2).
	# ES: T1, C2 -- el arbitro semanal (solo puntaje; T2 suma apoyo a la guerra, ocupacion y
	#     asedio). El unico lugar que ofrece una paz negociada. Prende gg_peace_offered en el mismo
	#     efecto que agenda el menu, asi ningun otro camino ofrece un segundo. Desde 1851 la guerra
	#     esta cansada y alcanza la mitad de la ventaja.
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					has_global_variable = guerra_grande_open_war
					has_global_variable = guerra_grande_split_done
					NOT = { has_global_variable = gg_peace_offered }
					NOT = { has_global_variable = gg_peace_settled }
					NOT = { has_global_variable = gg_peace_cooldown }
					# EN/ES: T7, I15 -- Howden's truce / la tregua de Howden
					NOT = { has_global_variable = gg_truce }
					exists = c:CER
					exists = c:URU
				}
				# EN/ES: T2 -- who is ripe (common/scripted_triggers) / quien esta maduro
				# EN/ES: review 2026-09-17 -- a Cerrito that asked for terms (F4-3 b/c) goes first
				#     / un Cerrito que pidio condiciones va primero
				if = {
					limit = { has_global_variable = gg_cerrito_negotiates }
					set_global_variable = gg_peace_offered
					random_country = {
						limit = { has_variable = guerra_grande_defensa_government }
						trigger_event = { id = gg_peace.1 days = 3 popup = yes }
					}
				}
				else_if = {
					limit = { gg_defensa_ripe = yes }
					set_global_variable = gg_peace_offered
					# EN: playtest 2026-09-17 -- the side that is beaten is asked first. It used to be annexed by
					#     the winner's menu without seeing anything; now it can ask for terms or fight on.
					# ES: al que va perdiendo se le pregunta primero. Antes lo anexaba el menu del ganador sin que
					#     viera nada; ahora puede pedir condiciones o seguir peleando.
					random_country = {
						limit = { has_variable = guerra_grande_defensa_government }
						trigger_event = { id = gg_peace.11 days = 1 popup = yes }
					}
				}
				else_if = {
					limit = { gg_cerrito_ripe = yes }
					set_global_variable = gg_peace_offered
					random_country = {
						limit = { has_variable = guerra_grande_cerrito_government }
						trigger_event = { id = gg_peace.12 days = 1 popup = yes }
					}
				}
				# EN/ES: T7, F3-2 -- a long stalemate brings back the powers' proposal, once
				else_if = {
					limit = { gg_powers_stalemate = yes }
					set_global_variable = gg_hood_retried
					random_country = {
						limit = { has_variable = guerra_grande_cerrito_government }
						trigger_event = { id = gg_powers.12 days = 3 popup = yes }
					}
				}
			}
		}
	}

	# EN: T1, C11 -- the entry closes on a signed peace, or when only one government is left and
	#     Montevideo is still oriental (a conquest, a conceded play or a third country), or at the
	#     1860 safety net. It never asks for guerra_grande_open_war, which its own closing clears.
	# ES: T1, C11 -- se cierra con la paz firmada, o cuando queda un solo gobierno y Montevideo sigue
	#     siendo oriental (conquista, jugada cedida o tercer pais), o en el tope de 1860. Nunca pide
	#     guerra_grande_open_war, que su propio cierre borra.
	complete = {
		custom_tooltip = {
			text = je_guerra_grande_complete_tt
			AND = {
				has_global_variable = guerra_grande_split_done
				OR = {
					c:URU ?= { any_scope_state = { state_region = s:STATE_MONTEVIDEO } }
					c:CER ?= { any_scope_state = { state_region = s:STATE_MONTEVIDEO } }
				}
				OR = {
					has_global_variable = gg_peace_settled
					NAND = {
						exists = c:CER
						exists = c:URU
					}
					year >= 1860
				}
			}
		}
	}

	# EN: T1, C12 -- Montevideo in foreign hands is the failure, and it wins over complete (complete
	#     requires an oriental Montevideo). The dead pre-split chain keeps its own branch.
	# ES: T1, C12 -- Montevideo extranjera es el fracaso, y manda sobre complete. La cadena muerta
	#     previa a la particion conserva su rama.
	fail = {
		custom_tooltip = {
			text = je_guerra_grande_fail_tt
			OR = {
				AND = {
					has_global_variable = guerra_grande_split_done
					NOR = {
						c:URU ?= { any_scope_state = { state_region = s:STATE_MONTEVIDEO } }
						c:CER ?= { any_scope_state = { state_region = s:STATE_MONTEVIDEO } }
					}
				}
				AND = {
					NOT = { has_global_variable = guerra_grande_split_done }
					year >= 1844
					OR = {
						NOT = { exists = c:URU }
						c:URU ?= { NOT = { exists = s:STATE_BANDA_ORIENTAL.region_state:URU } }
						c:URU ?= { NOT = { any_scope_state = { state_region = s:STATE_MONTEVIDEO } } }
					}
				}
			}
		}
	}

	on_fail_all_involved = {
		add_modifier = {
			name = guerra_grande_dismembered_republic
			years = 15
			is_decaying = yes
		}
		hidden_effect = {
			# EN/ES: T1, C14 -- the loans' watcher reads this / el vigia de los prestamos lee esto
			set_global_variable = gg_war_failed
			remove_global_variable = guerra_grande_open_war
		}
	}

	# EN: T1, C11 -- a war closed without a signed peace ends in gg_peace.5. The mark goes first, so
	#     the second involved country's pass does nothing (I5, I6).
	# ES: T1, C11 -- una guerra cerrada sin paz firmada termina en gg_peace.5. La marca va primero,
	#     asi la pasada del segundo pais involucrado no hace nada.
	on_complete_all_involved = {
		if = {
			limit = { NOT = { has_global_variable = gg_peace_settled } }
			set_global_variable = gg_peace_settled
			set_global_variable = gg_victory_pending
			if = {
				limit = {
					exists = c:CER
					exists = c:URU
				}
				set_global_variable = gg_victory_timeout
			}
			else_if = {
				limit = {
					OR = {
						AND = {
							NOT = { exists = c:CER }
							c:URU ?= { NOT = { any_scope_state = { state_region = s:STATE_SAN_JOSE } } }
						}
						AND = {
							NOT = { exists = c:URU }
							c:CER ?= { NOT = { any_scope_state = { state_region = s:STATE_MONTEVIDEO } } }
						}
					}
				}
				set_global_variable = gg_victory_third_party
			}
			if = {
				limit = {
					exists = c:CER
					OR = {
						NOT = { exists = c:URU }
						guerra_grande_blanco_lead > 0
					}
				}
				random_country = {
					limit = { has_variable = guerra_grande_cerrito_government }
					trigger_event = { id = gg_peace.5 days = 1 popup = yes }
				}
			}
			else = {
				random_country = {
					limit = { has_variable = guerra_grande_defensa_government }
					trigger_event = { id = gg_peace.5 days = 1 popup = yes }
				}
			}
		}
		# EN/ES: T8, C16 -- the postwar, once / la posguerra, una vez
		if = {
			limit = {
				NOT = { has_global_variable = gg_postwar_started }
				NOT = { has_global_variable = gg_reincorporated }
			}
			set_global_variable = gg_postwar_started
			gg_postwar_dispatch = yes
		}
		hidden_effect = {
			remove_global_variable = guerra_grande_open_war
		}
	}

	weight = 100
