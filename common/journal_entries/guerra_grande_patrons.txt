# EN: Shared international situation. Day-one activation is independent of the opening popup;
#     peace, an averted war or failure closes it for everyone.
# ES: Situacion internacional compartida. Abre el primer dia sin esperar el popup;
#     la paz, una guerra evitada o el fracaso la cierran para todos.
je_gg_patrons = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = no

	scripted_progress_bar = gg_sea_bar
	# EN/ES: calling in the squadrons is a request to the powers / llamar a las escuadras es un pedido a las potencias
	scripted_button = guerra_grande_foreign_intervention_button

	should_be_involved = {
		gg_international_participant = yes
	}

	should_show_when_not_involved = {
		has_interest_marker_in_region = sr:region_la_plata
	}

	immediate = {
		scope:journal_entry = { set_bar_progress = { name = gg_sea_bar value = gg_sea_bar_value } }
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_gg_patrons_status
				trigger = { always = yes }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			gg_international_sync = yes
		}
	}

	complete = {
		custom_tooltip = {
			text = je_gg_patrons_complete_tt
			# EN/ES: or the war that never became one: without a siege there is nothing left for the
			#     squadrons to decide, and this entry cannot stay open for a war that did not happen.
			#     / o la guerra que no llego a serlo: sin sitio no queda nada que las escuadras puedan
			#     decidir, y esta entrada no puede quedar abierta por una guerra que no fue.
			OR = {
				has_global_variable = gg_peace_settled
				has_global_variable = gg_war_never_crossed
			}
		}
	}

	fail = {
		custom_tooltip = {
			text = je_gg_patrons_fail_tt
			has_global_variable = gg_war_failed
		}
	}

	weight = 99
}
