# EN: playtest 2026-09-17 -- "Doctores y caudillos" (Juan's decision): the struggle between the city
#     doctors and the rural caudillos inside BOTH governments, one balance each (-100 caudillos,
#     100 doctores). Display only; the effects live in the journal's monthly pulse and the events.
# ES: "Doctores y caudillos": la puja entre doctores y caudillos dentro de LOS DOS gobiernos, una
#     balanza cada uno. Solo muestra; los efectos estan en el pulso del journal y en los eventos.
je_gg_factions = {
	group = je_group_guerra_grande
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes

	scripted_progress_bar = gg_cerrito_balance_bar
	scripted_progress_bar = gg_defensa_balance_bar

	should_be_involved = {
		OR = {
			c:URU ?= this
			c:CER ?= this
			has_variable = guerra_grande_cerrito_government
			has_variable = guerra_grande_defensa_government
		}
	}

	should_show_when_not_involved = {
		always = no
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_gg_factions_status
				trigger = { always = yes }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			scope:journal_entry = {
				set_bar_progress = { name = gg_cerrito_balance_bar value = global_var:gg_cerrito_balance }
				set_bar_progress = { name = gg_defensa_balance_bar value = global_var:gg_defensa_balance }
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_gg_patrons_complete_tt
			has_global_variable = gg_peace_settled
		}
	}

	fail = {
		custom_tooltip = {
			text = je_gg_patrons_fail_tt
			has_global_variable = gg_war_failed
		}
	}

	weight = 97
}
