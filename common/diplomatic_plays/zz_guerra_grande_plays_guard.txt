# EN: 23/9/2026 (Juan: "every war play against the Oriental governments guarded until the signed peace").
#     The vanilla plays that zz_guerra_grande_plays.txt and zz_gg_argentine_plays.txt did not already cover and
#     whose target can be the Republic or either wartime government, copied VERBATIM from
#     game/common/diplomatic_plays/00_diplomatic_plays.txt (1.13) with exactly one line added at the TOP LEVEL of
#     each possible block: guerra_grande_play_is_allowed (the same guard, same place, same reason as the other two
#     files; inserted by brace counting). No other mod of the playset defines these plays.
#     Not copied, on purpose (tools/validate_mod.py "war play guard" keeps the list): the engine's civil wars
#     (dp_revolution, dp_secession, dp_taiping_rebellion, dp_sepoy_mutiny, dp_hungarian_revolution: a
#     country against its own rebels) and the unifications of other regions
#     (Germany, Italy, Scandinavia, Ethiopia, China, Arabia, India, Indonesia, Yugoslavia, the Balkans, the
#     trialist manifesto). The Andes ones ARE here (geographic_region_andean_federation
#     includes region_la_plata), and so is dp_native_uprising: a decentralized people against its colonizer.
#     Behind these, every war goal type carries gg_war_goal_is_allowed (common/war_goal_types/).
# ES: las jugadas de vanilla que no cubrian zz_guerra_grande_plays.txt ni zz_gg_argentine_plays.txt y cuyo
#     objetivo puede ser la Republica o cualquiera de los dos gobiernos, copiadas TAL CUAL con una sola linea
#     agregada en el NIVEL SUPERIOR de cada possible: guerra_grande_play_is_allowed. No se copian las guerras
#     civiles del motor ni las unificaciones de otras regiones; las de los Andes si, porque la Federacion Andina
#     incluye la region del Plata, y el levantamiento nativo tambien (un pueblo contra su colonizador). Detras
#     de estas, cada tipo de objetivo de guerra lleva gg_war_goal_is_allowed.

REPLACE:dp_restore_union = {
	war_goal = restore_union

	texture = "gfx/interface/icons/war_goals/enforce_treaty_article.dds"

	selectable_in_lens = {
		has_law = law_type:law_monarchy
		has_variable = por_seeking_restore_union_var
	}

	possible = {
		aggressive_diplomatic_plays_permitted = yes
		guerra_grande_play_is_allowed = yes # EN/ES: 23/9 (Juan) -- the Oriental governments until the signed peace / los gobiernos orientales hasta la paz firmada
	}

	ai_acceptance_max = 90

	on_weekly_pulse = {}

	on_war_begins = {
		if = {
			limit = {
				any_scope_play_involved = {
					has_variable = por_seeking_restore_union_var
				}
			}
			random_scope_play_involved = { # This is typically a temp variable. Starting this war makes it permanent for the duration of the war.
				limit = {
					has_variable = por_seeking_restore_union_var
				}
				remove_variable = por_seeking_restore_union_var
				set_variable = por_seeking_restore_union_var
			}
		}
	}

	on_war_end = {
		if = {
			limit = {
				any_scope_play_involved = {
					has_variable = por_seeking_restore_union_var
				}
			}
			random_scope_play_involved = { # Removes variable upon war conclusion, regardless of outcome
				limit = {
					has_variable = por_seeking_restore_union_var
				}
				remove_variable = por_seeking_restore_union_var
			}
		}
	}
}
