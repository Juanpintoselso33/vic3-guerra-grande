REPLACE:dp_annex_war = {
	war_goal = annex_country

	texture = "gfx/interface/icons/war_goals/annex_country.dds"

	selectable_in_lens = {
		aggressive_diplomatic_plays_permitted = yes
	}

	possible = {
		NOT = { is_country_type = decentralized }
		aggressive_diplomatic_plays_permitted = yes
		NOT = { is_subject_of = scope:target_country }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			NOT = { is_subject_of = root }
			country_rank <= rank_value:insignificant_power
		}
		diplomatic_play_has_no_relations_blockers = yes
		gg_argentine_play_is_allowed = yes
		guerra_grande_play_is_allowed = yes # EN/ES: nor against the two oriental governments / ni contra los dos gobiernos orientales
	}

	ai_acceptance_max = 75

	# EN/ES: 23/9 (AUD-FARRAPOS-A) -- vanilla has no involvement trigger here (anyone may join); only the Brazil-PNI guard
	#     / vanilla no tiene condicion de ingreso aca; solo la guarda Brasil-PNI
	additional_involvement_trigger = {
		gg_farrapos_involvement_is_allowed = yes # EN/ES: 23/9 (AUD-FARRAPOS-A) -- no Brazil-PNI war by joining before Ponche Verde / nadie se suma a una guerra Brasil-PNI antes de Ponche Verde
	}

	on_weekly_pulse = {}

	on_war_begins = {}

	on_war_end = {}
}

REPLACE:dp_annex_subject = {
	war_goal = annex_country

	texture = "gfx/interface/icons/war_goals/annex_country.dds"
	confirmation_sound = "event:/SFX/UI/SubjectInteractions/subject_button_small_annex"
	possible = {
		trigger_if = {
			limit = {
				c:BIC ?= this
				is_at_war = no
				NOT = { has_variable = doctrine_of_lapse_cooldown }
				NOT = { has_variable = doctrine_of_lapse_ended }
				scope:target_country = {
					is_subject_of = ROOT
					has_variable = princely_state
					NOT = { country_tier = empire }
					is_player = no
				}
			}
			custom_tooltip = {
				text = CAN_USE_DOCTRINE_OF_LAPSE_INSTEAD
				always = no
			}
		}

		NOT = { is_country_type = decentralized }
		aggressive_diplomatic_plays_permitted = yes
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			is_direct_subject_of = root
			OR = {
				has_diplomatic_pact = { who = root type = colony }
				has_diplomatic_pact = { who = root type = puppet }
				has_diplomatic_pact = { who = root type = vassal }
				has_diplomatic_pact = { who = root type = crown_land }
			}
		}
		NAND = {
			is_involved_in_journal_entry = je_peru_bolivia
			scope:target_country = {
				OR = {
					country_has_primary_culture = cu:quechua
					country_has_primary_culture = cu:south_andean
				}
			}
		}
		NAND = {
			is_involved_in_journal_entry = je_ryukyu_rivalry
			scope:target_country = {
				c:RYU ?= this
			}
		}
		NAND = {
			c:JAP ?= this
			has_law_or_variant = law_type:law_serfdom # Proxy for feudalism
		}
		gg_argentine_play_is_allowed = yes
		guerra_grande_play_is_allowed = yes # EN/ES: nor against the two oriental governments / ni contra los dos gobiernos orientales
	}

	ai_acceptance_max = 75

	# EN/ES: 23/9 (AUD-FARRAPOS-A) -- vanilla has no involvement trigger here (anyone may join); only the Brazil-PNI guard
	#     / vanilla no tiene condicion de ingreso aca; solo la guarda Brasil-PNI
	additional_involvement_trigger = {
		gg_farrapos_involvement_is_allowed = yes # EN/ES: 23/9 (AUD-FARRAPOS-A) -- no Brazil-PNI war by joining before Ponche Verde / nadie se suma a una guerra Brasil-PNI antes de Ponche Verde
	}

	on_weekly_pulse = {}

	on_war_begins = {
		scope:initiator = {
			every_direct_subject = {
				limit = {
					NOT = { this = scope:target_country }
				}
				add_liberty_desire = annex_subject_base_liberty_desire_change
				add_modifier = {
					name = modifier_has_annexed_another_subject
					months = 60
					is_decaying = no
				}
			}
		}
	}

	on_war_end = {}

	on_demand_accepted = {
		every_direct_subject = {
			limit = {
				NOT = { this = scope:target_country }
			}
			add_liberty_desire = annex_subject_base_liberty_desire_change
			add_modifier = {
				name = modifier_has_annexed_another_subject
				months = 60
				is_decaying = no
			}
		}
	}
}
