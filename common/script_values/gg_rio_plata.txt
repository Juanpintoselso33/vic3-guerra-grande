# EN: C. The river in the war balance (rio-plata-spec.md section 4). One signed counter replaces the
#     old siege and relief stocks, which could both be positive and cancel out: +1 for each month the
#     river is shut, -1 for each month it is open, between -16 and +16. Four months per point, capped
#     at four -- below every threshold the arbiter reads, so the river never decides the war alone.
#     Positive helps the blancos, negative the colorados.
# ES: C. El rio en la balanza de la guerra. Un solo contador con signo reemplaza a los stocks del
#     sitio y del auxilio, que podian ser positivos a la vez y cancelarse: +1 por mes de rio cerrado,
#     -1 por mes abierto, entre -16 y +16. Cuatro meses por punto, techo cuatro: debajo de todos los
#     umbrales del arbitro, asi el rio nunca decide la guerra solo. Positivo ayuda a los blancos,
#     negativo a los colorados.
gg_plata_pressure_blanco = {
	value = global_var:gg_plata_months
	divide = 4
	floor = yes
	min = 0
	max = 4
}

gg_plata_pressure_colorado = {
	value = global_var:gg_plata_months
	multiply = -1
	divide = 4
	floor = yes
	min = 0
	max = 4
}

# EN/ES: B -- eighteen months of patience, twelve once the Defensa called the squadrons
#     / dieciocho meses de paciencia, doce si la Defensa llamo a las escuadras
gg_plata_patience_threshold = {
	value = 18
	if = {
		limit = { has_global_variable = gg_intervention_called }
		value = 12
	}
}

# EN/ES: -1 colorados, 0 even, 1 blancos / -1 colorados, 0 parejo, 1 blancos
gg_ui_plata_weight_side = {
	value = 0
	if = { limit = { gg_plata_pressure_blanco > 0 } value = 1 }
	if = { limit = { gg_plata_pressure_colorado > 0 } value = -1 }
}
gg_ui_plata_weight_points = {
	value = gg_plata_pressure_blanco
	add = gg_plata_pressure_colorado
}

# EN: The phase of the question (22/9/2026 redesign). 0 the open river, 1 the blockade, 2 the
#     ultimatum, 3 the intervention, 4 the settlement. Later phases win.
# ES: La fase de la cuestion. 0 el rio abierto, 1 el bloqueo, 2 el ultimatum, 3 la intervencion,
#     4 el arreglo. Mandan las fases posteriores.
# EN: A blockade after the settlement is the blockade again (phase 1), not the settlement.
# ES: Un bloqueo despues del arreglo vuelve a ser el bloqueo (fase 1), no el arreglo.
gg_plata_phase = {
	value = 0
	if = { limit = { has_global_variable = gg_plata_intervention_over } value = 4 }
	# EN/ES: 22/9/2026 (Juan) -- the French blockade of Buenos Aires, 28/3/1838 to Mackau (29/10/1840)
	#     / el bloqueo frances de Buenos Aires, del 28/3/1838 a Mackau
	# EN: River audit 22/9 -- below the closure: if Rosas shuts his shore during the French blockade,
	#     the band says so (and the city's mortality applies); the Buenos Aires card still names France.
	# ES: debajo del cierre: si Rosas cierra su orilla durante el bloqueo frances, la franja lo dice (y
	#     rige la mortalidad de la plaza); la tarjeta de Buenos Aires sigue nombrando a Francia.
	if = { limit = { has_global_variable = gg_plata_french_blockade } value = 5 }
	if = { limit = { gg_plata_closed = yes } value = 1 }
	if = { limit = { has_global_variable = gg_plata_ultimatum_pending } value = 2 }
	if = {
		limit = {
			has_global_variable = gg_plata_intervention
			NOT = { has_global_variable = gg_plata_intervention_over }
		}
		value = 3
	}
}
# EN: River audit 22/9/2026 -- which reading of the blockade phase the band's tooltip gives, so it
#     never claims a count the code is not keeping: 0 hunger and patience run; 1 no open war yet (the
#     monthly counters live in the open war, gg_plata_monthly); 2 a truce freezes them; 3 the two
#     months Rosas has to open his shore after yielding; 4 no power counts by its level (this includes
#     both having recognised the right of blockade); 5 only Paris counts. The later checks win.
# ES: que lectura de la fase del bloqueo da el tooltip de la franja, para que nunca diga que corre una
#     cuenta que el codigo no lleva: 0 corren hambre y paciencia; 1 todavia no hay guerra abierta; 2 la
#     tregua las congela; 3 los dos meses de Rosas para abrir su orilla; 4 ninguna potencia cuenta; 5
#     solo cuenta Paris.
gg_ui_plata_phase1_mode = {
	value = 0
	if = { limit = { gg_plata_london_out = yes gg_plata_france_counts = yes } value = 5 }
	if = {
		limit = {
			NOT = { gg_plata_france_counts = yes }
			NOT = { gg_plata_britain_counts = yes }
		}
		value = 4
	}
	if = { limit = { has_global_variable = gg_plata_guarantee_grace } value = 3 }
	if = { limit = { has_global_variable = gg_truce } value = 2 }
	if = { limit = { NOT = { has_global_variable = guerra_grande_open_war } } value = 1 }
}
# EN: River audit 22/9/2026 -- whose squadrons are in the intervention: 0 both, blockading Buenos
#     Aires; 1 France alone from the start (London out); 2 both still in the question, but London
#     lifted its blockade with Howden (15/7/1847, confirmed by article 1 of Arana-Southern,
#     SOURCES_powers.md sections 3 and 6), read at Howden's month, as gg_ui_plata_envoy does; 3 London
#     signed (Southern) and only France stays.
# ES: de quien son las escuadras de la intervencion: 0 las dos, bloqueando Buenos Aires; 1 Francia
#     sola desde el principio; 2 las dos siguen en la cuestion, pero Londres levanto su bloqueo con
#     Howden (15/7/1847, confirmado por el articulo 1 de Arana-Southern), al mes de Howden como el
#     enviado; 3 Londres firmo (Southern) y queda solo Francia.
gg_ui_plata_phase3_mode = {
	value = 0
	if = {
		limit = {
			has_global_variable = gg_plata_intervention
			global_var:gg_plata_intervention_months >= 23
			# EN/ES: Codex 22/9 -- and Howden has actually come (gg_powers.14 sets the mark)
			#     / y Howden vino de verdad (gg_powers.14 pone la marca)
			has_global_variable = gg_cal_done_15
		}
		value = 2
	}
	# EN/ES: 22/9/2026, decided with Juan -- 4 France lifted her blockade (Gros, June 1848) with London
	#     still in the question; 3 London signed (always after the lifting: month 36 > 34); 5 a French
	#     intervention alone after the lifting / 4 Francia levanto su bloqueo con Londres todavia en la
	#     cuestion; 3 Londres firmo; 5 intervencion solo francesa tras el levantamiento
	if = { limit = { has_global_variable = gg_plata_ba_blockade_lifted } value = 4 }
	if = { limit = { has_global_variable = gg_plata_britain_settled } value = 3 }
	if = { limit = { has_global_variable = gg_plata_france_alone } value = 1 }
	if = { limit = { has_global_variable = gg_plata_france_alone has_global_variable = gg_plata_ba_blockade_lifted } value = 5 }
}

# EN/ES: 22/9/2026 -- what the powers have promised about the river: 1 London and Paris guarantee it
#     open, 2 London has recognised the right of blockade and only Paris watches, 3 both settled after
#     their intervention / lo que las potencias prometieron sobre el rio
gg_ui_plata_powers_word = {
	# EN/ES: 1 guaranteed by London and Paris, 2 guaranteed by Paris alone, 4 London has recognised the
	#     right of blockade and only Paris's patience counts, 3 both have recognised it (after their
	#     intervention or without it) / 1 garantia de los dos, 2 solo Paris, 4 Londres reconocio, 3 los dos
	value = 0
	if = { limit = { has_global_variable = gg_plata_guarantee } value = 1 }
	if = {
		limit = {
			has_global_variable = gg_plata_britain_recognizes
			NOT = { has_global_variable = gg_plata_intervention }
		}
		value = 4
	}
	# EN/ES: 22/9/2026, the powers' levels -- London out of the river by standing aside: 5 if Paris's
	#     patience still counts, 6 if no power's does / Londres al margen: 5 si cuenta Paris, 6 si nadie
	# EN/ES: River audit 22/9 -- 5 and 6 speak of the powers' patience, which only runs in the open war:
	#     not before the split / 5 y 6 hablan de la paciencia de las potencias, que solo corre con la
	#     guerra abierta: no antes de la particion
	if = {
		limit = {
			has_global_variable = guerra_grande_split_done
			NOT = { has_global_variable = gg_plata_britain_recognizes }
			NOT = { has_global_variable = gg_plata_intervention }
			NOT = { has_global_variable = gg_plata_guarantee }
			gg_plata_london_out = yes
		}
		value = 5
	}
	# EN/ES: «a closure brings no ultimatum» is false with one already pending / falso con uno pendiente
	if = {
		limit = {
			has_global_variable = guerra_grande_split_done
			NOT = { has_global_variable = gg_plata_ultimatum_pending }
			NOT = { has_global_variable = gg_plata_intervention }
			NOT = { has_global_variable = gg_plata_guarantee }
			NOT = { has_global_variable = gg_plata_france_recognizes }
			NOT = { gg_plata_france_counts = yes }
			NOT = { gg_plata_britain_counts = yes }
		}
		value = 6
	}
	if = {
		limit = {
			has_global_variable = gg_plata_guarantee
			OR = {
				has_global_variable = gg_plata_britain_recognizes
				has_global_variable = gg_plata_guarantee_france_only
			}
		}
		value = 2
	}
	if = {
		limit = {
			OR = {
				has_global_variable = gg_plata_intervention_over
				has_global_variable = gg_plata_france_recognizes
			}
		}
		value = 3
	}	# EN/ES: 22/9/2026 -- while France blockades Buenos Aires, that is the powers' word
	#     / mientras Francia bloquea Buenos Aires, esa es la palabra de las potencias
	if = { limit = { has_global_variable = gg_plata_french_blockade } value = 7 }
}

# EN/ES: 22/9/2026 (Juan: the bars do not appear until they matter) -- the patience from the first
#     month of a shut river, the hunger while a run of shut months is on
#     / las barras no aparecen hasta que corresponden
gg_ui_plata_patience_shown = {
	value = 0
	if = { limit = { has_global_variable = gg_plata_closed_total global_var:gg_plata_closed_total > 0 } value = 1 }
	# EN/ES: once the ministers have spoken the clock has done its work / cuando los ministros ya
	#     hablaron, el reloj cumplio su funcion
	if = {
		limit = {
			OR = {
				has_global_variable = gg_plata_ultimatum_pending
				has_global_variable = gg_plata_intervention
				# EN/ES: 22/9/2026 -- both powers recognised the right of blockade: no ultimatum left
				#     / las dos potencias reconocieron el derecho de bloqueo: ya no hay ultimatum
				has_global_variable = gg_plata_france_recognizes
				# EN/ES: 22/9/2026 -- no power counts by its level / ninguna potencia cuenta por su nivel
				AND = {
					NOT = { gg_plata_france_counts = yes }
					NOT = { gg_plata_britain_counts = yes }
				}
			}
		}
		value = 0
	}
}

#     a human Rosas closes it in the strait panel or not at all (the 1842 save: decreed, never closed).

#     tiene bateria, y los barcos igual pasan (la IA tarda semanas en aplicar una estrategia; un Rosas

		limit = {

			gg_plata_ba_fort = yes
			gg_plata_open = yes

	}

		limit = {

		}
		value = 2
	}
