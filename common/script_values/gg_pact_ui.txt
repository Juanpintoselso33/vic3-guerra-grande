# EN/ES: a province in arms against Buenos Aires exists / existe una provincia alzada contra Buenos Aires
gg_ui_pact_front_shown = {
 value = 0
 if = {
  limit = {
   exists = c:ARG
   OR = {
    AND = { exists = c:CRR has_global_variable = gg_delegation_gone_corrientes }
    AND = { exists = c:CLN has_global_variable = gg_delegation_gone_coalicion_tucuman }
    AND = { exists = c:ENR has_global_variable = gg_delegation_gone_entre_rios }
   }
  }
  value = 1
 }
}
# EN/ES: unrepaired grievances, while the province still delegates / agravios sin reparar, mientras delega

}
