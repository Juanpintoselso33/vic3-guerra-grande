# EN: Thresholds for the war bar. The bar used to be decoration: it moved and nothing followed.
#     These turn a lead of five battles into a real advantage, and ten into a decision.
# ES: Umbrales de la barra de la guerra. La barra era decorativa: se movia y no pasaba nada.
#     Con esto, una ventaja de cinco batallas pesa de verdad, y una de diez decide la guerra.

guerra_grande_blanco_lead = {
	value = global_var:guerra_grande_blanco_score
	subtract = global_var:guerra_grande_colorado_score
}

guerra_grande_colorado_lead = {
	value = global_var:guerra_grande_colorado_score
	subtract = global_var:guerra_grande_blanco_score
}

# EN: C10 (22/9/2026) -- the river adds its weight through the signed counter of the River Plate
#     (common/script_values/gg_rio_plata.txt), which replaced the siege and relief stocks.
# ES: C10 -- el rio suma su peso por el contador con signo del Plata, que reemplazo a los stocks del
#     sitio y del auxilio.
gg_war_balance_blanco = {
	value = guerra_grande_blanco_lead
	add = gg_plata_pressure_blanco
	subtract = gg_plata_pressure_colorado
}

gg_war_balance_colorado = {
	value = guerra_grande_colorado_lead
	add = gg_plata_pressure_colorado
	subtract = gg_plata_pressure_blanco
}

# EN: T3a. Total backing of each side (to-be.md section 2); T3b makes it drain war support.
# ES: T3a. Sosten total de cada bando (to-be.md seccion 2); en T3b drena el apoyo a la guerra.
# EN/ES: 22/9/2026, the powers' levels -- only a power that is Protector or Guarantor backs the war;
#     a Spectator does not (and Brazil is a Spectator before 1850) / solo respalda la guerra la potencia
#     Protectora o Garante; la Espectadora no (y el Brasil lo es antes de 1850)
gg_defensa_backing = {
	value = 0
	if = { limit = { gg_patron_level_francia >= 1 } add = global_var:gg_patron_francia }
	if = { limit = { gg_patron_level_britania >= 1 } add = global_var:gg_patron_britania }
	if = { limit = { gg_patron_level_brasil >= 1 } add = global_var:gg_patron_brasil }
	# EN/ES: T7 -- Urquiza counts for the side he is on / Urquiza cuenta para el lado donde esta
	if = {
		limit = { has_global_variable = gg_urquiza_flipped }
		add = global_var:gg_patron_urquiza
	}
}

gg_cerrito_backing = {
	value = global_var:gg_patron_rosas
	if = {
		limit = { NOT = { has_global_variable = gg_urquiza_flipped } }
		add = global_var:gg_patron_urquiza
	}
}

# EN: T5, F0-5. How hard Rivera hits at Yucutuja: 2, +1 if the border stayed open to the
#     riograndenses, -1 if Oribe helped the Empire close it, +1 if the northern caudillos were
#     passed over by the government's political chiefs, -1 if they were left in charge.
# ES: T5, F0-5. Cuanto pega Rivera en Yucutuja: 2, +1 si la frontera quedo abierta a los
#     riograndenses, -1 si Oribe ayudo al Imperio a cerrarla, +1 si los caudillos del norte fueron
#     desplazados por jefes politicos, -1 si se los dejo al frente.
gg_rivera_1837_strength = {
	value = 2
	if = {
		limit = { has_global_variable = gg_farrapos_closed }
		add = -1
	}
	else = {
		add = 1
	}
	if = {
		limit = { has_global_variable = gg_depts_state_chiefs }
		add = 1
	}
	else_if = {
		limit = { has_global_variable = gg_depts_local_caudillos }
		add = -1
	}
	# EN/ES: balance 2026-09-17 -- capped at 1..3 (was 0..4); min/max run in order, so last / acotado a 1..3 (era 0..4); van al final
	min = 1
	max = 3
}

# EN: T3b, C19. Monthly war-support drain for a side whose patrons fall below 40 of backing:
#     one point for every 5 missing (negative, ready for add_war_war_support). Provisional (T10).
# ES: T3b, C19. Drenaje mensual de apoyo a la guerra para el bando cuyos padrinos suman menos de
#     40: un punto por cada 5 que faltan (negativo). Provisorio hasta T10.
gg_defensa_drain = {
	value = gg_defensa_backing
	subtract = 40
	divide = 5
}

gg_cerrito_drain = {
	value = gg_cerrito_backing
	subtract = 40
	divide = 5
}

# EN: T3b, C19. Monthly money from the patrons who pay: France and Brazil to the Defensa, Rosas to
#     the Cerrito. 30 per point of support is provisional (T10).
# ES: T3b, C19. Plata mensual de los padrinos que pagan: Francia y Brasil a la Defensa, Rosas al
#     Cerrito. 30 por punto de apoyo es provisorio (T10).
# EN/ES: 22/9/2026 -- one payment, not two: a power pays only as Protector or Guarantor; the historical
#     subsidies (the French convention of 1848, the Brazilian treaties of 1851, Rosas's backing) no
#     longer mint money of their own: their money is this payment / un solo pago: la potencia paga solo
#     como Protectora o Garante; los subsidios historicos ya no acunan plata propia: su plata es este pago
gg_defensa_subsidy = {
	value = 0
	# EN/ES: renouncing the subsidy (May 1851) is renouncing France's money, even while her squadron or
	#     her guarantee keeps her Guarantor / renunciar al subsidio es renunciar a la plata francesa
	if = {
		limit = {
			gg_patron_level_francia >= 1
			NOT = { has_global_variable = gg_french_subsidy_renounced }
		}
		add = global_var:gg_patron_francia
	}
	if = { limit = { gg_patron_level_brasil >= 1 } add = global_var:gg_patron_brasil }
	multiply = 30
}

gg_cerrito_subsidy = {
	value = global_var:gg_patron_rosas
	multiply = 30
	# EN/ES: explicit withdrawal stops the monthly cash / el retiro expreso corta la caja mensual
	if = {
		limit = { has_global_variable = gg_int_arg_withdrawn }
		multiply = 0
	}
}

# EN: balance 2026-09-17. What the split takes off the lead: half of it, rounded up, so half of
#     the lead (rounded down) is left. / ES: lo que la particion le saca a la ventaja: la mitad
#     redondeada para arriba, asi queda la mitad redondeada para abajo.
gg_split_trim = {
	value = guerra_grande_blanco_lead
	if = {
		limit = { guerra_grande_blanco_lead < 0 }
		multiply = -1
	}
	divide = 2
	round = yes
}

# EN/ES: the war bar's value, clamped to its range. The bar reads the COLORADO lead (colorados on
#     the left since 2026-09-17) / el valor de la barra; lee la ventaja COLORADA
gg_war_score_bar_value = {
	value = gg_war_balance_colorado
	min = -10
	max = 10
}

# EN: T2, the arbiter's tie-break (playtest 2026-09-19). Positive means the Defensa is the more
#     exhausted of the two; negative, the Cerrito. / ES: positivo, la Defensa esta mas agotada que
#     el Cerrito; negativo, al reves.
gg_exhaustion_gap = {
	value = global_var:gg_defensa_exhausted
	subtract = global_var:gg_cerrito_exhausted
}

# EN: what is left of the Ejercito Unido de Vanguardia when it crosses, read from how the interior
#     campaign ended. Five battalions if Rosas had the interior quiet, three if it cost him.
# ES: lo que queda del Ejercito Unido de Vanguardia cuando cruza, leido de como termino la campania
#     del interior. Cinco batallones si Rosas tuvo el interior quieto, tres si le costo.
gg_vanguardia_line = {
	value = 3
	if = {
		limit = { global_var:gg_interior_outcome = 2 }
		subtract = 1
	}
	# EN: AND ONE FEWER IF ROSAS NEVER CALLED HIM PRESIDENT. On the day Oribe landed in Buenos Aires
	#     the player chose between receiving him as the lawful president of the Oriental Republic --
	#     which is what Rosas did, and what cost him abroad for the rest of his life -- and taking him
	#     in as a soldier and saying nothing. Until 20/9/2026 that choice wrote a mark that NOBODY
	#     READ: the two roads did exactly the same thing. An army lent to a foreign president in exile
	#     is a different thing from an army lent to a general on the payroll, and the difference shows
	#     up here, where the Vanguardia is counted out.
	# ES: Y UNO MENOS SI ROSAS NUNCA LO LLAMO PRESIDENTE. El dia que Oribe desembarco en Buenos Aires
	#     el jugador eligio entre recibirlo como presidente legal de la Republica Oriental --que es lo
	#     que hizo Rosas, y lo que le costo afuera el resto de su vida-- y recibirlo de militar sin
	#     decir nada de la presidencia. Hasta el 20/9/2026 esa eleccion escribia una marca que NO LEIA
	#     NADIE: los dos caminos hacian exactamente lo mismo. Un ejercito prestado a un presidente
	#     extranjero en el exilio no es lo mismo que un ejercito prestado a un general a sueldo, y la
	#     diferencia se ve aca, cuando se cuenta la Vanguardia.
	if = {
		limit = { NOT = { has_global_variable = gg_oribe_recognised } }
		subtract = 1
	}
	if = {
		limit = { global_var:gg_vanguardia_cruce = 2 }
		subtract = 1
	}
	# EN/ES: y un piso de uno: dos recortes sobre dos caballerias dan cero, y un contingente de cero
	#     unidades no es un contingente / and a floor of one: two cuts on two would give zero
	min = 1
}

# EN: 23/9/2026 (Juan: "the war should end on historical dates, give or take one or two years") -- the
#     lead the arbiter needs to call a government beaten. The siege did not break for eight years: before
#     1849 only a rout counts (15); from 1849, when both governments were at the end of their means, a
#     clear lead (10); from 1851, the year Urquiza's pronouncement and the Brazilian alliance broke it, a
#     modest one (5, as before). Headless run 21 ended by score in November 1847 at 10.
# ES: la ventaja que el arbitro exige para dar por batido a un gobierno. El sitio no se quebro en ocho
#     anios: antes de 1849 solo una paliza (15); desde 1849, una ventaja clara (10); desde 1851, el anio de
#     Urquiza y la alianza con el Brasil, una modesta (5, como antes).
gg_arbiter_balance_needed = {
	value = 15
	if = { limit = { year >= 1849 } value = 10 }
	if = { limit = { year >= 1851 } value = 5 }
}
