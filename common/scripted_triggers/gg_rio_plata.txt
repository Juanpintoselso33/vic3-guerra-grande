# EN: THE RIVER PLATE QUESTION (C10, decided 22/9/2026, docs/guerra-grande-flujo/rio-plata-spec.md).
#     The river has one state, and the engine computes it: whoever controls each bank of the
#     Montevideo - Buenos Aires strait, whether that bank has a naval fortification (without one no
#     restriction applies), and what its controller set in the strait panel. The most severe bank
#     wins. This file only READS that state; nothing here adds up patrons or thresholds (the 60-point
#     sea was rejected on 20/9/2026 and is gone).
# ES: LA CUESTION DEL PLATA. El rio tiene un solo estado, y lo calcula el motor: quien controla cada
#     orilla del estrecho Montevideo - Buenos Aires, si esa orilla tiene fortificacion naval (sin ella
#     no se aplica ninguna restriccion) y que configuro su controlador. Manda la orilla mas severa.
#     Este archivo solo LEE ese estado; aca no se suman padrinos ni umbrales.

# EN: The Buenos Aires bank (x365A5B) and the Montevideo bank (x302DDA) are the land endpoints of
#     strait_rio_de_la_plata.
# ES: La orilla de Buenos Aires (x365A5B) y la de Montevideo (x302DDA) son los extremos del estrecho.
# EN/ES: only Buenos Aires's battery counts; Montevideo never shuts the river / solo cuenta la bateria
#     de Buenos Aires; Montevideo nunca cierra el rio
gg_plata_ba_fort = {
	p:x365A5B.state ?= { has_building = building_naval_fortification }
}
# EN: Vic3 has no state-controller trigger (occupation is per province), so this reads the owner.
#     has_any_strait_control was tried and rejected at load (22/9/2026: it takes a country target,
#     not "yes"). The closure itself is the engine's answer (gg_plata_closed), which already folds
#     occupation in; only the Overview's label can lag behind an occupation.
# ES: Vic3 no tiene trigger de controlador de estado, asi que se lee el dueno. has_any_strait_control
#     se probo y el motor lo rechazo al cargar (lleva un pais como objetivo, no "yes"). El cierre en
#     si es la respuesta del motor, que ya incluye la ocupacion.
gg_plata_ba_held_by_rosas = {
	exists = c:ARG
	p:x365A5B.state ?= { owner ?= c:ARG }
}

# EN: Is the River Plate shut to foreign flags? Rosas holds his bank, it is fortified, and British
#     warships cannot pass Argentina's straits. The last clause is the engine's own answer, and it
#     already folds in the controller's setting, its permissions and occupation. This is the ONE
#     place that decides it: if in-game test T3 shows that trigger does not follow the closure, only
#     this block changes.
# ES: ¿Esta el Plata cerrado a las banderas extranjeras? Rosas tiene su orilla, esta fortificada y
#     los barcos de guerra britanicos no pasan por los estrechos argentinos. Lo ultimo es la
#     respuesta del propio motor, que ya incluye la configuracion, los permisos y la ocupacion. Es el
#     UNICO lugar que lo decide: si la prueba T3 muestra que ese trigger no sigue al cierre, solo
#     cambia este bloque.
gg_plata_closed = {
	gg_plata_ba_held_by_rosas = yes
	gg_plata_ba_fort = yes
	exists = c:GBR
	c:GBR = { NOT = { has_military_strait_access_through = c:ARG } }
}

gg_plata_open = {
	NOT = { gg_plata_closed = yes }
}

# EN: A port blockade is not a river closure. The French (1838-1840) and Anglo-French (1845-1848)
#     blockades shut Rosas's own port; they never isolated Montevideo (C10).
# ES: Un bloqueo de puerto no es un cierre del rio. Los bloqueos frances y anglo-frances cerraban el
#     puerto de Rosas; nunca aislaron a Montevideo (C10).
gg_plata_ba_port_blockaded = {
	p:x365A5B.state ?= { blockade_level > 0 }
}

# EN: B. The powers' patience runs out: enough months of a shut river, and London or Paris no longer
#     cordial with Rosas. Twelve months instead of eighteen once the Defensa has called the
#     squadrons. Relations are read toward prev, the form vanilla uses.
# ES: B. Se acaba la paciencia de las potencias: bastantes meses de rio cerrado, y Londres o Paris
#     ya no cordiales con Rosas. Doce meses en vez de dieciocho si la Defensa llamo a las escuadras.
gg_plata_patience_spent = {
	global_var:gg_plata_closed_total >= gg_plata_patience_threshold
	# EN/ES: relations toward prev (= Argentina): "relations:scope:x" does not load (22/9/2026, "More
	#     than one colon in event target link"); vanilla only uses root/prev/this / relaciones hacia prev
	# EN/ES: 22/9/2026 -- once London has recognised the right of blockade (Southern without a war,
	#     gg_plata_britain_recognizes), only Paris's patience counts / cuando Londres ya reconocio el
	#     derecho de bloqueo, solo cuenta la paciencia de Paris
	c:ARG ?= {
		# EN/ES: 22/9/2026 -- only the powers that count by their level (gg_patron_levels.txt): a
		#     recognised right of blockade makes that power a Spectator / solo las potencias que
		#     cuentan por su nivel: reconocer el derecho de bloqueo la vuelve Espectadora
		OR = {
			AND = {
				gg_plata_britain_counts = yes
				c:GBR ?= { relations:prev < relations_threshold:cordial }
			}
			AND = {
				gg_plata_france_counts = yes
				c:FRA ?= { relations:prev < relations_threshold:cordial }
			}
		}
	}
}

gg_plata_intervention_due = {
	has_global_variable = guerra_grande_split_done
	has_global_variable = guerra_grande_open_war
	NOT = { has_global_variable = gg_plata_intervention }
	NOT = { has_global_variable = gg_plata_ultimatum_pending }
	# EN/ES: under the guarantee a closure is a broken word, not a matter of patience (gg_plata.26)
	#     / bajo la garantia un cierre es faltar a la palabra, no cuestion de paciencia
	NOT = { has_global_variable = gg_plata_guarantee }
	# EN/ES: 23/9/2026 (Juan, D2) -- no ultimatum before 1 July 1845: the notes of May and June 1845 and the
	#     ultimatum of 21 July (SOURCES_powers.md section 8) / ningun ultimatum antes del 1/7/1845
	game_date >= 1845.7.1
	# EN/ES: D3 -- after the settlement the answer is Brazil's, not an ultimatum / despues del arreglo responde el Brasil
	NOT = { gg_plata_settled = yes }
	gg_plata_patience_spent = yes
}

# EN: The intervention chain. Each dated event of the powers waits for the intervention and for its
#     own distance from the ultimatum, so an intervention that comes late keeps the historical
#     order. The relays already retry every month until the recipient's trigger lets them through.
# ES: La cadena de la intervencion. Cada evento fechado de las potencias espera a la intervencion y a
#     su propia distancia desde el ultimatum, asi una intervencion tardia conserva el orden
#     historico. Los relevos ya reintentan cada mes hasta que el trigger del destinatario los deja.
# EN: 23/9/2026 (Juan, D2) -- ONE CLOCK FOR BOTH BANKS. Every link also waits for its historical DATE:
#     the intervention's events never fire before the day they happened, on either bank, and a late
#     intervention still keeps its months between links.
# ES: UN SOLO RELOJ PARA LAS DOS ORILLAS. Cada eslabon espera ademas su FECHA historica: los eventos de la
#     intervencion nunca salen antes del dia en que pasaron, en ninguna orilla, y una intervencion tardia
#     conserva sus meses entre eslabones.
gg_plata_intervention_at = {
	has_global_variable = gg_plata_intervention
	global_var:gg_plata_intervention_months >= $MONTHS$
	game_date >= $DATE$
	# EN/ES: 22/9/2026 -- a French intervention alone (after Southern without a war) runs none of the
	#     British links / una intervencion solo francesa no corre ningun eslabon britanico
	NOT = { has_global_variable = gg_plata_france_alone }
}
# EN/ES: the links that also belong to a French intervention alone: the blockade of Buenos Aires and
#     the French settlement / los eslabones que tambien son de una intervencion solo francesa
gg_plata_intervention_at_any = {
	has_global_variable = gg_plata_intervention
	global_var:gg_plata_intervention_months >= $MONTHS$
	game_date >= $DATE$
}

# EN: 22/9/2026 -- THE POWERS' GUARANTEE (decided with Juan, alternative A of the review of the
#     close -> ultimatum -> yield -> close cycle). Rosas yielded to the ultimatum: London and Paris
#     stand as guarantors of the open river. It ends with Southern without a war: London recognises
#     the right of blockade when it is cordial with Buenos Aires again, or at the date of the
#     Arana-Southern treaty (24/11/1849, SOURCES_powers.md section 6).
# ES: LA GARANTIA DE LAS POTENCIAS. Rosas cedio al ultimatum: Londres y Paris quedan como garantes
#     del rio abierto. Termina con Southern sin guerra: Londres reconoce el derecho de bloqueo cuando
#     vuelve a estar cordial con Buenos Aires, o en la fecha del tratado Arana-Southern.
gg_plata_southern_without_war_due = {
	has_global_variable = gg_plata_guarantee
	# EN/ES: a guarantee London never gave is not London's to end / una garantia que Londres no dio
	NOT = { has_global_variable = gg_plata_guarantee_france_only }
	NOT = { has_global_variable = gg_plata_intervention }
	NOT = { has_global_variable = gg_plata_britain_recognizes }
	NOT = { has_global_variable = gg_plata_ultimatum_pending }
	OR = {
		game_date >= 1849.11.24
		c:ARG ?= { c:GBR ?= { relations:prev >= relations_threshold:cordial } }
	}
}

# EN: Paris settles too, without a war: the Arana-Lepredour convention (31/8/1850, SOURCES_powers.md
#     section 6), or Paris cordial with Buenos Aires again. From then on no power answers a closure.
# ES: Paris tambien arregla, sin guerra: la convencion Arana-Lepredour, o Paris otra vez cordial con
#     Buenos Aires. Desde ahi ninguna potencia responde a un cierre.
gg_plata_lepredour_without_war_due = {
	# EN/ES: London out of the river, by Southern or by standing aside / Londres fuera del rio
	OR = {
		has_global_variable = gg_plata_britain_recognizes
		has_global_variable = gg_plata_guarantee_france_only
		gg_plata_london_out = yes
	}
	NOT = { has_global_variable = gg_plata_france_recognizes }
	NOT = { has_global_variable = gg_plata_intervention }
	NOT = { has_global_variable = gg_plata_ultimatum_pending }
	OR = {
		game_date >= 1850.8.31
		c:ARG ?= { c:FRA ?= { relations:prev >= relations_threshold:cordial } }
	}
}

# EN: 23/9/2026 (Juan, D3) -- THE SETTLEMENT. Both powers have recognised the right of blockade (Lepredour,
#     after the intervention or without it). A closure is still possible, but it is no longer free: the
#     Littoral wants to navigate the Parana (Urquiza's backing and grievance) and, from 1850, the Empire of
#     Brazil counts the shut months as the powers once did (SOURCES_powers.md section 12; Brazil broke with
#     Rosas in 1850 and its squadron forced the Parana at Tonelero on 17/12/1851).
# ES: EL ARREGLO. Las dos potencias reconocieron el derecho de bloqueo. Cerrar sigue siendo posible, pero ya
#     no sale gratis: el Litoral quiere navegar el Parana y, desde 1850, el Imperio cuenta los meses cerrados.
gg_plata_settled = {
	OR = {
		has_global_variable = gg_plata_intervention_over
		has_global_variable = gg_plata_france_recognizes
	}
}
gg_plata_brazil_counts = {
	gg_plata_settled = yes
	year >= 1850
	exists = c:BRZ
	NOT = { has_global_variable = gg_plata_brazil_forced }
}
# EN/ES: Brazil's patience is spent: its squadron forces the passage (gg_plata_brazil_forces)
#     / se acabo la paciencia del Brasil: su escuadra fuerza el paso
gg_plata_brazil_answer_due = {
	gg_plata_brazil_counts = yes
	gg_plata_closed = yes
	has_global_variable = gg_plata_closed_total
	global_var:gg_plata_closed_total >= gg_plata_patience_threshold
}

# EN: 23/9/2026 (consequence of D2) -- THE CITY IS NOT STARVING while the powers stand by it. With the
#     ultimatum floored at 1/7/1845, a closure from mid-1843 would reach 24 months of hunger by mid-1845,
#     before London and Paris could answer. Once their patience is spent, their stations escort the
#     merchantmen into Montevideo while the ministers wait for instructions (the notes of May and June
#     1845, SOURCES_powers.md section 12), and while their squadrons are in the river the city is supplied
#     by them: the run of hunger holds, it neither grows nor resets. Calling the squadrons (patience 12)
#     brings this sooner, so the Defensa's call is a real lever.
# ES: LA PLAZA NO SE MUERE DE HAMBRE mientras las potencias la sostienen. Con el ultimatum desde el 1/7/1845,
#     un cierre desde mediados de 1843 llegaba a 24 meses de hambre antes de que Londres y Paris pudieran
#     contestar. Agotada su paciencia, sus estaciones escoltan a los mercantes mientras los ministros esperan
#     instrucciones, y con sus escuadras en el rio la plaza se abastece: la racha del hambre se sostiene, ni
#     crece ni se corta. Llamar a las escuadras lo adelanta: el llamado de la Defensa es una palanca real.
gg_plata_hunger_held = {
	OR = {
		AND = {
			NOT = { gg_plata_settled = yes }
			NOT = { has_global_variable = gg_plata_intervention }
			gg_plata_patience_spent = yes
		}
		has_global_variable = gg_plata_ultimatum_pending
		AND = {
			has_global_variable = gg_plata_intervention
			NOT = { has_global_variable = gg_plata_intervention_over }
		}
	}
}
