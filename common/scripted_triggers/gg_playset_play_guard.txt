# EN: 25/9/2026 (AUD-PLAYSET-PLAYS) -- a civil war: the country against its own rebels, not a foreign power reaching
#     into the Guerra Grande. The same exemption as gg_war_goal_is_allowed and as the static list (dp_revolution,
#     dp_secession...), applied to an optional mod's revolution play. Scope: a diplomatic play (its initiator/target
#     links, as in gg_farrapos_play_side_has).
# ES: una guerra civil: el pais contra sus propios rebeldes, no una potencia extranjera. La misma excepcion que
#     gg_war_goal_is_allowed y que la lista estatica, para la jugada de revolucion de un mod opcional. Scope: una jugada.
gg_play_is_civil_war = {
	OR = {
		initiator ?= {
			OR = {
				is_revolutionary = yes
				is_secessionist = yes
			}
		}
		target ?= {
			OR = {
				is_revolutionary = yes
				is_secessionist = yes
			}
		}
	}
}

# EN: 25/9/2026 (AUD-PLAYSET-PLAYS) -- one of the mod's own plays (script only; each has its own rules, and
#     dp_gg_farrapos is Ponche Verde's). Scope: a diplomatic play.
# ES: una de las jugadas propias del mod (solo por script). Scope: una jugada.
gg_play_is_mod_own = {
	OR = {
		is_diplomatic_play_type = dp_guerra_grande
		is_diplomatic_play_type = dp_gg_pacto_federal
		is_diplomatic_play_type = dp_gg_buenos_aires
		is_diplomatic_play_type = dp_gg_cisplatina
		is_diplomatic_play_type = dp_gg_farrapos
	}
}

# EN: 25/9/2026 (AUD-PLAYSET-PLAYS) -- a country no war goal may be aimed at by scope:actor (the goal holder): an
#     Oriental government (or the Republic before the partition) until the signed peace, and Brazil / the
#     Riograndense Republic against each other until Ponche Verde -- the rules of gg_war_goal_is_allowed. Argentina's
#     clause is left out on purpose: it only forbids annexing or conquering Argentina outright, which no goal of the
#     optional mods of the playset does, and Argentina backs plays all over the continent. Country scope.
# ES: un pais contra el que scope:actor (el dueno del objetivo) no puede apuntar un objetivo de guerra: un gobierno
#     oriental hasta la paz, y el Brasil y la Republica Riograndense entre si hasta Ponche Verde. La clausula de la
#     Argentina queda afuera a proposito: solo prohibe anexarla o conquistarla entera, cosa que ningun objetivo de los
#     mods opcionales hace, y la Argentina respalda jugadas en todo el continente. Scope de pais.
gg_goal_enemy_is_protected = {
	OR = {
		AND = {
			gg_is_oriental_side = yes
			NOT = { has_global_variable = gg_peace_settled }
		}
		AND = {
			NOT = { has_global_variable = gg_farrapos_ponche_verde }
			exists = c:PNI
			OR = {
				AND = {
					c:PNI ?= this
					scope:actor ?= { c:BRZ ?= this }
				}
				AND = {
					c:BRZ ?= this
					scope:actor ?= { c:PNI ?= this }
				}
			}
		}
	}
}
