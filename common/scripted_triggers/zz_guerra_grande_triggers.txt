# EN: Is this diplomatic play a foreign power reaching into the Guerra Grande?
#     Juan's design rule for this mod, stated twice and without room for doubt: "que no haya
#     intervenciones directas de Brasil ni Argentina, sino mediante tropas o divisiones" and
#     "que sea siempre por contingentes, pq si entra argentina o brasil la guerra termina
#     enseguida". We had blocked third parties from JOINING our play, but nothing stopped them
#     from opening their own -- and in a test game Brazil opened a return_state play on the
#     Cerrito in 1847 demanding Maldonado, Paysandu, Cerro Largo and San Jose back for the
#     Defensa, 55 mobilised against 14. That is the war ending in an afternoon, exactly as he said.
#     This allows the play only when the two governments are fighting each other, or when it is
#     1851 and the intervention is the historical one: Caxias crossing the border in September of
#     that year, which is told by events (gg_notice.2, after the Convention of Montevideo).
# ES: esta jugada diplomatica, es una potencia extranjera metiendose en la Guerra Grande?
#     La regla de diseño de Juan para este mod, dicha dos veces y sin lugar a dudas: "que no haya
#     intervenciones directas de Brasil ni Argentina, sino mediante tropas o divisiones" y "que sea
#     siempre por contingentes, pq si entra argentina o brasil la guerra termina enseguida".
#     Teniamos bloqueado que terceros se SUMARAN a nuestra jugada, pero nada les impedia abrir la
#     suya: en una partida de prueba Brasil le abrio al Cerrito una jugada de return_state en 1847
#     exigiendo Maldonado, Paysandu, Cerro Largo y San Jose para la Defensa, 55 movilizados contra
#     14. Eso es la guerra terminada en una tarde, exactamente lo que el dijo.
#     Esto permite la jugada solo si los dos gobiernos se pelean entre si, o si ya es 1851 y la
#     intervencion es la historica: Caxias cruzando la frontera en setiembre de ese anio, que la
#     cuentan los eventos (gg_notice.2, despues de la Convencion de Montevideo).
guerra_grande_play_is_allowed = {
	# EN/ES: 24/9/2026 (Juan, AUD-ARG-1850) -- nobody annexes or conquers Argentina while its dated calendar is
	#     pending (common/scripted_triggers/gg_arg_identity.txt) / nadie anexa ni conquista a la Argentina
	#     mientras su calendario fechado este pendiente
	gg_arg_target_is_allowed = yes
	# EN/ES: 23/9/2026 (Juan, AUD-FARRAPOS-A) -- no engine war Brazil-Riograndense Republic before Ponche Verde
	#     (common/scripted_triggers/gg_farrapos.txt) / ninguna guerra del motor Brasil-Republica Riograndense antes de Ponche Verde
	gg_farrapos_play_is_allowed = yes
	# EN: bug hunt 23/9/2026 (CB-01) -- the other direction. The Republic holds a claim on Rio Grande do
	#     Sul from 1836 (guerra_grande_on_actions.txt), so an Oriental AI could open return_state on
	#     Brazil and hand the Empire the war the target-side guard forbids. Until the peace, neither the
	#     Republic before the partition nor either wartime government opens one of these plays.
	# ES: la otra direccion. La Republica tiene un reclamo sobre Rio Grande do Sul desde 1836, asi que
	#     una IA oriental podia abrirle un return_state al Brasil y darle al Imperio la guerra que la
	#     guarda del objetivo prohibe. Hasta la paz, ni la Republica antes de la particion ni ninguno de
	#     los dos gobiernos abre una de estas jugadas.
	NOT = {
		AND = {
			gg_is_oriental_side = yes
			NOT = { has_global_variable = gg_peace_settled }
		}
	}
	OR = {
		# the target is nobody's wartime government
		scope:target_country = {
			NOR = {
				has_variable = guerra_grande_cerrito_government
				has_variable = guerra_grande_defensa_government
				# EN: bug hunt 23/9/2026 (CB-01) -- nor the Republic before the partition. The marks
				#     above only exist from guerra_grande.3, and in run 211244 Brazil took all five
				#     Oriental states on 31/1/1842, before there was any Guerra Grande to protect.
				# ES: ni la Republica antes de la particion. Las marcas de arriba existen recien desde
				#     guerra_grande.3, y en la corrida 211244 el Brasil se quedo con los cinco estados
				#     orientales el 31/1/1842, antes de que hubiera Guerra Grande que proteger.
				AND = {
					c:URU ?= this
					NOT = { has_global_variable = guerra_grande_split_done }
				}
			}
		}
		# EN/ES: 23/9/2026 (audit R5) -- no longer "or it is the other government": the siege is not an
		#     engine war, and a play between the two governments would reopen it / ya no se permite la
		#     jugada entre los dos gobiernos: reabriria la guerra del motor que el sitio saco
		# EN: Codex review 2026-09-19 -- this said only "year >= 1851", which from 1 January 1851 let
		#     ANY country in the world open a play against either oriental government, not just the
		#     historical intervention. That is also the main way a foreign power ends up holding
		#     Montevideo, which is the war's failure state. The hour is Brazil's: Caxias crossing the
		#     border in September, which is told by events (gg_notice.2).
		# ES: decia solo "year >= 1851", y desde el 1 de enero de 1851 cualquier pais del mundo podia
		#     abrirle una jugada a cualquiera de los dos gobiernos orientales, no solo la intervencion
		#     historica. Es ademas el camino principal por el que una potencia extranjera termina con
		#     Montevideo, que es el fracaso de la guerra. La hora es la del Brasil: Caxias cruzando la
		#     frontera en setiembre, que lo cuentan los eventos (gg_notice.2).
		# EN: CB-01, Codex review 23/9/2026 -- the 1851 door for Brazil is closed: decisiones.md (23/9, "the siege
		#     without the engine's war") keeps Brazil out until the signed peace, not until 1851, and Caxias's
		#     crossing is told by gg_notice.2, which opens no play. After the peace the Republic is an ordinary
		#     country again: the marks never go away, so without this it stayed untouchable for ever.
		# ES: se cierra la puerta de 1851 para el Brasil: decisiones.md deja al Brasil afuera hasta la paz firmada,
		#     no hasta 1851, y el cruce de Caxias lo cuenta gg_notice.2, que no abre ninguna jugada. Despues de la
		#     paz la Republica vuelve a ser un pais comun: las marcas no se borran, y sin esto quedaba intocable.
		has_global_variable = gg_peace_settled
	}
}

# EN: CB-01 (23/9/2026) -- a country that is one side of the Guerra Grande: either wartime government,
#     or the Republic itself before the partition. Country scope.
# ES: un pais que es un bando de la Guerra Grande: cualquiera de los dos gobiernos, o la Republica
#     misma antes de la particion. Scope de pais.
gg_is_oriental_side = {
	OR = {
		has_variable = guerra_grande_cerrito_government
		has_variable = guerra_grande_defensa_government
		AND = {
			c:URU ?= this
			NOT = { has_global_variable = guerra_grande_split_done }
		}
	}
}

# EN: 23/9/2026 (Juan: every war play against the Oriental governments guarded until the signed peace) -- the
#     same guard for WAR GOALS, which is what an enemy adds against a country already inside a play (an ally
#     called in, a defensive pact). Scopes as in a war goal type: root = the holder, scope:target_country.
#     Civil wars are the country against its own rebels, not a foreign power: they pass, as does a goal with
#     no target country. Both directions, like the play guard: before the peace an Oriental government adds no
#     goal either (the war is fought with contingents, never as a belligerent of the engine).
# ES: la misma guarda para los OBJETIVOS DE GUERRA, que es lo que un enemigo agrega contra un pais que ya esta
#     en una jugada (un aliado llamado, un pacto defensivo). Las guerras civiles son el pais contra sus propios
#     rebeldes, no una potencia extranjera: pasan, igual que un objetivo sin pais objetivo. En los dos sentidos,
#     como la guarda de jugadas: antes de la paz un gobierno oriental tampoco agrega objetivos.
gg_war_goal_is_allowed = {
	OR = {
		NOT = { exists = scope:target_country }
		# EN/ES: 24/9/2026 (AUD-ARG-1850) -- the goal of one of the mod's own wars on Argentina (Cisplatine, Buenos
		#     Aires up to Pavon) / el objetivo de una de las guerras propias del mod contra la Argentina
		scope:diplomatic_play ?= { gg_is_mod_play_on_arg = yes }
		is_revolutionary = yes
		is_secessionist = yes
		scope:target_country = {
			OR = {
				is_revolutionary = yes
				is_secessionist = yes
			}
		}
		guerra_grande_play_is_allowed = yes
	}
}

# EN: 23/9/2026 (Juan: after the peace the elections stay free, but foreigners cannot govern) -- character
#     scope. A foreigner is anyone whose culture is not the Oriental one (platinean: the Republic's only primary
#     culture) -- the legion colonels, Garibaldi, Brown, any French or Italian the engine spawns -- or an
#     Argentine the mod brought to the war (gg_foreign_commander: Paz, platinean by culture).
# ES: despues de la paz las elecciones siguen libres, pero un extranjero no gobierna. Extranjero es quien no es
#     de cultura oriental (platinean, la unica primaria de la Republica) o un argentino que el mod trajo a la
#     guerra (gg_foreign_commander: Paz, rioplatense de cultura).
gg_is_foreigner = {
	OR = {
		NOT = { has_culture = cu:platinean }
		has_variable = gg_foreign_commander
	}
}

#     told there were neither victors nor vanquished, and the same position could hand the winner's
#     menu to the government that had just lost its capital. The military result is what decides
#     now, and it is read with occupation included: enemy_occupation is the engine's own weighted
#     share of a country under enemy control, so a capital held but not yet annexed still counts.
#     Winning also requires not being just as beaten: half of your own ground in enemy hands is a
#     stalemate, not a victory.
# ES: T1/T9, quien gano la guerra. El cierre leia al ganador del puntaje politico, asi que un
#     Cerrito con Montevideo tomada podia recibir «ni vencedores ni vencidos», y en la misma
#     posicion el menu del ganador podia caerle al gobierno que acababa de perder su capital.

#     pais en manos enemigas segun el motor, asi que una capital tomada y todavia no anexada

	exists = c:CER
	OR = {
		NOT = { exists = c:URU }
	}

# EN: 24/9/2026 (Juan) -- the Defensa is under pressure: the only moment the AI reinforces a legion it already raised
#     (guerra_grande_italian_legion_button, guerra_grande_french_legion_button). Before this the AI pressed both
#     reinforcements with 50-60 % a month and reinforced in every story (France -5 each). Three of the mod's own measures,
#     chosen with the balance lab (tools/lab_balance, «legion_pressure»): its war support is falling (endurance 50 or
#     less), the blancos lead the war's balance by 12 or more, or the river has been shut for 18 months in a row (three
#     quarters of the hunger that ripens the Defensa). With lower marks (8 and 12) the lab still saw a reinforcement in
#     96 % of the AI's stories: the river's early blockade and an ordinary blanco lead are the siege's normal state.
# ES: la Defensa esta bajo presion: el unico momento en que la IA refuerza una legion ya levantada. Antes la IA apretaba
#     los dos refuerzos con 50-60 % por mes y reforzaba en toda historia. Tres medidas del propio mod, elegidas con el
#     laboratorio: el aguante cae (50 o menos), los blancos le sacan 12 o mas en la balanza, o el rio lleva 18 meses
#     seguidos cerrado (tres cuartos del hambre que madura a la Defensa). Con 8 y 12 el laboratorio seguia viendo un
#     refuerzo en el 96 % de las historias de la IA: el bloqueo temprano y una ventaja blanca comun son el sitio normal.
gg_defensa_under_pressure = {
	OR = {
		AND = {
			has_global_variable = gg_endurance_defensa
			global_var:gg_endurance_defensa <= 50
		}
		gg_war_balance_blanco >= 12
		AND = {
			has_global_variable = gg_plata_closed_streak
			global_var:gg_plata_closed_streak >= 18
		}
	}
}

# EN: AUD-CAL-POSTWAR (23/9/2026, Juan: after the Defensa won, the dated calendar kept delivering
#     wartime events). The dated calendar's relays (events/guerra_grande_calendar.txt, through
#     gg_cal_fire and its variants) are wartime: every event they carry speaks of the siege, the two
#     governments or the powers' patronage of one of them. This is the one gate they all read. The war
#     is over when any ending was written (gg_peace_settled is set by every peace, the closing dispatch,
#     the reincorporation and the Cisplatina), when it failed or never crossed, or -- the window before
#     the closing mark, or an engine war that annexed a government -- once the country has split and one
#     of the two governments no longer exists. Before the split it is never over: a late partition must
#     still receive what was dated before it (the retry).
# ES: Los relevos del calendario fechado son de guerra: todo lo que entregan habla del sitio, de los dos
#     gobiernos o del padrinazgo de las potencias. Esta es la unica guarda que leen todos. La guerra
#     termino cuando se escribio cualquier final (gg_peace_settled lo prende toda paz, el despacho de
#     cierre, la reincorporacion y la Cisplatina), cuando fracaso o no llego a cruzar, o --la ventana
#     antes de la marca de cierre, o una guerra del motor que anexo a un gobierno-- cuando el pais ya se
#     partio y uno de los dos gobiernos ya no existe. Antes de la particion nunca termino: una particion
#     tardia tiene que recibir igual lo fechado antes (el reintento).
gg_cal_war_over = {
	OR = {
		has_global_variable = gg_peace_settled
		has_global_variable = gg_war_failed
		has_global_variable = gg_war_never_crossed
		AND = {
			has_global_variable = guerra_grande_split_done
			OR = {
				NOT = { any_country = { has_variable = guerra_grande_defensa_government } }
				NOT = { any_country = { has_variable = guerra_grande_cerrito_government } }
			}
		}
	}
}
