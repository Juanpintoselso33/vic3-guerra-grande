# EN: THE PROVINCES AS COUNTRIES (22/9/2026, decided with Juan). One list of the province tags, so the
#     journal, the settlement, the government type and the AI read the same thirteen and nobody keeps
#     a second copy. CRR and ENR are born by pronouncing; the other eleven only when Rosas beats the
#     country that holds them (common/scripted_effects/gg_provinces.txt). The Coalition (CLN) is NOT a
#     province: it is a league of them, and it has its own tests.
# ES: LAS PROVINCIAS COMO PAISES. Una sola lista de los tags de provincia, para que el journal, el
#     arreglo, el tipo de gobierno y la IA lean las mismas trece y nadie lleve una segunda copia. CRR y
#     ENR nacen al pronunciarse; las otras once solo cuando Rosas bate al pais que las tiene. La
#     Coalicion (CLN) NO es una provincia: es una liga de ellas, y tiene sus propias pruebas.
# -------------------------------------------------------------------------------------------------
gg_is_province_country = {
	OR = {
		c:CRR ?= this
		c:ENR ?= this
		c:TUC ?= this
		c:SLT ?= this
		c:JUJ ?= this
		c:CTM ?= this
		c:LRJ ?= this
		c:CBA ?= this
		c:SFE ?= this
		c:SDE ?= this
		c:MZA ?= this
		c:SJN ?= this
		c:SNL ?= this
	}
}

# EN: 24/9/2026 (hotfix 0.1.2, AUD-ORIENTAL-PROVINCE) -- the Banda Oriental taken in as a province of the Pact
#     (gg_oriental_make_province: the URU tag, subject_type_gg_pact_province). It is NOT in gg_is_province_country:
#     that list is the thirteen Argentine provinces, read by the journal, the Pact's rows and the AI, and URU is also
#     the independent Republic. This trigger is true only while it is Buenos Aires's province, so everything keyed on
#     it (the governor, the annexation, the side in Urquiza's war) turns off by itself when it is independent again.
# ES: la Banda Oriental recibida como provincia del Pacto. NO esta en gg_is_province_country (las trece provincias
#     argentinas, que leen el journal, las filas y la IA; URU es tambien la Republica independiente). Vale solo
#     mientras es provincia de Buenos Aires: lo que depende de esto se apaga solo si vuelve a ser independiente.
# EN: 25/9/2026 (AUD-POSTWAR-PROVINCE, Juan's decision: option a) -- the Uruguayan postwar (events/guerra_grande_postwar.txt:
#     Oribe's four-year mandate, the handovers, «the caudillos come back», the elections and the governing interest
#     groups it fixes) belongs to the independent Republic. It does not run while the Banda Oriental is a Province of
#     the Pact (gg_oriental_province) or once Argentina absorbed it (gg_reincorporated). In Juan's game it ran on Oribe's
#     Oriental Province: «the caudillos come back» on 20/3/1853, eight years of gg_postwar_caudillos and the landowners
#     and armed forces made its government.
# ES: la posguerra uruguaya es de la Republica independiente: no corre con la Banda Oriental provincia del Pacto ni
#     absorbida por la Argentina.
gg_postwar_republic = {
	NOT = { has_global_variable = gg_oriental_province }
	NOT = { has_global_variable = gg_reincorporated }
}

gg_is_oriental_pact_province = {
	c:URU ?= this
	exists = c:ARG
	has_global_variable = gg_oriental_province
	is_subject_of = c:ARG
}

# EN/ES: 24/9/2026 (AUD-ORIENTAL-PROVINCE) -- a country whose head is a governor (common/government_types/):
#     the thirteen province countries and the Oriental Province / un pais cuya cabeza es un gobernador
gg_has_province_governor = {
	OR = {
		gg_is_province_country = yes
		gg_is_territory_province_country = yes # EN/ES: 24/9 (AUD-CONST-TERR) -- a territory made a province has a governor / un territorio hecho provincia tiene gobernador
		gg_is_oriental_pact_province = yes
	}
}

# EN: 24/9/2026 (Juan, AUD-ERR-CUARTELES) -- a country born from Argentina: the province countries, the Coalition
#     and Buenos Aires (BUE). It holds Argentina's army model (common/laws/zz_gg_local_caudillos.txt,
#     gg_argentina_inherit_army_model). By tag, so it is true at the moment of create_country.
# ES: un pais nacido de la Argentina: los paises provinciales, la Coalicion y Buenos Aires. Tiene el modelo de
#     ejercito de la Argentina. Por tag, asi vale en el momento del create_country.
gg_is_born_of_argentina = {
	OR = {
		gg_is_province_country = yes
		gg_is_territory_province_country = yes # EN/ES: 24/9 (AUD-CONST-TERR) -- the territories made provinces / los territorios hechos provincia
		c:CLN ?= this
		c:BUE ?= this
	}
}

# EN: 24/9/2026 (Juan, AUD-ERR-IDENTIDAD) -- countries that never open vanilla's South American national identity
#     (je_south_american_national_identity_emergence, common/journal_entries/zzz_gg_south_american_identity.txt):
#     its events map only vanilla's tags, so for these the old culture was never set (culture_south_america.txt:386
#     and :890, ~210 error lines a run) and the conversion did nothing. The born countries of Argentina and the
#     Cerrito for good; the Republic of the Defensa while the Guerra Grande lasts (guerra_grande_open_war).
#     Argentina and Uruguay keep it.
# ES: paises que nunca abren la identidad nacional sudamericana de vanilla: sus eventos solo conocen los tags de
#     vanilla. Los nacidos de la Argentina y el Cerrito siempre; la Defensa mientras dure la guerra. La Argentina y
#     el Uruguay la conservan.
gg_no_south_american_identity_tag = {
	OR = {
		gg_is_born_of_argentina = yes
		c:CER ?= this
	}
}
gg_no_south_american_identity = {
	OR = {
		gg_no_south_american_identity_tag = yes
		AND = {
			c:URU ?= this
			has_global_variable = guerra_grande_open_war
		}
	}
}

# EN: The eleven born only by defeat: they have no pronouncement, so they get the quiet AI.
# ES: Las once que nacen solo por derrota: no tienen pronunciamiento, asi que llevan la IA quieta.
gg_is_beaten_province_country = {
	gg_is_province_country = yes
	NOT = { c:CRR ?= this }
	NOT = { c:ENR ?= this }
}

# EN: Who sees and moves the Federal Pact: Buenos Aires, the Coalition and every province country.
# ES: Quien ve y mueve el Pacto Federal: Buenos Aires, la Coalicion y todo pais provincial.
gg_is_pact_country = {
	OR = {
		has_variable = gg_arg_rosas_seat
		c:CLN ?= this
		gg_is_province_country = yes
	}
}

# EN: A country that may hold an Argentine province, for where a governor is born: Argentina, the
#     Coalition or a province country -- never Bolivia, which shares Salta and Jujuy from 1836.
# ES: Un pais que puede tener una provincia argentina, para saber donde nace un gobernador: la
#     Argentina, la Coalicion o un pais provincial -- nunca Bolivia, que comparte Salta y Jujuy.
gg_is_argentine_holder = {
	OR = {
		c:ARG ?= this
		gg_is_pact_country = yes
	}
}

# EN: 23/9/2026 -- in the possible block of the vanilla plays and of da_decrease_autonomy (scope: the
#     initiator; scope:target_country the target). False against the Coalition or a province country:
#     those are fought only with dp_gg_pacto_federal (zz_gg_argentine_plays.txt).
# ES: en el possible de las jugadas de vanilla y de da_decrease_autonomy. Falso contra la Coalicion o un
#     pais provincial: a esos se los pelea solo con dp_gg_pacto_federal.
gg_argentine_play_is_allowed = {
	scope:target_country = {
		NOR = {
			gg_is_province_country = yes
			c:CLN ?= this
		}
	}
}

# EN: 24/9/2026 (Juan, «two governments», common/scripted_effects/gg_pact_sides.txt) -- state scope: a state of one of

# EN: 24/9/2026 (Juan, AUD-PROV-INCORP: «they were Argentine provinces, they should be incorporated at once») -- state
#     scope: a state of a historical Argentine province, the ground a reincorporation makes incorporated
#     (gg_incorporate_province_states). The fourteen provinces of the Federal Pact plus Misiones (Corrientes's
#     lands). The Banda Oriental's five states count only once it rejoined the Confederation as a province
#     (gg_reincorporated, set before its annexation by gg_oriental_incorporate and gg_oriental_province_absorb).
#     Left out on purpose: the frontier (La Pampa, Rio Negro, the Chaco), Tarija (Bolivian ground) and la Argentina's
#     Parana and Santa Catarina (in Brazil's region): none was a province of the Confederation.
# ES: scope estado: un estado de una provincia argentina historica. Las catorce provincias del Pacto Federal y
#     Misiones; la Banda Oriental solo despues de volver como provincia (gg_reincorporated). Afuera a proposito: la
#     frontera (La Pampa, Rio Negro, el Chaco), Tarija y Parana y Santa Catarina: ninguna fue provincia.
gg_arg_historical_province_state = {
	OR = {
		state_region = s:STATE_BUENOS_AIRES
		state_region = s:STATE_SANTA_FE
		state_region = s:STATE_ENTRE_RIOS
		state_region = s:STATE_CORRIENTES
		state_region = s:STATE_MISIONES
		state_region = s:STATE_CORDOBA
		state_region = s:STATE_SANTIAGO_DEL_ESTERO
		state_region = s:STATE_TUCUMAN
		state_region = s:STATE_SALTA
		state_region = s:STATE_JUJUY
		state_region = s:STATE_CATAMARCA
		state_region = s:STATE_LA_RIOJA
		state_region = s:STATE_SAN_JUAN
		state_region = s:STATE_MENDOZA
		state_region = s:STATE_SAN_LUIS
		# EN/ES: 24/9/2026 (AUD-CONST-TERR) -- a national territory once it was declared a province under the confederation
		#     (gg_territory_promote_one) / un territorio nacional despues de ser declarado provincia
		AND = { state_region = s:STATE_LA_PAMPA has_global_variable = gg_terr_province_la_pampa }
		AND = { state_region = s:STATE_NEUQUEN has_global_variable = gg_terr_province_neuquen }
		AND = { state_region = s:STATE_RIO_NEGRO has_global_variable = gg_terr_province_rio_negro }
		AND = { state_region = s:STATE_CHUBUT has_global_variable = gg_terr_province_chubut }
		AND = { state_region = s:STATE_SANTA_CRUZ_ARG has_global_variable = gg_terr_province_santa_cruz }
		AND = { state_region = s:STATE_TIERRA_DEL_FUEGO has_global_variable = gg_terr_province_tierra_del_fuego }
		AND = { state_region = s:STATE_FORMOSA_ARG has_global_variable = gg_terr_province_formosa }
		AND = { state_region = s:STATE_CHACO has_global_variable = gg_terr_province_chaco }
		AND = {
			has_global_variable = gg_reincorporated
			OR = {
				state_region = s:STATE_MONTEVIDEO
				state_region = s:STATE_SAN_JOSE
				state_region = s:STATE_MALDONADO
				state_region = s:STATE_PAYSANDU
				state_region = s:STATE_BANDA_ORIENTAL
			}
		}
	}
}

# EN: 24/9/2026 (Juan, AUD-CONST-CONFED) -- state scope: a state the confederation (gg_constitution.1.g) hands to a
#     province: every state of gg_conf_province_state but the Chaco. Buenos Aires keeps its own province, La Pampa and
#     Rio Negro as the «two governments» partition does, and the Chaco too: in 1851 no province governed it (Indian land
#     beyond the frontier lines), so it stays with the government that keeps the foreign relations and the frontier.
#     In the partition it follows the provinces because there they are one country; here there is no one province to
#     hand it to, and naming one would be invented.
# ES: scope estado: un estado que la confederacion le entrega a una provincia: todos los de gg_conf_province_state menos
#     el Chaco, que ninguna provincia gobernaba en 1851 y queda, como La Pampa y Rio Negro, con el gobierno que lleva las
#     relaciones exteriores y la frontera.
gg_confed_province_state = {
	gg_conf_province_state = yes
	NOT = { state_region = s:STATE_CHACO }
}
