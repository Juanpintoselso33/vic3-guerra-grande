#     el ejercito de Urquiza antes del Ejercito Grande. La historia no vio esa guerra: Rosas espero a la defensiva en

# EN/ES: country scope / scope de pais
gg_pact_rosas_side = {
	exists = c:ARG
	OR = {
		c:ARG ?= this
		is_subject_of = c:ARG
	}
}

# EN/ES: country scope / scope de pais
gg_pact_urquiza_side = {
	exists = c:ARG
	OR = {
		c:ENR ?= this
		AND = {
			c:CRR ?= this
			has_war_with = c:ARG
		}
	}
}

# EN: 26/9/2026 (Juan, AUD-PACT-HOLD-MILESTONES; «secuencia antes que fechas») -- THE HOLDS FOLLOW THE CAMPAIGN, NOT THE
#     CALENDAR, AND THE DAY OF CASEROS IS CONSOLIDATED. In the six caseros runs of 26/9 (mod 1d7c3d7) Entre Rios came to the
#     day with 1-9 battalions in three of five wars (LOST_ENR 89-100, OCC_ENR up to 42), and in 030959 Urquiza carried the
#     day and still lost the war six weeks later (gg_pact_wargoal_enforced for Rosas, 18/3/1852). Three changes:
#       - the general's OWN front (the character link: the front he is at OR travelling towards), not his formation's (the
#         front it is at): the AI gives its order when it sends a general, and a general still on the road had no
#         formation front, so the hold did not apply to him;
#       - Rosas's side is held until the day of Caseros is fought (gg_caseros_battle), not until a date; and, once Urquiza
#         carried it (gg_caseros_battle = 1), for the rest of this war: Rosas's army dispersed that afternoon and never took
#         the field again. Mirror: once Buenos Aires held the day (= 2), Urquiza's side is held (the allied contingents went
#         home and the Ejercito Grande broke up);
#       - Urquiza's side is held until the Ejercito Grande crosses the Parana (gg_ejercito_grande_crossed, gg_caseros.11,
#         33 days after it is raised: 21/11 -> 24/12/1851 at Diamante): its battalions muster first and are not spent
#         piecemeal before the crossing.
#     A war with no Ejercito Grande keeps the old fallback dates (the crossing 24/12/1851, the day 3/2/1852); the holds
#     before the day end on 1/1/1853 at the latest (a safety cap); the holds after the day last as long as that war
#     (Codex review 26/9).
# ES: LAS RETENCIONES SIGUEN A LA CAMPANIA, NO AL CALENDARIO, Y LA JORNADA SE CONSOLIDA. En las seis corridas del 26/9
#     Entre Rios llego a la jornada con 1-9 batallones en tres de cinco guerras, y en 030959 Urquiza gano la jornada y
#     perdio la guerra seis semanas despues. El frente PROPIO del general (al que esta o hacia el que marcha), no el de su
#     formacion; Rosas retenido hasta que se pelea la jornada y, si la gano Urquiza, por el resto de la guerra (su ejercito
#     se disperso esa tarde), y el espejo si la sostuvo Buenos Aires; Urquiza retenido hasta que el Ejercito Grande cruza
#     el Parana. Sin Ejercito Grande valen las fechas viejas; antes de la jornada nada retiene despues del 1/1/1853;
#     despues de la jornada la retencion dura lo que dure esa guerra.
# SOURCE / FUENTE: es.wikipedia «Batalla de Caseros» (the Rosista army dispersed; Rosas resigned the same day);
#     es.wikipedia «Ejército Grande» (the crossing of the Parana at Diamante, 24/12/1851-6/1/1852).

# EN/ES: character scope: the front this general is at or marching to holds ground of `SIDE` (gg_pact_rosas_side /
#     gg_pact_urquiza_side) / el frente de este general tiene suelo de ese lado
gg_pact_general_faces = {
	front ?= {
		any_scope_state = {
			OR = {
				owner ?= { $SIDE$ = yes }
				controller ?= { $SIDE$ = yes }
			}
		}
	}
}

# EN: 26/9/2026 (Juan, AUD-PACT-HOLD-LEAK) -- character scope: a general with no front at all (in his headquarters, just raised
#     or just recalled). In the six caseros runs on 6cb1e2f the probe found held generals WITH an advance order on the Pact
#     front: ADV_ROSAS=1 from 31/5/1851 to January 1852 in traza_20260926-134931 (Entre Rios occupied 100 %, LOST_ENR 82 at
#     Caseros), ADV_ROSAS=1 from 27/12/1851 in -145836 (OCC_ENR 90, LOST_ENR 100) and after the day Urquiza carried in -134933,
#     ADV_URQ=1 on 31/5/1851 in three runs (Rosas's LOST_ARG 100 in -134933). Each first shows right after the script sent an army
#     to the front (GG_MOB deployed on 2, 9 and 22/5/1851): the AI picks a general's order while he has no front, when the hold
#     could not see where he was going, and the order goes with him to the front. So a held side's general with no front may not
#     take an advance order either; the probe's HELD_ADV/NF_ADV say whether that closes it.
# ES: un general sin ningun frente (en su cuartel, recien levantado o recien llamado). En las seis corridas de caseros sobre
#     6cb1e2f la sonda encontro generales retenidos CON orden de avance en el frente del Pacto; cada caso aparece justo despues
#     de que el script mando un ejercito al frente: la IA elige la orden mientras el general no tiene frente, cuando la
#     retencion no podia ver adonde iba, y la orden viaja con el. Asi que un general sin frente del lado retenido tampoco puede
#     tomar una orden de avance; HELD_ADV/NF_ADV de la sonda dicen si eso lo cierra.
#     Only when the Pact war is the owner's ONLY war (Codex review 26/9: a general with no front may be bound for another war, and
#     every other war keeps vanilla's orders): `$ENEMY$` is the other side's leader (c:ENR for Rosas's side, c:ARG for Urquiza's).
#     / Solo si la guerra del Pacto es la UNICA del duenio: un general sin frente puede ir a otra guerra, y las demas guerras
#     conservan las ordenes de vanilla.
gg_pact_general_no_front = {
	NOT = { front ?= { always = yes } }
	owner ?= {
		NOT = { any_scope_war = { NOT = { is_war_participant = $ENEMY$ } } }
	}
}

# EN/ES: global: the day of Caseros is still to be fought / la jornada de Caseros todavia no se peleo
gg_pact_before_caseros_day = {
	NOT = { has_global_variable = gg_caseros_battle }
	game_date < 1853.1.1 # EN/ES: a safety cap, never a milestone / un tope de seguridad, nunca un hito
	OR = {
		has_global_variable = gg_ejercito_grande_created
		game_date < 1852.2.3
	}
}

# EN/ES: global: Urquiza's side has not crossed yet / el lado de Urquiza todavia no cruzo
gg_pact_before_crossing = {
	NOT = { has_global_variable = gg_ejercito_grande_crossed }
	NOT = { has_global_variable = gg_caseros_battle }
	game_date < 1853.1.1 # EN/ES: a safety cap / un tope de seguridad
	OR = {
		has_global_variable = gg_ejercito_grande_created
		game_date < 1851.12.24
	}
}

# EN: 26/9/2026 (Juan, AUD-CASEROS-FORCES) -- global: the day of Caseros went to `VALUE` (gg_caseros_battle: 1 Urquiza, 2 Buenos
#     Aires) by a clear lead, so the loser's army is broken and its generals are held for the rest of the war. A near-tie carried
#     by fortune (gg_caseros_tie) breaks neither army: both fight on, and the tilt (gg_caseros.8) still leans to the side fortune
#     favoured. Why: with the holds, no battle is fought before the day, so a fortune's day that also froze the loser decided the
#     whole war by the dice (traza_20260926-181813: Rosas by fortune against twice his battalions, then his free army against a
#     held Entre Rios, LOST_ENR 100 and OCC_ENR 27 by 27/3, and Rosas's war goal on 1/4/1852); now only a clear balance of forces ends the fighting, as at Caseros, where Rosas's army dispersed.
# ES: la jornada fue de `VALUE` por una ventaja clara: el ejercito del que perdio queda deshecho y sus generales retenidos el resto
#     de la guerra. Un empate que decidio la suerte no deshace ninguno: los dos siguen peleando y la inclinacion sigue del lado
#     que favorecio la suerte. Con las retenciones ninguna batalla se pelea antes de la jornada, asi que una jornada de la suerte
#     que ademas congelaba al que perdio decidia la guerra entera a los dados.
gg_pact_day_broke_army = {
	global_var:gg_caseros_battle ?= $VALUE$
	NOT = { has_global_variable = gg_caseros_tie }
}

# EN/ES: character scope: this general may not take an advance order now / este general no puede tomar una orden de avance
gg_pact_front_hold = {
	has_role_of_type = general
	has_global_variable = gg_war_against_buenos_aires
	NOT = { has_global_variable = gg_pact_settled }
	exists = c:ENR
	exists = c:ARG
	c:ARG = { has_war_with = c:ENR }
	owner ?= { is_player = no }
	OR = {
		# EN/ES: Rosas waits for the attack until the day, and after a lost day his army is gone
		#     / Rosas espera el ataque hasta la jornada, y despues de perderla su ejercito ya no esta
		AND = {
			owner ?= { gg_pact_rosas_side = yes }
			OR = {
				gg_pact_before_caseros_day = yes
				gg_pact_day_broke_army = { VALUE = 1 } # EN/ES: 26/9/2026 (AUD-CASEROS-FORCES) -- a clear day, not fortune's / una jornada clara, no la de la suerte
			}
			OR = {
				gg_pact_general_faces = { SIDE = gg_pact_urquiza_side }
				gg_pact_general_no_front = { ENEMY = c:ENR } # EN/ES: 26/9/2026 (AUD-PACT-HOLD-LEAK)
			}
		}
		# EN/ES: Urquiza keeps his army for the crossing, and after a lost day the Ejercito Grande is gone
		#     / Urquiza guarda su ejercito para el cruce, y despues de perder la jornada el Ejercito Grande ya no esta
		AND = {
			owner ?= { gg_pact_urquiza_side = yes }
			OR = {
				gg_pact_before_crossing = yes
				gg_pact_day_broke_army = { VALUE = 2 } # EN/ES: 26/9/2026 (AUD-CASEROS-FORCES)
			}
			OR = {
				gg_pact_general_faces = { SIDE = gg_pact_rosas_side }
				gg_pact_general_no_front = { ENEMY = c:ARG } # EN/ES: 26/9/2026 (AUD-PACT-HOLD-LEAK)
			}
		}
	}
}

# EN/ES: 26/9/2026 (AUD-PACT-ARMY-MELTS) -- Entre Rios or Corrientes while the war of the Pronouncement lasts: its AI keeps the
#     army it has (gg_pact_field_army_wanted, common/script_values/gg_caseros.txt). Country scope
#     / Entre Rios o Corrientes mientras dura la guerra del Pronunciamiento: su IA conserva el ejercito que tiene
gg_pact_keeps_field_army = {
	exists = c:ARG
	has_global_variable = gg_war_against_buenos_aires
	OR = {
		AND = {
			OR = {
				c:ENR ?= this
				c:CRR ?= this
			}
			has_war_with = c:ARG
		}
		# EN/ES: 26/9/2026 (Juan, option A of AUD-PACT-ROSAS-SMALL) -- and Buenos Aires, the same rule, through its Plata
		#     strategies (common/ai_strategies/gg_rio_plata.txt) / y Buenos Aires, la misma regla
		AND = {
			c:ARG ?= this
			exists = c:ENR
			has_war_with = c:ENR
		}
	}
}
