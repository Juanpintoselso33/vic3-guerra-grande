# EN/ES: country scope / scope de pais
gg_pact_rosas_side = {
	exists = c:ARG
	OR = {
		c:ARG ?= this
		is_subject_of = c:ARG
	}
}

# EN/ES: country scope / scope de pais
gg_pact_urquiza_side = {
	exists = c:ARG
	OR = {
		c:ENR ?= this
		AND = {
			c:CRR ?= this
			has_war_with = c:ARG
		}
	}
}

# EN/ES: character scope: this general may not take an advance order now / este general no puede tomar una orden de avance
gg_pact_front_hold = {
	has_role_of_type = general
	has_global_variable = gg_war_against_buenos_aires
	NOT = { has_global_variable = gg_pact_settled }
	exists = c:ENR
	exists = c:ARG
	c:ARG = { has_war_with = c:ENR }
	owner ?= { is_player = no }
	OR = {
		# EN/ES: Rosas waits for the attack until the day of Caseros / Rosas espera el ataque hasta el dia de Caseros
		AND = {
			NOT = { has_global_variable = gg_caseros_battle }
			game_date < 1852.2.3
			owner ?= { gg_pact_rosas_side = yes }
			commander_military_formation ?= {
				front ?= {
					any_scope_state = {
						OR = {
							owner ?= { gg_pact_urquiza_side = yes }
							controller ?= { gg_pact_urquiza_side = yes }
						}
					}
				}
			}
		}
		# EN/ES: Urquiza keeps his field army for the Ejercito Grande / Urquiza guarda su ejercito para el Ejercito Grande
		AND = {
			NOT = { has_global_variable = gg_ejercito_grande_created }
			game_date < 1851.12.24
			owner ?= { gg_pact_urquiza_side = yes }
			commander_military_formation ?= {
				front ?= {
					any_scope_state = {
						OR = {
							owner ?= { gg_pact_rosas_side = yes }
							controller ?= { gg_pact_rosas_side = yes }
						}
					}
				}
			}
		}
	}
}
