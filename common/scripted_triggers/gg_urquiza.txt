# EN: 23/9/2026 (Juan: "a Rosas that consolidated should not lose to Entre Rios... my decisions to appease Entre
#     Rios do not count") -- the state of things on 1 May 1851, read by Urquiza's choice (gg_urquiza_discontent).
# ES: el estado de las cosas el 1/5/1851, que lee la decision de Urquiza.

# EN/ES: Rosas won the Guerra Grande: peace signed, not "no victors", and the Defensa's government is gone
#     / Rosas gano la Guerra Grande: paz firmada, no "ni vencidos ni vencedores", y la Defensa ya no existe
gg_rosas_won_guerra_grande = {
	has_global_variable = gg_peace_settled
	NOT = { has_global_variable = gg_peace_no_victors }
	NOT = { any_country = { has_variable = guerra_grande_defensa_government } }
}

# EN/ES: Brazil went to war over the Convention of 1828 and did not get its way (gg_cisplatina.3, outcome 0)
#     / el Brasil fue a la guerra por la Convencion y no se salio con la suya
gg_brazil_beaten = {
	has_global_variable = gg_cis_outcome
	global_var:gg_cis_outcome = 0
}

# EN/ES: Urquiza has nobody to sign a convention with: Rosas holds the Banda Oriental or won its war, or Brazil
#     is beaten / Urquiza no tiene con quien firmar una convencion
gg_urquiza_alone = {
	OR = {
		gg_rosas_won_guerra_grande = yes
		has_global_variable = gg_oriental_in_pact
		gg_brazil_beaten = yes
	}
}

# EN/ES: 23/9 (text audit) -- no Montevideo convention to sign: Urquiza alone, or the Guerra Grande already over
#     (gg_international_open false: peace, never crossed, failed) / sin convencion: Urquiza solo o la guerra terminada
gg_urquiza_no_convention = {
	OR = {
		gg_urquiza_alone = yes
		NOT = { gg_international_open = yes }
	}
}

# EN: 23/9/2026 (Juan, §2 of decisiones-argentina.md) -- Entre Rios is still Rosas's on the eve of the Pronouncement:
#     Argentine ground with no country of its own, or the loyal Entre Rios a player took at Alcaraz
#     (gg_argentina_alcaraz_seat), still a subject. The Pronouncement chain (gg_argentina_cal.4, gg_urquiza.1,
#     gg_argentina.71) reads this instead of "ENR does not exist", so the seat of 1846 does not switch 1851 off.
# ES: Entre Rios sigue siendo de Rosas en la vispera del Pronunciamiento: suelo argentino sin pais propio, o el Entre
#     Rios leal que un jugador tomo en Alcaraz, todavia sujeto. La cadena del Pronunciamiento lee esto en vez de «no
#     existe ENR», asi el asiento de 1846 no apaga el de 1851.
gg_entre_rios_with_rosas = {
	OR = {
		AND = {
			NOT = { exists = c:ENR }
			exists = s:STATE_ENTRE_RIOS.region_state:ARG
		}
		c:ENR ?= {
			has_variable = gg_enr_alcaraz_seat
			is_subject_of = c:ARG
		}
	}
}

# EN: 23/9/2026 (Juan, AUD-SEAT-ALCARAZ) -- Urquiza governs the loyal Entre Rios of Alcaraz and Buenos Aires no longer
#     holds the province's ground. The 1847-1851 orders of Rosas (Vences gg_argentina.8, the port order .28, the two
#     orders .67, La Regeneracion .9) are then delivered to c:ENR, from Urquiza's side; with Argentine Entre Rios they
#     go to Buenos Aires as before.
# ES: Urquiza gobierna el Entre Rios leal de Alcaraz y Buenos Aires ya no tiene el suelo de la provincia. Las ordenes
#     de Rosas de 1847-1851 se entregan entonces a c:ENR, del lado de Urquiza; con Entre Rios argentino van a Buenos
#     Aires como antes.
gg_entre_rios_alcaraz_loyal = {
	NOT = { exists = s:STATE_ENTRE_RIOS.region_state:ARG }
	c:ENR ?= {
		has_variable = gg_enr_alcaraz_seat
		is_subject_of = c:ARG
		exists = s:STATE_ENTRE_RIOS.region_state:ENR
	}
}

# EN: 24/9/2026 (Juan: the representation of Entre Rios has to come every game) -- when Urquiza's letter of
#     22/2/1851 (gg_urquiza.1, relay gg_argentina_cal.35) can reach Buenos Aires: the Pact still open and Urquiza
#     not yet broken with Rosas (Entre Rios has not taken back its delegation and has no country of its own outside
#     the loyal seat of Alcaraz). Wider than gg_entre_rios_with_rosas on purpose: if Corrientes, the Coalition or a
#     foreign army holds the ground of Entre Rios, Urquiza is still Rosas's general and his grievances still stand,
#     so the letter still comes. Only what depends on the province (its claim, its grievance count) waits for
#     gg_entre_rios_with_rosas inside the event; the 1 May decision (gg_argentina_cal.4) keeps its own stricter test.
# ES: cuando la carta de Urquiza del 22/2/1851 puede llegar a Buenos Aires: el Pacto sigue abierto y Urquiza todavia
#     no rompio con Rosas (Entre Rios no reasumio la delegacion ni tiene pais propio fuera del asiento leal de
#     Alcaraz). Mas ancho que gg_entre_rios_with_rosas a proposito: si Corrientes, la Coalicion o un ejercito
#     extranjero tiene el suelo de Entre Rios, Urquiza sigue siendo general de Rosas y sus agravios siguen en pie,
#     asi que la carta llega igual. Lo que depende de la provincia (su reclamo, su cuenta de agravios) espera
#     gg_entre_rios_with_rosas adentro del evento; la decision del 1/5 conserva su propia prueba, mas estricta.
gg_urquiza_prelude_possible = {
	exists = c:ARG
	NOT = { has_global_variable = gg_pact_settled }
	NOT = { has_global_variable = gg_delegation_gone_entre_rios }
	OR = {
		gg_entre_rios_with_rosas = yes
		NOT = { exists = c:ENR }
	}
}
