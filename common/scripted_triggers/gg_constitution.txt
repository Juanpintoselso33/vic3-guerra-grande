# EN: 24/9/2026 (AUD-CONST-CHOICE) -- which answers gg_constitution.1 offers, by how the Pact was settled
#     (gg_pact_settlement_*, gg_pact_resolve). The options and the trace (gg_constitution_offer_log) read the same ones.
#     a) the Constitution of 1853: not after Rosas's victory (AUD-CONST-ROSAS) nor with two governments;
#     b) the Pact without a constitution: not after Urquiza's victory;
#     c) the Congress of the Pact, given by the Restorer: only after Rosas's victory (the AI never takes it);
#     e) the constitution of the State of Buenos Aires: only with two governments (the AI never takes it);
#     g) the confederation, every province a Province of the Pact (24/9, AUD-CONST-CONFED): not after Urquiza's
#        victory nor with two governments.
#     Rosas therefore always has two answers: b and c after his victory, b and e with two governments.
# ES: que respuestas ofrece gg_constitution.1 segun como se cerro el Pacto. Rosas siempre tiene dos: b y c tras su
#     victoria, b y e con dos gobiernos.
gg_constitution_offers_1853 = {
	NOT = { has_global_variable = gg_pact_settlement_rosas }
	NOT = { has_global_variable = gg_pact_settlement_stalemate }
}
gg_constitution_offers_no_constitution = {
	NOT = { has_global_variable = gg_pact_settlement_urquiza }
}
gg_constitution_offers_pact_congress = {
	has_global_variable = gg_pact_settlement_rosas
}
gg_constitution_offers_state_constitution = {
	has_global_variable = gg_pact_settlement_stalemate
	NOT = { has_global_variable = gg_pact_settlement_rosas }
}
# EN: 24/9/2026 (Juan: «a confederation with every province kept as a subject»; AUD-CONST-CONFED) -- g) the Pact
#     itself as the only charter: every province its own country, a Province of the Pact under Buenos Aires, which keeps
#     the foreign relations. Offered after Rosas's victory, the negotiated confederation and the fallen mandate (and so
#     under the dynamic rule whenever it ends in one of those); NOT after Urquiza's victory (his cause was the National
#     Organisation, and gg_argentina.80.b is closed there too) nor with two governments (the provinces are Urquiza's:
#     Buenos Aires has none to confederate). It no longer needs a subject province: an Argentina that incorporated them
#     at gg_argentina.80.a hands their ground back (gg_constitution_confederation). Scope: the Rosas seat.
# ES: g) el Pacto como unica ley: cada provincia su propio pais, provincia del Pacto bajo Buenos Aires, que conserva las
#     relaciones exteriores. Tras la victoria de Rosas, la confederacion negociada y el mandato caido; no tras la de
#     Urquiza (su causa era la Organizacion Nacional) ni con dos gobiernos (las provincias son de Urquiza). Ya no pide una
#     provincia sujeta: si .80.a las incorporo, les devuelve su suelo.
gg_constitution_offers_confederation = {
	NOT = { has_global_variable = gg_pact_settlement_urquiza }
	NOT = { has_global_variable = gg_pact_settlement_stalemate }
	OR = {
		any_scope_state = { gg_confed_province_state = yes }
		any_subject_or_below = {
			is_subject_of = c:ARG
			OR = {
				gg_is_province_country = yes
				c:CLN ?= this
				gg_is_oriental_pact_province = yes
			}
		}
		# EN/ES: Codex review -- an independent province at peace with Buenos Aires and not a player's is taken in too
		#     / una provincia independiente en paz con Buenos Aires y que no es de un jugador tambien se recibe
		any_country = {
			OR = {
				gg_is_province_country = yes
				c:CLN ?= this
			}
			is_subject = no
			is_player = no
			NOT = { has_war_with = c:ARG }
		}
	}
}

# EN: 24/9/2026 (AUD-CONST-CHOICE) -- what holds the constitution off: Buenos Aires at war with Brazil, or the
#     Cisplatine play still on (its war goal is a humiliation of Argentina; between the play and the war the
#     question waits too). Scope: the Rosas seat.
# ES: lo que frena la constitucion: Buenos Aires en guerra con el Brasil, o la jugada cisplatina todavia abierta.
gg_constitution_brazil_war = {
	OR = {
		AND = {
			exists = c:BRZ
			has_war_with = c:BRZ
		}
		any_diplomatic_play = { is_diplomatic_play_type = dp_gg_cisplatina }
		# EN: 24/9/2026 (AUD-END-A3, Juan) -- and while Brazil waits for its moment (gg_cis_wait, after gg_cisplatina.2.b/.c/.d):
		#     the Cisplatine question is still open. The wait cannot hang: gg_cisplatina.7 clears it when the war opens, when
		#     the Banda Oriental leaves the Pact, or on 1/1/1853.
		# ES: y mientras el Brasil espera su ocasion: la cuestion cisplatina sigue abierta. La espera no queda colgada:
		#     gg_cisplatina.7 la corta al abrir la guerra, si la Banda Oriental deja el Pacto o el 1/1/1853.
		# EN/ES: Codex review -- only while Brazil stands and before 1/1/1853, the wait's own end: its only clearer
		#     (gg_cisplatina.7) lives on Brazil, and a lost copy must not hold the question forever
		#     / solo mientras exista el Brasil y antes del 1/1/1853: quien la borra vive en el Brasil
		AND = {
			has_global_variable = gg_cis_waiting
			exists = c:BRZ
			game_date < 1853.1.1
		}
	}
}

# EN: 24/9/2026 (AUD-END-A6) -- the constitution still has provinces to join: a province country (or the Coalition, or the
#     Oriental Province) under Buenos Aires, or -- after Urquiza's victory or the negotiated confederation -- an independent
#     Corrientes, Entre Rios or Coalition at peace with it (gg_argentina_province_join, ALLY = yes). Scope: Argentina.
#     Only for the tooltip: an Argentina that already incorporated them (gg_argentina.80.a) is not promised it again.
# ES: a la constitucion todavia le quedan provincias por sumar. Solo para el tooltip: no se promete lo ya hecho.
gg_constitution_provinces_to_join = {
	OR = {
		any_subject_or_below = {
			is_subject_of = c:ARG
			OR = {
				gg_is_province_country = yes
				c:CLN ?= this
				gg_is_oriental_pact_province = yes
			}
		}
		AND = {
			OR = {
				has_global_variable = gg_pact_settlement_urquiza
				has_global_variable = gg_pact_settlement_constitutional
			}
			any_country = {
				OR = {
					c:CRR ?= this
					c:ENR ?= this
					c:CLN ?= this
				}
				is_subject = no
				NOT = { has_war_with = c:ARG }
			}
		}
	}
}

# EN: 24/9/2026 (AUD-END-A1, Juan: «one term») -- Rosas's first term under the Constitution of 1853 is running: six years
#     from gg_constitution.1.a/.c (art. 74: «El Presidente y Vicepresidente duran en sus empleos el termino de seis anos»).
#     gg_rosas_first_term expires by itself (days = 2190, the engine's six years). gg_rosas_first_term_legacy is the
#     one-time repair of a save that answered before this existed (gg_rosas_first_term_migration): its date is unknown,
#     so the term is counted from the earliest day the old flow could have asked, 31/5/1851 -- it can end a little early,
#     never late. After it, the engine's elections decide (AUD-END-B4).
# ES: corre el primer periodo de Rosas bajo la Constitucion de 1853: seis anios (art. 74). La marca vence sola. La de
#     partidas viejas cuenta desde el primer dia en que el flujo viejo pudo preguntar (31/5/1851): puede terminar un poco
#     antes, nunca despues. Despues deciden las elecciones del motor.
# SOURCE / FUENTE: Constitucion de la Nacion Argentina (1 May 1853), art. 74:
#     https://es.wikisource.org/wiki/Constituci%C3%B3n_de_la_Naci%C3%B3n_Argentina_(1853)
gg_rosas_first_term_running = {
	OR = {
		has_global_variable = gg_rosas_first_term
		AND = {
			has_global_variable = gg_rosas_first_term_legacy
			game_date < 1857.5.31
		}
	}
}

# EN: 25/9/2026 (Juan's decision, AUD-ROSAS-TERM; his Greece game: the term ended ~31/3/1858 and Rosas still ruled in

#     term is over (gg_rosas_term_limit starts gg_rosas_no_reelection, six years, on the first check after it ends), Rosas
#     may not hold the seat: whoever seats him -- an election, the engine, a script -- he is replaced that same day. Only

# SOURCE / FUENTE: Constitucion de la Nacion Argentina (1 May 1853), art. 74 (art. 77 after the reform of 1860):

gg_rosas_reelection_barred = {

	gg_rosas_first_term_running = no
}
