# EN/ES: any scope -- Rosas rules the country that holds Buenos Aires (ARG, or what it became: PLT, Great_ARG). The
#     seat keeper (gg_rosas_seat_on_new_ruler) puts him back the same day while he holds it by right, so a ruler who is
#     not Rosas means he is out: his term is over (AUD-ROSAS-TERM), he died, or he fell
#     / Rosas gobierna el pais que tiene Buenos Aires; si no, termino su periodo, murio o cayo
gg_mitre_rosas_rules_buenos_aires = {
	s:STATE_BUENOS_AIRES = {
		any_scope_state = {
			owner = { ruler ?= { has_template = ARG_Juan_Manuel_de_Rosas } }
		}
	}
}
