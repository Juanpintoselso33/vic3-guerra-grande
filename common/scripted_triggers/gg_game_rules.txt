# EN: 23/9/2026 (Juan, P2 of the endings audit, AUD-RULES-CLIENT2) -- Rosas has fallen: the Pact was settled with
#     Urquiza's victory or the collapsed mandate (the two settlements that call gg_rosista_client_release). There is
#     nobody left to be the client of, so the Cerrito's peace does not offer it (gg_peace.2.e) and the rule stops
#     forcing it. Defined here, before its first reader below.
# ES: Rosas cayo: el Pacto se cerro con la victoria de Urquiza o el mandato deshecho. No queda de quien ser cliente,
#     asi que la paz del Cerrito no lo ofrece y la regla deja de forzarlo.
gg_rosas_fallen = {
	OR = {
		has_global_variable = gg_pact_settlement_urquiza
		has_global_variable = gg_pact_settlement_mandate_collapsed
	}
}
# EN: 23/9/2026 (Codex audit, AUD-RULES-AI) -- a setting of the war's rule forces one option of the peace menu, and
#     that option is open (its country exists). Used to give every other option AI weight zero, so the AI cannot take
#     a contradictory ending; a human player still sees and may choose them.
# ES: una opcion de la regla de la guerra fuerza una salida del menu de paz, y esa salida esta abierta. Sirve para
#     darle peso cero a las demas en la IA; el jugador humano las sigue viendo y eligiendo.
gg_defensa_terms_forced = {
	OR = {
		gg_outcome_historical = yes
		gg_terms_defensa_punish = yes
		AND = { gg_terms_defensa_brazil_treaties = yes exists = c:BRZ }
		gg_terms_defensa_europe = yes
		AND = { gg_terms_defensa_war_rosas = yes exists = c:ARG }
		AND = { gg_terms_defensa_brazil_vassal = yes exists = c:BRZ }
		AND = { gg_terms_defensa_cisplatina = yes exists = c:BRZ }
	}
}
gg_cerrito_terms_forced = {
	OR = {
		gg_terms_cerrito_amnesty = yes
		gg_terms_cerrito_presidency = yes
		gg_terms_cerrito_punish = yes
		AND = { gg_terms_cerrito_pact_province = yes exists = c:ARG }
		AND = { gg_terms_cerrito_rosas_client = yes exists = c:ARG NOT = { gg_rosas_fallen = yes } }
		AND = { gg_terms_cerrito_reincorporation = yes exists = c:ARG }
	}
}
