# 1/4/1846 -- Rivera lands in Montevideo, heard in Buenos Aires. The Defensa's gg_rivera.3.a (the mutiny
#     wins, the decree is annulled) writes gg_rivera_restored.
# SOURCE / FUENTE: SOURCES_rivera.md, chronology 1845-1846 and 18.III / 1.IV.1846 (S1, S10, S11): prison in
#     Rio, the Paraguay appointment, the schooner Perla, the landing banned, the rising of 1 April (the
#     Basque battalion, the freedmen of the infantry, Cesar Diaz and Venancio Flores), Pacheco y Obes
#     resigns and sails for Europe, Rivera lands on Bernardina Fragoso's arm, general in chief of the
#     Ejercito de Operaciones and gran mariscal.
gg_arg_notice.1 = {
	type = country_event
	placement = ROOT
	title = gg_arg_notice.1.t
	desc = gg_arg_notice.1.d
	flavor = gg_arg_notice.1.f
	# EN/ES: provisional vanilla video (video lane) / video vanilla provisorio
	event_image = { video = "gg_argn_iln_montevideo" }
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 12
	is_popup = yes

	trigger = {
		has_variable = gg_arg_rosas_seat
		has_global_variable = guerra_grande_open_war
		has_global_variable = gg_rivera_restored
	}

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_notice.1|" # GG_TRAZA
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = gg_arg_notice_done_1 } }
				set_global_variable = gg_arg_notice_done_1
			}
		}
	}

	# EN: the Confederation watches the Uruguay river: it pays, and stands closer to Oribe.
	# ES: la Confederacion vigila el rio Uruguay: paga, y queda mas cerca de Oribe.
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_notice.1.a|" # GG_TRAZA
		name = gg_arg_notice.1.a
		default_option = yes
		custom_tooltip = gg_arg_notice.1.a_tt
		add_treasury = -10000
		gg_patron_change = { PATRON = rosas DELTA = 5 }
		ai_chance = { base = 70 }
	}

	# EN: Rivera is the Oriental president's business. ES: Rivera es asunto del presidente oriental.
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_arg_notice.1.b|" # GG_TRAZA
		name = gg_arg_notice.1.b
		gg_patron_change = { PATRON = rosas DELTA = -5 }
		ai_chance = { base = 30 }
	}
}
