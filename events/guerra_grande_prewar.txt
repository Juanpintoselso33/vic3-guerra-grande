Yucutuja (gg_prewar.5) reads the answers to the Farrapos (gg_prewar.2) and the departments (gg_prewar.4): it waits, a week at a time, while either is unanswered. / Yucutuja lee las respuestas de los Farrapos y de los departamentos: espera, de a una semana, mientras alguna siga sin contestar.
# -------------------------------------------------------------------------------------------------
gg_prewar.20 = {
	type = country_event
	hidden = yes

	# EN: capped wait (22/9/2026). Each retry counts a week in gg_prewar_20_weeks; after 8 weeks
	#     Yucutuja is delivered anyway, answered or not, and the counter is removed.
	# ES: espera con tope (22/9/2026). Cada reintento suma una semana en gg_prewar_20_weeks; a las 8
	#     semanas Yucutuja se entrega igual, haya respuesta o no, y el contador se borra.
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_prewar.20|" # GG_TRAZA
		if = {
			limit = { NOT = { has_global_variable = gg_prewar_20_weeks } }
			set_global_variable = { name = gg_prewar_20_weeks value = 0 }
		}
		if = {
			limit = {
				OR = {
					has_global_variable = gg_pending_prewar_2
					has_global_variable = gg_pending_prewar_4
				}
				global_var:gg_prewar_20_weeks < 8
			}
			change_global_variable = { name = gg_prewar_20_weeks add = 1 }
			trigger_event = { id = gg_prewar.20 days = 7 }
		}
		else = {
			remove_global_variable = gg_prewar_20_weeks
			trigger_event = { id = gg_prewar.5 popup = yes }
		}
	}
}
