	}

	duration = 3

		this = c:URU
	}

	option = {
		name = gg_prewar.1.a

		gg_patron_change = { PATRON = francia DELTA = -5 }

	}

	option = {
		name = gg_prewar.1.b
		# EN: asylum respected. Rosas lower, France higher, Lavalle stays out for a while (read in T6).
		# ES: se respeta el asilo. Rosas mas bajo, Francia mas alta, Lavalle queda neutral un tiempo (T6).

		gg_patron_change = { PATRON = francia DELTA = 5 }
		set_global_variable = gg_unitarian_asylum
		ai_chance = { base = 35 }
	}

}
