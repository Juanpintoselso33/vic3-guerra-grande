namespace = gg_troops

# EN: gg_troops.1 -- the armies at the split, two days after guerra_grande.3 (both options). Hidden
#     and with no trigger, so it cannot be discarded when it is queued; by then the engine has moved
#     every battalion with its barracks and dropped the empty ones. See scripted_effects/gg_troops.txt.
# ES: gg_troops.1 -- los ejercitos en el corte, dos dias despues de guerra_grande.3 (las dos
#     opciones). Oculto y sin trigger, asi no se descarta al encolarse; para entonces el motor ya
#     movio cada batallon con su cuartel y descarto los vacios.
gg_troops.1 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_troops.1|" # GG_TRAZA
		gg_troops_split_armies_defensa = yes
		# EN/ES: 23/9 -- the city arms fast after the split (~6,500 in Feb 1843) / la ciudad se arma rapido
		add_modifier = { name = gg_defensa_city_in_arms months = 6 }
		if = { limit = { NOT = { has_modifier = gg_siege_no_aggression } } add_modifier = { name = gg_siege_no_aggression } }
		# EN/ES: 23/9 -- the Empire stays out until 1851 / el Imperio se queda afuera hasta 1851
		c:BRZ ?= { if = { limit = { NOT = { has_modifier = gg_brazil_stays_out } } add_modifier = { name = gg_brazil_stays_out } } }
	}
}

# EN: gg_troops.2 -- the same for the Cerrito, fired on the Cerrito itself (see the note in
#     scripted_effects/gg_troops.txt).
# ES: gg_troops.2 -- lo mismo para el Cerrito, disparado sobre el propio Cerrito.
gg_troops.2 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_troops.2|" # GG_TRAZA
		gg_troops_split_armies_cer = yes
		if = { limit = { NOT = { has_modifier = gg_siege_no_aggression } } add_modifier = { name = gg_siege_no_aggression } }
		# EN: 22/9/2026 (Juan, playtest as Argentina) -- the Cerrito's battalions are born empty and
		#     fill over months; in that game it had 16 battalions and 894 men against the Defensa's
		#     5,522, was half occupied by July 1843 and lost the war before the siege began. A
		#     government born in the middle of a war trains fast: the same answer vanilla gives the
		#     countries born of an uprising (static modifier "uprising", building_training_rate_mult 2)
		#     and the Taiping (training rate +1000, x2).
		# ES: los batallones del Cerrito nacen vacios; en esa partida tenia 16 batallones y 894
		#     hombres contra 5.522 de la Defensa y perdio antes del sitio. Un gobierno nacido en
		#     plena guerra entrena rapido: lo mismo que vanilla les da a los paises nacidos de un
		#     levantamiento y a los Taiping.
		add_modifier = { name = gg_cerrito_army_forming months = 6 }
		# EN/ES: 23/9 -- the siege lives off its countryside / el sitio vive de su campania
		if = { limit = { NOT = { has_modifier = gg_cerrito_lives_off_the_land } } add_modifier = { name = gg_cerrito_lives_off_the_land } }
	}
}

# EN: The next-day mobilizer (gg_mobilize_next_day): on the owner, the day after its armies were created.
# ES: El movilizador del dia siguiente: sobre el duenio, al dia siguiente de crear sus ejercitos.
gg_troops.10 = {
	type = country_event
	hidden = yes
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_troops.10|" # GG_TRAZA
		remove_variable = gg_mob_pending
		if = {
			limit = { has_variable = gg_mob_enemy }
			var:gg_mob_enemy ?= { save_scope_as = gg_mob_enemy_scope }
			if = {
				limit = { exists = scope:gg_mob_enemy_scope }
				random_scope_war = {
					limit = { is_war_participant = scope:gg_mob_enemy_scope }
					random_scope_front = { save_scope_as = gg_mob_front }
				}
			}
			remove_variable = gg_mob_enemy
		}
		every_in_list = {
			variable = gg_mob_queue
			if = { limit = { is_mobilized = no } fully_mobilize_army = yes }
			if = { limit = { exists = scope:gg_mob_front } deploy_to_front = scope:gg_mob_front }
		}
		clear_variable_list = gg_mob_queue
	}
}
