# EN: Both governments of the Guerra Grande abolished slavery, and both did it to fill their
#     ranks. The Defensa by law of 12 December 1842 -- Ley N. 242, six days after the defeat of
#     Arroyo Grande -- whose article 2 sends the freed men "al servicio de las armas por el
#     tiempo que crea necesario". The Cerrito by decree of 26 October 1846, whose text does not
#     mention conscription but whose regulation did: days after it was promulgated the blanco
#     authorities were drawing up lists of former slaves fit to serve.
#     The men were not free: the slaveholders were the estancieros and the saladeristas who
#     backed the government, and many of them were Brazilian subjects on the frontier. That is
#     the whole loop -- battalions bought with the support of the class that owned the country.
# ES: Los dos gobiernos de la Guerra Grande abolieron la esclavitud, y los dos lo hicieron para
#     llenar sus filas. La Defensa por ley del 12 de diciembre de 1842 -- Ley N. 242, seis dias
#     despues de la derrota de Arroyo Grande -- cuyo articulo 2 destina a los libertos "al
#     servicio de las armas por el tiempo que crea necesario". El Cerrito por decreto del 26 de
#     octubre de 1846, que no menciona el reclutamiento en su texto pero si en su reglamento:
#     dias despues de promulgado ya se hacian listas de ex esclavos utiles para el ejercito.
#     Los hombres no salieron gratis: los duenios de esclavos eran los estancieros y los
#     saladeristas que sostenian al gobierno, y muchos eran subditos brasilenios de la frontera.
#     Ese es el circuito entero: batallones pagados con el apoyo de la clase que tenia el pais.
#
# Sources: Ley N. 242, 12.XII.1842 (IMPO, Registro Nacional de Leyes y Decretos, t. II);
# Keila Grinberg, "Emancipacion y guerra en el Rio de la Plata, 1840-1865", Historia Mexicana
# 69:2, 2019; Florencia Thul Charbonier, "Mercado de trabajo y movilizacion militar en
# Montevideo durante la Guerra Grande (1838-1851)", Claves 5:8, 2019; Mario Etchechury Barrera,
# "'Defensores de la humanidad y la civilizacion'", Historia (PUC Chile) 50:II, 2017; Magdalena
# Candioti, "Abolicion y post-abolicion de la esclavitud en la America Hispana", Anuario IHA,
# 2022, on Alex Borucki, "Abolicionismo y trafico de esclavos en Montevideo", 2009; Ministerio
# de Educacion y Cultura (Uruguay), "Esclavitud y afrodescendientes en Uruguay".
# Full citations, bilingual, in SOURCES_abolition.md.

# EN: Hidden dispatcher. The split event fires on ROOT, which is URU in one branch and CER in
#     the other, so a decree event scheduled there directly would reach the wrong government
#     half the time. This one fires on whoever ROOT is and then hands each decree to the country
#     that actually made it, by tag, once both exist.
# ES: Despachador oculto. El evento del corte dispara sobre ROOT, que es URU en una rama y CER
#     en la otra, asi que un decreto agendado ahi le llegaria al gobierno equivocado la mitad de
#     las veces. Este dispara sobre quien sea ROOT y le pasa cada decreto al pais que de verdad
#     lo dicto, por tag, cuando los dos ya existen.
gg_abolition.1 = {
	type = country_event
	hidden = yes

	# EN/ES: T6 -- on 12 December 1842 the Cerrito does not exist yet / el Cerrito todavia no existe
	trigger = {
		exists = c:URU
	}

	immediate = {
		hidden_effect = {
			# EN: 1842.12.12 -- only if the defeat that produced the law already happened, or if this
			#     run has no Arroyo Grande at all (gg_partition_delayed). Otherwise Arroyo Grande
			#     itself schedules it six days later.
			# ES: 12.12.1842 -- solo si la derrota que produjo la ley ya paso, o si en esta partida no
			#     hay Arroyo Grande. Si no, la programa el propio Arroyo Grande seis dias despues.
			# EN: playtest 2026-09-19 -- Ley 242 is the Defensa's law, and it was being passed by the
			#     government that existed on 12 December 1842, which in the blanco path is Oribe's.
			#     It now waits for the partition; if the country has not split yet, the law is not
			#     lost -- gg_partition_calendar passes it two days after the Defensa exists.
			# ES: la Ley 242 es la ley de la Defensa, y la votaba el gobierno que existiera el 12 de
			#     diciembre de 1842, que en el camino blanco es el de Oribe. Ahora espera a la
			#     particion; si el pais todavia no se partio la ley no se pierde: la pasa
			#     gg_partition_calendar dos dias despues de que exista la Defensa.
			if = {
				limit = {
					has_global_variable = guerra_grande_split_done
					OR = {
						has_global_variable = gg_rivera_arroyo_grande_settled
						has_global_variable = gg_partition_delayed
					}
				}
				c:URU ?= {
					trigger_event = { id = gg_abolition.2 days = 0 popup = yes }
				}
			}
			# EN/ES: T6 -- the Cerrito's decree (1846.10.26) is scheduled by gg_partition_calendar
		}
	}
}

# 1842.12.12 - Ley N. 242. Article 1: "Desde la promulgacion de la presente resolucion, no hay
# esclavos en todo el territorio de la Republica." Article 2 sends the able-bodied men to the
# army. Around 1,800 libertos form the line battalions that the garrison's own commanders
# consider the only real soldiers on the walls (Etchechury 2017, note 28, on Borucki 2015).
gg_abolition.2 = {
	type = country_event
	placement = ROOT

	title = gg_abolition.2.t
	# EN: playtest 2026-09-17 -- the law's date is fixed, but the war around it follows the player:
	#     Arroyo Grande can come a year early (gg_partition_early) or not at all (gg_partition_delayed).
	# ES: la fecha de la ley es fija, pero la guerra alrededor sigue al jugador: Arroyo Grande puede
	#     llegar un anio antes (gg_partition_early) o no pasar (gg_partition_delayed).
	desc = {
		first_valid = {
			triggered_desc = {
				desc = gg_abolition.2.d_early_delayed
				trigger = {
					NOT = { has_global_variable = gg_rivera_arroyo_grande_settled }
					has_global_variable = guerra_grande_split_done
				}
			}
			triggered_desc = {
				desc = gg_abolition.2.d_delayed
				trigger = { NOT = { has_global_variable = gg_rivera_arroyo_grande_settled } }
			}
			triggered_desc = {
				desc = gg_abolition.2.d_early
				trigger = { has_global_variable = guerra_grande_split_done }
			}
			triggered_desc = {
				desc = gg_abolition.2.d
				trigger = { always = yes }
			}
		}
	}
	flavor = gg_abolition.2.f

	# EN: Ley N.o 242, 12 Dec 1842: the Defensa frees the slaves to fill its ranks.
	event_image = {
		video = "gg_abol_defensa"
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	# EN/ES: playtest 2026-09-17 -- 3 days is vanilla's default and at speed 4-5 it is one second:
	#     the game answered for the player. Every event of this mod has consequences, so they wait.
	#     / 3 dias es el default de vanilla y a velocidad 4-5 es un segundo: el juego elegia solo.
	duration = 1825

	# EN/ES: playtest 2026-09-19 -- the Defensa's law needs the Defensa / la ley de la Defensa
	#     necesita a la Defensa
	trigger = {
		c:URU ?= this
		has_variable = guerra_grande_defensa_government
		NOT = { has_variable = guerra_grande_abolition_settled }
	}

	# The decree as it was written: freedom in article 1, the army in article 2.
	option = {
		name = gg_abolition.2.a
		default_option = yes
		set_variable = guerra_grande_abolition_settled
		set_variable = guerra_grande_abolition_decreed
		activate_law = law_type:law_slavery_banned
		create_military_formation = {
			type = army
			hq_region = capital.region
			combat_unit = {
				type = unit_type:combat_unit_type_line_infantry
				state_region = s:STATE_MONTEVIDEO
				count = 2
			}
		}
		add_modifier = {
			name = guerra_grande_abolition_liberto_battalions
			years = 10
			is_decaying = yes
		}
		# EN: and the bill. The owners were the estancieros and the saladeristas -- the class the
		#     government rests on -- and the men who walked into the battalions were the hands
		#     that salted the hides. They were promised compensation "en tiempo oportuno" and
		#     most of them never saw it.
		# ES: y la cuenta. Los duenios eran los estancieros y los saladeristas, la clase en la que
		#     se apoya el gobierno, y los hombres que entraron a los batallones eran los brazos
		#     que salaban los cueros. Les prometieron indemnizacion "en tiempo oportuno" y la
		#     mayoria no la vio nunca.
		ig:ig_landowners ?= {
			add_modifier = {
				name = guerra_grande_abolition_slaveholders_alienated
				days = long_modifier_time
				is_decaying = yes
			}
		}
		add_modifier = {
			name = guerra_grande_abolition_saladeros_short_handed
			years = 5
			is_decaying = yes
		}
		add_radicals = {
			value = medium_radicals
			strata = upper
		}
		hidden_effect = {
			trigger_event = { id = gg_abolition.4 days = 365 popup = yes }
		}
		ai_chance = {
			base = 70
		}
	}

	# EN: the measure that came first, in mid-1842: slaves drawn by lot for the army, 300 pesos
	#     paid to the owner for each man, and the man freed on the spot with four years to serve.
	#     It did not raise enough men, which is why the law of December followed.
	# ES: la medida que vino antes, a mediados de 1842: sorteo de esclavos para el servicio, 300
	#     pesos al duenio por cada uno, y el hombre libre en el acto con cuatro anios por delante.
	#     No alcanzo a juntar hombres, y por eso vino la ley de diciembre.
	option = {
		name = gg_abolition.2.b
		set_variable = guerra_grande_abolition_settled
		create_military_formation = {
			type = army
			hq_region = capital.region
			combat_unit = {
				type = unit_type:combat_unit_type_line_infantry
				state_region = s:STATE_MONTEVIDEO
				count = 1
			}
		}
		add_treasury = -20000
		add_modifier = {
			name = guerra_grande_abolition_bought_recruits
			years = 5
			is_decaying = yes
		}
		ai_chance = {
			base = 20
		}
	}

	# Property before men: the ranks stay empty and the owners stay loyal.
	option = {
		name = gg_abolition.2.c
		set_variable = guerra_grande_abolition_settled
		ig:ig_landowners ?= {
			add_modifier = {
				name = guerra_grande_abolition_slaveholders_reassured
				days = long_modifier_time
				is_decaying = yes
			}
		}
		add_loyalists = {
			value = medium_radicals
			strata = upper
		}
		add_modifier = {
			name = guerra_grande_abolition_ranks_unfilled
			years = 5
			is_decaying = yes
		}
		ai_chance = {
			base = 10
		}
	}
}

# 1846.10.26 - the Cerrito's decree, total and unrestricted, its application watched by Oribe

# EN: the bill from across the frontier, and it came to both governments. Most of the slaves on
#     Uruguayan soil belonged to Brazilian owners with estancias astride the border; in some
#     districts, Tacuarembo among them, the enslaved were a third of the population. In December
#     1842, days after the Defensa's law, more than two hundred slaves were carried out of
#     Montevideo aboard the corvette Siete de Abril to Santa Catarina, to save the property of
#     the Empire's subjects. After the Cerrito's decree the claims began in earnest in 1847, and
#     Oribe's government refused to hand back fugitives who were serving in its army -- a refusal
#     that went on hardening relations with Rio de Janeiro until 1851 (Grinberg 2019).
# ES: la cuenta que llego de la otra orilla, y les llego a los dos gobiernos. La mayoria de los
#     esclavos en suelo oriental eran de duenios brasilenios con estancias a caballo de la
#     frontera; en algunos distritos, Tacuarembo entre ellos, llegaban a un tercio de la
#     poblacion. En diciembre de 1842, dias despues de la ley de la Defensa, mas de doscientos
#     esclavos salieron de Montevideo en la corbeta Siete de Abril rumbo a Santa Catarina, para
#     salvar la propiedad de los subditos del Imperio. Despues del decreto del Cerrito las
#     reclamaciones arreciaron en 1847, y el gobierno de Oribe se nego a devolver fugitivos que
#     servian en su ejercito: una negativa que fue endureciendo la relacion con Rio de Janeiro
#     hasta 1851.
gg_abolition.4 = {
	type = country_event
	placement = ROOT

	title = gg_abolition.4.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = gg_abolition.4.d_defensa
				trigger = {
					has_variable = guerra_grande_defensa_government
				}
			}
			triggered_desc = {
				desc = gg_abolition.4.d_cerrito
				trigger = {
					has_variable = guerra_grande_cerrito_government
				}
			}
			triggered_desc = {
				desc = gg_abolition.4.d_generic
				trigger = { always = yes }
			}
		}
	}
	flavor = gg_abolition.4.f

	# EN: Brazil's claims for the fugitives who crossed the border.
	event_image = {
		video = "gg_money_brazilian_claims"
	}

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 1825

	trigger = {
		has_variable = guerra_grande_abolition_decreed
		NOT = { has_variable = guerra_grande_abolition_claims_answered }
	}

	# "El negro paso de cosa a hombre porque podia cambiarsele esa condicion; y sin gran
	# injusticia no puede volver al estado de esclavitud." Free soil, and the Empire's enmity.
	option = {
		name = gg_abolition.4.a
		# EN/ES: T4 -- FE-4: free soil sours Rio / el suelo libre agria a Rio
		gg_patron_change = { PATRON = brasil DELTA = -5 }
		default_option = yes
		set_variable = guerra_grande_abolition_claims_answered
		add_modifier = {
			name = guerra_grande_abolition_free_soil
			years = 10
			is_decaying = yes
		}
		if = {
			limit = { exists = c:BRZ }
			change_relations = {
				country = c:BRZ
				value = -20
			}
		}
		ai_chance = {
			base = 60
		}
	}

	# Hand them back, the way the Siete de Abril did. The Empire is placated; the decree is a lie.
	option = {
		name = gg_abolition.4.b
		# EN/ES: T4 -- FE-4: returning them buys the Empire / devolverlos compra al Imperio
		gg_patron_change = { PATRON = brasil DELTA = 10 }
		set_variable = guerra_grande_abolition_claims_answered
		add_modifier = {
			name = guerra_grande_abolition_slaves_returned
			years = 10
			is_decaying = yes
		}
		if = {
			limit = { exists = c:BRZ }
			change_relations = {
				country = c:BRZ
				value = 20
			}
		}
		add_radicals = {
			value = medium_radicals
			strata = lower
		}
		ai_chance = {
			base = 40
		}
	}
}
