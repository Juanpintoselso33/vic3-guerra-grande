# EN: THE STATE OF BUENOS AIRES, 1852-1861 (23/9/2026, Juan). After Urquiza's victory in the Pact, or with two
#     governments, Buenos Aires and the Confederation live apart until Cepeda and Pavon settle which of them makes the
#     Republic. Sides and wars in common/scripted_effects/gg_buenos_aires.txt; the plan and the dated sources in
#     docs/guerra-grande-argentina/buenos-aires-separada-plan.md (design repo).
#     The relays (.101-.108) are scheduled on ARG on the first day (common/on_actions/guerra_grande_on_actions.txt):
#     no trigger on them, because a scheduled event's trigger is checked when it is scheduled.
# ES: EL ESTADO DE BUENOS AIRES, 1852-1861. Despues de la victoria de Urquiza en el Pacto, o con dos gobiernos,
#     Buenos Aires y la Confederacion viven aparte hasta que Cepeda y Pavon deciden cual de las dos hace la Republica.
#     Los relevos se agendan sobre ARG el primer dia, sin trigger.
# SOURCE / FUENTE: Canedo, M. (coord.), Un pais para los portenos (UNLP, 2023), https://doi.org/10.24215/978-950-34-2319-6 ;
#     Garcia, C. F., «El tratado del 11 de noviembre de 1859», Humanidades 28 (1940); Moreno, G. R., «El Pacto de San
#     Jose de Flores y la Union Nacional», Derechos en Accion 3 (2017); argentinahistorica.com.ar (Scobie-based chapter
#     on the convivencia treaties and the differential duties); es.wikipedia, «Estado de Buenos Aires».

gg_ba.101 = {
	type = country_event
	hidden = yes
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_ba.101|" # GG_TRAZA
		if = {
			limit = {
				c:ARG ?= this
				NOT = { has_global_variable = gg_ba_split_active }
				NOT = { has_global_variable = gg_ba_done }
				has_global_variable = gg_pact_settlement_urquiza
				has_state_in_state_region = STATE_BUENOS_AIRES
				any_scope_state = { NOT = { state_region = s:STATE_BUENOS_AIRES } }
				# EN: 23/9/2026 (AUD-BA-WAR) -- was `is_at_war = no`: a war with Brazil over the Banda Oriental (the second
				#     Cisplatine war, gg_cisplatina.2) kept the split waiting until it expired in 1855. Only a war with the
				#     provinces themselves holds the revolution back; any other war goes on with the Confederation.
				# ES: era `is_at_war = no`: una guerra con el Brasil por la Banda Oriental dejaba la particion esperando
				#     hasta vencer en 1855. Solo una guerra con las provincias frena la revolucion; otra guerra sigue con
				#     la Confederacion.
				# EN/ES: the province countries (gg_is_province_country), the Coalition and the State of Buenos Aires
				#     / las provincias-pais, la Coalicion y el Estado de Buenos Aires
				NOT = {
					any_country = {
						OR = {
							gg_is_province_country = yes
							c:CLN ?= this
							c:BUE ?= this
						}
						has_war_with = c:ARG
					}
				}
			}
			gg_ba_split_historical = yes
			trigger_event = { id = gg_ba.1 days = 0 popup = yes }
		}
		else_if = {
			limit = {
				c:ARG ?= this
				NOT = { has_global_variable = gg_ba_split_active }
				NOT = { has_global_variable = gg_ba_done }
				has_global_variable = gg_pact_settlement_stalemate
				c:ENR ?= { is_subject = no }
			}
			gg_ba_split_stalemate = yes
			trigger_event = { id = gg_ba.2 days = 0 popup = yes }
			c:ENR = { trigger_event = { id = gg_ba.2 days = 0 popup = yes } }
		}
		# EN: 23/9/2026 (AUD-BA-SUBJECT, buenos-aires-separada-plan.md) -- «two governments» with Entre Rios already a
		#     subject: there are not two governments. Rosas beat Entre Rios in the real war before the rule closed the
		#     Pact, so the Confederation has one government and nothing splits. Recorded once, and no retry.
		#     24/9/2026 (AUD-BA-STALEMATE-ENR) -- the Pact war's own protectorate no longer reaches here: the draw undoes it
		#     (gg_pact_stalemate_frees_enr). What is left is an Entre Rios that is gone, or a subject by another road.
		# ES: «dos gobiernos» con Entre Rios ya sometido: no hay dos gobiernos. Rosas batio a Entre Rios en la guerra real
		#     antes de que la regla cerrara el Pacto; la Confederacion tiene un solo gobierno y no se parte nada.
		#     24/9: el protectorado de la propia guerra del Pacto ya no llega aca (las tablas lo deshacen); queda Entre Rios
		#     desaparecido o sujeto por otro camino.
		else_if = {
			limit = {
				c:ARG ?= this
				NOT = { has_global_variable = gg_ba_split_active }
				NOT = { has_global_variable = gg_ba_done }
				has_global_variable = gg_pact_settlement_stalemate
				OR = {
					NOT = { exists = c:ENR }
					c:ENR ?= { is_subject_of = c:ARG }
				}
			}
			debug_log = "GG_EF|[TimeKeeper.GetCurrentDate.GetString]|gg_ba.101|no_split_enr_subject|" # GG_TRAZA
		}
		# EN/ES: the Pact may close later (a war still running): retry monthly through 1854 / se reintenta
		else_if = {
			limit = {
				NOT = { has_global_variable = gg_ba_split_active }
				NOT = { has_global_variable = gg_ba_done }
				game_date < 1855.1.1
				OR = {
					NOT = { has_global_variable = gg_pact_settled }
					has_global_variable = gg_pact_settlement_urquiza
					has_global_variable = gg_pact_settlement_stalemate
				}
			}
			trigger_event = { id = gg_ba.101 days = 30 }
		}
	}
}

gg_ba.9 = {
	type = country_event
	hidden = yes
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_ba.9|" # GG_TRAZA
		if = {
			limit = {
				has_global_variable = gg_ba_split_active
				NOT = { has_global_variable = gg_ba_unifying }
				has_global_variable = gg_ba_phase
			}
			# EN/ES: the Pact's Historical rule keeps the history: Lagos and Pavon to Buenos Aires, Cepeda to the
			#     Confederation / la regla historica: Lagos y Pavon para Buenos Aires, Cepeda para la Confederacion
			gg_ba_save_sides = yes
			if = {
				limit = { gg_pact_historical = yes }
				if = {
					limit = { global_var:gg_ba_phase = 2 }
					set_global_variable = { name = gg_ba_conf_won value = 1 }
					set_global_variable = { name = gg_ba_ba_won value = 0 }
				}
				else = {
					set_global_variable = { name = gg_ba_ba_won value = 1 }
					set_global_variable = { name = gg_ba_conf_won value = 0 }
				}
			}
			if = {
				limit = { global_var:gg_ba_conf_won = 1 }
				# EN/ES: Cepeda won: San Jose de Flores; Lagos or Pavon won: the State is absorbed
				if = {
					limit = { global_var:gg_ba_phase = 2 }
					global_var:gg_ba_side ?= { trigger_event = { id = gg_ba.8 days = 0 popup = yes } }
					global_var:gg_conf_side ?= { trigger_event = { id = gg_ba.8 days = 0 popup = yes } }
					if = {
						limit = { exists = scope:gg_ba_state }
						# EN: 24/9/2026 (AUD-VASSAL-RECOGNIZED) -- was type = vassal, which the engine dropped at once: the
						#     Confederation is a recognized country and vanilla's vassal takes only an unrecognized overlord.
						#     San Jose de Flores (11/11/1859) made Buenos Aires a province of the Confederation again: the
						#     Province of the Pact, without the rank rule (the State may outrank Parana). Watched thirty days.
						# ES: era type = vassal, que el motor tiraba enseguida (la Confederacion es un pais reconocido). San Jose
						#     de Flores volvio a hacer de Buenos Aires una provincia de la Confederacion: la Provincia del Pacto,
						#     sin la regla del rango (el Estado puede tener mas rango que Parana). Vigilado treinta dias.
						global_var:gg_conf_side ?= {
							create_diplomatic_pact = {
								country = scope:gg_ba_state
								type = gg_pact_province
							}
							gg_subject_watch = { SUBJECT = scope:gg_ba_state }
						}
					}
					global_var:gg_ba_side ?= {
						every_scope_state = {
							limit = { state_region = s:STATE_BUENOS_AIRES }
							add_modifier = { name = gg_ba_flores_customs years = 5 }
						}
					}
					set_global_variable = gg_ba_flores
				}
				else = {
					gg_ba_unify = { WINNER = conf }
				}
			}
			else_if = {
				limit = { global_var:gg_ba_ba_won = 1 global_var:gg_ba_phase = 3 }
				gg_ba_unify = { WINNER = ba }
			}
			else = {
				# EN/ES: Buenos Aires held (or nobody won): it stays apart / Buenos Aires aguanto: sigue aparte
				global_var:gg_ba_side ?= { trigger_event = { id = gg_ba.12 days = 0 popup = yes } }
				global_var:gg_conf_side ?= { trigger_event = { id = gg_ba.12 days = 0 popup = yes } }
			}
			remove_global_variable = gg_ba_phase
		}
	}
}

# EN: two days after the end: the country that remains absorbs the other, with its people. If Buenos Aires won in the
#     Urquiza branch, Mitre governs the Republic from Buenos Aires (he took the national executive on 12/12/1861 and
#     the presidency on 12/10/1862). If Entre Rios won with two governments, it takes the ARG tag the next day.
# ES: dos dias despues: el pais que queda absorbe al otro, con su gente. Si gano Buenos Aires en la rama de Urquiza,
#     Mitre gobierna la Republica desde Buenos Aires. Si gano Entre Rios con dos gobiernos, toma el tag ARG al dia
#     siguiente.
gg_ba.20 = {
	type = country_event
	hidden = yes
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_ba.20|" # GG_TRAZA
		if = {
			limit = { has_global_variable = gg_ba_unifying NOT = { has_global_variable = gg_ba_done } }
			set_global_variable = gg_ba_done
			save_scope_as = gg_ba_keeper
			if = {
				limit = { global_var:gg_ba_side ?= this }
				global_var:gg_conf_side ?= { save_scope_as = gg_ba_absorbed }
			}
			else = {
				global_var:gg_ba_side ?= { save_scope_as = gg_ba_absorbed }
			}
			# EN: 23/9/2026 (AUD-SEAT-ANNEX, the rule of gg_argentina_province_join) -- both countries are players': nobody
			#     is annexed and nobody changes seat (gg_ba_unify sends no gg_ba.13). The side that lost the last war
			#     becomes the winner's Province of the Pact (a protectorate until 24/9/2026), which is what one country with two governments can be; the capital,
			#     the ruler and the ARG tag are not touched.
			# ES: los dos paises son de jugadores: no se anexa a nadie ni nadie cambia de sede. El lado que perdio la
			#     ultima guerra queda provincia del Pacto del ganador; no se tocan capital, gobernante ni tag.
			if = {
				limit = {
					exists = scope:gg_ba_absorbed
					is_player = yes
					scope:gg_ba_absorbed = { is_player = yes }
				}
				if = {
					limit = { global_var:gg_ba_winner_ba = 1 }
					global_var:gg_ba_side ?= { save_scope_as = gg_ba_overlord }
					global_var:gg_conf_side ?= { save_scope_as = gg_ba_protected }
				}
				else = {
					global_var:gg_conf_side ?= { save_scope_as = gg_ba_overlord }
					global_var:gg_ba_side ?= { save_scope_as = gg_ba_protected }
				}
				if = {
					limit = {
						exists = scope:gg_ba_overlord
						exists = scope:gg_ba_protected
						scope:gg_ba_protected = { is_subject = no }
					}
					# EN/ES: 24/9 (AUD-VASSAL-RECOGNIZED) -- was a protectorate, which needs the winner to outrank the loser
					#     (both halves of one country are close in rank): the Province of the Pact, without the rank rule
					#     / era un protectorado, que pide mas rango al ganador; la Provincia del Pacto, sin esa regla
					scope:gg_ba_overlord = {
						create_diplomatic_pact = { country = scope:gg_ba_protected type = gg_pact_province }
						gg_subject_watch = { SUBJECT = scope:gg_ba_protected }
					}
				}
				clear_saved_scope = gg_ba_overlord
				clear_saved_scope = gg_ba_protected
				clear_saved_scope = gg_ba_absorbed
			}
			if = {
				limit = { exists = scope:gg_ba_absorbed }
				# EN: 24/9/2026 (the report's open question) -- the loser's subjects are handed to the country that remains
				#     BEFORE the annexation, so they never depend on what the engine does with an annexed country's
				#     subjects (with two governments, Rosas's Argentina may still hold the Banda Oriental as a client).
				# ES: los sujetos del perdedor pasan al que queda ANTES de la anexion, asi no dependen de lo que haga el
				#     motor con los sujetos de un pais anexado.
				gg_ba_take_subjects = yes
				scope:gg_ba_absorbed = { gg_bring_characters_to = { TARGET = scope:gg_ba_keeper } }
				annex = scope:gg_ba_absorbed
			}
			if = {
				limit = { global_var:gg_ba_winner_ba = 1 exists = scope:gg_ba_absorbed }
				if = {
					limit = { has_state_in_state_region = STATE_BUENOS_AIRES }
					set_capital = STATE_BUENOS_AIRES
				}
				if = {
					limit = { has_global_variable = gg_ba_from_urquiza }
					# EN/ES: 24/9 -- the one Mitre, fetched, never a second copy (gg_mitre_fetch: the mod's own, marched
					#     with the Ejercito Grande) / el unico Mitre, traido, nunca una segunda copia
					gg_mitre_fetch = yes
					random_scope_character = {
						limit = { gg_is_the_mitre = yes }
						set_character_as_ruler = yes
					}
				}
			}
			if = {
				limit = { NOT = { c:ARG ?= this } exists = scope:gg_ba_absorbed }
				trigger_event = { id = gg_ba.21 days = 1 }
			}
			remove_global_variable = gg_ba_split_active
			clear_saved_scope = gg_ba_absorbed
			clear_saved_scope = gg_ba_keeper
		}
	}
}
