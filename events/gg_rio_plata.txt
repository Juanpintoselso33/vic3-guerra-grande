# -------------------------------------------------------------------------------------------------
# EN: gg_plata.4 -- the intervention's clock (Codex review 22/9/2026). One month per tick, independent
#     of the war journal and of truces, until France settles. The links of the chain read it through
#     gg_plata_intervention_at. 23/9/2026 (B1) -- two copies run, on Rosas's Argentina and on Britain,
#     each rescheduling itself on its own country; gg_plata_clock_ticked (29 days) lets only one of them
#     count each month, and gg_plata_clock_alive tells the watchdog in gg_plata_situation_monthly that
#     the clock still runs. A British revolution no longer stops it.
# ES: gg_plata.4 -- el reloj de la intervencion. Un mes por vuelta, independiente del journal de la
#     guerra y de las treguas, hasta que Francia arregla. Corren dos copias, sobre la Argentina de Rosas
#     y sobre Gran Bretania; una marca de 29 dias deja contar a una sola por mes, y otra le avisa al vigia
#     que el reloj sigue vivo. Una revolucion britanica ya no lo frena.
# -------------------------------------------------------------------------------------------------
gg_plata.4 = {
	type = country_event
	hidden = yes
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_plata.4|" # GG_TRAZA
		if = {
			limit = {
				has_global_variable = gg_plata_intervention
				NOT = { has_global_variable = gg_plata_intervention_over }
				NOT = { has_global_variable = gg_plata_clock_ticked }
			}
			set_global_variable = { name = gg_plata_clock_ticked days = 29 }
			set_global_variable = { name = gg_plata_clock_alive days = 65 }
			change_global_variable = { name = gg_plata_intervention_months add = 1 }
			# EN: 22/9/2026, decided with Juan -- France lifts her blockade of Buenos Aires with Gore and
			#     Gros: the convention of 12/6/1848 was arranged by Baron Gros to compensate Montevideo for
			#     the lifting of the French blockade (SOURCES_powers.md section 7, Duvergier, Collection
			#     complete des lois, t. 50). 34 months after the passports (31/7/1845 -> 12/6/1848). France
			#     stays in the question until Lepredour (month 45). The same month applies to a French
			#     intervention alone, which has no Gros link of its own.
			# ES: decidido con Juan -- Francia levanta su bloqueo de Buenos Aires con Gore y Gros: la
			#     convencion del 12/6/1848 la arreglo el baron Gros para compensar a Montevideo por el
			#     levantamiento del bloqueo frances. 34 meses despues de los pasaportes. Francia sigue en la
			#     cuestion hasta Lepredour. El mismo mes vale para una intervencion solo francesa.
			if = {
				limit = {
					global_var:gg_plata_intervention_months >= 34
					# EN/ES: 23/9/2026 (D2) -- never before Gros's convention / nunca antes del convenio de Gros
					game_date >= 1848.6.12
					NOT = { has_global_variable = gg_plata_ba_blockade_lifted }
				}
				set_global_variable = gg_plata_ba_blockade_lifted
				# EN/ES: Codex 22/9 -- lifted now, not at the next monthly upkeep / levantado ahora, no en el pulso siguiente
				p:x365A5B.state ?= { if = { limit = { has_modifier = gg_plata_phase_intervention_port } remove_modifier = gg_plata_phase_intervention_port } }
				gg_plata_notify = { EVENT = gg_plata.46 }
			}
		}
		# EN/ES: every copy reschedules itself while the intervention runs (the tick above counts once a month)
		#     / cada copia se reagenda mientras dure la intervencion (el tic de arriba cuenta una vez por mes)
		if = {
			limit = {
				has_global_variable = gg_plata_intervention
				NOT = { has_global_variable = gg_plata_intervention_over }
				year < 1862
			}
			trigger_event = { id = gg_plata.4 days = 30 }
		}
	}
}
