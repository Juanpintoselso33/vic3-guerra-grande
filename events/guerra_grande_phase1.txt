# F1-0 - the blanco path converges (1 February 1839)
gg_phase1.9 = {
	type = country_event
	placement = ROOT

	title = gg_phase1.9.t
	desc = gg_phase1.9.d
	flavor = gg_phase1.9.f

	# EN/ES: T6 reuses an existing video; the new picture lands in T9 (SOURCES_flujo_images.md).
	event_image = {
		video = "gg_oribe_fall"
	}

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		this = c:URU
		has_variable = guerra_grande_blanco_path
		NOT = { has_global_variable = guerra_grande_split_done }
	}

	immediate = {
		# EN: what the resistance of 1838 bought: Palmar avoided and a deal with France.
		# ES: lo que compro la resistencia de 1838: Palmar evitado y un trato con Francia.
		if = {
			limit = { has_global_variable = gg_palmar_avoided }
			gg_score_blanco = { AMOUNT = 2 }
		}
		if = {
			limit = { has_global_variable = gg_french_deal_1838 }
			gg_score_blanco = { AMOUNT = 1 }
		}
	}

	option = {
		name = gg_phase1.9.a
		default_option = yes
		# EN: F1-0 -- the blanco path converges: Oribe falls anyway, and leaves with his army.
		# ES: F1-0 -- el camino blanco converge: Oribe cae igual, y se va con su ejercito.
		set_ruling_interest_groups = { ig_petty_bourgeoisie ig_intelligentsia }
		random_scope_character = {
			limit = { has_template = URU_Fructuoso_Rivera }
			set_character_as_ruler = yes
		}
		gg_score_blanco = { AMOUNT = 1 }
		gg_patron_change = { PATRON = rosas DELTA = 5 }
		ai_chance = { base = 60 }
	}

	option = {
		name = gg_phase1.9.b
		# EN: negotiate the exit: fewer men go, but Rivera grants an amnesty the doctores remember.
		# ES: negociar la salida: se van menos hombres, pero Rivera da una amnistia que los doctores recuerdan.
		set_ruling_interest_groups = { ig_petty_bourgeoisie ig_intelligentsia }
		random_scope_character = {
			limit = { has_template = URU_Fructuoso_Rivera }
			set_character_as_ruler = yes
		}
		gg_cerrito_balance_change = { DELTA = 5 }
		ai_chance = { base = 40 }
	}

}


# F1-1 - Rivera declares war on Rosas (21 February 1839)
gg_phase1.1 = {
	type = country_event
	placement = ROOT

	title = gg_phase1.1.t
	desc = gg_phase1.1.d
	flavor = gg_phase1.1.f

	# EN/ES: T6 reuses an existing video; the new picture lands in T9 (SOURCES_flujo_images.md).
	event_image = {
		video = "gg_fl_rivera_rugendas"
	}

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		this = c:URU
		NOT = { has_global_variable = guerra_grande_split_done }
	}

	option = {
		name = gg_phase1.1.a
		default_option = yes
		# EN: historical: total war. France up, Rosas angry, Echague invades.
		# ES: lo historico: guerra total. Francia sube, Rosas se enoja, Echague invade.
		gg_patron_change = { PATRON = francia DELTA = 10 }
		gg_patron_change = { PATRON = rosas DELTA = 10 }
		hidden_effect = {
			trigger_event = { id = gg_phase1.2 days = 158 popup = yes }   # 1839.07.29 Echague cruza
			if = {
				# EN/ES: F0-2 b -- with asylum respected, Lavalle comes half a year later / Lavalle llega medio anio despues
				limit = { has_global_variable = gg_unitarian_asylum }
				trigger_event = { id = gg_phase1.4 days = 372 popup = yes }   # 1840.02.28
			}
			else = {
				trigger_event = { id = gg_phase1.4 days = 192 popup = yes }   # 1839.09.01 Lavalle
			}
			trigger_event = { id = gg_phase1.5 days = 616 popup = yes }   # 1840.10.29 Arana-Mackau
			trigger_event = { id = gg_phase1.6 days = 967 popup = yes }   # 1841.10.15 Famailla
			trigger_event = { id = gg_phase1.7 days = 1342 popup = yes }  # 1842.10.25 el Uruguay mayor
			trigger_event = { id = gg_phase1.8 days = 1368 popup = yes }  # 1842.11.20 Arroyo Grande
		}
		ai_chance = { base = 60 }
	}

	option = {
		name = gg_phase1.1.b
		# EN: only the alliance with Corrientes: less France, a smaller invasion.
		# ES: solo la alianza con Corrientes: menos Francia, una invasion mas chica.
		gg_patron_change = { PATRON = francia DELTA = 5 }
		gg_patron_change = { PATRON = rosas DELTA = 5 }
		set_global_variable = gg_small_invasion
		hidden_effect = {
			trigger_event = { id = gg_phase1.2 days = 158 popup = yes }   # 1839.07.29 Echague cruza
			if = {
				# EN/ES: F0-2 b -- with asylum respected, Lavalle comes half a year later / Lavalle llega medio anio despues
				limit = { has_global_variable = gg_unitarian_asylum }
				trigger_event = { id = gg_phase1.4 days = 372 popup = yes }   # 1840.02.28
			}
			else = {
				trigger_event = { id = gg_phase1.4 days = 192 popup = yes }   # 1839.09.01 Lavalle
			}
			trigger_event = { id = gg_phase1.5 days = 616 popup = yes }   # 1840.10.29 Arana-Mackau
			trigger_event = { id = gg_phase1.6 days = 967 popup = yes }   # 1841.10.15 Famailla
			trigger_event = { id = gg_phase1.7 days = 1342 popup = yes }  # 1842.10.25 el Uruguay mayor
			trigger_event = { id = gg_phase1.8 days = 1368 popup = yes }  # 1842.11.20 Arroyo Grande
		}
		ai_chance = { base = 25 }
	}

	option = {
		name = gg_phase1.1.c
		# EN: neutrality: France cools and nobody invades in 1839, but Rosas prepares anyway and
		#     Arroyo Grande comes a year early -- and with it the split (gg_partition_calendar).
		# ES: neutralidad: Francia se enfria y nadie invade en 1839, pero Rosas se prepara igual y
		#     Arroyo Grande llega un anio antes, y con el la particion.
		gg_patron_change = { PATRON = francia DELTA = -10 }
		set_global_variable = gg_partition_early
		hidden_effect = {
			trigger_event = { id = gg_phase1.5 days = 616 popup = yes }   # 1840.10.29 Arana-Mackau
			trigger_event = { id = gg_phase1.6 days = 967 popup = yes }   # 1841.10.15 Famailla
			trigger_event = { id = gg_phase1.8 days = 1003 popup = yes }  # 1841.11.20 Arroyo Grande
		}
		ai_chance = { base = 15 }
	}

}

# F1-4 - Arroyo Grande: the news that Oribe is 'defenceless' (20 November 1842)

	placement = ROOT

	title = gg_phase1.8.t
	desc = gg_phase1.8.d
	flavor = gg_phase1.8.f

	# EN/ES: T6 reuses an existing video; the new picture lands in T9 (SOURCES_flujo_images.md).
	event_image = {
		video = "gg_rivera_arroyo_grande"
	}

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		this = c:URU
		NOT = { has_global_variable = guerra_grande_split_done }
	}

	option = {
		name = gg_phase1.8.a
		default_option = yes
		# EN: historical: cross the Uruguay and look for him. Defeat; the split follows in 88 days.
		# ES: lo historico: cruzar el Uruguay y buscarlo. Derrota; la particion llega a los 88 dias.

		if = {
			limit = { has_global_variable = gg_invaded_entre_rios }
			gg_score_blanco = { AMOUNT = 1 }
		}
		if = {
			limit = { has_global_variable = gg_greater_uruguay }
			gg_score_blanco = { AMOUNT = 1 }
		}
		hidden_effect = {
			trigger_event = { id = gg_rivera.1 days = 16 popup = yes }       # la derrota, 6 de diciembre

	}

	option = {
		name = gg_phase1.8.b

		#     colorado army is intact.
		# ES: esperar en la costa: Oribe invade igual y la particion llega seis meses despues, pero el
		#     ejercito colorado queda entero.
		gg_score_colorado = { AMOUNT = 1 }
		set_global_variable = gg_partition_delayed

		set_global_variable = gg_rivera_arroyo_grande_settled
		set_global_variable = gg_rivera_reinforced
		hidden_effect = {
			trigger_event = { id = guerra_grande.3 days = 270 popup = yes }
		}
		ai_chance = { base = 40 }
	}

}
