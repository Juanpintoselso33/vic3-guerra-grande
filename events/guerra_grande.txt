# 1836.9.19 - The battle of Carpinteria. Oribe's men wear white ribbons, Rivera's red ones:
# the two parties of Uruguay are born here, three years before the war itself.
guerra_grande.1 = {
	type = country_event
	placement = ROOT

	title = guerra_grande.1.t
	desc = guerra_grande.1.d
	flavor = guerra_grande.1.f

	# Carlos Morel, "Combate de caballeria en la epoca de Rosas" (c. 1840), public domain.
	event_image = {
		video = "gg_carpinteria"
	}

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		this = c:URU
	}

	immediate = {
		add_contextless_journal_entry = je_guerra_grande
		# EN: T5 -- phase 0 already scored before the journal existed; bring the bar up to date.
		# ES: T5 -- la fase 0 ya puntuo antes de que existiera el journal; se pone la barra al dia.
		je:je_guerra_grande ?= {
			add_progress = { value = guerra_grande_blanco_lead name = guerra_grande_war_score_bar }
		}
		hidden_effect = {
			# EN: the ribbons name the parties. Vanilla groups only: Oribe sits with the
			#     landowners and the countryside, Rivera with the city and the port. BPM renames
			#     its own conservatives and liberals from the compatch.
			# ES: las divisas le dan nombre a los partidos. Solo grupos de vanilla: Oribe va con
			#     los terratenientes y la campania, Rivera con la ciudad y el puerto. BPM renombra
			#     sus propios conservadores y liberales desde el compatch.
			ig:ig_landowners ?= {
				set_interest_group_name = ig_blanco_party
			}
			ig:ig_petty_bourgeoisie ?= {
				set_interest_group_name = ig_colorado_party
			}
			# 1838.10.24 - Rivera marches on Montevideo.
			trigger_event = { id = guerra_grande.2 days = 765 popup = yes }
			# The four men the war is about, one every six months of the wait.
			trigger_event = { id = guerra_grande.11 days = 100 popup = yes }
			trigger_event = { id = guerra_grande.12 days = 280 popup = yes }
			trigger_event = { id = guerra_grande.13 days = 460 popup = yes }
			trigger_event = { id = guerra_grande.14 days = 640 popup = yes }
		}
	}

	option = {
		name = guerra_grande.1.a
		# EN: this is what the choice is worth when the war finally starts.
		# ES: esto es lo que vale la eleccion cuando la guerra por fin empiece.
		change_global_variable = {
			name = guerra_grande_blanco_score
			add = 2
		}
		je:je_guerra_grande ?= {
			add_progress = { value = 2 name = guerra_grande_war_score_bar }
		}
		default_option = yes
		set_variable = guerra_grande_blanco_path
		add_modifier = {
			name = guerra_grande_blanco_order
			years = 10
		}
		ai_chance = {
			base = 50
		}
	}

	option = {
		name = guerra_grande.1.b
		# EN: this is what the choice is worth when the war finally starts.
		# ES: esto es lo que vale la eleccion cuando la guerra por fin empiece.
		change_global_variable = {
			name = guerra_grande_colorado_score
			add = 2
		}
		je:je_guerra_grande ?= {
			add_progress = { value = -2 name = guerra_grande_war_score_bar }
		}
		set_variable = guerra_grande_colorado_path
		add_modifier = {
			name = guerra_grande_colorado_order
			years = 10
		}
		ai_chance = {
			base = 40
		}
	}

	option = {
		name = guerra_grande.1.c
		add_modifier = {
			name = guerra_grande_armed_truce
			years = 10
			is_decaying = yes
		}
		add_radicals = {
			value = small_radicals
		}
		ai_chance = {
			base = 10
		}
	}
}

			# ES: T6 -- la particion ya no sigue a la caida a los 110 dias: la agenda Arroyo Grande

			trigger_event = { id = gg_phase1.1 days = 120 popup = yes }   # 1839.02.21 guerra a Rosas
		}

	# Historical: Oribe resigns and sails to Buenos Aires, and Rivera takes the government.

		name = guerra_grande.2.a

		# ES: esto es lo que vale la eleccion cuando la guerra por fin empiece.

			name = guerra_grande_colorado_score
			add = 3
		}
		je:je_guerra_grande ?= {
			add_progress = { value = -3 name = guerra_grande_war_score_bar }
		}
		default_option = yes
		set_variable = guerra_grande_colorado_path
		# EN: a resignation has to actually remove him. set_ruling_interest_groups replaces the
		#     whole coalition -- add_ruling_interest_group only adds a partner and leaves the
		#     blancos in power, which is why nothing seemed to happen.
		# ES: una renuncia tiene que sacarlo de verdad. set_ruling_interest_groups reemplaza la
		#     coalicion entera; add_ruling_interest_group solo suma un socio y deja a los blancos
		#     en el gobierno, que es por lo que no pasaba nada.
		# EN: vanilla interest groups, because this mod's baseline is vanilla -- Better
		#     Politics Mod support belongs in la_argentina_bpm_compatch, not here. Naming
		#     ig_liberals in the base mod made it fail PostValidate on every load without
		#     BPM, and a resignation that fails validation is a government that never changes.
		#     In vanilla, the colorados are the city: merchants, doctores and the port.
		# ES: grupos de interes de vanilla, porque la base de este mod es vanilla: el soporte
		#     de Better Politics Mod va en la_argentina_bpm_compatch, no aca. Nombrar
		#     ig_liberals en el mod base hacia fallar el PostValidate en cada carga sin BPM,
		#     y una renuncia que falla la validacion es un gobierno que no cambia nunca.
		#     En vanilla, los colorados son la ciudad: comerciantes, doctores y el puerto.
		set_ruling_interest_groups = { ig_petty_bourgeoisie ig_intelligentsia }
		# Rivera takes over as head of state. Oribe sails for Buenos Aires.
		random_scope_character = {
			limit = { has_template = URU_Fructuoso_Rivera }
			set_character_as_ruler = yes
		}
		add_modifier = {
			name = guerra_grande_colorado_order
			years = 10
		}
		ai_chance = {
			base = 60
		}
	}

	# The blanco government digs in instead: Oribe stays, and the blancos take the state alone.
	option = {

# 1839 - The country splits: the Defensa keeps Montevideo, the Cerrito takes the countryside.
guerra_grande.3 = {
	type = country_event
	placement = ROOT

	title = guerra_grande.3.t
	desc = guerra_grande.3.d
	flavor = guerra_grande.3.f

	# "Vista de la Matriz", Montevideo lithograph, 1840, public domain.
	event_image = {
		video = "gg_two_governments"
	}

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 3

	trigger = {
		this = c:URU
		NOT = { exists = c:CER }
		exists = s:STATE_BANDA_ORIENTAL.region_state:URU
		any_scope_state = {
			state_region = s:STATE_MONTEVIDEO
		}
	}

	immediate = {
		hidden_effect = {
			# EN/ES: T1, C13 -- open_war and Rosas's event moved into the options, after the Cerrito exists.
			# EN: the foreign contingents, each on the day it actually arrived, counted from
			#     the split of February 1839. ES: los contingentes extranjeros, cada uno el dia
			#     en que de verdad llegaron, contados desde el corte de febrero de 1839.
			# EN: replaces the old gg_contingent.2, which handed the Defensa the same 1842 decree
			#     as two free battalions with no choice, no cost and no blanco counterpart. The
			#     dispatcher routes each decree to the government that actually issued it, and
			#     1400 days lands on 1842.12.12 exactly (1401 landed a day late).
			# ES: reemplaza al viejo gg_contingent.2, que le daba a la Defensa ese mismo decreto
			#     de 1842 como dos batallones gratis, sin eleccion, sin costo y sin contraparte
			#     blanca. El despachador le da cada decreto al gobierno que lo firmo, y 1400 dias
			#     caen justo en el 12.12.1842 (1401 caia un dia tarde).
			# EN/ES: T6 -- the abolitions are scheduled from on_game_start and gg_partition_calendar
		}
	}

	option = {
		name = guerra_grande.3.a
		default_option = yes
		# EN: you stay in Montevideo. The Cerrito secedes with the countryside and Oribe goes
		#     with it -- the character is moved, never cloned.
		# ES: te quedas en Montevideo. El Cerrito se separa con la campania y Oribe se va con
		#     el: el personaje se muda, no se clona.
		set_variable = guerra_grande_defensa_government
		# EN: grab Oribe BEFORE the country is created. create_country copies characters from the
		#     origin, so afterwards there can be two of him and the new government ends up ruled
		#     by whoever the engine picked -- which is how Rivera ended up at the Cerrito.
		# ES: hay que agarrar a Oribe ANTES de crear el pais. create_country copia personajes del
		#     origen, asi que despues puede haber dos y el gobierno nuevo queda mandado por el que
		#     el motor haya elegido, que es como Rivera terminaba en el Cerrito.
		random_scope_character = {
			limit = { has_template = URU_Manuel_Oribe }
			save_scope_as = gg_oribe
		}
		# EN: same reason, same fix, for the rest of the divisa transferred below (every_scope_character
		#     at line ~327): grab each ORIGINAL now, while ROOT is still URU and has never held a copy of
		#     itself (create_country only clones the roster onto the NEW country, never back onto the
		#     origin), so a random_scope_character here always resolves to the one real character,
		#     whether it runs before or after create_country. Used after the transfer to tell the real
		#     transferred man apart from the create_country clone that is about to join him in c:CER --
		#     see the cleanup block after the transfers.
		# ES: misma razon, mismo arreglo, para el resto de la divisa que se transfiere mas abajo
		#     (every_scope_character en la linea ~327): agarrar cada ORIGINAL ahora, mientras ROOT todavia
		#     es URU y nunca tuvo una copia de si mismo (create_country solo clona la lista hacia el pais
		#     NUEVO, nunca de vuelta al origen), asi que un random_scope_character aca siempre resuelve al
		#     unico personaje real, corra antes o despues de create_country. Se usa despues de la
		#     transferencia para distinguir al hombre de verdad del clon de create_country que esta por
		#     acompaniarlo en c:CER -- ver el bloque de limpieza despues de las transferencias.
		random_scope_character = {
			limit = { has_template = URU_Ignacio_Oribe }
			save_scope_as = gg_ignacio_oribe
		}
		random_scope_character = {
			limit = { has_template = URU_bernardo_prudencio_berro }
			save_scope_as = gg_berro
		}
		random_scope_character = {
			limit = { has_template = URU_timoteo_aparicio }
			save_scope_as = gg_aparicio
		}
		random_scope_character = {
			limit = { has_template = URU_Juan_Antonio_Lavalleja }
			save_scope_as = gg_lavalleja
		}
		create_country = {
			tag = CER
			origin = this
			state = s:STATE_BANDA_ORIENTAL.region_state:URU
			on_created = {
				set_variable = guerra_grande_cerrito_government
			}
		}
		if = {
			limit = { exists = s:STATE_SAN_JOSE.region_state:URU }
			s:STATE_SAN_JOSE.region_state:URU = { set_state_owner = c:CER }
		}
		if = {
			limit = { exists = s:STATE_MALDONADO.region_state:URU }
			s:STATE_MALDONADO.region_state:URU = { set_state_owner = c:CER }
		}
		# EN: everything that is not Montevideo goes to the Cerrito. Oribe held the whole country
		#     except the walled city -- the Defensa never governed an inch of countryside.
		# ES: todo lo que no es Montevideo va al Cerrito. Oribe controlaba el pais entero menos la
		#     plaza amurallada: la Defensa no goberno una cuadra de campania.
		if = {
			limit = { exists = s:STATE_PAYSANDU.region_state:URU }
			s:STATE_PAYSANDU.region_state:URU = { set_state_owner = c:CER }
		}
		if = {
			limit = { exists = scope:gg_oribe }
			scope:gg_oribe = { transfer_character = c:CER }
		}
		# EN: and the rest of the divisa goes with him. Ignacio Oribe commanded at Carpinteria,
		#     Berro served the Cerrito government and Aparicio fought the whole war on that side.
		#     Everyone else -- Suarez, Rivera, Flores, Cesar Diaz, Medina, Lorenzo Batlle, Anaya --
		#     stays inside the walls, which is where they were.
		# ES: y el resto de la divisa se va con el. Ignacio Oribe mando en Carpinteria, Berro
		#     sirvio al gobierno del Cerrito y Aparicio peleo toda la guerra de ese lado. Los
		#     demas -- Suarez, Rivera, Flores, Cesar Diaz, Medina, Lorenzo Batlle, Anaya -- se
		#     quedan intramuros, que es donde estuvieron.
		every_scope_character = {
			limit = {
				OR = {
					has_template = URU_Manuel_Oribe
					has_template = URU_Ignacio_Oribe
					has_template = URU_bernardo_prudencio_berro
					has_template = URU_timoteo_aparicio
					# EN: Lavalleja rode with Oribe. He was in the .b list and not this one, so on the
					#     default branch a blanco caudillo stayed inside the Defensa's government --
					#     the Aparicio bug again, on the side nobody re-checked.
					# ES: Lavalleja cabalgo con Oribe. Estaba en la lista de .b y no en esta, asi que en
					#     la rama por defecto un caudillo blanco se quedaba adentro del gobierno de la
					#     Defensa: el bug de Aparicio otra vez, del lado que nadie reviso.
					has_template = URU_Juan_Antonio_Lavalleja
				}
			}
			transfer_character = c:CER
		}
		# EN: create_country cloned Oribe, Ignacio Oribe, Berro, Aparicio and Lavalleja onto CER along
		#     with everyone else, and the transfers above then moved each ORIGINAL into CER too -- so
		#     right now CER holds two of each of these five. The enemy-copy exile block further down
		#     (has_template = URU_joaquin_suarez, etc.) does not touch this: those are a different set
		#     of templates. Both instances share has_template, so filtering by template alone cannot
		#     tell them apart anymore; the saved scopes from before create_country are what makes the
		#     surviving original identifiable, via the this != scope:x idiom vanilla uses to exclude one
		#     specific character from a limit (see events/commander_events.txt, arguing_commander_two's
		#     limit: NOT = { this = scope:arguing_commander_one }). Excluding all five saved scopes here
		#     leaves only the create_country clones matching, so this exiles exactly the five copies.
		# ES: create_country clono a Oribe, Ignacio Oribe, Berro, Aparicio y Lavalleja hacia CER junto
		#     con todos los demas, y las transferencias de arriba despues movieron cada ORIGINAL tambien
		#     a CER -- asi que ahora mismo CER tiene dos de cada uno de estos cinco. El bloque de exilio
		#     de copias enemigas de mas abajo (has_template = URU_joaquin_suarez, etc.) no toca esto: es
		#     otro conjunto de templates. Las dos instancias comparten has_template, asi que filtrar solo
		#     por template ya no alcanza para distinguirlas; los scopes guardados antes de create_country
		#     son lo que permite identificar al original que sobrevive, con el modismo this != scope:x
		#     que vanilla usa para excluir a un personaje puntual de un limit (ver
		#     events/commander_events.txt, el limit de arguing_commander_two: NOT = { this =
		#     scope:arguing_commander_one }). Excluir aca los cinco scopes guardados deja matcheando solo
		#     a los clones de create_country, asi que esto exilia justo a las cinco copias.
		c:CER ?= {
			every_scope_character = {
				limit = {
					OR = {
						has_template = URU_Manuel_Oribe
						has_template = URU_Ignacio_Oribe
						has_template = URU_bernardo_prudencio_berro
						has_template = URU_timoteo_aparicio
						has_template = URU_Juan_Antonio_Lavalleja
					}
					NOT = { this = scope:gg_oribe }
					NOT = { this = scope:gg_ignacio_oribe }
					NOT = { this = scope:gg_berro }
					NOT = { this = scope:gg_aparicio }
					NOT = { this = scope:gg_lavalleja }
				}
				exile_character = yes
			}
		}
		# EN: the Defensa was presided over by Joaquin Suarez -- an interim presidency that lasted
		#     eight years. Without naming him the engine picks a random character.
		# ES: la Defensa la presidio Joaquin Suarez, una interinidad que duro ocho anios. Si no se
		#     lo nombra, el motor elige un personaje cualquiera.
		if = {
			limit = { any_scope_character = { has_template = URU_joaquin_suarez } }
			random_scope_character = {
				limit = { has_template = URU_joaquin_suarez }
				set_character_as_ruler = yes
			}
		}
		else = {
			create_character = {
				template = URU_joaquin_suarez
				ruler = yes
			}
		}
		c:CER ?= {
			if = {
				limit = { any_scope_state = { state_region = s:STATE_SAN_JOSE } }
				set_capital = STATE_SAN_JOSE
			}
			else = {
				set_capital = STATE_BANDA_ORIENTAL
			}
			if = {
				limit = { exists = scope:gg_oribe }
				scope:gg_oribe = { set_character_as_ruler = yes }
			}
			else_if = {
				limit = { any_scope_character = { has_template = URU_Manuel_Oribe } }
				random_scope_character = {
					limit = { has_template = URU_Manuel_Oribe }
					set_character_as_ruler = yes
				}
			}
			else = {
				create_character = {
					template = URU_Manuel_Oribe
					ruler = yes
				}
			}
			# EN: the new government needs the journal too. The battle score requires BOTH sides to
			#     have it (scope:enemy_country ?= { has_journal_entry }), so without this the bar
			#     never moved and no battle ever counted.
			# ES: el gobierno nuevo tambien necesita el journal. El marcador de batallas exige que
			#     LOS DOS lo tengan (scope:enemy_country ?= { has_journal_entry }), asi que sin
			#     esto la barra no se movia y ninguna batalla contaba.
			add_contextless_journal_entry = je_guerra_grande
			create_diplomatic_play = {
				name = guerra_grande_play
				target_country = c:URU
				type = dp_guerra_grande
			}
		}
		# EN: create_country copies the whole character roster from origin, so the new government
		#     is born holding a duplicate of every man in the country -- including the enemy's.
		#     That is how Brigadier General Timoteo Aparicio, a blanco, turned up serving the
		#     Defensa. Transferring the other side's men is not enough: the transfers move the
		#     originals, and these copies stay behind. They are exiled, which is also what actually
		#     happened to a man caught on the wrong side of the lines in 1839.
		# ES: create_country copia la lista entera de personajes del origen, asi que el gobierno
		#     nuevo nace con un duplicado de cada hombre del pais, incluidos los del enemigo. Asi
		#     fue como el general Timoteo Aparicio, blanco, aparecio sirviendo a la Defensa.
		#     Transferir a los del otro bando no alcanza: las transferencias mueven los originales y
		#     estas copias quedan. Se los exilia, que es ademas lo que le pasaba de verdad a un
		#     hombre que quedaba del lado equivocado de las lineas en 1839.
		c:CER ?= {
			every_scope_character = {
				limit = {
					OR = {
					has_template = URU_joaquin_suarez
					has_template = URU_Fructuoso_Rivera
					has_template = URU_venancio_flores
					has_template = URU_cesar_diaz
					has_template = URU_anacleto_medina
					has_template = URU_lorenzo_batlle
					has_template = URU_manuel_herrera_y_obes
					has_template = URU_andres_lamas
					}
				}
				exile_character = yes
			}
		}
		# EN: THE ARMIES, SIZED FROM THE SOURCES. See SOURCES_order_of_battle.md.
		#     Two things the research changed. First, the arm mix: Uruguay was a cavalry country
		#     (Guardia Nacional 1836: 1,500 foot to 7,000 horse, as Oribe reported it himself), but
		#     the men in front of Montevideo were mostly foot and guns -- Antonio Diaz, whose father
		#     was Oribe's War Minister, lists the besieging infantry battalion by battalion at 3,430
