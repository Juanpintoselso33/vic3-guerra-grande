# =================================================================================================
# EN: THE PEACE MENU OF THE GUERRA GRANDE
#
#     The war ends in one of several ways, and the two governments do not end it the same way,
#     because they did not want the same thing. Everything offered below is something one of the
#     two actually proposed, decreed, signed or asked a foreign power for, between 1838 and 1852.
#     The sources are in ~/dev/juegos/victoria3-uruguay/SOURCES_peace.md, with URLs, and nothing
#     in this file goes past them. Where the record is disputed -- most of all on whether Oribe
#     ever meant to hand the Republic to Rosas -- the text says so instead of picking a side.
#
#     The menus are deliberately NOT symmetrical. The Defensa won by surviving until someone else
#     broke Rosas, and its documented programme is foreign: Brazilian treaties, an Anglo-French
#     protectorate, and a campaign across the river. The Cerrito would have won by walking into
#     the city, and its documented programme is domestic and legalist: the interrupted
#     presidency, the Constitution of 1830, the acts of its own government declared lawful --
#     with reincorporation into the Confederation sitting there as an offer it never answered.
#
# ES: EL MENU DE PAZ DE LA GUERRA GRANDE
#
#     La guerra termina de varias maneras, y los dos gobiernos no la terminan igual, porque no
#     querian lo mismo. Todo lo que se ofrece abajo es algo que uno de los dos propuso, decreto,
#     firmo o le pidio a una potencia extranjera entre 1838 y 1852. Las fuentes estan en
#     ~/dev/juegos/victoria3-uruguay/SOURCES_peace.md, con URLs, y nada en este archivo va mas
#     alla de ellas. Donde el registro esta en disputa -- sobre todo si Oribe penso alguna vez en
#     entregarle la Republica a Rosas -- el texto lo dice en vez de elegir bando.
#
#     Los menus NO son simetricos a proposito. La Defensa gano sobreviviendo hasta que otro
#     rompio a Rosas, y su programa documentado es de afuera: tratados con el Brasil, un
#     protectorado anglofrances y una campania del otro lado del rio. El Cerrito habria ganado
#     entrando a la ciudad, y su programa documentado es de adentro y legalista: la presidencia
#     interrumpida, la Constitucion de 1830, los actos de su propio gobierno declarados legales
#     -- con la reincorporacion a la Confederacion ahi, como una oferta que nunca contesto.
# =================================================================================================
# SOURCE / FUENTE: SOURCES_peace.md (with URLs); nothing in this file goes past it.

				exists = s:STATE_MONTEVIDEO.region_state:URU
			}
			c:URU = { set_capital = STATE_MONTEVIDEO }
		}
		# EN/ES: and the subject pact that change_tag dropped, rebuilt here where c:URU resolves.
		#     The overlord makes the pact, which is the same direction the peace options use.
		#     / y el pacto de sujecion que se llevo el change_tag, rehecho aca donde c:URU resuelve.
		#     Lo hace el señor, que es la misma direccion que usan las opciones de paz.
		if = {
			limit = {
				# EN/ES: I17 -- no subject pact without a signed peace, checked here because this
				#     relay has no trigger of its own (a scheduled event's trigger is read when it is
				#     scheduled, and this one is queued a day ahead: a trigger here would be a place
				#     to lose it). / I17 -- ningun pacto de sujecion sin paz firmada, chequeado aca
				#     porque este relevo no tiene trigger propio (el trigger de un evento agendado se
				#     lee cuando se agenda, y este se encola un dia antes: un trigger aca seria un
				#     lugar donde perderlo).
				has_global_variable = gg_peace_settled
				has_global_variable = gg_cerrito_overlord
				has_global_variable = gg_cerrito_subject_kind
				exists = c:URU
				c:URU = { is_subject = no }
			}
			global_var:gg_cerrito_overlord ?= {
				if = {
					limit = { global_var:gg_cerrito_subject_kind = 1 }
					create_diplomatic_pact = { country = c:URU type = vassal }
				}
				else_if = {
					limit = { global_var:gg_cerrito_subject_kind = 2 }
					create_diplomatic_pact = { country = c:URU type = puppet }
				}
				else_if = {
					limit = { global_var:gg_cerrito_subject_kind = 3 }
					create_diplomatic_pact = { country = c:URU type = protectorate }
				}
			}
		}
		if = {
			limit = { has_global_variable = gg_cerrito_overlord }
			remove_global_variable = gg_cerrito_overlord
		}
		if = {
			limit = { has_global_variable = gg_cerrito_subject_kind }
			remove_global_variable = gg_cerrito_subject_kind
		}
	}
}
