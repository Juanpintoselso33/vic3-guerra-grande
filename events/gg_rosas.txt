# EN/ES: relay for gg_rosas.1 (1843-02-16); the conditions live here (a scheduled event reads its trigger when
#     queued) / relevo; las condiciones viven aca
gg_rosas.101 = {
	type = country_event
	hidden = yes
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_rosas.101|" # GG_TRAZA
		if = {
			limit = {
				has_global_variable = guerra_grande_open_war
				NOT = { has_global_variable = gg_peace_settled }
				exists = c:ARG
				any_country = { has_variable = guerra_grande_cerrito_government }
			}
			c:ARG = { trigger_event = { id = gg_rosas.1 days = 0 popup = yes } }
		}
		# EN/ES: headless 23/9 -- the partition may come after 16/2/1843: retry monthly through 1843
		#     / la particion puede llegar despues: se reintenta cada mes durante 1843
		else_if = {
			limit = { game_date < 1843.12.31 NOT = { has_global_variable = gg_peace_settled } }
			trigger_event = { id = gg_rosas.101 days = 30 }
		}
	}
}
