# ES: Las potencias. T3b solo trae el retiro de una intervencion llamada por el boton; los eventos
#     con fecha llegan en T4 y T7.

# EN: five years after the button, the squadrons thin out (C10, "whichever comes first" with the
#     historical dates of T4).
# ES: cinco anios despues del boton, las escuadras se ralean.

	type = country_event
	hidden = yes

	immediate = {
		gg_patron_change = { PATRON = francia DELTA = -20 }
		gg_patron_change = { PATRON = britania DELTA = -20 }
	}

# EN: T4, C20. 1 March 1843 -- the French monthly subsidy begins (SOURCES_siege.md).
# ES: T4, C20. 1 de marzo de 1843 -- empieza el subsidio mensual frances.
gg_powers.2 = {
	type = country_event
	hidden = yes
	# EN/ES: review -- no split requirement: a delayed partition must not lose it
	#     / sin exigir la particion: una particion atrasada no puede perderlo
	immediate = {
		# EN/ES: balance, first playtest 2026-09-17: the sea was historically open from the start of
		#     the siege (the French station, Purvis). 20 here plus Purvis's 15 opens it in 1843.

# EN: T4, C20/FE-8. 18 September 1845 -- the Anglo-French blockade of the Confederation. If
#     nobody called the squadrons earlier, they come now, and the Defensa decides how to receive
#     them (guerra_grande.9).
# ES: T4, C20/FE-8. 18 de septiembre de 1845 -- el bloqueo anglo-frances. Si nadie llamo antes a
#     las escuadras, llegan ahora, y la Defensa decide como recibirlas.
gg_powers.3 = {
	type = country_event
	hidden = yes
	trigger = {
		has_global_variable = guerra_grande_split_done
		NOT = { has_global_variable = gg_intervention_called }
	}
	immediate = {
		set_global_variable = gg_intervention_called
		gg_patron_change = { PATRON = francia DELTA = 30 }
		gg_patron_change = { PATRON = britania DELTA = 30 }
		c:URU ?= {
			trigger_event = { id = guerra_grande.9 popup = yes }
		}
	}
}

# EN: T4, C20/FE-2. June 1848 -- the February revolution reaches the Plata. A signed convention
#     (gg_loan.1.a) softens it; without one, France falls hard, and T7's departures follow.
# ES: T4, C20/FE-2. Junio de 1848 -- la revolucion de febrero llega al Plata. Un convenio firmado
#     la amortigua; sin el, Francia cae fuerte, y en T7 siguen las salidas.
gg_powers.4 = {
	type = country_event
	hidden = yes
	trigger = { has_global_variable = guerra_grande_split_done }

		}
		else = {
			gg_patron_change = { PATRON = francia DELTA = -20 }
		}
	}
}

		name = gg_powers.10.b

# F3-4 - the Gore-Gros mission: the presidency for Oribe (March-May 1848)

	type = country_event
	placement = ROOT

	title = gg_powers.16.t
	desc = gg_powers.16.d
	flavor = gg_powers.16.f

	event_image = {
		video = "gg_fl_oribe"
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	trigger = {
		c:CER ?= this
		has_global_variable = guerra_grande_open_war
		NOT = { has_global_variable = gg_peace_settled }
		NOT = { has_global_variable = gg_peace_offered }
		NOT = { has_global_variable = gg_reunification_pledged }
	}

	immediate = {
		# EN/ES: FS-2 -- if the Defensa blamed Rosas for Varela, the Cerrito listens less / si la Defensa culpo a Rosas por Varela, el Cerrito escucha menos
		if = {
			limit = { has_global_variable = gg_varela_blamed }
			gg_cerrito_balance_change = { DELTA = -5 }
		}
	}

	option = {
		name = gg_powers.16.a
		custom_tooltip = gg_powers.16.a_tt
		# EN: accept the presidency under a European guarantee and leave Rosas (final F13). With the
		#     caudillos in charge, the camp resents it.
		# ES: aceptar la presidencia con garantia europea y dejar a Rosas (F13). Con los caudillos al
		#     frente, el campamento lo resiente.
		if = {
			limit = { global_var:gg_cerrito_balance < 0 }
			add_radicals = { value = medium_radicals }
		}
		set_global_variable = gg_peace_offered
		trigger_event = { id = gg_peace.7 days = 3 popup = yes }
		ai_chance = { base = 15 }
	}

	option = {
		name = gg_powers.16.b
		default_option = yes
		# EN: historical (18 May 1848): refuse after Rosas's warning. Rosas rises, the powers leave sooner.
		# ES: lo historico: rechazar tras la advertencia de Rosas. Rosas sube, las potencias se van antes.
		gg_patron_change = { PATRON = rosas DELTA = 10 }
		gg_patron_change = { PATRON = francia DELTA = -10 }
		gg_patron_change = { PATRON = britania DELTA = -10 }
		ai_chance = { base = 65 }
	}

	option = {
		name = gg_powers.16.c
		trigger = { NOT = { has_global_variable = gg_gore_gros_stalled } }
		# EN: stall. The powers stay a little longer, Rosas grows suspicious, and the offer returns once.
		# ES: dar largas. Las potencias se quedan un poco mas, Rosas desconfia, y la oferta vuelve una vez.
		set_global_variable = gg_gore_gros_stalled
		gg_patron_change = { PATRON = rosas DELTA = -5 }
		trigger_event = { id = gg_powers.16 days = 90 popup = yes }
		ai_chance = { base = 20 }
	}

}
