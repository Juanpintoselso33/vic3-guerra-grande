#     historical dates of T4).

	hidden = yes

	immediate = {

# EN: T4, C20. 1 March 1843 -- the French monthly subsidy begins (SOURCES_siege.md).

	type = country_event
	hidden = yes
	# EN/ES: review -- no split requirement: a delayed partition must not lose it
	#     / sin exigir la particion: una particion atrasada no puede perderlo
	immediate = {

# EN: T4, C20/FE-8. 18 September 1845 -- the Anglo-French blockade of the Confederation. If
#     nobody called the squadrons earlier, they come now, and the Defensa decides how to receive
#     them (guerra_grande.9).
# ES: T4, C20/FE-8. 18 de septiembre de 1845 -- el bloqueo anglo-frances. Si nadie llamo antes a
#     las escuadras, llegan ahora, y la Defensa decide como recibirlas.
gg_powers.3 = {
	type = country_event
	hidden = yes
	# EN: C10 (22/9/2026) -- a link of the intervention chain: it waits for Rosas to return the
	#     passports and comes two months after (historically 31 July -> 18 September 1845). The
	#     Defensa's button no longer brings the squadrons by itself: it shortens the powers' patience.
	# ES: C10 -- un eslabon de la cadena de la intervencion: espera a que Rosas devuelva los
	#     pasaportes y llega dos meses despues. El boton de la Defensa ya no trae las escuadras por su
	#     cuenta: acorta la paciencia de las potencias.
	trigger = {
		has_global_variable = guerra_grande_split_done
		gg_plata_intervention_at_any = { MONTHS = 2 }
	}
	immediate = {
		gg_patron_change = { PATRON = francia DELTA = 30 }
		# EN/ES: 22/9/2026 -- not when France intervenes alone / no cuando Francia interviene sola
		if = {
			limit = { NOT = { has_global_variable = gg_plata_france_alone } }
			gg_patron_change = { PATRON = britania DELTA = 30 }
		}
		# EN: the blockade shuts Rosas's ports, so the Cerrito's trade walks north (SOURCES_siege.md):
		#     moved here from the Defensa's button, because this is the blockade that caused it.
		# ES: el bloqueo cierra los puertos de Rosas y el comercio del Cerrito se va al norte: pasado
		#     aca desde el boton de la Defensa, porque es este bloqueo el que lo causo.
		if = {
			limit = { exists = s:STATE_BANDA_ORIENTAL.region_state:CER }
			s:STATE_BANDA_ORIENTAL.region_state:CER = {
				add_modifier = { name = gg_siege_the_land_frontier_stays_open years = 5 }
			}
		}
		if = {
			limit = { exists = s:STATE_PAYSANDU.region_state:CER }
			s:STATE_PAYSANDU.region_state:CER = {
				add_modifier = { name = gg_siege_the_land_frontier_stays_open years = 5 }
			}
		}
		# EN/ES: the Anglo-French mediation event, not when France intervenes alone
		#     / el evento de la mediacion anglo-francesa, no cuando Francia interviene sola
		if = {
			limit = { NOT = { has_global_variable = gg_plata_france_alone } }
			c:URU ?= {
				trigger_event = { id = guerra_grande.9 popup = yes }
			}
		}
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = gg_cal_done_31 } }
				set_global_variable = gg_cal_done_31
			}
		}
	}
}

# F2-1 - Purvis does not recognise the blockade (April 1843)
gg_powers.10 = {
	type = country_event
	placement = ROOT

	title = gg_powers.10.t
	desc = gg_powers.10.d
	flavor = gg_powers.10.f

	event_image = {
		video = "gg_f2_brown"
	}

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	# EN/ES: playtest 2026-09-17 -- 3 days is vanilla's default and at speed 4-5 it is one second:
	#     the game answered for the player. Every event of this mod has consequences, so they wait.
	#     / 3 dias es el default de vanilla y a velocidad 4-5 es un segundo: el juego elegia solo.
	duration = 1825

	trigger = {
		c:CER ?= this
		has_global_variable = guerra_grande_open_war
	}

	immediate = {
		# EN/ES: Purvis refuses to recognise the blockade whatever the Cerrito answers (historical).
		#     / Purvis no reconoce el bloqueo conteste lo que conteste el Cerrito.
		gg_patron_change = { PATRON = britania DELTA = 20 }
		# EN: C10 -- while Purvis keeps supplying the city, Rosas's blockade only stops the Defensa's
		#     own ships; London disavows him in June 1843 and the blockade is reimposed (gg_plata.3,
		#     70 days later). SOURCES_powers.md section 1.
		# ES: C10 -- mientras Purvis abastece la ciudad, el bloqueo de Rosas solo frena a los barcos
		#     de la Defensa; Londres lo desautoriza en junio de 1843 y el bloqueo se reimpone.
		if = {
			limit = { global_var:gg_plata_rosas_stance = 1 }
			gg_plata_rosas_neutrals_pass = yes
			c:ARG ?= { trigger_event = { id = gg_plata.3 days = 70 } }
		}
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = gg_cal_done_3 } }
				set_global_variable = gg_cal_done_3
			}
		}
	}

	option = {
		name = gg_powers.10.a
		custom_tooltip = gg_powers.10.a_tt
		default_option = yes
		# EN: keep the blockade: Britain takes offence, and the ships are at risk.
		# ES: sostener el bloqueo: Gran Bretania se ofende, y los barcos corren riesgo.
		gg_patron_change = { PATRON = britania DELTA = 10 }
		random_list = {
			60 = {}
			40 = { add_modifier = { name = gg_powers_ships_lost years = 5 is_decaying = yes } }
		}
		ai_chance = { base = 40 }
	}

	option = {
		name = gg_powers.10.b
		# EN: historical: open trade with the city to drain its specie; the sea stays open.
		# ES: lo historico: comercio abierto con la ciudad para sacarle el metalico; el mar queda abierto.
		c:URU ?= {
			add_modifier = { name = gg_powers_specie_drain years = 3 is_decaying = yes }
		}
		ai_chance = { base = 60 }
	}

}

# F3-1 - the allies seize Brown's squadron (July-August 1845)

	flavor = gg_powers.11.f

	event_image = {
		video = "gg_fl_barry"
	}

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	duration = 1825

	trigger = {
		c:CER ?= this
		has_global_variable = guerra_grande_open_war
		# EN/ES: C10 -- the first link of the intervention: the squadron is detained the day after the
		#     ultimatum / el primer eslabon: la escuadra queda detenida al dia siguiente del ultimatum
		gg_plata_intervention_at = { MONTHS = 0 }
	}

	option = {
		name = gg_powers.11.a
		# EN/ES: playtest 2026-09-19 -- the war bar has to move during the war: the squadron stays off the port /
		#     la barra tiene que moverse durante la guerra: la escuadra queda frente al puerto
		gg_score_blanco = { AMOUNT = 1 }
		# EN: lift the blockade and save the squadron: Rosas opens the river, the fort and ships stay.
		#     Historically Rosas ordered Brown to lift the blockade on 22 July 1845.
		# ES: levantar el bloqueo y salvar la escuadra: Rosas abre el rio, quedan el fuerte y los barcos.
		#     Historicamente Rosas ordeno a Brown levantar el bloqueo el 22 de julio de 1845.
		gg_plata_rosas_open = yes
		ai_chance = { base = 45 }
	}

	option = {
		name = gg_powers.11.b
		# EN/ES: playtest 2026-09-19 -- the war bar has to move during the war: with no squadron, the river is the city's again /
		#     la barra tiene que moverse durante la guerra: sin escuadra, el rio vuelve a ser de la ciudad
		gg_score_colorado = { AMOUNT = 1 }
		custom_tooltip = gg_powers.11.b_tt2
		default_option = yes
		# EN: historical: stand firm. The squadron is taken and the blockade dies until the powers leave,
		#     but Rosas and American opinion rise.
		# ES: lo historico: resistir. Se pierde la escuadra y el bloqueo muere hasta que se vayan las
		#     potencias, pero suben Rosas y la opinion americana.
		set_global_variable = gg_brown_fleet_lost
		# EN/ES: C10, A -- without the squadron the blockade dies: the Buenos Aires bank cannot restrict
		#     anything until Arana-Southern returns the ships / sin la escuadra muere el bloqueo
		gg_plata_fort_remove = yes
		# EN/ES: FN-1 -- Purvis took Brown's ships away, so the squadron leaves the map as well
		#     / Purvis se llevo los barcos de Brown, asi que la escuadra tambien se va del mapa
		c:CER ?= {
			# EN: Codex review 2026-09-19 -- disband_military_formation is not a script effect (the log
			#     says "Unknown effect"), so the ships stayed on the map. They are handed over instead,
			#     which is what happened: Purvis and the French took Brown's squadron.
			# ES: disband_military_formation no es un efecto de script (el log dice "Unknown effect"),
			#     asi que los barcos se quedaban en el mapa. Ahora se entregan, que es lo que paso:
			#     Purvis y los franceses se llevaron la escuadra de Brown.
			if = {
				limit = { has_variable = gg_brown_squadron_ref }
				if = {
					limit = { exists = c:GBR }
					var:gg_brown_squadron_ref ?= {
						every_scope_ship = { set_ship_owner_multiple = c:GBR }
					}
					# EN/ES: Codex review 22/9 -- on the receiving country / en el pais que recibe
					c:GBR = { clear_ownership_transfer_fleet = yes }
				}
				remove_variable = gg_brown_squadron_ref
				remove_variable = gg_brown_squadron_raised
			}
		}
		gg_patron_change = { PATRON = rosas DELTA = 10 }
		add_modifier = { name = gg_powers_american_cause years = 10 is_decaying = yes }
		ai_chance = { base = 55 }
	}


	immediate = {
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = gg_cal_done_10 } }
				set_global_variable = gg_cal_done_10
			}
		}
	}
}

# F3-2a - the Hood bases reach the Cerrito (July 1846)
gg_powers.12 = {
	type = country_event
	placement = ROOT

	title = gg_powers.12.t
	desc = gg_powers.12.d
	flavor = gg_powers.12.f

	# EN/ES: reused until T9 (SOURCES_flujo_images.md) / reusado hasta T9
	event_image = {
		video = "gg_f2_oribe_gallino"
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 1825

	trigger = {
		c:CER ?= this
		has_global_variable = guerra_grande_open_war
		NOT = { has_global_variable = gg_peace_settled }
		NOT = { has_global_variable = gg_peace_offered }
		# EN: With the Argentine side loaded, Buenos Aires owns the answer and hands it here. Without
		#     that side, this is exactly the old gate and the oriental game is unchanged.
		# ES: Con el lado argentino, Buenos Aires decide y entrega la respuesta aca. Sin ese lado, esta
		#     es exactamente la guarda anterior y la partida oriental no cambia.
		OR = {
			AND = {
				NOT = { exists = s:STATE_ENTRE_RIOS }
				NOT = { has_global_variable = gg_hood_done }
			}
			AND = {
				has_global_variable = gg_hood_argentina_pending
				NOT = { has_global_variable = gg_hood_cerrito_received }
			}
			# EN: AND THE NET. With the Argentine side loaded, Buenos Aires owns the answer -- so if
			#     Buenos Aires is not there to give it, the Hood bases would vanish from the oriental
			#     game without a line in any log. This branch reopens the old gate when there is no
			#     Argentina left to answer. It can only ADD a way for the event to fire, never remove
			#     one.
			# ES: Y LA RED. Con el lado argentino cargado la respuesta es de Buenos Aires -- asi que
			#     si Buenos Aires no esta para darla, las bases Hood desaparecerian de la partida
			#     oriental sin una linea en ningun log. Esta rama reabre la guarda vieja cuando no
			#     queda Argentina que conteste. Solo puede AGREGAR una via de disparo, nunca sacar
			#     una.
			AND = {
				NOT = { exists = c:ARG }
				NOT = { has_global_variable = gg_hood_done }
			}
		}
		OR = {
			# EN/ES: C10 -- a year into the intervention (historically July 1846) / al anio de la intervencion
			gg_plata_intervention_at = { MONTHS = 12 }
			has_global_variable = gg_powers_mediators
		}
	}

	option = {
		name = gg_powers.12.a
		custom_tooltip = gg_powers.12.a_tt
		trigger = {
			OR = {
				NOT = { has_global_variable = gg_hood_argentina_pending }
				global_var:gg_hood_argentina_answer = 1
			}
		}
		default_option = yes
		# EN: historical: accept with amendments (11 August 1846). The Defensa decides next.
		# ES: lo historico: aceptar con retoques (11 de agosto de 1846). Despues decide la Defensa.
		set_global_variable = gg_hood_done
		if = {
			limit = { has_global_variable = gg_hood_argentina_pending }
			set_global_variable = gg_hood_cerrito_received
		}
		# EN/ES: the table is ours for those 29 days / la mesa es nuestra esos 29 dias
		gg_peace_chain_hold = yes
		c:URU ?= { trigger_event = { id = gg_powers.13 days = 29 popup = yes } }
		ai_chance = { base = 70 }
	}

	option = {
		name = gg_powers.12.b
		custom_tooltip = gg_powers.12.b_tt2
		trigger = {
			OR = {
				NOT = { has_global_variable = gg_hood_argentina_pending }
				global_var:gg_hood_argentina_answer = 2
			}
		}
		# EN: refuse: it ends here, and the powers harden against the Cerrito.
		# ES: rechazar: se corta aca, y las potencias se endurecen con el Cerrito.
		set_global_variable = gg_hood_done
		if = {
			limit = { has_global_variable = gg_hood_argentina_pending }
			set_global_variable = gg_hood_cerrito_received
		}
		add_modifier = { name = gg_powers_hardened years = 5 is_decaying = yes }
		ai_chance = { base = 30 }
	}


	immediate = {
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = gg_cal_done_12 } }
				set_global_variable = gg_cal_done_12
			}
		}
	}
}
