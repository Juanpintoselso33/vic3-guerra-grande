# EN: CASEROS, THE THREE WAYS OUT (22/9/2026, decided with Juan). gg_argentina.41 offers Rosas the three
#     roads historical criticism lists for 3 February 1852; each now changes how the war of the
#     Federal Pact can end (common/scripted_effects/gg_pact_resolution.txt):
#       (a) give battle -- if Rosas loses, he falls at once and goes into exile;
#       (b) manoeuvre   -- each month Rosas holds wears Entre Rios down; at six, a negotiated way out;
#       (c) hold the city -- a siege of Buenos Aires of six months at least before any ending.
#     SOURCE / FUENTE: the battle of Caseros, 3/2/1852, and Rosas's embarkation for England on HMS
#     Centaur (SOURCES_peace.md, Caseros section). DESIGN: (b) and (c) are counterfactuals; their
#     six-month thresholds and the -5 monthly war support are game-scale choices, not history.
# ES: CASEROS, LAS TRES SALIDAS. Cada una cambia ahora como puede terminar la guerra del Pacto.
#     DISENO: (b) y (c) son contrafacticos; los seis meses y el -5 mensual son escala de juego.
# -------------------------------------------------------------------------------------------------

# EN: The monthly clock of (b) and (c), on Argentina. Hidden; reschedules itself while it matters.
# ES: El reloj mensual de (b) y (c), sobre la Argentina. Oculto; se reagenda mientras importe.
gg_caseros.1 = {
	type = country_event
	hidden = yes
	trigger = {
		c:ARG ?= this
	}
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros.1|" # GG_TRAZA
		change_global_variable = { name = gg_caseros_months add = 1 }
		# (b) manoeuvre / maniobrar
		if = {
			limit = {
				global_var:gg_caseros_choice ?= 2
				NOT = { has_global_variable = gg_pact_settled }
				exists = c:ENR
				has_war_with = c:ENR
			}
			every_scope_war = {
				limit = {
					is_war_participant = c:ENR
					is_war_participant = c:ARG
				}
				add_war_war_support = { target = c:ENR value = -5 }
			}
			if = {
				limit = { global_var:gg_caseros_months >= 6 }
				set_global_variable = { name = gg_pact_war_white_peace value = 1 }
				# EN/ES: only the Buenos Aires - Entre Rios play; another Pact war goes on
				#     / solo la jugada Buenos Aires - Entre Rios; otra guerra del Pacto sigue
				every_diplomatic_play = {
					limit = {
						is_diplomatic_play_type = dp_gg_pacto_federal
						initiator_is = c:ARG
						target_is = c:ENR
					}
					end_play = yes
				}
				gg_pact_resolve = yes
			}
			else = {
				trigger_event = { id = gg_caseros.1 days = 30 }
			}
		}
		# EN/ES: (b) with Entre Rios gone and nothing written yet: the ending is written now, never lost
		#     / (b) sin Entre Rios y sin final escrito: se escribe ahora, nunca se pierde
		else_if = {
			limit = {
				global_var:gg_caseros_choice ?= 2
				NOT = { has_global_variable = gg_pact_settled }
				NOT = { exists = c:ENR }
			}
			gg_pact_resolve = yes
		}
		# (c) hold the city / sostener la plaza
		else_if = {
			limit = {
				global_var:gg_caseros_choice ?= 3
				has_global_variable = gg_caseros_siege
			}
			if = {
				limit = { global_var:gg_caseros_months >= 6 }
				remove_global_variable = gg_caseros_siege
				if = {
					limit = { is_player = yes }
					trigger_event = { id = gg_caseros.3 popup = yes }
				}
				gg_caseros_notify_urquiza = { EVENT = gg_caseros.3 }
				# EN/ES: a pending ending is written now; so is one whose war already ended without
				#     leaving a mark (Entre Rios gone) / el final pendiente se escribe ahora, y tambien el
				#     de una guerra que termino sin dejar marca (Entre Rios desaparecida)
				if = {
					limit = {
						OR = {
							has_global_variable = gg_pact_resolve_pending
							NOT = { exists = c:ENR }
							c:ENR ?= { NOT = { has_war_with = c:ARG } }
						}
					}
					if = {
						limit = { has_global_variable = gg_pact_resolve_pending }
						remove_global_variable = gg_pact_resolve_pending
					}
					gg_pact_resolve = yes
				}
			}
			else = {
				trigger_event = { id = gg_caseros.1 days = 30 }
			}
		}
	}
}

	}
}

# EN: 24/9/2026 (AUD-BA-STALEMATE-ENR) -- the day after every «two governments» reading (since 24/9, Juan point 4, it
#     also runs the partition again, gg_pact_sides.txt); for a Pact war Rosas won: the war's
#     protectorate over Entre Rios is undone again, in case the engine applied the war goal after gg_pact_resolve read
#     the war (common/scripted_effects/gg_pact_resolution.txt, gg_pact_stalemate_frees_enr). Hidden, on Argentina.
# ES: el dia despues de leer en tablas una guerra del Pacto que gano Rosas: se deshace otra vez el protectorado sobre
#     Entre Rios, por si el motor aplico el objetivo despues de la lectura. Oculto, sobre la Argentina.
gg_caseros.7 = {
	type = country_event
	hidden = yes
	trigger = {
		c:ARG ?= this
	}
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_caseros.7|" # GG_TRAZA
		if = {
			limit = { has_global_variable = gg_pact_settlement_stalemate }
			if = {
				limit = { has_global_variable = gg_pact_war_enforced_rosas }
				gg_pact_stalemate_frees_enr = yes
			}
			# EN/ES: 24/9 (Juan, point 4) -- and the partition again, now that Entre Rios is free: idempotent
			#     / y otra vez la particion, ya libre Entre Rios: idempotente
			gg_pact_stalemate_partition = yes
		}
		# EN/ES: 24/9 (Codex, round 2) -- Rosas won: the protectorates the same-tick reading could not make (gg_pact_resolve,
		#     already settled, only reconciles) / gano Rosas: los protectorados que la lectura no pudo hacer
		if = {
			limit = { has_global_variable = gg_pact_settlement_rosas }
			gg_pact_resolve = yes
		}
		if = {
			limit = { has_global_variable = gg_pact_war_enforced_rosas }
			remove_global_variable = gg_pact_war_enforced_rosas
		}
	}
}
