# EN: 23/9/2026 (Juan) -- the Oriental side of two gaps of docs/grafo/historia-por-pais-2026-09-23.md:
#     C4, the Defensa never heard how its war ended (the Convention of 29 May 1851 and Caxias's crossing,
#     which with la Argentina loaded is switched off in gg_contingent.8), and G1, the Cerrito had almost
#     nothing from late 1846 to 1850. Each event here tells, in its own side's voice, something that the
#     other side already gets, and only if it happened in this game: the trigger reads the mark the
#     source event wrote. Delivery goes through gg_notice_relay (common/scripted_effects/gg_notices.txt),
#     which stops at the end of the war (gg_cal_war_over).
# ES: el lado oriental de dos huecos del informe de historia por pais: C4, la Defensa no se enteraba de
#     como terminaba su guerra, y G1, el Cerrito casi no tenia nada de fines de 1846 a 1850. Cada evento
#     cuenta, con la voz de su lado, algo que el otro lado ya recibe, y solo si paso en esta partida: el
#     trigger lee la marca que escribio el evento fuente. Lo entrega gg_notice_relay, que se corta al
#     terminar la guerra.
# VIDEO: all images here are provisional vanilla videos; the list for the video lane is in
#     AUDITORIAS.md (AUD-NOTICES-VIDEO) / todas las imagenes son videos vanilla provisorios.

# 29/5/1851 -- the Convention of Montevideo, from the Defensa's side.
# SOURCE / FUENTE: text of the convention (Brasil, Republica Oriental, Entre Rios, Montevideo 29/5/1851),
#     https://www.dipublico.org/122098/ -- signatories, arts. I, IV, V, VII, XIV, XIX, XXI, XXIV quoted
#     literally; SOURCES_powers.md section 8.
# EN: signed = the mark of the signature: Entre Rios's gg_argentina.13.a (gg_argentina_sign_montevideo_convention
#     writes gg_montevideo_free_navigation, and nothing else does). Only with la Argentina: without it the
#     Pronunciamiento that enables the convention never comes (its relay needs s:STATE_ENTRE_RIOS; Codex 23/9),
#     and gg_contingent.8 carries the alliance and the crossing in one decision.
# ES: firmada = la marca de la firma (gg_argentina.13.a). Solo con la Argentina: sin ella no hay Pronunciamiento
#     que habilite la convencion, y gg_contingent.8 lleva la alianza y el cruce en una sola decision.
gg_notice.1 = {
	type = country_event
	placement = ROOT
	title = gg_notice.1.t
	desc = gg_notice.1.d
	flavor = gg_notice.1.f
	# EN/ES: provisional vanilla video (video lane) / video vanilla provisorio
	event_image = { video = "gg_ntc_suarez" }
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 12
	is_popup = yes

	trigger = {
		has_variable = guerra_grande_defensa_government
		has_global_variable = guerra_grande_open_war
		any_country = { has_variable = guerra_grande_cerrito_government }
		has_global_variable = gg_montevideo_free_navigation
	}

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_notice.1|" # GG_TRAZA
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = gg_notice_done_1 } }
				set_global_variable = gg_notice_done_1
			}
		}
	}

	# EN: art. IV -- the Empire's army marches to the frontier. The city sees the end; Brazil weighs more.
	# ES: art. IV -- el ejercito del Imperio marcha a la frontera. La ciudad ve el final; el Brasil pesa mas.
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_notice.1.a|" # GG_TRAZA
		name = gg_notice.1.a
		default_option = yes
		custom_tooltip = gg_notice.1.a_tt
		gg_endurance_change = { SIDE = defensa VALUE = 10 }
		gg_patron_change = { PATRON = brasil DELTA = 10 }
		ai_chance = { base = 70 }
	}

	# EN: art. VII -- the Oriental emigres come back under the general in chief, "por cuenta de la
	#     Republica": the treasury pays, and the men count for the Defensa at the table.
	# ES: art. VII -- vuelven los emigrados, por cuenta de la Republica: paga la caja, y cuentan en la mesa.
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_notice.1.b|" # GG_TRAZA
		name = gg_notice.1.b
		custom_tooltip = gg_notice.1.b_tt
		add_treasury = -20000
		gg_endurance_change = { SIDE = defensa VALUE = 5 }
		gg_score_colorado = { AMOUNT = 1 }
		ai_chance = { base = 30 }
	}
}

# 9/1851 -- the Empire's army crosses, from the Defensa's side. Only with la Argentina loaded: without it
#     gg_contingent.8 is this crossing and its treaties.
# SOURCE / FUENTE: pt.wikipedia, Campanha do Uruguai (16,200 men in four divisions, by the Santana pass;
#     the figures of gg_contingent.8); convention of 29/5/1851, arts. IV, V, VI, https://www.dipublico.org/122098/
# EN: crossing = the convention signed (gg_montevideo_free_navigation) and the Empire in the game; the
#     event itself is the crossing, and its first option lists the Allied Army in the siege journal, as
#     gg_contingent.8.a does.
# ES: el cruce = la convencion firmada y el Imperio en la partida; el evento es el cruce, y su primera
#     opcion lista al Ejercito Aliado en el journal del sitio, como gg_contingent.8.a.
gg_notice.2 = {
	type = country_event
	placement = ROOT
	title = gg_notice.2.t
	desc = gg_notice.2.d
	flavor = gg_notice.2.f
	# EN/ES: provisional vanilla video (video lane) / video vanilla provisorio
	event_image = { video = "gg_ntc_caxias_orden" }
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 12
	is_popup = yes

	trigger = {
		has_variable = guerra_grande_defensa_government
		has_global_variable = guerra_grande_open_war
		any_country = { has_variable = guerra_grande_cerrito_government }
		exists = s:STATE_ENTRE_RIOS
		exists = c:BRZ
		has_global_variable = gg_montevideo_free_navigation
		NOT = { has_variable = gg_allied_army_arrived }
	}

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_notice.2|" # GG_TRAZA
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = gg_notice_done_2 } }
				set_global_variable = gg_notice_done_2
			}
		}
	}

	# EN: the Empire enters with everything it brought (art. VI, its own general in chief).
	# ES: el Imperio entra con todo lo que trajo (art. VI, con su propio general en jefe).
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_notice.2.a|" # GG_TRAZA
		name = gg_notice.2.a
		default_option = yes
		custom_tooltip = gg_notice.2.a_tt
		set_variable = gg_allied_army_arrived
		gg_endurance_change = { SIDE = cerrito VALUE = -15 }
		gg_patron_change = { PATRON = brasil DELTA = 10 }
		ai_chance = { base = 80 }
	}

	# EN: the Empire keeps to the twenty leagues of art. V; the campaign stays with Orientals and Entre Rios.
	# ES: el Imperio se atiene a las veinte leguas del art. V; la campania queda en manos orientales y entrerrianas.
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_notice.2.b|" # GG_TRAZA
		name = gg_notice.2.b
		custom_tooltip = gg_notice.2.b_tt
		gg_endurance_change = { SIDE = cerrito VALUE = -5 }
		gg_patron_change = { PATRON = brasil DELTA = -5 }
		ai_chance = { base = 20 }
	}
}
