# EN: playtest 2026-09-19 (Juan's request) -- the opening used to be three popups in three days,
#     each one an OK button that opened a journal, and the first real decision only came on the
#     fifth day. One event opens both journals now. The siege one still fires for both governments
#     the day after the split.
# ES: la apertura eran tres popups en tres dias, cada uno un boton de OK que abria un journal, y la
#     primera decision de verdad recien llegaba el quinto dia. Ahora un solo evento abre los dos
#     journals. El del sitio sigue saliendo para los dos gobiernos al dia siguiente de la particion.

# EN/ES: day 1 -- the opening: it opens both journals / la apertura: abre los dos journals
gg_intro.1 = {
	type = country_event
	placement = ROOT

	title = gg_intro.1.t
	desc = gg_intro.1.d
	flavor = gg_intro.1.f

	event_image = {
		video = "gg_in_montevideo_plan"
	}

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	# EN: 22/9/2026 -- duration is in MONTHS (save: utilitarian.4 duration = 3 -> expires 3 months on).
	#     The 1825 of 17/9 (152 years) left minimized events pending forever, and vanilla's 3 answers for
	#     the player within seconds at speed 5 (Juan, 17/9): 12 reopens it at 11 months, then resolves.
	# ES: la duracion va en MESES (en el save, 3 = vence a los 3 meses). El 1825 del 17/9 (152 anos)
	#     dejaba colgados para siempre los eventos minimizados, y el 3 de vanilla contesta por el jugador
	#     en segundos a velocidad 5: 12 lo reabre a los 11 meses y recien ahi resuelve.
	duration = 12
	is_popup = yes

	trigger = {
		NOT = { has_journal_entry = je_guerra_grande }
	}

	option = {
		name = gg_intro.1.a
		default_option = yes
		# EN/ES: 22/9/2026 (Juan) -- the Guerra Grande opens at Carpinteria (guerra_grande.1), not on day 1;
		#     the reckoning already runs on globals / la Guerra Grande abre en Carpinteria, no el dia 1
		gg_international_ensure = yes
		ai_chance = { base = 100 }
	}
}

# EN/ES: hidden -- the partition calendar, the day after the split / el calendario de la particion
gg_intro.5 = {
	type = country_event
	hidden = yes

	# EN: playtest 2026-09-17, SECOND round. The event never fired, in either branch of the split:
	#     the engine checks a scheduled event's trigger WHEN IT IS SCHEDULED, and at that moment
	#     guerra_grande_split_done is not set yet (the option sets it at its end). So the trigger
	#     asked for a mark that could not exist, and the event was dropped -- silently, which is why
	#     the Cerrito's calendar and its war went missing. The guard lives in the immediate now.
	# ES: playtest, SEGUNDA vuelta. El evento no salia en ninguna de las dos ramas del corte: el
	#     motor chequea el trigger de un evento agendado CUANDO SE AGENDA, y ahi todavia no esta
	#     guerra_grande_split_done (la opcion la prende al final). El trigger pedia una marca que no
	#     podia existir y el evento se descartaba, en silencio. La guarda ahora va en el immediate.
	trigger = {
		NOT = { has_global_variable = gg_partition_calendar_done }
	}

	immediate = {
		set_global_variable = gg_partition_calendar_done
		gg_partition_calendar = yes
		# EN/ES: playtest 2026-09-17 -- the war of option guerra_grande.3.b, opened once CER resolves
		#     / la guerra de la opcion .3.b, que se abre cuando CER ya se resuelve
		if = {
			limit = { has_global_variable = gg_play_pending }
			remove_global_variable = gg_play_pending
			c:CER ?= {
				create_diplomatic_play = {
					name = guerra_grande_play
					target_country = c:URU
					type = dp_guerra_grande
					# EN: 23/9/2026 (headless runs 15-16) -- the play opens AS a war. Without it the Defensa's AI
					#     backed down within weeks and the Cerrito's annex_country goal was enforced with no war at all
					#     (the Defensa gone in April 1843, not one battle). Vanilla does the same for the wars already
					#     burning in 1836 (common/history/diplomatic_plays/00_south_american_wars.txt).
					# ES: la jugada abre COMO guerra. Sin esto la IA de la Defensa cedia en semanas y se aplicaba la
					#     anexion del Cerrito sin guerra (la Defensa desaparecia en abril de 1843, sin una batalla).
					war = yes
				}
			}
		}
	}
}
