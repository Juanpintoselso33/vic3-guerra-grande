namespace = gg_postwar

# -------------------------------------------------------------------------------------------------
# EN: T8 -- THE POSTWAR (C16, FX-2, FX-3). How the peace was made decides what comes after it:
#     a punished side keeps fighting in other ways; a peace without victors opens elections and
#     the fusion policy. The router reads the marks the peace menus left.
# ES: T8 -- LA POSGUERRA (C16, FX-2, FX-3). Como se hizo la paz decide lo que viene despues: un
#     bando castigado sigue peleando por otros medios; una paz sin vencidos abre las elecciones y
#     la politica de fusion. El router lee las marcas que dejaron los menus de paz.
# -------------------------------------------------------------------------------------------------

gg_postwar.1 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = { has_global_variable = gg_peace_punished }
			trigger_event = { id = gg_postwar.2 days = 1 popup = yes }
			# EN/ES: FX-2 -- the caudillos come back early / los caudillos vuelven antes
			trigger_event = { id = gg_postwar.5 days = 730 popup = yes }
		}
		else_if = {
			limit = { has_global_variable = gg_peace_no_victors }
			trigger_event = { id = gg_postwar.3 days = 1 popup = yes }
			trigger_event = { id = gg_postwar.4 days = 150 popup = yes }
			# EN/ES: FX-3 -- contained: later and softer / contenidos: mas tarde y mas blandos
			set_global_variable = gg_postwar_caudillos_contained
			trigger_event = { id = gg_postwar.5 days = 1825 popup = yes }
			trigger_event = { id = gg_postwar.6 days = 1460 popup = yes }
		}
	}
}

# -------------------------------------------------------------------------------------------------
# EN: gg_postwar.2 -- FX-2. THE VICTOR'S RECKONING. Nobody recognises the debts of the beaten
#     government, its men go back to the countryside with their grudges, and fusion is off the
#     table. The only choice is whether to soften it late.
# ES: gg_postwar.2 -- FX-2. LA CUENTA DEL VENCEDOR. Nadie reconoce las deudas del gobierno
#     vencido, sus hombres vuelven a la campaña con el rencor, y la fusion queda descartada. Lo
#     unico que se elige es si se ablanda tarde.
# -------------------------------------------------------------------------------------------------
gg_postwar.2 = {
	type = country_event
	placement = ROOT

	title = gg_postwar.2.t
	desc = gg_postwar.2.d
	flavor = gg_postwar.2.f

	# EN/ES: reused until T9 / reusado hasta T9
	event_image = {
		video = "gg_money_reckoning"
	}

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	duration = 3

	immediate = {
		set_global_variable = gg_fusion_blocked
		add_modifier = { name = gg_postwar_unrecognised_debts years = 15 is_decaying = yes }
		every_scope_state = {
			limit = { NOT = { state_region = s:STATE_MONTEVIDEO } }
			add_modifier = { name = gg_postwar_unrest years = 10 is_decaying = yes }
		}
	}

	# a -- let them pay / que paguen
	option = {
		name = gg_postwar.2.a
		default_option = yes
		if = {
			limit = { has_variable = guerra_grande_cerrito_government }
			add_radicals = { value = large_radicals interest_group = ig:ig_intelligentsia }
			add_radicals = { value = medium_radicals interest_group = ig:ig_petty_bourgeoisie }
		}
		else = {
			add_radicals = { value = large_radicals interest_group = ig:ig_landowners }
			add_radicals = { value = medium_radicals interest_group = ig:ig_rural_folk }
		}
		add_loyalists = { value = medium_radicals interest_group = ig:ig_armed_forces }
		ai_chance = { base = 60 }
	}

	# b -- a late amnesty: the unrest eases, the army resents it / una amnistia tardia
	option = {
		name = gg_postwar.2.b
		add_treasury = -50000
		every_scope_state = {
			limit = { has_modifier = gg_postwar_unrest }
			remove_modifier = gg_postwar_unrest
		}
		add_radicals = { value = medium_radicals interest_group = ig:ig_armed_forces }
		add_modifier = { name = gg_postwar_late_amnesty years = 5 is_decaying = yes }
		ai_chance = { base = 40 }
	}
}

# -------------------------------------------------------------------------------------------------
# EN: gg_postwar.3 -- FX-3. SHARED DEBTS. As at the Pantanoso, the State takes on what both
#     governments owed. If the Defensa pledged the customs again (gg_powers.21), the bill is larger.
# ES: gg_postwar.3 -- FX-3. DEUDAS COMPARTIDAS. Como en el Pantanoso, el Estado asume lo que
#     debian los dos gobiernos. Si la Defensa volvio a hipotecar la aduana (gg_powers.21), la
#     cuenta es mayor.
# -------------------------------------------------------------------------------------------------
gg_postwar.3 = {
	type = country_event
	placement = ROOT

	title = gg_postwar.3.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = gg_postwar.3.d_mortgaged
				trigger = { has_global_variable = gg_customs_mortgaged }
			}
			triggered_desc = {
				desc = gg_postwar.3.d
				trigger = { always = yes }
			}
		}
	}
	flavor = gg_postwar.3.f

	# EN/ES: reused until T9 / reusado hasta T9
	event_image = {
		video = "gg_money_cerrito_debts"
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	# a -- both governments' debts / las deudas de los dos gobiernos
	option = {
		name = gg_postwar.3.a
		default_option = yes
		add_modifier = { name = gg_postwar_shared_debts years = 15 is_decaying = yes }
		if = {
			limit = { has_global_variable = gg_customs_mortgaged }
			add_treasury = -100000
		}
		add_loyalists = { value = medium_radicals interest_group = ig:ig_landowners }
		add_loyalists = { value = medium_radicals interest_group = ig:ig_petty_bourgeoisie }
		ai_chance = { base = 70 }
	}

	# b -- only our own / solo las propias
	option = {
		name = gg_postwar.3.b
		if = {
			limit = { has_variable = guerra_grande_cerrito_government }
			add_radicals = { value = large_radicals interest_group = ig:ig_petty_bourgeoisie }
		}
		else = {
			add_radicals = { value = large_radicals interest_group = ig:ig_landowners }
		}
		add_modifier = { name = gg_postwar_debts_disowned years = 10 is_decaying = yes }
		ai_chance = { base = 30 }
	}
}

# -------------------------------------------------------------------------------------------------
# EN: gg_postwar.4 -- FX-3. THE ELECTIONS OF 1852. Everyone expected Garzon; he died in December
#     1851 and the Chambers chose Giro, a blanco acceptable to both sides. With a powers' peace
#     (F12) the vote is held under European eyes.
# ES: gg_postwar.4 -- FX-3. LAS ELECCIONES DE 1852. Todos esperaban a Garzon; murio en diciembre
#     de 1851 y las Camaras eligieron a Giro, un blanco aceptable para los dos bandos. Con la paz
#     de las potencias (F12), se vota bajo la mirada europea.
# -------------------------------------------------------------------------------------------------
gg_postwar.4 = {
	type = country_event
	placement = ROOT

	title = gg_postwar.4.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = gg_postwar.4.d_powers
				trigger = { has_global_variable = gg_powers_peace }
			}
			triggered_desc = {
				desc = gg_postwar.4.d
				trigger = { always = yes }
			}
		}
	}
	flavor = gg_postwar.4.f

	event_image = {
		video = "gg_fl_giro"
	}

	icon = "gfx/interface/icons/event_icons/event_election.dds"

	duration = 3

	immediate = {
		# EN/ES: T9 -- Garzon died on 1 December 1851 / Garzon murio el 1 de diciembre de 1851
		every_scope_character = {
			limit = { has_template = URU_gg_eugenio_garzon }
			kill_character = yes
		}
		if = {
			limit = { has_global_variable = gg_powers_peace }
			add_modifier = { name = gg_postwar_observed_election years = 5 is_decaying = yes }
		}
	}

	# a -- a compromise blanco (Giro) / un blanco de compromiso (Giro)
	option = {
		name = gg_postwar.4.a
		default_option = yes
		set_ruling_interest_groups = { ig_landowners ig_intelligentsia }
		# EN/ES: T9 -- Giro takes office / Giro asume
		if = {
			limit = { NOT = { any_scope_character = { has_template = URU_gg_juan_francisco_giro } } }
			create_character = { template = URU_gg_juan_francisco_giro }
		}
		random_scope_character = {
			limit = { has_template = URU_gg_juan_francisco_giro }
			set_character_as_ruler = yes
		}
		add_radicals = { value = small_radicals interest_group = ig:ig_armed_forces }
		ai_chance = { base = 50 }
	}

	# b -- the Defensa's candidate / el candidato de la Defensa
	option = {
		name = gg_postwar.4.b
		set_ruling_interest_groups = { ig_petty_bourgeoisie ig_intelligentsia }
		add_radicals = { value = medium_radicals interest_group = ig:ig_landowners }
		ai_chance = { base = 25 }
	}

	# c -- the countryside's candidate / el candidato de la campaña
	option = {
		name = gg_postwar.4.c
		set_ruling_interest_groups = { ig_landowners ig_rural_folk }
		add_radicals = { value = medium_radicals interest_group = ig:ig_intelligentsia }
		ai_chance = { base = 25 }
	}
}
