# -------------------------------------------------------------------------------------------------
# EN: T8 -- THE POSTWAR (C16, FX-2, FX-3). How the peace was made decides what comes after it:
#     a punished side keeps fighting in other ways; a peace without victors opens elections and
#     the fusion policy. The router reads the marks the peace menus left.
# ES: T8 -- LA POSGUERRA (C16, FX-2, FX-3). Como se hizo la paz decide lo que viene despues: un
#     bando castigado sigue peleando por otros medios; una paz sin vencidos abre las elecciones y
#     la politica de fusion. El router lee las marcas que dejaron los menus de paz.
# -------------------------------------------------------------------------------------------------

gg_postwar.1 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_postwar.1|" # GG_TRAZA
		# EN: the handover of 1852, when the Defensa's government survives the war: Suarez resigned
		#     his command to Bernardo Berro, president of the Senate, on 15 February 1852, and Juan
		#     Francisco Giro took office as constitutional president (Wikipedia ES, Joaquin Suarez).
		# ES: el traspaso de 1852, si el gobierno de la Defensa sobrevive a la guerra: Suarez resigno
		#     el mando en Bernardo Berro, presidente del Senado, el 15 de febrero de 1852, y Juan
		#     Francisco Giro asumio como presidente constitucional.
		# EN: WHO GOVERNS AFTER THE WAR (Juan, 2026-09-17). History only rules while history holds:
		#     - the Cerrito wins: the Assembly confirms Oribe for a new four-year term, and elections
		#       come after it;
		#     - the Defensa survives a peace with no victors: the handover of 1852, Suarez to Berro
		#       and then to Giro, as it happened;
		#     - the Defensa wins outright: Suarez finishes his interim and hands over to a colorado,
		#       Flores, in 1852; from there the ballot decides.
		# ES: QUIEN GOBIERNA DESPUES DE LA GUERRA. La historia manda mientras la historia se sostiene:
		#     gana el Cerrito y la Asamblea confirma a Oribe por cuatro anios mas; sobrevive la
		#     Defensa en una paz sin vencidos y va el traspaso de 1852 a Berro y a Giro; gana la
		#     Defensa y Suarez termina su interinato y entrega a un colorado, Flores, en 1852.
		if = {
			limit = { has_variable = guerra_grande_cerrito_government }
			set_global_variable = gg_president_railroad
			gg_ruler_set = { TEMPLATE = URU_Manuel_Oribe }
			hidden_effect = {
				trigger_event = { id = gg_postwar.9 days = 1460 }
			}
		}
		else_if = {
			limit = {
				has_variable = guerra_grande_defensa_government
				has_global_variable = gg_peace_no_victors
			}
			gg_ruler_set = { TEMPLATE = URU_bernardo_prudencio_berro }
			hidden_effect = {
				trigger_event = { id = gg_postwar.7 days = 15 }
			}
		}
		else_if = {
			limit = { has_variable = guerra_grande_defensa_government }
			set_global_variable = gg_president_railroad
			gg_ruler_set = { TEMPLATE = URU_joaquin_suarez }
			hidden_effect = {
				# EN/ES: 23/9 (N5, decisiones.md §4) -- gg_postwar.8 walks to 15/2/1852 by itself / camina solo hasta la fecha
				trigger_event = { id = gg_postwar.8 days = 1 }
			}
		}
		if = {
			limit = { has_global_variable = gg_peace_punished }
			trigger_event = { id = gg_postwar.2 days = 1 popup = yes }
			# EN/ES: FX-2 -- the caudillos come back early / los caudillos vuelven antes
			trigger_event = { id = gg_postwar.5 days = 730 popup = yes }
		}
		else_if = {
			limit = { has_global_variable = gg_peace_no_victors }
			trigger_event = { id = gg_postwar.3 days = 1 popup = yes }
			# EN/ES: 22/9/2026 (event audit, decided with Juan) -- not under Oribe's four-year term:
			#     gg_postwar.9 calls this election when the term ends / no con el mandato de Oribe en curso:
			#     lo convoca gg_postwar.9 al vencer el mandato
			if = {
				limit = { NOT = { has_variable = guerra_grande_cerrito_government } }
				trigger_event = { id = gg_postwar.4 days = 150 popup = yes }
			}
			# EN/ES: FX-3 -- contained: later and softer / contenidos: mas tarde y mas blandos
			set_global_variable = gg_postwar_caudillos_contained
			trigger_event = { id = gg_postwar.5 days = 1825 popup = yes }
			trigger_event = { id = gg_postwar.6 days = 1460 popup = yes }
		}
		# EN/ES: review -- a peace with neither mark (the presidency restored, Rosas's vassal or client):
		#     elections and the caudillos, no fusion / una paz sin ninguna marca: elecciones y caudillos
		else = {
			# EN/ES: 22/9/2026 (event audit, decided with Juan) -- not under Oribe's four-year term:
			#     gg_postwar.9 calls this election when the term ends / no con el mandato de Oribe en curso:
			#     lo convoca gg_postwar.9 al vencer el mandato
			if = {
				limit = { NOT = { has_variable = guerra_grande_cerrito_government } }
				trigger_event = { id = gg_postwar.4 days = 150 popup = yes }
			}
			trigger_event = { id = gg_postwar.5 days = 1095 popup = yes }
		}
	}
}

# EN: gg_postwar.2 -- FX-2. THE VICTOR'S RECKONING. Nobody recognises the debts of the beaten
#     government, its men go back to the countryside with their grudges, and fusion is off the
#     table. The only choice is whether to soften it late.
# ES: gg_postwar.2 -- FX-2. LA CUENTA DEL VENCEDOR. Nadie reconoce las deudas del gobierno
#     vencido, sus hombres vuelven a la campaña con el rencor, y la fusion queda descartada. Lo
#     unico que se elige es si se ablanda tarde.
# -------------------------------------------------------------------------------------------------
# SOURCE / FUENTE: SOURCES_peace.md s.4 (punishment decreed and threatened by both sides).
#     design (a punitive peace did not happen; this branch is counterfactual).
gg_postwar.2 = {
	type = country_event
	placement = ROOT

	title = gg_postwar.2.t
	desc = gg_postwar.2.d
	flavor = gg_postwar.2.f

	# EN/ES: reused until T9 / reusado hasta T9
	event_image = {
		video = "gg_f2_medina"
	}

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	# EN: 22/9/2026 -- duration is in MONTHS (save: utilitarian.4 duration = 3 -> expires 3 months on).
	#     The 1825 of 17/9 (152 years) left minimized events pending forever, and vanilla's 3 answers for
	#     the player within seconds at speed 5 (Juan, 17/9): 12 reopens it at 11 months, then resolves.
	# ES: la duracion va en MESES (en el save, 3 = vence a los 3 meses). El 1825 del 17/9 (152 anos)
	#     dejaba colgados para siempre los eventos minimizados, y el 3 de vanilla contesta por el jugador
	#     en segundos a velocidad 5: 12 lo reabre a los 11 meses y recien ahi resuelve.
	duration = 12

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_postwar.2|" # GG_TRAZA
		# EN/ES: 23/9 (hidden consequences audit, row 10) -- gg_fusion_blocked removed: it only ever repeated
		#     gg_peace_punished, the mark that brings this event / se saco: solo repetia gg_peace_punished
		# EN/ES: punitive repudiation is one outcome, never debt-free plus inherited debt.
		if = { limit = { has_modifier = guerra_grande_owes_nobody } remove_modifier = guerra_grande_owes_nobody }
		if = { limit = { has_modifier = guerra_grande_debts_of_the_cerrito } remove_modifier = guerra_grande_debts_of_the_cerrito }
		add_modifier = { name = gg_postwar_unrecognised_debts years = 15 is_decaying = yes }
		every_scope_state = {
			limit = { NOT = { state_region = s:STATE_MONTEVIDEO } }
			add_modifier = { name = gg_postwar_unrest years = 10 is_decaying = yes }
		}
	}

	# a -- let them pay / que paguen
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_postwar.2.a|" # GG_TRAZA
		name = gg_postwar.2.a
		default_option = yes
		if = {
			limit = { has_variable = guerra_grande_cerrito_government }
			add_radicals = { value = large_radicals interest_group = ig:ig_intelligentsia }
			add_radicals = { value = medium_radicals interest_group = ig:ig_petty_bourgeoisie }
		}
		else = {
			add_radicals = { value = large_radicals interest_group = ig:ig_landowners }
			add_radicals = { value = medium_radicals interest_group = ig:ig_rural_folk }
		}
		add_loyalists = { value = medium_radicals interest_group = ig:ig_armed_forces }
		ai_chance = { base = 60 }
	}

	# b -- a late amnesty: the unrest eases, the army resents it / una amnistia tardia
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_postwar.2.b|" # GG_TRAZA
		name = gg_postwar.2.b
		add_treasury = -50000
		every_scope_state = {
			limit = { has_modifier = gg_postwar_unrest }
			remove_modifier = gg_postwar_unrest
		}
		add_radicals = { value = medium_radicals interest_group = ig:ig_armed_forces }
		add_modifier = { name = gg_postwar_late_amnesty years = 5 is_decaying = yes }
		ai_chance = { base = 40 }
	}
}

# EN/ES: gg_postwar.8 -- the Defensa won: Suarez holds the interim until 15 February 1852 and hands
#     it to Flores (hidden) / la Defensa gano: Suarez sostiene el interinato hasta el 15/2/1852 y entrega a Flores
# EN: 23/9/2026 (N5 of the design-vs-code audit, decisiones.md §4: "15 de febrero de 1852") -- ON the date, not on the
#     anniversary of the peace: it used to check once a year from the peace, so a peace in June 1849 handed over in
#     June 1852. Now it steps down to the day (a year, a month, a week, a day), never past the date; a peace signed
#     after it hands over the next day.
# ES: EN la fecha, no en el aniversario de la paz: chequeaba una vez por anio desde la paz, asi que una paz en junio
#     de 1849 entregaba en junio de 1852. Ahora baja hasta el dia (anio, mes, semana, dia) sin pasarse; una paz
#     firmada despues de la fecha entrega al dia siguiente.
gg_postwar.8 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_postwar.8|" # GG_TRAZA
		if = {
			limit = { game_date >= 1852.2.15 }
			# EN/ES: only Suarez's historical interim may hand the office to Flores.
			if = {
				limit = { ruler = { has_template = URU_joaquin_suarez } }
				gg_ruler_set = { TEMPLATE = URU_venancio_flores }
			}
			remove_global_variable = gg_president_railroad
		}
		else_if = {
			limit = { game_date < 1851.2.15 }
			trigger_event = { id = gg_postwar.8 days = 365 }
		}
	}
		else_if = {
			limit = { game_date < 1852.1.16 }
			trigger_event = { id = gg_postwar.8 days = 30 }
		}
		else_if = {
			limit = { game_date < 1852.2.8 }
			trigger_event = { id = gg_postwar.8 days = 7 }
		}
		else = {
			trigger_event = { id = gg_postwar.8 days = 1 }
		}
}
