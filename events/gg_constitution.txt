# EN: THE CONSTITUTION (23/9/2026, Juan: "the end of the Federal Pact, the annexation or the confederation,
#     also means choosing a constitution, and that has to be there"). A month after the settlement of the Pact
#     (gg_argentina.80, incorporation or confederation), Buenos Aires's government writes it -- or does not.
#     a) the Constitution of 1853, after Alberdi's Bases: a presidential republic; freedom of worship (art. 14);
#        no slaves (art. 15); immigration encouraged (art. 25); the press without prior censorship and the right
#        of association (art. 14). Free navigation of the rivers (art. 26) is already the mod's modifier.
#     b) the Pact without a constitution: the Restorer's regime, the sum of public power (law of 7 March 1835),
#        the Catholic religion of the State and the press under control. Not open if Urquiza won.
# ES: LA CONSTITUCION. Un mes despues del arreglo del Pacto, el gobierno de Buenos Aires la escribe, o no.
# SOURCE / FUENTE: Constitucion de la Nacion Argentina (1 May 1853), arts. 14, 15, 25, 26:
#     https://es.wikisource.org/wiki/Constituci%C3%B3n_de_la_Naci%C3%B3n_Argentina_(1853)
#     Ley de la Sala de Representantes de Buenos Aires, 7 March 1835 (suma del poder publico):
#     docs/guerra-grande-argentina/historia-argentina.md.
#     Pacto Federal, 4 Jan 1831, art. 16.5 (the General Federative Congress; the flavour quote):
#     https://es.wikisource.org/wiki/Pacto_Federal_(1831). Urquiza's decree of 1 May 1851 changing
#     the motto to "Mueran los enemigos de la Organizacion Nacional": es.wikisource
#     "Pronunciamiento de Urquiza"; argentinahistorica.com.ar doc 131. Urquiza read, praised and
#     promoted the Bases, which shaped the Santa Fe Congress: CONICET study, ri.conicet.gov.ar
#     handle 11336/113669. 23/9: the Bases (1852) left the pre-1852 descs and the flavour.
# -------------------------------------------------------------------------------------------------

	placement = ROOT

	desc = {

			triggered_desc = { desc = gg_constitution.1.d_urquiza trigger = { has_global_variable = gg_pact_settlement_urquiza } }

		}

	icon = "gfx/interface/icons/event_icons/event_election.dds"
	duration = 12
	is_popup = yes

	trigger = {
		has_variable = gg_arg_rosas_seat
		NOT = { has_global_variable = gg_constitution_chosen }
	}

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution.1|" # GG_TRAZA
	}

	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution.1.a|" # GG_TRAZA
		name = gg_constitution.1.a
		default_option = yes
		# EN: 23/9/2026 (Juan's decision, decisiones.md; bug hunt run 205411, AUD-CONST-ROSAS) -- a Rosas victory means
		#     no constitution: autocracy while he lives, the question reopened at his death (gg_constitution.2). The
		#     AI took this option about a third of the time, and the elections it brought kept putting Rosas back.
		# ES: la victoria de Rosas es sin constitucion: autocracia mientras viva, la cuestion se reabre a su muerte.

		set_global_variable = { name = gg_constitution_chosen value = 1 }
		custom_tooltip = gg_constitution.1.a_tt

		add_loyalists = { value = medium_radicals interest_group = ig:ig_intelligentsia }
		add_radicals = { value = medium_radicals interest_group = ig:ig_devout }
		add_radicals = { value = small_radicals interest_group = ig:ig_landowners }
		ai_chance = {
			base = 50
			modifier = { trigger = { has_global_variable = gg_pact_settlement_urquiza } add = 100 }
			modifier = { trigger = { has_global_variable = gg_pact_settlement_constitutional } add = 30 }
			modifier = { trigger = { OR = { gg_pact_historical = yes gg_pact_negotiated = yes } } add = 10000 } # EN/ES: the Pact's game rule / la regla del Pacto
			modifier = { trigger = { gg_pact_rosas = yes } factor = 0 } # EN/ES: 23/9 (AUD-RULES-AI) -- another setting of the rule: never for the AI / otra opcion de la regla: nunca para la IA
		}
	}
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution.1.b|" # GG_TRAZA
		name = gg_constitution.1.b
		trigger = { NOT = { has_global_variable = gg_pact_settlement_urquiza } }
		set_global_variable = { name = gg_constitution_chosen value = 2 }
		custom_tooltip = gg_constitution.1.b_tt
		activate_law = law_type:law_autocracy
		activate_law = law_type:law_state_religion
		activate_law = law_type:law_censorship
		# EN/ES: 23/9 (Juan) -- the question stays open under him / la cuestion sigue abierta por debajo
		add_modifier = { name = gg_pais_sin_constitucion }
		add_loyalists = { value = medium_radicals interest_group = ig:ig_landowners }
		add_loyalists = { value = small_radicals interest_group = ig:ig_devout }
		add_radicals = { value = medium_radicals interest_group = ig:ig_intelligentsia }
		ai_chance = {
			base = 30
			modifier = { trigger = { has_global_variable = gg_pact_settlement_rosas } add = 60 }
			modifier = { trigger = { gg_pact_rosas = yes } add = 10000 } # EN/ES: the Pact's game rule / la regla del Pacto
			modifier = { trigger = { OR = { gg_pact_historical = yes gg_pact_negotiated = yes } } factor = 0 } # EN/ES: 23/9 (AUD-RULES-AI) -- another setting of the rule: never for the AI / otra opcion de la regla: nunca para la IA
		}
	}
}

# EN: 23/9/2026 -- the day after the war between federales and unitarios ends (or at once if nobody rose): the winner
#     writes the constitution, the country is one either way (a federation or a unitary state), the autocracy ends.
# ES: el dia despues del fin de la guerra entre federales y unitarios: el que gana escribe la constitucion.
gg_constitution.3 = {
	type = country_event
	hidden = yes
	trigger = { c:ARG ?= this }
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution.3|" # GG_TRAZA
		if = { limit = { has_global_variable = gg_unitario_war } remove_global_variable = gg_unitario_war }
		# EN: 23/9/2026 (Codex audit, AUD-CONST-PEACE) -- the winner is the one recorded when the war ended
		#     (gg_unitario_* in common/on_actions/gg_rosas_succession.txt), as the Pact's wars do. It used to be read
		#     from the map, and a white peace (the league still independent, no war) passed for a unitario victory.
		#     A WHITE PEACE: the federales keep the government and write the federal constitution (design choice,
		#     decisiones.md 23/9) -- the league rose to overturn them and did not.
		# ES: el ganador es el anotado al terminar la guerra, como en las del Pacto. Se leia del mapa, y una paz blanca
		#     pasaba por victoria unitaria. PAZ BLANCA: los federales siguen en el gobierno y escriben la federal.
		if = {
			limit = { global_var:gg_unitario_unitarios_won ?= 1 }
			set_global_variable = { name = gg_constitution_outcome value = 2 }
			gg_constitution_unitaria = yes
		}
		else_if = {
			limit = {
				has_global_variable = gg_unitario_war_ended
				NOT = { global_var:gg_unitario_federales_won ?= 1 }
			}
			set_global_variable = { name = gg_constitution_outcome value = 3 }
			gg_constitution_federal = yes
		}
		else = {
			set_global_variable = { name = gg_constitution_outcome value = 1 }
			gg_constitution_federal = yes
		}
		# EN/ES: the league's ground joins the one country too / la tierra de la liga tambien se incorpora
		if = { limit = { exists = c:CLN } annex_with_incorporation = c:CLN }
		trigger_event = { id = gg_constitution.4 days = 0 popup = yes }
	}
}

# EN: 23/9/2026 (Codex audit, AUD-SEAT-ANNEX) -- the constitution makes one country, and the province this player
#     governs goes into it (gg_argentina_province_join). As vanilla does with Cuba (cuba_independencia.6), the player
#     takes the seat of the country that absorbs his, Argentina; the annexation runs the next day (gg_constitution.6),
#     with the player already there. One option, so the default is the same one. play_as last, through a saved
#     scope (gg_argentina.1.b, gg_ba.13).
# ES: la constitucion hace un solo pais, y la provincia que gobierna este jugador entra en el. Como hace vanilla con
#     Cuba, el jugador pasa a la sede del pais que absorbe al suyo; la anexion corre al dia siguiente.
gg_constitution.5 = {
	type = country_event
	placement = ROOT
	title = gg_constitution.5.t
	desc = gg_constitution.5.d
	flavor = gg_constitution.5.f
	event_image = { video = "unspecific_ruler_speaking_to_people" }
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 12
	is_popup = yes

	# EN/ES: once per country: gg_constitution.6 runs the incorporation again / una vez por pais: .6 vuelve a correr la incorporacion
	trigger = {
		exists = c:ARG
		NOT = { has_variable = gg_join_seat_offered }
	}

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution.5|" # GG_TRAZA
		set_variable = gg_join_seat_offered
	}

	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_constitution.5.a|" # GG_TRAZA
		name = gg_constitution.5.a
		default_option = yes
		custom_tooltip = gg_play_as_argentina_tt
		ai_chance = { base = 100 }
		hidden_effect = {
			c:ARG ?= {
				trigger_event = { id = gg_constitution.6 days = 1 }
				save_scope_as = gg_new_seat
			}
			play_as = scope:gg_new_seat
		}
	}
}
