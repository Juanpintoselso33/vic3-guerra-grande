# EN: URQUIZA WEIGHS IT (23/9/2026, Juan). On 1 May 1851 Entre Rios no longer pronounces by calendar: its

# ES: URQUIZA LO PESA. El 1/5/1851 Entre Rios ya no se pronuncia por calendario: se suma su descontento y hay
#     tres salidas. Aca la intermedia (.1, una representacion sin armas) y el caso en que no tiene con quien
#     firmar (.2, Urquiza solo). El pronunciamiento y la confederacion negociada siguen.
# SOURCE / FUENTE: the representations the provinces addressed to Buenos Aires under the Pact of 1831 (the

# EN: 1 May 1851, discontent 2-3. Entre Rios keeps the mandate in Buenos Aires but sends a formal representation:
#     the constitutional question, the rivers, the orders of 1850. Rosas concedes (the negotiated confederation) or
#     refuses (the pronouncement comes in three months).
# ES: 1/5/1851, descontento 2-3. Entre Rios mantiene el encargo pero manda una representacion formal. Rosas cede
#     (confederacion negociada) o se niega (el pronunciamiento llega en tres meses).
gg_urquiza.1 = {
	type = country_event
	placement = ROOT
	title = gg_urquiza.1.t
	desc = gg_urquiza.1.d
	flavor = gg_urquiza.1.f
	event_image = { video = "unspecific_politicians_arguing" }
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 12
	is_popup = yes

	trigger = {
		c:ARG ?= this
		gg_entre_rios_with_rosas = yes # EN/ES: 23/9 (Juan, §2) -- or the Entre Rios of Alcaraz / o el de Alcaraz
	}

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza.1|" # GG_TRAZA
		gg_reclamo_abre = { KEY = entre_rios }
	}

	# EN: concede: the constituent route, and Entre Rios stays / ceder: la ruta constituyente, y Entre Rios se queda
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza.1.a|" # GG_TRAZA
		name = gg_urquiza.1.a
		default_option = yes
		custom_tooltip = gg_urquiza.1.a_tt
		gg_patron_change = { PATRON = urquiza DELTA = 10 }
		trigger_event = { id = gg_argentina.71 days = 0 popup = yes }
		ai_chance = { base = 60 }
	}
	# EN: refuse: the representation becomes a pronouncement / negarse: la representacion se vuelve pronunciamiento
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza.1.b|" # GG_TRAZA
		name = gg_urquiza.1.b
		custom_tooltip = gg_urquiza.1.b_tt
		gg_patron_change = { PATRON = urquiza DELTA = -10 }
		trigger_event = { id = gg_argentina.3 days = 90 popup = yes }
		ai_chance = {
			base = 40
			modifier = { trigger = { gg_rosas_won_guerra_grande = yes } add = 20 }
		}
	}
}

# EN: 4/2/1852, the day after Caseros -- under a game rule that ends the Pact by war, a Pact still open is closed
#     today the rule's way: the war with Entre Rios, if it lasts, ends, and the settlement is read (gg_pact_resolve).
#     Hidden; no trigger, because a scheduled event's trigger is checked when it is scheduled. A siege of Buenos
#     Aires (Caseros option c) resolves itself when it ends. 23/9 (rules/AI audit): a day after Caseros
#     (gg_argentina_cal.10, day 5873) because the order of two events on the same day is not guaranteed.
# ES: 4/2/1852, el dia despues de Caseros -- con una regla que cierra el Pacto por la guerra, el Pacto que siga
#     abierto se cierra hoy a su modo: termina la guerra con Entre Rios si dura, y se lee el arreglo. Oculto; sin
#     trigger. Un sitio de Buenos Aires se resuelve solo al terminar. Un dia despues de Caseros porque el orden de
#     dos eventos del mismo dia no esta garantizado.
gg_urquiza.3 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza.3|" # GG_TRAZA
		if = {
			limit = {
				exists = s:STATE_ENTRE_RIOS # EN/ES: the «la Argentina» guard, as gg_argentina_cal.4 / la guarda de «la Argentina»
				c:ARG ?= this
				gg_pact_forced_war = yes
				NOT = { has_global_variable = gg_pact_settled }
				NOT = { has_global_variable = gg_caseros_siege }
				NOT = { has_global_variable = gg_confederacion_negociada }
			}
			# EN/ES: 23/9 (Codex audit, AUD-URQ-NET) -- only the Buenos Aires - Entre Rios play, as gg_caseros.1; another
			#     Pact war (Corrientes, the Coalition) goes on / solo la jugada Buenos Aires - Entre Rios; otra sigue
			if = {
				limit = { exists = c:ENR }
				every_diplomatic_play = {
					limit = {
						is_diplomatic_play_type = dp_gg_pacto_federal
						initiator_is = c:ARG
						target_is = c:ENR
					}
					end_play = yes
				}
			}
			gg_pact_resolve = yes
		}
		# EN: 23/9/2026 (P5 of the endings audit, AUD-PACT-NEG-NET) -- the negotiated rule's net: if the Pronouncement
		#     never came (Entre Rios was not Argentine ground on 1/5/1851: Corrientes or the Coalition held it), the
		#     Pact is closed today as the rule says, with the negotiated confederation. Otherwise it stayed open for
		#     good: no constitution and no Buenos Aires apart.
		# ES: la red de la regla negociada: si no hubo Pronunciamiento (Entre Rios no era suelo argentino el 1/5/1851),
		#     el Pacto se cierra hoy como dice la regla, con la confederacion negociada. Si no, quedaba abierto para siempre.
		else_if = {
			limit = {
				exists = s:STATE_ENTRE_RIOS # EN/ES: the «la Argentina» guard / la guarda de «la Argentina»
				c:ARG ?= this
				gg_pact_negotiated = yes
				NOT = { has_global_variable = gg_pact_settled }
				NOT = { has_global_variable = gg_caseros_siege }
				NOT = { has_global_variable = gg_confederacion_negociada }
			}
			gg_argentina_open_negotiated_confederation = yes
		}
	}
}
