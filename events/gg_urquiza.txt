# EN: 4/2/1852, the day after Caseros -- under a game rule that ends the Pact by war, a Pact still open is closed
#     today the rule's way: the war with Entre Rios, if it lasts, ends, and the settlement is read (gg_pact_resolve).
#     Hidden; no trigger, because a scheduled event's trigger is checked when it is scheduled. A siege of Buenos
#     Aires (Caseros option c) resolves itself when it ends. 23/9 (rules/AI audit): a day after Caseros
#     (gg_argentina_cal.10, day 5873) because the order of two events on the same day is not guaranteed.
# ES: 4/2/1852, el dia despues de Caseros -- con una regla que cierra el Pacto por la guerra, el Pacto que siga
#     abierto se cierra hoy a su modo: termina la guerra con Entre Rios si dura, y se lee el arreglo. Oculto; sin
#     trigger. Un sitio de Buenos Aires se resuelve solo al terminar. Un dia despues de Caseros porque el orden de
#     dos eventos del mismo dia no esta garantizado.
gg_urquiza.3 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza.3|" # GG_TRAZA
		if = {
			limit = {
				c:ARG ?= this
				gg_pact_forced_war = yes
				NOT = { has_global_variable = gg_pact_settled }
				NOT = { has_global_variable = gg_caseros_siege }
				NOT = { has_global_variable = gg_confederacion_negociada }
			}
			# EN/ES: 23/9 (Codex audit, AUD-URQ-NET) -- only the Buenos Aires - Entre Rios play, as gg_caseros.1; another
			#     Pact war (Corrientes, the Coalition) goes on / solo la jugada Buenos Aires - Entre Rios; otra sigue
			if = {
				limit = { exists = c:ENR }
				every_diplomatic_play = {
					limit = {
						is_diplomatic_play_type = dp_gg_pacto_federal
						initiator_is = c:ARG
						target_is = c:ENR
					}
					end_play = yes
				}
			}
			gg_pact_resolve = yes
		}
		# EN: 23/9/2026 (P5 of the endings audit, AUD-PACT-NEG-NET) -- the negotiated rule's net: if the Pronouncement
		#     never came (Entre Rios was not Argentine ground on 1/5/1851: Corrientes or the Coalition held it), the
		#     Pact is closed today as the rule says, with the negotiated confederation. Otherwise it stayed open for
		#     good: no constitution and no Buenos Aires apart.
		# ES: la red de la regla negociada: si no hubo Pronunciamiento (Entre Rios no era suelo argentino el 1/5/1851),
		#     el Pacto se cierra hoy como dice la regla, con la confederacion negociada. Si no, quedaba abierto para siempre.
		else_if = {
			limit = {
				c:ARG ?= this
				gg_pact_negotiated = yes
				NOT = { has_global_variable = gg_pact_settled }
				NOT = { has_global_variable = gg_caseros_siege }
				NOT = { has_global_variable = gg_confederacion_negociada }
			}
			gg_argentina_open_negotiated_confederation = yes
		}
	}
}

				}
				trigger_event = { id = gg_urquiza.6 days = 30 }
			}
			else_if = {
				limit = { gg_entre_rios_with_rosas = yes }
				trigger_event = { id = gg_urquiza.5 days = 0 }
			}
			else = {
				set_global_variable = gg_urquiza_1851_answered
				debug_log = "GG_URQ1851|[TimeKeeper.GetCurrentDate.GetString]|overtaken: Entre Rios broke away before answering|" # GG_TRAZA
			}
		}
	}
