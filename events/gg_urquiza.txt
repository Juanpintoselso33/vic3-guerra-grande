# EN: 22/2/1851 (relay gg_argentina_cal.35), every game, under every Pact rule. 24/9/2026 (Juan: «almost never
#     fires, that is a bug») -- it was the 2-3 band of 1 May and a base discontent of 9 almost never reached it.
#     Urquiza writes to Rufino de Elizalde a letter meant for Rosas: the Gaceta Mercantil gives Arroyo Grande to
#     Oribe and Pacheco alone, Baldomero Garcia left Urquiza out of his speech of 2 November, the
#     customs of Buenos Aires is deaf to its claims (the export of gold), and the port captain takes a declaration
#     from every skipper coming from Entre Rios, as if it were an enemy. He keeps the mandate in Buenos Aires.
#     Historically Elizalde never showed the letter to Rosas and nothing was answered; here Rosas answers.
#     THE ANSWER IS WHAT 1 MAY READS. Heard: one grievance less, and in the middle band (discontent 2-3) the
#     negotiated confederation. Unanswered: one grievance more, and in the middle band the pronouncement. The
#     Pact's game rules still decide first, and 4+ or 1 or less decide by themselves (gg_argentina_cal.4).
# ES: 22/2/1851, en toda partida y con cualquier regla del Pacto. Era la banda 2-3 del 1/5 y con un descontento
#     base de 9 casi nunca se llegaba. Urquiza le escribe a Rufino de Elizalde una carta para Rosas: la Gaceta le da
#     Arroyo Grande solo a Oribe y Pacheco, Baldomero Garcia omitio a Urquiza el 2 de noviembre, la
#     aduana es sorda a sus reclamos (la extraccion del oro) y el capitan del puerto les toma declaracion a los
#     patrones que vienen de Entre Rios como si fuera enemiga. Mantiene el encargo. En la historia Elizalde no le
#     mostro la carta a Rosas y nadie contesto; aca Rosas contesta. LA RESPUESTA ES LO QUE LEE EL 1/5: atendida, un
#     agravio menos y, en la banda media, la confederacion negociada; sin respuesta, un agravio mas y, en la banda
#     media, el pronunciamiento. Las reglas del Pacto siguen decidiendo primero.
# SOURCE / FUENTE: Jose Maria Rosa, «El pronunciamiento de Urquiza», cap. III, «En Entre Rios (enero, febrero de
#     1851)»: the letter of 22/2/1851 and its quotations (his note 43) / la carta y sus citas:
#     https://institutorosas.cultura.gob.ar/media/uploads/site-35/multimedia/rosa-jose-maria-el-pronunciamiento-de-urquiza.pdf
#     Elizalde's letter of 15/1/1851 and the context / la carta de Elizalde y el contexto: «Una lectura del
#     pronunciamiento de Justo Jose de Urquiza», Almanack (SciELO), https://www.scielo.br/j/alm/a/sjhGsFChMMt49y8yC99B4yc/?lang=es
gg_urquiza.1 = {
	type = country_event
	placement = ROOT
	title = gg_urquiza.1.t
	desc = gg_urquiza.1.d
	flavor = gg_urquiza.1.f
	event_image = { video = "gg_urq_general" }
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 12
	is_popup = yes

	trigger = {
		c:ARG ?= this
		NOT = { has_global_variable = gg_arg_cal_done_35 }
		# EN/ES: 24/9 -- wider than gg_entre_rios_with_rosas: the letter comes every game / la carta sale en toda partida
		gg_urquiza_prelude_possible = yes
	}

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza.1|" # GG_TRAZA
		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = gg_arg_cal_done_35 } }
				set_global_variable = gg_arg_cal_done_35
			}
		}
		# EN/ES: the claim is the province's: only while Entre Rios is still Rosas's / el reclamo es de la provincia
		if = {
			limit = { gg_entre_rios_with_rosas = yes }
			gg_reclamo_abre = { KEY = entre_rios }
		}
	}

	# EN: hear it: the grievance answered, and Urquiza takes note / atenderla: el agravio contestado, y Urquiza lo anota
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza.1.a|" # GG_TRAZA
		name = gg_urquiza.1.a
		default_option = yes
		# EN/ES: the grievance count moves only while Entre Rios is Rosas's / la cuenta se mueve solo si Entre Rios es de Rosas
		if = { limit = { gg_entre_rios_with_rosas = yes } custom_tooltip = gg_urquiza.1.a_tt }
		else = { custom_tooltip = gg_urquiza.1.a_far_tt }
		gg_patron_change = { PATRON = urquiza DELTA = 10 }
		# EN/ES: read by gg_argentina_cal.4 in the middle band / la lee el 1/5 en la banda media
		set_global_variable = gg_urquiza_representation_heard
		# EN/ES: the gg_pact_all_grievances_down pattern / el patron de gg_pact_all_grievances_down
		if = {
			limit = {
				gg_entre_rios_with_rosas = yes
				has_global_variable = gg_entre_rios_agravio
			}
			if = {
				limit = { global_var:gg_entre_rios_agravio > 0 }
				change_global_variable = { name = gg_entre_rios_agravio add = -1 }
			}
			if = {
				limit = { global_var:gg_entre_rios_agravio = 0 }
				gg_reclamo_cierra = { KEY = entre_rios }
			}
		}
		ai_chance = { base = 60 }
	}
	# EN: refuse: the representation goes unanswered / negarse: la representacion queda sin respuesta
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza.1.b|" # GG_TRAZA
		name = gg_urquiza.1.b
		# EN/ES: the grievance count moves only while Entre Rios is Rosas's / la cuenta se mueve solo si Entre Rios es de Rosas
		if = { limit = { gg_entre_rios_with_rosas = yes } custom_tooltip = gg_urquiza.1.b_tt }
		else = { custom_tooltip = gg_urquiza.1.b_far_tt }
		gg_patron_change = { PATRON = urquiza DELTA = -10 }
		if = {
			limit = {
				gg_entre_rios_with_rosas = yes
				has_global_variable = gg_entre_rios_agravio
			}
			change_global_variable = { name = gg_entre_rios_agravio add = 1 }
		}
		ai_chance = {
			base = 40
			modifier = { trigger = { gg_rosas_won_guerra_grande = yes } add = 20 }
		}
	}
}

# EN: 4/2/1852, the day after Caseros -- under a game rule that ends the Pact by war, a Pact still open is closed
#     today the rule's way: the war with Entre Rios, if it lasts, ends, and the settlement is read (gg_pact_resolve).
#     Hidden; no trigger, because a scheduled event's trigger is checked when it is scheduled. A siege of Buenos
#     Aires (Caseros option c) resolves itself when it ends. 23/9 (rules/AI audit): a day after Caseros
#     (gg_argentina_cal.10, day 5873) because the order of two events on the same day is not guaranteed.
# ES: 4/2/1852, el dia despues de Caseros -- con una regla que cierra el Pacto por la guerra, el Pacto que siga
#     abierto se cierra hoy a su modo: termina la guerra con Entre Rios si dura, y se lee el arreglo. Oculto; sin
#     trigger. Un sitio de Buenos Aires se resuelve solo al terminar. Un dia despues de Caseros porque el orden de
#     dos eventos del mismo dia no esta garantizado.
gg_urquiza.3 = {
	type = country_event
	hidden = yes

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_urquiza.3|" # GG_TRAZA
		if = {
			limit = {
				exists = s:STATE_ENTRE_RIOS # EN/ES: the «la Argentina» guard, as gg_argentina_cal.4 / la guarda de «la Argentina»
				c:ARG ?= this
				gg_pact_forced_war = yes
				NOT = { has_global_variable = gg_pact_settled }
				NOT = { has_global_variable = gg_caseros_siege }
				NOT = { has_global_variable = gg_confederacion_negociada }
			}
			# EN/ES: 23/9 (Codex audit, AUD-URQ-NET) -- only the Buenos Aires - Entre Rios play, as gg_caseros.1; another
			#     Pact war (Corrientes, the Coalition) goes on / solo la jugada Buenos Aires - Entre Rios; otra sigue
			if = {
				limit = { exists = c:ENR }
				every_diplomatic_play = {
					limit = {
						is_diplomatic_play_type = dp_gg_pacto_federal
						initiator_is = c:ARG
						target_is = c:ENR
					}
					end_play = yes
				}
			}
			gg_pact_resolve = yes
		}
		# EN: 23/9/2026 (P5 of the endings audit, AUD-PACT-NEG-NET) -- the negotiated rule's net: if the Pronouncement
		#     never came (Entre Rios was not Argentine ground on 1/5/1851: Corrientes or the Coalition held it), the
		#     Pact is closed today as the rule says, with the negotiated confederation. Otherwise it stayed open for
		#     good: no constitution and no Buenos Aires apart.
		# ES: la red de la regla negociada: si no hubo Pronunciamiento (Entre Rios no era suelo argentino el 1/5/1851),
		#     el Pacto se cierra hoy como dice la regla, con la confederacion negociada. Si no, quedaba abierto para siempre.
		else_if = {
			limit = {
				exists = s:STATE_ENTRE_RIOS # EN/ES: the «la Argentina» guard / la guarda de «la Argentina»
				c:ARG ?= this
				gg_pact_negotiated = yes
				NOT = { has_global_variable = gg_pact_settled }
				NOT = { has_global_variable = gg_caseros_siege }
				NOT = { has_global_variable = gg_confederacion_negociada }
			}
			gg_argentina_open_negotiated_confederation = yes
		}
	}
}
