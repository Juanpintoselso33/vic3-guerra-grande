# EN: 11 August 1845. Buenos Aires decides how hard to push the reunification it offers the Cerrito.
# ES: 11 de agosto de 1845. Buenos Aires decide cuanto empujar la reunificacion que le ofrece al Cerrito.
gg_cisplatina.1 = {
	type = country_event
	placement = ROOT
	title = gg_cisplatina.1.t
	desc = gg_cisplatina.1.d
	flavor = gg_cisplatina.1.f
	event_image = { video = "unspecific_politicians_arguing" }
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 12
	is_popup = yes

	trigger = {
		c:ARG ?= this
		has_global_variable = guerra_grande_open_war
		any_country = { has_variable = guerra_grande_cerrito_government }
	}

	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_cisplatina.1|" # GG_TRAZA
	}

	# EN: press him. Oribe weighs it with Rosas's whole weight behind the request.
	# ES: insistir. Oribe lo pesa con todo el respaldo de Rosas detras del pedido.
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_cisplatina.1.a|" # GG_TRAZA
		name = gg_cisplatina.1.a
		set_global_variable = gg_arg_reincorporation_asked
		custom_tooltip = gg_cisplatina.1.a_tt
		# EN/ES: the Litoral fears a bigger Buenos Aires; London and Paris want the Republic and the river
		#     open / el Litoral teme una Buenos Aires mas grande; Londres y Paris quieren la Republica
		gg_patron_change = { PATRON = urquiza DELTA = -10 }
		c:GBR ?= { change_relations = { country = ROOT value = -25 } }
		c:FRA ?= { change_relations = { country = ROOT value = -25 } }
		gg_cis_offer_go = yes
		ai_chance = {
			base = 40
			modifier = { trigger = { global_var:gg_patron_rosas >= 80 } add = 40 }
		}
	}
	# EN: leave it as an offer, as it was historically: Oribe sent it to a commission.
	# ES: dejarlo como oferta, como fue: Oribe la mando a una comision.
	option = {
		debug_log = "GG_OPC|[TimeKeeper.GetCurrentDate.GetString]|gg_cisplatina.1.b|" # GG_TRAZA
		name = gg_cisplatina.1.b
		default_option = yes
		custom_tooltip = gg_cisplatina.1.b_tt
		gg_cis_offer_go = yes
		ai_chance = { base = 60 }
	}
}

		custom_tooltip = gg_cisplatina.2.d_tt
		gg_cis_open_war = yes
		ai_chance = {
			base = 20
			modifier = { trigger = { global_var:gg_patron_brasil >= 50 } add = 10 }

# EN: The day after the war ends (on_war_end fires before the winner is recorded; gg_cis_on_war_end).
# ES: El dia despues del fin de la guerra (on_war_end corre antes de que se anote el ganador).
gg_cisplatina.3 = {
	type = country_event
	hidden = yes
	trigger = { c:BRZ ?= this }
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_cisplatina.3|" # GG_TRAZA
		remove_global_variable = gg_cis_war
		if = {
			limit = { has_global_variable = gg_cis_brazil_won }
			if = {
				limit = { global_var:gg_cis_goal = 2 }
				gg_cis_annex = yes
				set_global_variable = { name = gg_cis_outcome value = 2 }
			}
			else = {
				gg_cis_release = yes
				if = {
					limit = { global_var:gg_cis_goal = 3 c:URU ?= { is_subject = no } }
					create_diplomatic_pact = { country = c:URU type = protectorate }
					set_global_variable = { name = gg_cis_outcome value = 3 }
				}
				else = { set_global_variable = { name = gg_cis_outcome value = 1 } }
			}
		}
		else = { set_global_variable = { name = gg_cis_outcome value = 0 } }
		trigger_event = { id = gg_cisplatina.4 days = 0 popup = yes }
		c:ARG ?= { trigger_event = { id = gg_cisplatina.4 days = 0 popup = yes } }
	}
}
