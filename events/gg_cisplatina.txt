# EN: The day after the war ends (on_war_end fires before the winner is recorded; gg_cis_on_war_end).
# ES: El dia despues del fin de la guerra (on_war_end corre antes de que se anote el ganador).
gg_cisplatina.3 = {
	type = country_event
	hidden = yes
	trigger = { c:BRZ ?= this }
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_cisplatina.3|" # GG_TRAZA
		remove_global_variable = gg_cis_war
		if = {
			limit = { has_global_variable = gg_cis_brazil_won }
			if = { limit = { has_global_variable = gg_oriental_in_pact } remove_global_variable = gg_oriental_in_pact }
			if = {
				limit = { global_var:gg_cis_goal = 2 }
				# EN/ES: the Pact province is a country: Brazil annexes it whole / la provincia del Pacto es un pais
				if = { limit = { c:URU ?= { is_subject_of = c:ARG } } c:URU = { gg_bring_characters_to = { TARGET = root } } annex = c:URU } # EN/ES: 25/9 (AUD-CHAR-ANNEX) -- its men come along / sus hombres vienen con ella
				else = { gg_cis_annex = yes }
				remove_global_variable = gg_oriental_province
				set_global_variable = { name = gg_cis_outcome value = 2 }
			}
			else = {
				if = {
					limit = { c:URU ?= { is_subject_of = c:ARG } }
					c:URU = { make_independent = yes gg_subject_unwatch = yes } # EN/ES: 24/9 (AUD-VASSAL-RECOGNIZED) -- a release on purpose / liberacion a proposito
					remove_global_variable = gg_oriental_province
				}
				else = { gg_cis_release = yes }
				if = {
					limit = { global_var:gg_cis_goal = 3 c:URU ?= { is_subject = no } }
					create_diplomatic_pact = { country = c:URU type = protectorate }
					set_global_variable = { name = gg_cis_outcome value = 3 }
				}
				else = { set_global_variable = { name = gg_cis_outcome value = 1 } }
			}
		}
		else = { set_global_variable = { name = gg_cis_outcome value = 0 } }
		trigger_event = { id = gg_cisplatina.4 days = 0 popup = yes }
		c:ARG ?= { trigger_event = { id = gg_cisplatina.4 days = 0 popup = yes } }
	}
}

# EN: 23/9/2026 (Juan) -- Brazil's monthly look at the Plata while it waits (gg_cis_wait).
# ES: la mirada mensual del Brasil sobre el Plata mientras espera.
gg_cisplatina.7 = {
	type = country_event
	hidden = yes
	trigger = { c:BRZ ?= this }
	immediate = {
		debug_log = "GG_EV|[TimeKeeper.GetCurrentDate.GetString]|gg_cisplatina.7|" # GG_TRAZA
		if = {
			limit = {
				has_global_variable = gg_cis_waiting
				OR = {
					NOT = { has_global_variable = gg_oriental_in_pact }
					NOT = { exists = c:ARG }
					game_date >= 1853.1.1
				}
			}
			remove_global_variable = gg_cis_waiting
		}
		else_if = {
			limit = {
				has_global_variable = gg_cis_waiting
				OR = {
					c:ARG ?= { is_at_war = yes }
					army_size > c:ARG.army_size
				}
			}
			remove_global_variable = gg_cis_waiting
			gg_cis_open_war = yes
		}
		else_if = {
			limit = { has_global_variable = gg_cis_waiting }
			trigger_event = { id = gg_cisplatina.7 days = 30 }
		}
	}
}
